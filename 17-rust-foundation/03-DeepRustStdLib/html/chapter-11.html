典型实操

use std::sync::{Arc, Mutex, mpsc};
use std::thread;

// Mutex 共享内存
let data = Arc::new(Mutex::new(0));
let mut handles = vec![];
for _ in 0..4 {
    let d = Arc::clone(&data);
    handles.push(thread::spawn(move || {
        let mut val = d.lock().unwrap();
        *val += 1;
    }));
}
for h in handles { h.join().unwrap(); }

// MPSC 通信
let (tx, rx) = mpsc::channel();
let tx2 = tx.clone();
thread::spawn(move || { tx.send(1).unwrap(); });
thread::spawn(move || { tx2.send(2).unwrap(); });
println!("{} {}", rx.recv().unwrap(), rx.recv().unwrap());

11.1

Futex 分析

章索引：第 11 章 · 前：11.0 · 总览 · 后：11.2 · Mutex

一句话

Futex（Fast Userspace muTEX）— 无竞争 时 纯用户态 CAS；有竞争 才 futex_wait/wake 进内核 — 是 Mutex/RwLock/Condvar/MPSC 阻塞 的 OS 层底座。

原书要点

锁状态字（典型 AtomicU32）

值 含义
0 无锁 · 空闲
1 已加锁 · 无等待线程
≥2 已加锁 · 有线程在等

加锁 / 解锁

lock():
  CAS 0→1 成功 → 纯用户态，返回
  CAS 失败   → futex_wait（内核阻塞）

unlock():
  原子重置状态
  若存在等待者 → futex_wake（唤醒一个或多个）

平台差异

平台 机制
Linux 原生 syscall(SYS_futex, …)
macOS futex 类接口（版本依 OS）
Windows 无原生 futex · WaitOnAddress / SRWLock / Event 模拟

适配层统一 wait / wake · 上层 零 cfg。

↔ 4.1.1 · atomic · 05-atomic

与第 8 章

Arc 管 内存生命周期 · Futex+Mutex 管 数据竞争 — 无畏并发两条腿。

HFT

锁 热点竞争 → 慢路径 futex 唤醒 · 实盘常改 无锁/分片 减 wake 成本。

一句话

结构组成

pub struct Mutex<T> {
    inner: sys::Mutex,       // 底层 futex 锁
    poison: AtomicBool,      // panic 持锁 → 中毒

章索引：第 11 章 · 前：11.2.3 · 后：11.3.1 · OS 层

一句话

原书要点

notify_one() 唤醒 一个 等待线程

线程 A: lock → 条件不满足 → wait（释放锁 + 睡）

一句话

锁 word 状态（回顾）

值 含义

1 locked · 无 waiter

A. 加锁快路径 + 慢路径

// 概念：std Mutex 内部 lock 逻辑
fn lock(&self) {
    loop {
        // 快路径：尝试 0 → 1
        match self.state.compare_exchange_weak(0, 1, AcqRel, Acquire) {
            Ok(_) => return, // 获锁，零 syscall
            Err(actual) => {
                // 慢路径：已锁或有竞争
                self.contended_lock(actual);
            }
        }
    }
}

fn contended_lock(&self, mut state: u32) {
    loop {
        // 标记存在等待者（示意：1 → 2 或类似编码）
        if state == 1 {
            match self.state.compare_exchange_weak(1, 2, Relaxed, Relaxed) {
                Ok(_) => {}
                Err(s) => { state = s; continue; }
            }
        }
        // 内核睡眠：仅当 state 仍为「已锁+等待」时才 block
        futex_wait(&self.state, 2); // SAFETY: 合法 futex 地址
        state = self.state.load(Acquire);
        // 被唤醒后再 CAS 抢锁
        match self.state.compare_exchange_weak(0, 1, AcqRel, Acquire) {
            Ok(_) => return,
            Err(s) => state = s,
        }
    }
}

B. 解锁 + 唤醒

fn unlock(&self) {
    let prev = self.state.fetch_sub(1, Release); // 1→0 或 2→1
    if prev != 1 {
        // 有 waiter：重置并 wake
        self.state.store(0, Release);
        futex_wake(&self.state, 1); // 唤醒一个等待线程
    }
}

MPSC 队列极简实现走读

章索引：第 11 章 · 前：11.10.2 · 阻塞唤醒 · 相关：11.10 · MPSC 总览

一句话

std::sync::mpsc 内部 = FIFO 队列 + Sender 原子计数 + recv 空时 park + send 后 unpark — 下列为 教学用极简注释版（真实 std 实现更复杂，含 list/queue 分文件）。

核心组件关系

channel() 创建:
  Inner { queue: Mutex<VecDeque<T>>, receiver_count, sender_count, on_empty: Condvar/park }

Sender::send(msg):
  lock queue → push_back(msg) → signal receiver

Receiver::recv():
  lock queue → 若空且 senders>0 → wait
              → pop_front 或 Err(Disconnected)

A. 队列 + send

struct Inner<T> {
    queue: Mutex<VecDeque<T>>,
    senders: AtomicUsize,   // 存活 Sender 数
    receivers: AtomicUsize, // 通常 1
    // 阻塞：Condvar 或 thread::park 的 Packet
}

impl<T> Sender<T> {
    fn send(&self, msg: T) -> Result<(), SendError<T>> {
        let inner = &self.inner;
        if inner.receivers.load(Acquire) == 0 {
            return Err(SendError(msg)); // 无人收
        }
        {
            let mut q = inner.queue.lock().unwrap();
            q.push_back(msg);
        }
        inner.wake_receiver(); // unpark / condvar.notify
        Ok(())
    }
}

B. recv 阻塞与唤醒

impl<T> Receiver<T> {
    fn recv(&self) -> Result<T, RecvError> {
        loop {
            {
                let mut q = self.inner.queue.lock().unwrap();
                if let Some(msg) = q.pop_front() {
                    return Ok(msg);
                }
                if self.inner.senders.load(Acquire) == 0 {
                    return Err(RecvError); // 通道关闭且空
                }
                // 空且有 sender：释放锁并阻塞
                self.inner.wait_while_empty(&mut q); // park / condvar.wait
            }
        }
    }
}

// wake_receiver（send 侧调用）:
fn wake_receiver(&self) {
    // 若有线程在 park/condvar 上睡 → unpark_one / notify_one
}

C. Sender Drop → 通道关闭

impl<T> Drop for Sender<T> {
    fn drop(&mut self) {
        if self.inner.senders.fetch_sub(1, AcqRel) == 1 {
            // 最后一个 Sender 消失 → 唤醒 Receiver 使其 recv 返回 Err
            self.inner.wake_receiver();
        }
    }
