# 10.6.1 变体记录

## 本节讲什么

**变体记录（variant record）** 是 union 最有价值的应用：同一消息头后，载荷类型不同——用 **union + 类型判别字段** 建模。这是 FIX/SBE/Protobuf 等协议的"带标签联合"（tagged union）模式，也是 C 语言实现"多态"的传统方式。本节剖析变体记录的结构、判别字段的安全使用、匿名 union 简化、以及 HFT 协议的实际应用。

## 要点

### 基本结构

```c
struct msg {
    uint32_t magic;        /* 公共头部 */
    uint8_t  type;         /* 判别字段 */
    uint16_t len;
    union {                /* 变体载荷 */
        struct { uint64_t order_id; uint32_t qty; } order;
        struct { uint32_t seq; uint16_t price; } quote;
        struct { uint64_t cancel_id; } cancel;
    } body;
};
```

三个组成部分：

1. **公共头部**：所有变体共有的字段（magic、type、len）
2. **判别字段**：标识当前有效变体（type）
3. **union 载荷**：按 type 解读不同结构

### 判别字段的安全使用

```c
struct msg m;
memcpy(&m, buf, sizeof m);

switch (m.type) {
case PKT_ORDER:
    printf("order %lu qty %u\n",
           (unsigned long)m.body.order.order_id,
           m.body.order.qty);
    break;
case PKT_QUOTE:
    printf("quote seq %u price %u\n",
           m.body.quote.seq, m.body.quote.price);
    break;
case PKT_CANCEL:
    printf("cancel %lu\n",
           (unsigned long)m.body.cancel.cancel_id);
    break;
default:
    printf("unknown type %u\n", m.type);
}
```

**铁律**：

- **必须**先判 `type` 再访问 `body` 对应分支
- 禁止无判别地读 union 另一分支——静默 UB
- default 处理未知类型，防止未来协议扩展

### 判别字段缺失的陷阱

```c
/* 反例：无判别字段 */
union {
    struct { uint64_t order_id; } order;
    struct { uint32_t seq; } quote;
} body;

body.order.order_id = 1001;
/* 后续代码无法知道这个 union 当前是 order 还是 quote */
printf("%u\n", body.quote.seq);   /* UB：读错分支 */
```

无判别字段的 union 是**定时炸弹**——任何"读错"都是静默 UB。

### 匿名 union 简化（C11）

```c
struct msg {
    uint32_t magic;
    uint8_t  type;
    uint16_t len;
    union {                /* 匿名 union */
        struct { uint64_t order_id; uint32_t qty; } order;
        struct { uint32_t seq; uint16_t price; } quote;
    };                     /* 无成员名 */
};

struct msg m;
m.type = PKT_ORDER;
m.order.order_id = 1001;   /* 直接访问，无需 m.body.order */
```

- C11 标准化（GCC 早就支持为扩展）
- 简化访问语法
- 可读性更好

### 公共前缀模式

```c
struct pkt_base {
    uint32_t magic;
    uint8_t  type;
    uint16_t len;
};

struct order_pkt {
    struct pkt_base base;   /* 公共前缀 */
    uint64_t order_id;
    uint32_t qty;
};

struct quote_pkt {
    struct pkt_base base;   /* 相同前缀 */
    uint32_t seq;
    uint16_t price;
};

/* 通过 base.type 判别 */
void process(void *buf) {
    struct pkt_base *base = buf;
    switch (base->type) {
    case PKT_ORDER: {
        struct order_pkt *o = buf;
        /* ... */
        break;
    }
    case PKT_QUOTE: {
        struct quote_pkt *q = buf;
        /* ... */
        break;
    }
    }
}
```

- 公共前缀作为"基类"
- 强转后访问具体类型
- 是 C 的"继承"惯用法，Linux 内核大量使用

### 复合字面量构造

```c
struct msg make_order(uint64_t id, uint32_t qty) {
    return (struct msg){
        .magic = MAGIC,
        .type = PKT_ORDER,
        .len = sizeof(struct { uint64_t; uint32_t; }),
        .body.order = { .order_id = id, .qty = qty },
    };
}
```

C99 复合字面量让变体记录的构造简洁。

### 变体记录的内存布局

```c
struct msg {
    uint32_t magic;     /* 4 字节，偏移 0 */
    uint8_t  type;      /* 1 字节，偏移 4 */
    uint16_t len;       /* 2 字节，偏移 6（padding 后） */
    union {             /* 偏移 8 */
        struct { uint64_t order_id; uint32_t qty; } order;   /* 16 字节 */
        struct { uint32_t seq; uint16_t price; } quote;      /* 8 字节 */
    } body;
};
/* sizeof = 24（union 取最大 16，加前缀 8） */
```

union 让所有变体共享同一块载荷空间，大小等于最大变体。

### 变体记录 vs 继承

| 维度 | **变体记录** | **C++ 继承** |
|------|--------------|--------------|
| 判别 | 显式 type 字段 | vtable 隐式 |
| 内存 | 单块，union 共享 | 对象 + vtable 指针 |
| 扩展 | 加 union 分支 | 派生类 |
| 性能 | switch 分派，可预测 | 间接调用，分支预测 |
| 适用 | C、HFT、协议 | C++、动态多态 |

HFT 代码倾向变体记录——switch 比 vtable 更利于分支预测和内联。

## HFT 关联

- **FIX 协议**：消息类型字段（MsgType=35）判别不同消息体，用变体记录建模
- **SBE 协议**：schema 定义的变体消息，编译期生成 tagged union
- **DPDK rte_mbuf**：不同报文类型（Ethernet/IP/ARP）用 union 共享缓冲
- **订单事件**：`struct event { uint8_t type; union { Order; Quote; Cancel; } body; }`
- **性能**：switch 分派可被分支预测器优化，热路径比 vtable 快

### 实操：FIX 消息变体

```c
enum fix_msg_type {
    FIX_NEW_ORDER = 'D',
    FIX_CANCEL = 'F',
    FIX_EXECUTION_REPORT = '8',
};

struct fix_msg {
    uint8_t  type;
    uint16_t len;
    union {
        struct {
            char clordid[16];
            char symbol[8];
            uint32_t qty;
            uint32_t price;
        } new_order;
        struct {
            char clordid[16];
            char origclordid[16];
        } cancel;
        struct {
            char orderid[16];
            uint8_t exec_type;
            uint8_t status;
            uint32_t leaves_qty;
        } exec_report;
    } body;
};

void process_fix(const struct fix_msg *m) {
    switch (m->type) {
    case FIX_NEW_ORDER:
        handle_new_order(&m->body.new_order);
        break;
    case FIX_CANCEL:
        handle_cancel(&m->body.cancel);
        break;
    case FIX_EXECUTION_REPORT:
        handle_exec_report(&m->body.exec_report);
        break;
    default:
        log_unknown(m->type);
    }
}
```

### 实操：公共前缀分派

```c
/* 所有报文共享前 8 字节头部 */
struct hdr { uint32_t magic; uint8_t type; uint8_t ver; uint16_t len; };

void dispatch(void *buf, size_t len) {
    if (len < sizeof(struct hdr)) return;
    const struct hdr *h = buf;

    switch (h->type) {
    case TYPE_ORDER:
        handle_order(buf, len);   /* 内部强转 struct order_pkt * */
        break;
    case TYPE_QUOTE:
        handle_quote(buf, len);
        break;
    }
}
```

这是 HFT 报文分派的标准模式——先读公共头部，按 type 分派。
