15.1

错误报告

本节讲什么

I/O 操作可能失败——文件不存在、权限不足、磁盘满、网络断。C 标准库通过 errno + perror + strerror 三件套提供错误报告机制。这一节给出它们的用法、errno 的线程局部性、stderr 无缓冲特性，以及 HFT 工程里故障日志的实践。

认知锚点：errno 不是「错误码变量」，而是线程局部的可修改左值——每次系统调用或库函数失败时设置，成功时不保证清零。理解这一点才能正确使用：先保存 errno，再调用可能修改它的函数。

要点

perror：打印错误

FILE *fp = fopen("quote.dat", "rb");
if (!fp) {
    perror("fopen quote.dat");
    /* 输出：fopen quote.dat: No such file or directory */
    return -1;
}

perror(prefix) 输出 prefix: strerror(errno)\n 到 stderr。prefix 是用户提供的上下文，便于定位是哪个操作失败。

strerror：错误码转字符串

#include <string.h>
#include <errno.h>

int err = errno;   /* 先保存 */
fprintf(stderr, "fopen failed: %s\n", strerror(err));

strerror(errno) 返回错误码对应的可读字符串。调用前先保存 errno——fprintf 等函数可能修改 errno。

strerror_r：线程安全版本

char buf[256];
int ret = strerror_r(errno, buf, sizeof(buf));
/* POSIX: 返回 0 成功，buf 内是字符串 */
/* GNU:  返回字符串指针（可能不是 buf） */

strerror 返回静态字符串指针，多线程并发调用可能互相覆盖。strerror_r 是线程安全版本，但 POSIX 与 GNU 实现签名不同（详见 ch09）。

errno 的线程局部性

/* C11 起 errno 是线程局部的可修改左值 */
errno = 0;              /* 显式清零 */
FILE *fp = fopen(...);
if (!fp) {
    int err = errno;    /* 立即保存 */
    /* ... 其他函数调用可能修改 errno ... */
    log_err("fopen failed: %s", strerror(err));
}
特性 说明
线程局部 每个线程有自己的 errno，互不干扰
成功不清零 函数成功时 errno 不保证清零，不能靠 errno 判断成功
失败才设置 函数失败时设置为对应错误码
必须立即保存 后续函数调用可能覆盖 errno

正确模式：检查返回值判断成功/失败，失败时立即保存 errno 到局部变量。

stderr 无缓冲

fprintf(stderr, "fatal: %s\n", msg);   /* 立即输出，不等 \n */

stderr 默认无缓冲，每次 fprintf 立即输出。崩溃前的最后一条日志不会因缓冲丢失。致命错误日志必须用 stderr，不能用 stdout（行缓冲，崩溃可能丢失）。

常见 errno 值

错误码 含义 典型场景
ENOENT No such file or directory fopen 不存在的文件
EACCES Permission denied 权限不足
EEXIST File exists 创建已存在的文件
ENOSPC No space left on device 磁盘满
EBADF Bad file descriptor fd 无效
EINVAL Invalid argument 参数非法
ENOMEM Cannot allocate memory 内存不足
EAGAIN Resource temporarily unavailable 非阻塞 I/O 无数据
EINTR Interrupted system call 信号中断
EPIPE Broken pipe 写已关闭的管道

HFT 关联

故障日志规范

#define LOG_FATAL(fmt, ...) do { \
    int _err = errno; \
    fprintf(stderr, "[FATAL %s:%d %s] " fmt " (errno=%d: %s)\n", \
            __FILE__, __LINE__, __func__, ##__VA_ARGS__, \
            _err, strerror(_err)); \
    fflush(stderr); \
} while (0)

FILE *fp = fopen("/data/quote.dat", "rb");
if (!fp) {
    LOG_FATAL("cannot open quote file");
    exit(1);
}
/* 输出：[FATAL engine.c:42 init] cannot open quote file (errno=2: No such file or directory) */

致命错误日志集成 __FILE__/__LINE__/__func__ + errno + strerror，一行定位问题。fflush(stderr) 确保崩溃前落盘（虽然 stderr 无缓冲，但显式 flush 是保险）。

磁盘满降级

size_t n = fwrite(batch, sizeof(batch[0]), 32, log_fp);
if (n < 32 && ferror(log_fp)) {
    int err = errno;
    LOG_FATAL("log write partial: %d/%d written", n, 32);
    if (err == ENOSPC) {
        /* 磁盘满，切换到降级模式：只写内存环形缓冲 */
        log_mode = LOG_MODE_MEMORY_ONLY;
    }
}

HFT 系统对磁盘满要降级而非崩溃——行情日志写不下时切到内存缓冲，保证数据面继续运行。errno == ENOSPC 是磁盘满的标志。

errno 与多线程

/* 多线程下 errno 是线程局部的，安全 */
void *worker_thread(void *arg) {
    FILE *fp = fopen(...);
    if (!fp) {
        int err = errno;   /* 本线程的 errno */
        log_err("fopen failed: %s", strerror(err));
    }
    return NULL;
}

C11 起 errno 是线程局部的，多线程并发使用安全。但 strerror 返回静态字符串，多线程并发调用不安全——用 strerror_r。

数据面避免 perror

/* 数据面热路径避免 perror/fprintf */
/* 用内存日志缓冲 + 异步落盘 */
static char log_buf[64 * 1024];
static size_t log_pos = 0;

static inline void log_fast(const char *msg, size_t len) {
    if (log_pos + len < sizeof(log_buf)) {
        memcpy(log_buf + log_pos, msg, len);
        log_pos += len;
    }
}

HFT 数据面热路径不用 perror/fprintf——它们有锁、有格式化、有系统调用。日志写入内存缓冲，由专门日志线程异步落盘。

UB 边界

情形 后果 防护
不保存 errno 直接调用 strerror errno 可能被中间函数覆盖 先 int err = errno 保存
用 errno 判断成功 成功时 errno 不清零，误判 用返回值判断成功/失败
strerror 多线程并发 静态字符串互相覆盖 用 strerror_r
strerror 传入非法 errno 返回 "Unknown error N" 检查 errno 范围
perror 在 stdout 输出 perror 输出到 stderr，不是 stdout 理解 perror 输出流

errno 的「成功不清零」陷阱

errno = 0;
FILE *fp = fopen("exists.txt", "r");
/* fopen 成功，但 errno 可能被内部调用设为非 0 */
if (errno != 0) {
    /* 误判：fopen 成功但 errno 非 0 */
    log_err("fopen failed?");
}

正确做法：用返回值判断成功/失败，失败时才看 errno。

15.2

终止执行

本节讲什么

程序遇到不可恢复的错误时需要终止——exit 做正常清理退出，abort 做异常崩溃退出。这一节给出两者的行为差异、atexit 回调机制、退出时的 stdio 刷新顺序，以及 HFT 工程里数据面与控制面对「退出」的不同态度。

认知锚点：exit 和 abort 不是「同一件事的两种写法」，而是语义完全不同——exit 是「我知道出错了，但还能有序收尾」，abort 是「状态已损坏，不能再信任任何代码」。HFT 数据面绝不能因 I/O 错误退出，控制面启动失败可以 exit。

要点

exit：正常退出

#include <stdlib.h>

void exit(int status);

exit(status) 做以下事情，按顺序：

  1. 调用 atexit 注册的回调（按注册的逆序）。
  2. 刷新并关闭所有 stdio 流：fflush 所有打开的 FILE *，包括 stdout/stderr。
  3. 删除 tmpfile 创建的临时文件。
  4. 调用 _exit（POSIX）：关闭文件描述符，释放资源，结束进程。
  5. 返回 status 给父进程（$? 获取）。
atexit(cleanup_func);   /* 注册退出回调 */
exit(EXIT_SUCCESS);     /* 0 */
exit(EXIT_FAILURE);     /* 1 */

abort：异常退出

void abort(void);

abort() 做以下事情：

  1. 向自己发送 SIGABRT 信号。
  2. 不调用 atexit 回调。
  3. 不刷新 stdio 缓冲（实现相关，部分实现会刷新）。
  4. 生成 core dump（若系统启用）。
  5. 返回 SIGABRT 给父进程（退出码 134）。

abort 用于程序状态已损坏，不能再信任任何代码的场景。断言失败（assert 在 NDEBUG 未定义时）内部调用 abort。

exit vs abort 对比

维度 exit(status) abort()
调用 atexit 回调 是（逆序） 否
刷新 stdio 缓冲 是 否（实现相关）
关闭 FILE * 是 由 OS 回收 fd
生成 core dump 否 是（若启用）
信号 无 SIGABRT
退出码 status 134（128 + SIGABRT）
语义 有序收尾 异常崩溃
适用 可恢复错误 不可恢复错误

atexit 回调

#include <stdlib.h>

void cleanup_logs(void)    { flush_log_buffer(); }
void cleanup_shm(void)     { unlink_shm_files(); }
void cleanup_network(void) { close_sockets(); }

int main(void) {
    atexit(cleanup_logs);      /* 第三个执行 */
    atexit(cleanup_shm);       /* 第二个执行 */
    atexit(cleanup_network);   /* 第一个执行（逆序） */

    /* ... */

    if (fatal_error)
        exit(EXIT_FAILURE);    /* 触发回调 */

    return 0;                  /* return 从 main 等价 exit */
}

atexit 注册的回调在 exit 或 return from main 时按注册的逆序执行。用于资源清理：日志刷新、共享内存释放、网络关闭。

  • 注册上限：标准保证至少 32 个，实际通常更多。
  • 回调签名：void (*)(void)，无参无返回。
  • 回调中调用 exit：UB，可能导致死循环。

_exit：直接退出（POSIX）

#include <unistd.h>

void _exit(int status);   /* 不刷新缓冲，不调 atexit */

_exit 是最底层的退出，直接进入内核：

  • 不调用 atexit 回调。
  • 不刷新 stdio 缓冲。
  • 关闭文件描述符（由内核完成）。
  • 用于 fork 后的子进程，避免污染父进程缓冲。
pid_t pid = fork();
if (pid == 0) {
    /* 子进程 */
    _exit(127);   /* 不用 exit，避免刷新从父进程继承的 stdio 缓冲 */
}

stdio 刷新顺序

exit(status)
   |
   | 1. atexit 回调（逆序）
   v
   | 2. fflush 所有 FILE *（含 stdout/stderr）
   v
   | 3. fclose 所有 FILE *
   v
   | 4. _exit(status) → 内核回收资源

关键：exit 会刷新所有 stdio 缓冲。如果崩溃前想确保日志落盘，用 exit 而非 abort。但若状态已损坏，exit 的回调可能再次触发错误，此时用 abort + core dump 更安全。

return from main vs exit

int main(void) {
    /* ... */
    return 0;   /* 等价 exit(0)，触发 atexit 回调 */
}

return 从 main 返回等价于 exit——都会调用 atexit 回调、刷新 stdio。但 return 会先析构 main 的局部变量（C++ 有析构函数，C 无）。C 中两者基本等价。

HFT 关联

控制面启动失败用 exit

int main(int argc, char **argv) {
    if (load_config("hft.conf") < 0) {
        fprintf(stderr, "config load failed\n");
        exit(EXIT_FAILURE);   /* 启动失败，有序退出 */
    }

    FILE *fp = fopen("/data/quote.dat", "rb");
    if (!fp) {
        perror("fopen");
        exit(EXIT_FAILURE);   /* 数据文件不可用，退出 */
    }

    /* ... 启动数据面 ... */
    run_dataplane();
    return 0;
}

控制面启动阶段遇到不可恢复错误（配置错、数据文件缺失、依赖库初始化失败），用 exit 有序退出，刷新日志、释放资源。

数据面热路径避免 exit/abort

/* 数据面处理报文 */
void on_packet(struct rte_mbuf *m) {
    if (parse_failed(m)) {
        stats_inc(STATS_PARSE_ERR);
        rte_pktmbuf_free(m);
        return;   /* 继续处理下一个，不退出 */
    }

    /* 即使严重错误也不退出 */
    if (orderbook_corrupted()) {
        stats_inc(STATS_OB_CORRUPT);
        alert_admin();   /* 告警，但继续运行 */
        return;
    }
}

HFT 数据面对「退出」零容忍——行情不能停、订单不能丢。即使发生严重错误，也要降级运行、告警、继续处理。exit/abort 只用于启动阶段或人工主动停止。

abort 用于断言失败

#define ASSERT(cond) do { \
    if (!(cond)) { \
        fprintf(stderr, "ASSERT failed: %s at %s:%d in %s\n", \
                #cond, __FILE__, __LINE__, __func__); \
        fflush(stderr); \
        abort();   /* 生成 core dump 供调试 */ \
    } \
} while (0)

ASSERT(orderbook->levels > 0);   /* 不变式破坏，崩溃调试 */

断言失败意味着程序状态已损坏，继续运行可能导致更严重后果（错单、资损）。abort 生成 core dump 供事后调试。HFT 工程通常保留断言在生产环境——宁可崩溃也不能错单。

atexit 清理共享内存

static void cleanup_shm(void) {
    shm_unlink("/hft_orderbook");
    shm_unlink("/hft_strategy_state");
}

int main(void) {
    atexit(cleanup_shm);
    /* ... */
}

HFT 进程间通过共享内存通信，退出时要清理 /dev/shm 下的文件。atexit 确保正常退出时清理。异常崩溃（abort/segfault）不触发 atexit，需要运维脚本兜底清理。

fork 后子进程用 _exit

pid_t pid = fork();
if (pid == 0) {
    /* 子进程：执行辅助工具 */
    execl("/usr/bin/tcpdump", "tcpdump", "-i", "eth0", NULL);
    _exit(127);   /* execl 失败才到这里，用 _exit 避免刷新父进程缓冲 */
}

fork 后子进程继承父进程的 stdio 缓冲副本。若子进程用 exit，会刷新这份副本，导致缓冲内容重复输出。_exit 不刷新，安全。

UB 边界

情形 后果 防护
atexit 回调中调用 exit UB，可能死循环 回调只做清理，不调用 exit
exit 后继续执行 不可能，exit 不返回 不要期望 exit 后的代码执行
abort 不生成 core ulimit 限制 ulimit -c unlimited
fork 后子进程用 exit 父进程缓冲重复输出 子进程用 _exit
回调中取锁但持有者已退出 死锁 atexit 回调避免取锁

atexit 回调的注册上限

/* 标准保证至少 32 个 atexit 回调 */
/* 超过返回非 0 */
if (atexit(my_cleanup) != 0) {
    fprintf(stderr, "atexit registration failed\n");
}

大型项目可能注册几十个清理回调，超过实现限制时 atexit 返回非 0。关键清理逻辑应在主退出路径显式调用，不完全依赖 atexit。

abort 与 core dump

# 启用 core dump
ulimit -c unlimited

# 设置 core 路径（Linux）
echo "/var/core/core.%e.%p.%t" > /proc/sys/kernel/core_pattern

# 分析 core
gdb ./my_app core.12345

abort 生成的 core dump 是事后调试的关键。HFT 生产环境通常配置 core dump 路径，崩溃后自动收集供分析。

15.3

标准 I/O 函数库

本节讲什么

<stdio.h> 是 C 标准库的 I/O 接口，提供带缓冲的高层 I/O 抽象——FILE * 流、fopen/fclose、fread/fwrite、printf 族等。这一节给出 stdio 的函数分类总览、与底层 POSIX I/O（open/read/write）的关系，以及 HFT 工程里控制面与数据面对 stdio 的不同取舍。

认知锚点：stdio 不是「I/O 的唯一方式」，而是用户态缓冲层——它在底层系统调用之上加了一层缓冲，减少 syscall 次数，提高吞吐。但缓冲引入延迟抖动，HFT 数据面追求纳秒级延迟时必须绕过 stdio，直接用系统调用或 DPDK。

要点

FILE 结构（典型实现）

/* glibc 的 FILE 结构（简化） */
struct _IO_FILE {
    int    _flags;          /* 标志位 */
    char  *_IO_read_ptr;    /* 当前读位置 */
    char  *_IO_read_end;    /* 读缓冲末尾 */
    char  *_IO_read_base;   /* 读缓冲起始 */
    char  *_IO_write_base;  /* 写缓冲起始 */
    char  *_IO_write_ptr;   /* 当前写位置 */
    char  *_IO_write_end;   /* 写缓冲末尾 */
    char  *_IO_buf_base;    /* 缓冲区起始 */
    char  *_IO_buf_end;     /* 缓冲区末尾 */

    int    _mode;           /* 文本/二进制 */
    size_t _IO_buf_len;     /* 缓冲区长度 */
    /* ... 更多字段 ... */
};

FILE * 是指向这个结构的指针。每个 FILE 封装：

  • 文件描述符 fd：底层 POSIX fd。
  • 用户态缓冲区：减少 syscall。
  • 读写位置：当前偏移。
  • 错误/EOF 标志：状态查询。
  • 缓冲模式：全缓冲/行缓冲/无缓冲。

stdio 的缓冲机制

应用 fwrite(buf, 1, 100, fp)
   |
   | 数据拷贝到 fp 的用户态缓冲区
   v
fp 缓冲区（用户态，4KB 默认）
   |
   | 缓冲区满 或 fflush 或 流关闭
   v
write(fd, 缓冲区, 4096)   ← 一次 syscall
   |
   v
内核页缓存（Page Cache）
   |
   | fsync 或内核回写
   v
磁盘

缓冲的好处：100 次 fwrite 各 100 字节，只触发 1 次 write syscall（缓冲区 4KB）。

缓冲的代价：数据在用户态缓冲区停留，崩溃可能丢失；延迟不可预测（什么时候 flush 不确定）。

启动配置读取 stdio (fgets) 简单、低频、可读性高
日志写入 stdio (fprintf) + 自定义缓冲 格式化方便，批量减少 syscall
行情数据持久化 stdio (fwrite) 批量二进制读写高效
历史行情回放 stdio (fread) 批量读、缓冲友好
数据面报文收发 绕过 stdio，用 DPDK 纳秒级延迟，不能容忍缓冲
数据面网络 I/O 绕过 stdio，用 recv/send 或 epoll 直接 syscall，延迟可预测

口诀：控制面/管理面用 stdio，数据面绕过 stdio。stdio 的缓冲是为吞吐优化，与 HFT 数据面的延迟需求冲突。

日志系统的分层

/* 数据面热路径：内存缓冲，无 stdio */
static char log_ring[64 * 1024];
static size_t log_pos = 0;

static inline void log_fast(const char *msg, size_t len) {
    size_t pos = __atomic_fetch_add(&log_pos, len, __ATOMIC_RELAXED);
    pos &= (sizeof(log_ring) - 1);   /* 环形缓冲 */
    memcpy(log_ring + pos, msg, len);
}

/* 日志线程：用 stdio 异步落盘 */
void *log_thread(void *arg) {
    FILE *fp = fopen("/var/log/hft.log", "a");
    while (running) {
        /* 从 log_ring 取数据，fprintf 落盘 */
        fprintf(fp, "%s\n", log_entries[i]);
        fflush(fp);
    }
    fclose(fp);
    return NULL;
}

HFT 日志系统分层：数据面写内存环形缓冲（纳秒级），专门日志线程用 stdio 异步落盘（毫秒级可接受）。数据面绝不直接调用 fprintf。

行情数据持久化

/* 用 fwrite 批量持久化行情 */
struct quote batch[32];
/* ... 填充 batch ... */

size_t n = fwrite(batch, sizeof(batch[0]), 32, market_data_fp);
if (n != 32) {
    perror("fwrite market data");
}

fwrite 批量写 32 个报价，stdio 缓冲减少 syscall。延迟要求不高（落盘是异步操作），用 stdio 合适。

15.5

流 I/O 总览

本节讲什么

stdio 按数据粒度分层：字符 → 行 → 格式化 → 二进制块。不同粒度对应不同接口，选型直接影响性能与正确性。这一节给出四层接口的总览、选型对照表，以及 HFT 工程里「禁止使用」的接口清单。

认知锚点：选对 I/O 接口是工程效率的第一步。配置文件用 fgets、日志用 fprintf、行情数据用 fread/fwrite——各有专攻。用错接口（如用 fgetc 读大文件、用 sprintf 拼字符串）是 C 工程的低级错误。

要点

四层粒度

字符     fgetc/fputc          逐字节，最细粒度
  ↓
行       fgets/fputs          文本行，配置/日志
  ↓
格式化   fprintf/snprintf     格式化文本，日志核心
  ↓
二进制   fread/fwrite         原始内存块，行情/struct
粒度 接口 典型场景 性能
字符 fgetc/fputc 单字节解析、状态机 低（频繁函数调用）
行 fgets/fputs 配置文件、文本日志 中
格式化 fprintf/snprintf 日志、调试输出 中（格式化开销）
二进制 fread/fwrite 行情数据、struct 持久化 高（批量内存拷贝）

选型对照表

数据类型 推荐接口 理由
文本配置 fgets 安全读行，含 \n
日志输出 fprintf + fflush 格式化 + 落盘
二进制行情 fread/fwrite 批量内存拷贝
单字节解析 fgetc 状态机逐字节
字符串拼装 snprintf 限长安全
调试输出 fprintf(stderr, ...) 无缓冲立即输出

禁止使用的接口

接口 原因 替代
gets 无长度限制，缓冲区溢出 fgets
sprintf 无长度限制，缓冲区溢出 snprintf
scanf 直接读用户输入 格式串漏洞 fgets + sscanf
vsprintf 无长度限制 vsnprintf

gets 已在 C11 移除。sprintf 仍在标准但生产代码禁用——缓冲区溢出是安全漏洞的高频来源。

接口的返回值检查

/* 字符 I/O */
int c = fgetc(fp);
if (c == EOF) {
    if (feof(fp)) { /* 正常结束 */ }
    if (ferror(fp)) { perror("fgetc"); }
}

/* 行 I/O */
char *ret = fgets(buf, sizeof(buf), fp);
if (ret == NULL) {
    if (feof(fp)) { /* 正常结束 */ }
    if (ferror(fp)) { perror("fgets"); }
}

/* 二进制 I/O */
size_t n = fread(buf, sizeof(buf[0]), 32, fp);
if (n < 32 && ferror(fp)) {
    perror("fread");
}

/* 格式化 I/O */
int written = snprintf(buf, sizeof(buf), "x=%d", x);
if (written < 0 || (size_t)written >= sizeof(buf)) {
    /* 截断或错误 */
}

每个 I/O 接口都要检查返回值——不检查的 I/O 是 bug 来源。feof/ferror 区分正常结束与错误。

数据面绕过 stdio

15.6

打开流

本节讲什么

fopen(path, mode) 是 stdio 的入口——把文件路径绑定到 FILE * 流。这一节给出所有打开模式的对照、b 标志的跨平台意义、fopen 失败的错误处理，以及 HFT 工程里滚动日志、行情数据文件、配置文件的打开实践。

认知锚点：fopen 模式字符串看似简单（r/w/a），但每种模式对文件存在性、读写权限、文件指针位置有不同要求。选错模式会导致数据丢失（w 截断已有文件）或权限错误。

要点

打开模式对照表

模式 作用 文件不存在时 文件已存在时 读写
r 只读 失败 从头读 只读
w 只写 创建 截断为 0 只写
a 追加 创建 在末尾写 只写
r+ 读写 失败 从头读写 读写
w+ 读写 创建 截断为 0 读写
a+ 读写 创建 读从头，写末尾 读写

加 b：rb/wb/ab/r+b/w+b/a+b，二进制模式，不转换换行。跨平台统一加 b。

模式选择的陷阱

/* 陷阱 1：w 截断已有文件 */
FILE *fp = fopen("important.dat", "w");   /* 文件内容被清空！ */
/* 应该用 "r+" 或 "a" */

/* 陷阱 2：r 打开不存在的文件 */
FILE *fp = fopen("newfile.txt", "r");     /* 返回 NULL */
/* 应该用 "w" 或 "a" */

/* 陷阱 3：a 模式不能读 */
FILE *fp = fopen("log.txt", "a");
fgetc(fp);   /* 失败，a 是只写 */
/* 应该用 "a+" */

错误处理

FILE *fp = fopen("data.dat", "rb");
if (!fp) {
    perror("fopen data.dat");
    /* 输出：fopen data.dat: No such file or directory */
    return -1;
}

fopen 失败返回 NULL，必须检查。perror 输出错误原因（errno 对应的消息）。

freopen：重定向流

/* 把 stdout 重定向到文件 */
freopen("output.txt", "w", stdout);
printf("this goes to file\n");   /* 写入 output.txt */

/* 把 stderr 重定向到 /dev/null */
freopen("/dev/null", "w", stderr);

freopen 关闭原流，重新绑定到新文件。用于重定向标准流。

tmpfile：临时文件

FILE *tmp = tmpfile();   /* 二进制临时文件，关闭后自动删除 */
if (tmp) {
    fprintf(tmp, "temporary data\n");
    rewind(tmp);
    char buf[256];
    fgets(buf, sizeof(buf), tmp);
}
fclose(tmp);   /* 文件自动删除 */

tmpfile 创建匿名临时文件，fclose 后自动删除。详见 15.15。

HFT 关联

滚动日志用 a 追加

FILE *log_fp = fopen("/var/log/hft.log", "a");
/* a 模式：文件不存在则创建，存在则追加 */
/* 多进程写同一日志文件时，a 模式保证原子追加（POSIX） */
fprintf(log_fp, "[INFO] %s\n", msg);
fclose(log_fp);

滚动日志每天一个文件，a 追加模式不覆盖旧内容。POSIX 保证 a 模式的写操作原子追加，多进程并发写同一文件不会交错（但 stdio 缓冲可能破坏原子性，需配合 setvbuf 无缓冲或 fflush）。

行情数据用 wb 截断写

/* 每天新建行情文件 */
char path[256];
snprintf(path, sizeof(path), "/data/market_%04d%02d%02d.dat",
         tm->tm_year + 1900, tm->tm_mon + 1, tm->tm_mday);
FILE *fp = fopen(path, "wb");   /* 新文件，截断（文件不存在则创建） */
fwrite(batch, sizeof(batch[0]), n, fp);

行情数据每天新文件，wb 模式创建新文件。不要用 w 打开已有文件——会截断历史数据。

配置文件用 r 只读

FILE *cfg_fp = fopen("/etc/hft/hft.conf", "r");
if (!cfg_fp) {
    perror("cannot open config");
    exit(EXIT_FAILURE);
}
char line[256];
while (fgets(line, sizeof(line), cfg_fp)) {
    parse_config(line);
}
fclose(cfg_fp);

配置文件只读，r 模式。文件不存在则 fopen 失败，程序退出。

二进制文件强制 b

/* HFT 工程规范：所有 fopen 都加 b */

15.7

关闭流

本节讲什么

fclose(fp) 是 stdio 的出口——刷新缓冲、关闭底层 fd、释放 FILE 结构。这一节给出 fclose 的完整行为、忘记关闭的后果（句柄泄漏）、goto cleanup 模式，以及 HFT 长跑进程的句柄管理。

认知锚点：fclose 不是「可选的清理」，而是必须的收尾——未 fclose 的流缓冲区数据丢失，fd 泄漏。长跑进程（HFT 网关 7×24 运行）不 fclose 会在数小时/数天内耗尽 fd，无法新建文件或套接字。

要点

fclose 的行为

fclose(fp);

fclose 按顺序做：

  1. fflush 该流的用户态缓冲：把未写数据 write 到底层 fd。
  2. close(fd)：关闭底层文件描述符。
  3. 释放缓冲区：归还 FILE 内部缓冲区内存。
  4. 从全局 FILE 链表移除：避免 exit 时重复刷新。
  5. 释放 FILE 结构：归还结构体内存。

忘记 fclose 的后果

void bad_func(void) {
    FILE *fp = fopen("log.txt", "a");
    fprintf(fp, "something\n");
    /* 忘记 fclose */
}
后果 说明
缓冲数据丢失 未 fflush 的数据在程序崩溃时丢失
fd 泄漏 每次调用泄漏一个 fd，长跑进程耗尽 fd
文件锁不释放 某些系统对打开的文件加锁，不关闭不释放
磁盘空间不释放 已 unlink 但未 fclose 的文件 inode 不释放

fd 泄漏的阈值

/* 查看进程 fd 限制 */
ulimit -n    /* 通常 1024，可调到百万 */

/* 查看进程打开的 fd */
ls -la /proc/$(pidof my_app)/fd/

默认 fd 上限 1024。长跑进程每天泄漏 100 个 fd，10 天后耗尽，无法 fopen 新文件或 socket 新连接。HFT 进程通常把 fd 上限调到百万级（ulimit -n 1000000），但泄漏仍要修复。

goto cleanup 模式

int process_file(const char *path) {
    FILE *fp1 = NULL, *fp2 = NULL, *fp3 = NULL;
    int ret = -1;

    fp1 = fopen(path, "rb");
    if (!fp1) goto cleanup;

    fp2 = fopen("temp.tmp", "wb");
    if (!fp2) goto cleanup;

    fp3 = fopen("index.idx", "wb");
    if (!fp3) goto cleanup;

    /* ... 处理逻辑 ... */
    if (error) goto cleanup;

    ret = 0;   /* 成功 */

cleanup:
    /* 逆序关闭，NULL 安全 */
    if (fp3) fclose(fp3);
    if (fp2) fclose(fp2);
    if (fp1) fclose(fp1);
    return ret;
}

goto cleanup 是 C 工程处理多资源释放的标准模式：

  1. 所有资源指针初始化为 NULL。
  2. 分配失败或处理错误时 goto cleanup。
  3. cleanup 标签后逆序释放，if (ptr) fclose(ptr) 保证 NULL 安全。

详见 ch11.6 动态内存的 goto cleanup。

fclose 后置 NULL

fclose(fp);
fp = NULL;   /* 防止后续误用 */

/* 后续误用会被立即捕获 */
if (fp) {
    fprintf(fp, "...");   /* 不会执行 */
}

fclose 后置 NULL 是防御性编程——防止后续代码误用已关闭的 FILE *（UAF）。

fclose 与 exit 的关系

/* exit 会自动 fclose 所有打开的流 */
atexit(cleanup_func);
exit(0);   /* 触发 atexit 回调 + fflush 所有流 + fclose 所有流 */

exit 会遍历全局 FILE 链表，fflush + fclose 所有打开的流。正常退出不依赖手动 fclose，但异常退出（abort/segfault）不触发。关键数据仍需手动 fflush。

/* 致命错误处理：先 fflush 再 abort */

15.8

字符 I/O

本节讲什么

逐字节读写流的最薄一层：fgetc / fputc 与它们的 stdin/stdout 特化版 getchar / putchar，再加上宏实现 getc / putc 与回退接口 ungetc。这是 stdio 离底层 read(2) 最近的一层，也是 C 程序员最常踩 EOF 陷阱的地方。

要点

函数族对照

函数 原型 备注
fgetc(FILE*) int fgetc(FILE *stream); 真函数，可取地址
getc(FILE*) int getc(FILE *stream); 可能是宏，不能取地址作回调
getchar(void) int getchar(void); 等价 getc(stdin)
fputc(int,FILE*) int fputc(int c, FILE *stream); 写一字节，返回 c 或 EOF
putc(int,FILE*) int putc(int c, FILE *stream); 可能是宏
putchar(int) int putchar(int c); 等价 putc(c, stdout)
ungetc(int,FILE*) int ungetc(int c, FILE *stream); 回退一字节，仅保证 1 层

返回值必须用 int，绝不能用 char

这是字符 I/O 头号陷阱。fgetc 读到一个字节后返回 unsigned char 值（0..255）扩展为 int，遇到结尾或错误返回 EOF（宏定义为 -1）。如果把返回值塞进 char，三个分支会被压缩成两个：

/* 错误示范 */
char c = fgetc(fp);          /* 0xFF 会被截成 -1，与 EOF 撞车 */
if (c == EOF) break;         /* 二进制 0xFF 字节直接被当作结束 */

正确写法：

int c;
while ((c = fgetc(fp)) != EOF) {
    /* c 是 0..255，可安全当 unsigned char 用 */
    process_byte((unsigned char)c);
}
if (ferror(fp)) perror("read");
存储 0xFF 字节表现 与 EOF(-1) 比较
int 255 不相等 ✓
signed char -1 相等 ✗ 误判结束
unsigned char 255 永不相等 ✗ 永不结束

EOF 的双重含义

EOF == -1 同时表示两种情况：到达文件尾，或发生读错误。要区分必须配合 feof() / ferror()（见 15.14）。循环结束后的判别模板：

while ((c = fgetc(fp)) != EOF) { ... }
if (ferror(fp))  { /* 真出错了 */ perror("fgetc"); }
else if (feof(fp)) { /* 正常读完 */ }

getchar / putchar 是 stdin/stdout 特化

getchar() 等价于 getc(stdin)，putchar(c) 等价于 putc(c, stdout)。当 stdin 是行缓冲（终端）时，getchar() 会阻塞到用户按回车，缓冲区一次灌入整行，再逐字节返回——这点对交互程序和管道理解很关键。

getc/putc 可能是宏

POSIX 允许 getc / putc 实现为宏（带参数、可能多次求值 stream 参数）。后果：

  • 不能 &getc 取地址传给 qsort 之类的回调
  • getc(f++) 这类带副作用的 stream 参数可能被求值多次
  • 想要真函数行为，用 fgetc / fputc；性能差异在 stdio 已缓冲的情况下极小

ungetc：回退一字节

int c = fgetc(fp);
if (isdigit(c)) { ungetc(c, fp); parse_number(fp); }
  • 只保证回退 1 字节有效（实现通常允许更多但不保证）
  • 回退不改变底层文件偏移（影响 stdio 内部缓冲）
  • ungetc(EOF, fp) 行为由实现定义，应避免
  • 成功后 feof 标志被清除

性能模型：stdio 已缓冲

fgetc 并不是每次都 syscall。当流是 _IOFBF 全缓冲时，底层 read(2) 一次性填满整块缓冲区（典型 4KB-64KB），fgetc 只是从内存缓冲区取一字节。所以"逐字节 = 慢"在 stdio 下并不成立，瓶颈在 syscall 次数而不是字节次数。真要极致时绕过 stdio 直接 read + 自己管理缓冲（见 15.4.1）。

HFT 关联

  • 协议头逐字节解析：定长头未知长度时常用 fgetc 读 magic + 版本号 + 长度字段再切到 fread 读 body；ungetc 在向前看一字节判断分支时很方便
  • pcap 文件头解析：pcap 全局头是 24 字节定长，魔数 0xa1b2c3d4 用 fread 一次读更优，但若需要兼容大小端探测则逐字节读 magic 更直观

题目 1： 以下代码有什么问题？

int c;
while (c = getchar() != EOF) {

}
参考答案

运算符优先级错误。!= 优先级高于 =，所以 c = (getchar() != EOF)，c 的值是 0 或 1，不是读取的字符。putchar(c) 输出的是 0 或 1 对应的控制字符。正确写法：while ((c = getchar()) != EOF)。这也是为什么 getchar 返回 int 而非 char——EOF 通常是 -1，不能用 char 表示。

15.9

未格式化的行 I/O

本节讲什么

按"行"读写文本流：fgets / fputs。这是配置文件、日志解析、协议文本行的标准接口——既不像 fgetc 那样逐字节低效，也不像 fscanf 那样有格式串风险。本节重点讲 fgets 的安全语义、fputs 不加换行的反直觉行为，以及 gets 为什么被 C11 彻底移除。

要点

fgets：唯一安全的整行读取

char *fgets(char *s, int size, FILE *stream);

语义：

  • 最多读 size - 1 字节，遇 \n 停下，\n 也算进缓冲区
  • 末尾自动追加 \0
  • 成功返回 s，遇 EOF 无数据返回 NULL，出错返回 NULL
  • 缓冲区不够大时，行被截断，但保证 \0 终止
char buf[256];
if (fgets(buf, sizeof buf, fp) == NULL) {
    if (ferror(fp)) perror("fgets");
    /* EOF 正常结束 */
} else {
    /* buf 含一行（可能无 \n，说明被截断或最后一行无换行） */
    size_t len = strlen(buf);
    int truncated = (len > 0 && buf[len-1] != '\n');
}

判断截断 vs 正常一行

fgets 不告诉你"这行是否完整"。要判断必须看末尾：

size_t len = strlen(buf);
if (len == 0) { /* 异常，不该发生 */ }
else if (buf[len-1] == '\n') {
    buf[len-1] = '\0';   /* 去掉换行，方便后续处理 */
    /* 完整一行 */
} else {
    /* 行被截断，或文件最后一行没有 \n */
    /* 需要继续 fgets 读完剩余部分，或按需丢弃 */
}

fputs：写字符串，不加换行

int fputs(const char *s, FILE *stream);
  • 不在末尾加 \n，与 puts（加 \n）不同
  • 不写结尾 \0，只写字符串内容
  • 成功返回非负数，失败返回 EOF
  • 写 \0 中间的字符串？做不到——fputs 在第一个 \0 就停
fputs("[TRACE] new quote\n", fp);   /* 手动带 \n */

puts 的陷阱

puts(s) 等价 fputs(s, stdout) + putchar('\n')。多了一个 \n，写日志时容易重复换行。生产代码统一用 fputs 显式控制。

gets 为什么被移除

char *gets(char *s);   /* C99 弃用，C11 移除 */

gets 不接受长度参数，无法防止缓冲区溢出——经典缓冲区溢出漏洞源（Code Red 蠕虫就是 gets 类问题）。C11 标准正式从库里删除。代码里看到 gets 必须立即换成 fgets，注意 fgets 会保留 \n 而 gets 不保留，迁移时要手动去掉。

行缓冲与终端

当 stream 是终端时，stdio 通常是行缓冲（_IOLBF），用户按回车后整行才进入 stdio 缓冲区，fgets 才能返回。重定向到管道或文件时，stdio 切到全缓冲，fgets 会在缓冲区空时阻塞等 read(2) 填充。

HFT 关联

  • 配置文件解析：网关 gateway.conf 一行一个键值对，fgets + strtok/strtol 是标准模式；用 fscanf 反而难处理注释和空行
  • 文本日志读取：崩溃后回放 gateway.log，fgets 逐行解析比 fread 整块再切行更省内存
  • FIX 协议文本行：FIX 是 \001 分隔的文本协议，单条消息可能很长，fgets 按 \n 切不一定合适，但许多实现仍用 fgets 读定长块再自己切字段
  • 二进制 buf 禁止当 %s 打印：行情 packet 含 \0 字节，fprintf("%s", pkt) 会在第一个 \0 截断，必须用 fwrite(pkt, 1, len, fp)
  • 滚动日志写：高频写日志用 fputs 比 fprintf 快（无格式串解析），但格式化需求大时仍用 fprintf/snprintf

15.10

格式化的行 I/O

本节讲什么

printf 族：printf / fprintf / sprintf / snprintf / vfprintf 等，以及对应的 scanf 族。这是 C 程序员最熟也最常踩坑的接口——格式串与参数类型不匹配、缓冲区溢出、格式串注入攻击。HFT 场景里日志几乎全靠这一族，掌握它等于掌握了控制面输出主力。

要点

printf 族输出目标对照

函数 输出目标 长度限制 用途
printf stdout 无 控制台
fprintf(fp, ...) 指定流 无 落盘日志
sprintf(buf, ...) 数组 无界 禁用
snprintf(buf, size, ...) 数组 限长 生产强制
asprintf(...)（GNU） malloc 堆 自动 调用者负责 free
vfprintf(fp, fmt, ap) 指定流 无 可变参数转发
vsnprintf(buf, size, fmt, ap) 数组 限长 日志宏核心

snprintf 返回值的反直觉

int n = snprintf(buf, size, "seq=%llu", seq);
  • n < 0：编码错误
  • n < size：实际写入字节数（不含 \0）
  • n >= size：被截断，返回值是"如果缓冲区够大会写多少"

利用这一点可以两次调用：第一次 snprintf(NULL, 0, fmt, ...) 拿长度，malloc 后再写。但 HFT 数据面忌讳 malloc，常用栈上固定缓冲 + 截断接受。

scanf 族的返回值

scanf / fscanf / sscanf 返回成功匹配并赋值的字段数，遇文件尾返回 EOF。这是判断解析成功的关键，但被无数初学者忽略：

int n = fscanf(fp, "%u %u %u", &seq, &bid, &ask);
if (n != 3) {
    /* 字段不全，可能是格式错或 EOF */
    if (n == EOF) break;
    fprintf(stderr, "bad line, got %d fields\n", n);
    continue;
}

常用占位符速查

符 用途 HFT 典型
%u / %llu 无符号整型 订单 ID、时间戳
%d / %lld 有符号整型 价差、数量
%x / %X 十六进制 标志位、寄存器值
%p 指针 调试输出
%s 字符串 仅 \0 终止文本
%.10s 限长字符串 防越界读
%c 单字符 协议字段
%f / %lf 浮点 价位（不推荐用浮点存价格）
%zu size_t 字节数
%jd / %ju intmax_t/uintmax_t 跨平台大整数

长度修饰符陷阱

%u 对应 unsigned int，%llu 对应 unsigned long long。在 64 位 Linux 上 long 是 8 字节但 Windows LLP64 上 long 是 4 字节——跨平台代码不要用 %lu 表示 64 位，统一用 %llu 或 PRId64 宏（inttypes.h）：

#include <inttypes.h>
uint64_t ts = ...;
printf("ts=%" PRIu64 "\n", ts);   /* 跨平台安全 */

%s 的三大风险

  1. 必须 \0 终止：二进制 buf 里没有 \0 会一直读到越界
  2. 无长度限制：%s 不限长，攻击者可构造超长字符串导致缓冲区溢出（snprintf 也只保护 buf，不保护读取源）
  3. 格式串注入：fprintf(fp, user_input) 若 user_input 含 %x 会读栈数据——格式串必须是字面量或受信字符串
/* 错误：user_msg 可能含 %s */
fprintf(log_fp, user_msg);
/* 正确 */
fprintf(log_fp, "%s", user_msg);

整数溢出与 %d

int bid = 3500000;   /* 单位 0.0001 */
printf("bid=%d\n", bid * 100);   /* 可能溢出 int */

HFT 价格常用定点小数（uint64_t 或 int64_t，单位 0.0001），打印时务必用 %llu 或 %lld，并考虑溢出。

scanf 为什么少用

scanf 对错误输入极其脆弱：输入 abc 给 %d 会留下输入流污染，需要手动清空。生产代码通常 fgets 整行再 sscanf 或 strtol 解析，错误恢复更可控。

HFT 关联

  • 日志核心：rte_log 内部就是 vsnprintf + fputs/write，HFT 控制面日志全靠这一族
  • 行情落盘：snprintf 拼一行再 fputs，比直接 fprintf 更易控制缓冲和原子性
  • 崩溃前 fflush：信号处理里 vsnprintf 拼栈帧信息后立即 fflush(stderr)，确保崩溃栈不丢
  • 避免数据面 printf：每条日志的 vsnprintf 解析格式串 + 多次 va_arg 是微秒级开销，数据面禁止 printf，日志走异步队列由专门线程消费
  • 格式串攻击面：外部输入绝不能直接当格式串，FIX/二进制协议字段必须用 "%s" 包一层
  • 时间戳格式化：%lld + uint64_t 或 PRIu64 打印纳秒时间戳，注意时区与本地时间转换用 localtime_r（线程安全版本）

15.11

二进制 I/O

本节讲什么

fread / fwrite：按内存块原样读写，不做任何字符转换。这是 HFT 行情持久化、pcap 抓包、回放引擎的主力接口，也是 stdio 里最贴近 read(2)/write(2) 的一层。本节重点讲返回值语义、部分写的处理、跨平台二进制格式的坑（对齐、字节序、padding）。

要点

函数原型

size_t fread(void *ptr, size_t size, size_t nmemb, FILE *stream);
size_t fwrite(const void *ptr, size_t size, size_t nmemb, FILE *stream);
  • size：每个元素的字节数
  • nmemb：元素个数
  • 返回值：成功读写的完整元素个数，不是字节数

返回值语义详解

size_t r = fread(buf, sizeof(Quote), 32, fp);
r 值 含义
32 全部读成功
0 < r < 32 读到 r 个完整元素后遇 EOF 或错误，剩余元素不完整
0 一开始就 EOF 或错误，需要 feof/ferror 区分

关键：部分元素的字节不写入 buf。fread 保证要么读到完整元素，要么不读，不会留下"半截"数据污染缓冲区。这就是为什么 size 参数化成元素大小而不是直接字节数——库帮你做完整性检查。

写循环的标准模板

Quote batch[32];
size_t n = fill_batch(batch, 32);
size_t w = fwrite(batch, sizeof batch[0], n, fp);
if (w != n) {
    if (ferror(fp)) perror("fwrite");
    /* 磁盘满或断电，需要降级处理 */
}

fread 循环读的标准模板

Quote pkt;
size_t n;
while ((n = fread(&pkt, sizeof pkt, 1, fp)) == 1) {
    process_quote(&pkt);
}
if (ferror(fp)) {
    perror("fread");
} else if (feof(fp)) {
    /* 正常读完 */
} else {
    /* n == 0 但既不 EOF 也不 error？理论不该发生，防御性处理 */
}

部分读的尾巴处理

如果文件大小不是元素大小的整数倍，最后会留下不足一个元素的字节。fread 返回 0，但 stdio 内部缓冲可能已经消费了这些字节——这些字节丢失了，无法用 ungetc 全部回退。所以二进制协议设计时务必让头部包含长度字段，按长度读 body，避免"读到半截"的歧义。

跨平台二进制格式三坑

1. 对齐与 padding

struct Quote {
    uint32_t seq;       /* 4 字节 */
    char side;          /* 1 字节 */
    /* 3 字节 padding */
    uint64_t price;     /* 8 字节 */
};   /* sizeof = 16 */

直接 fwrite(&q, sizeof q, 1, fp) 会把 padding 字节也写进去——不同编译器、不同平台 padding 可能不同。规范做法：

#pragma pack(push, 1)
struct Quote {
    uint32_t seq;
    char side;
    uint64_t price;
};
#pragma pack(pop)
_Static_assert(sizeof(struct Quote) == 13, "wire format mismatch");

或手动序列化（最稳）：

void encode_quote(uint8_t *buf, const Quote *q) {
    memcpy(buf + 0, &q->seq, 4);
    buf[4] = q->side;
    memcpy(buf + 5, &q->price, 8);
}

2. 字节序

x86/ARM 小端，Power/SPARC/部分 ARM 大端。网络字节序大端（htonl/ntohl）。HFT 内部若全 x86 可直接小端，但跨机房或对接外部系统必须显式转换：

uint32_t net_seq = htonl(q.seq);
fwrite(&net_seq, sizeof net_seq, 1, fp);

3. 类型大小

long 在 LP64（Linux）是 8 字节，LLP64（Windows）是 4 字节。二进制格式必须用定长类型 uint32_t/uint64_t，禁用 long/int。

fread vs read：stdio 的价值

fread 内部维护缓冲区，对小块频繁读（如逐条读 quote）比裸 read(2) 少很多 syscall。但缓冲也意味着延迟——fwrite 后立即崩溃，缓冲里的数据会丢。对策见 15.12 的 fflush。

与文本 I/O 的本质区别

维度 文本（fprintf/fscanf） 二进制（fwrite/fread）
表示 ASCII/UTF-8 字符 原始字节
大小 数值 12345678 → 8 字节 uint32 → 4 字节
可读性 人眼可读 不可读
解析速度 需 strtol 解析 直接 memcpy
跨平台 自动转换 \n 需手动处理字节序/对齐
HFT 偏好 仅日志/配置 行情持久化主力

HFT 关联

  • 行情落盘：fwrite 批量写 quote 数组，每次 32-256 条，减少 syscall 次数
  • pcap 抓包：pcap 文件头 24 字节 + 每包 16 字节包头 + 原始以太帧，全二进制，fwrite/fread 直接对应
  • 回放引擎：从历史 .dat 用 fread 重建 tick 流，按 seq 索引文件 + fseek 跳转（见 15.12）
  • 订单簿快照：周期性 dump 全量订单簿到文件，重启时 fread 恢复，避免冷启动重建延迟
  • checksum 与定长头：每条记录前加 magic + length + checksum，fread 读头验证再读 body，防止文件损坏导致解析错位
  • mmap 替代：极致延迟场景连 fread 都嫌慢，直接 mmap 文件做内存映射，零拷贝访问（见 ch18）
  • DPDK 完全绕过 stdio：数据面用 PMD + hugepage + ring，fwrite 仅用于控制面日志和冷数据持久化

15.12

刷新和定位

本节讲什么

fflush 强制把 stdio 缓冲区刷到内核（不是磁盘！），fseek / ftell / rewind / fgetpos / fsetpos 在流里随机定位。这俩操作在 HFT 场景里关系到崩溃不丢日志、按索引跳读历史行情。

要点

fflush：刷到内核，不是落盘

int fflush(FILE *stream);

语义：把 stream 的 stdio 用户态缓冲区写到内核（通过 write(2)）。这不等于数据已经到磁盘——内核 page cache 还有一层。要真正落盘必须 fsync(fd) 或 fdatasync(fd)：

应用数据 ──fwrite──> stdio 缓冲 ──fflush──> 内核 page cache ──fsync──> 磁盘
层 接口 持久性
stdio 缓冲 fflush 进程崩溃不丢
内核 page cache fsync/fdatasync 系统崩溃不丢
磁盘缓存 指令 SYNCHRONIZE_CACHE 断电不丢

HFT 持久化要 fflush + fsync 才算真落盘。

fflush(NULL) 刷所有输出流

fflush(NULL);   /* 刷新所有可写流 */

常用于 fork 前冲洗所有缓冲，避免子进程重复 flush 导致输出重复。

fflush(stdin) 是未定义行为

C 标准没定义 fflush 对输入流的行为。POSIX 扩展允许在可定位的输入流上 fflush 丢弃未读缓冲，但跨平台代码不要依赖。要丢弃 stdin 缓冲用 fseek(stdin, 0, SEEK_END) 或 fpurge（BSD，非标准）。

fseek 三种锚点

int fseek(FILE *stream, long offset, int whence);
whence 锚点 offset 限制
SEEK_SET 文件开头 offset >= 0（标准要求）
SEEK_CUR 当前位置 offset 可正可负
SEEK_END 文件末尾 offset 通常 <= 0

返回值：成功 0，失败 -1 并设 errno。注意 fseek 会清除 EOF 标志但保留错误标志。

ftell 与 long 的限制

long pos = ftell(fp);

long 在 32 位 Linux 是 4 字节，最大 2GB——超过 2GB 的文件 ftell 返回 -1 失败。大文件必须用：

fpos_t pos;
fgetpos(fp, &pos);   /* 标准接口，类型不透明 */
fsetpos(fp, &pos);

/* POSIX 扩展 */
off_t off = ftello(fp);   /* off_t 64 位（_FILE_OFFSET_BITS=64） */
fseeko(fp, off, SEEK_SET);

Windows MSVC 用 _ftelli64 / _fseeki64。

rewind 的陷阱

void rewind(FILE *stream);

等价 fseek(stream, 0, SEEK_SET)，但不返回错误—— rewind 失败时你完全不知道。生产代码用 fseek 检查返回值：

if (fseek(fp, 0, SEEK_SET) != 0) {
    perror("rewind");
}

fseek 与缓冲的交互

fseek 后 stdio 缓冲区被丢弃，下次 fread 重新触发 read(2) 填充。这意味着频繁 fseek + 小 fread 会退化成每次 syscall 读几字节，性能崩塌。大文件随机访问应该：

  • 用 pread(2) / pwrite(2)（不改变文件偏移，无锁）
  • 或 mmap + 指针运算（见 ch18）
  • 或自己维护用户态缓冲 + 批量预读

二进制流才可定位

文本流在某些平台（Windows \r\n ↔ \n 转换）fseek 的 offset 不是字节偏移而是"实现定义的值"，只能用 fgetpos/fsetpos 保存恢复。HFT 二进制文件用 fseeko 即可。

典型用例：按索引跳读

/* 索引文件: 每条 8 字节, 存 quote 在数据文件的偏移 */
off_t off;
fseeko(idx_fp, seq * sizeof(off_t), SEEK_SET);
fread(&off, sizeof off, 1, idx_fp);

fseeko(dat_fp, off, SEEK_SET);
fread(&pkt, sizeof pkt, 1, dat_fp);

这是回放引擎按 seq 随机读历史行情的标准模式。

HFT 关联

  • 崩溃前 fflush(stderr)：信号处理里打印栈帧后立即 fflush(stderr)（stderr 无缓冲其实可省，但 fprintf 到 stderr 也可能被 glibc 缓冲，保险起见显式 flush）
  • 日志落盘策略：高频日志 fprintf + 周期性 fflush + 关键事件 fsync，权衡延迟与可靠性
  • 滚动日志切换：fclose 当前日志 → rename 归档 → fopen 新文件，避免 rename 打开的 fd 导致写入丢失
  • 回放引擎索引：fseeko 按 seq 跳到指定 tick，跳读比顺序读慢但支持时间旅行回测
  • 订单簿快照恢复：启动时 fseek 到快照段，fread 恢复全量订单簿，再 replay 增量日志
  • 避免数据面 fsync：fsync 是毫秒级 syscall，会阻塞数据线程；快照/落盘走专门线程，数据面只 fwrite 到缓冲队列
  • O_DIRECT 绕过 page cache：极致场景用 open(O_DIRECT) + 自己管理对齐缓冲，完全跳过 stdio 和 page cache，延迟可预测

参考答案

用于获取文件大小。先定位到文件末尾，再读 ftell 得到当前位置（即文件大小），然后回到开头。但注意：(1) 对文本模式打开的文件，ftell 返回值不一定等于字节数；(2) 对大文件（>2GB）需要用 fseeko/ftello；(3) 对非普通文件（如管道、设备），fseek 可能失败。

15.13

改变缓冲方式

本节讲什么

setvbuf / setbuf / setbuffer（BSD）/ setlinebuf（BSD）：在流打开后、第一次 I/O 前，调整 stdio 的缓冲模式与缓冲区大小。这层调优在 HFT 日志抖动排查里是常用工具——同一份 fprintf 日志代码，缓冲模式不同可能差出 10x 的延迟方差。

要点

setvbuf 三参数

int setvbuf(FILE *stream, char *buf, int mode, size_t size);
mode 行为 典型场景
_IONBF 无缓冲，每次 I/O 立即 syscall stderr、调试流、崩溃日志
_IOLBF 行缓冲，遇 \n 或缓冲满才 flush 终端 stdout、交互式日志
_IOFBF 全缓冲，缓冲满才 flush 文件日志、批量写盘
  • buf == NULL：stdio 自己 malloc 缓冲区
  • buf != NULL：用调用者提供的缓冲，该缓冲在流关闭前必须有效（典型用法：static 数组或长生命周期堆缓冲）
  • 返回值：成功 0，失败非 0（通常因为流已经有 I/O 发生）

调用时机铁律

setvbuf 必须在流打开后、第一次 I/O 前调用。一旦发生任何 fread/fwrite/fgetc，再调 setvbuf 是未定义行为。常见错误：

fp = fopen("log.txt", "w");
fprintf(fp, "started\n");      /* 已经 I/O 了 */
setvbuf(fp, NULL, _IONBF, 0);  /* UB！可能静默失败也可能崩溃 */

正确顺序：

fp = fopen("log.txt", "w");
if (!fp) { perror("fopen"); exit(1); }
setvbuf(fp, NULL, _IONBF, 0);  /* 先设缓冲 */
fprintf(fp, "started\n");      /* 再 I/O */

setbuf / setbuffer / setlinebuf 速记

setbuf(fp, NULL);          /* 等价 setvbuf(fp, NULL, _IONBF, 0) */
setbuf(fp, big_buf);       /* 等价 setvbuf(fp, big_buf, _IOFBF, BUFSIZ) */
setbuffer(fp, big_buf, n); /* BSD，全缓冲 + 自定义大小 */
setlinebuf(fp);            /* BSD，等价 setvbuf(fp, NULL, _IOLBF, 0) */

BUFSIZ 通常 8192，可跨平台查询 <stdio.h>。

默认缓冲模式

流 终端 重定向到文件/管道
stdout 行缓冲 _IOLBF 全缓冲 _IOFBF
stderr 无缓冲 _IONBF 无缓冲 _IONBF
普通 fopen 全缓冲 _IOFBF 全缓冲 _IOFBF

stderr 永远无缓冲——这是设计，保证错误信息立即输出。stdout 在重定向时切到全缓冲，这导致 printf 调试重定向到文件时数据不立即出现，需要 fflush(stdout) 或 setvbuf(stdout, NULL, _IONBF, 0)。

缓冲大小对性能的影响

static char big_buf[1 << 16];   /* 64KB */
setvbuf(log_fp, big_buf, _IOFBF, sizeof big_buf);
  • 缓冲越大，syscall 越少，吞吐越高
  • 但缓冲越大，崩溃丢失的数据越多（未 flush 的部分）
  • 缓冲越大，单次 flush 延迟越高（写 64KB 比写 8KB 慢）
  • HFT 日志常用 4KB-64KB，平衡 syscall 频次与延迟方差

HFT 日志抖动调优实例

症状：高频 fprintf 到 stdout，延迟偶发尖峰到 100us。

排查： 1. stdout 重定向到文件 → 全缓冲，平时攒 8KB 才写，但偶尔 flush 时阻塞 2. setvbuf(stdout, NULL, _IONBF, 0) → 每次 fprintf 都 syscall，平均延迟升但方差降 3. 改 fputs 到 stderr（无缓冲） → 同样每次 syscall 4. 用大缓冲 + 后台线程周期 flush → 平均最低，方差最小（异步日志模式）

最终方案：自定义日志队列，前台线程 memcpy 到 ring buffer（亚微秒），后台线程批量 write + fsync。这是 DPDK rte_log + 自建日志库的通用思路。

自定义缓冲的生命周期陷阱

void bad_init() {
    char buf[4096];
    setvbuf(fp, buf, _IOFBF, sizeof buf);
}   /* buf 离开作用域，stdio 仍引用它 → UB */

setvbuf 不复制 buf，只存指针。buf 必须是 static、全局、或堆分配且在 fclose 前 free。

缓冲与 fork

fork 后子进程继承 stdio 缓冲区的副本。如果缓冲区有未 flush 的数据，子进程 exit 时会再 flush 一次，导致输出重复。fork 前显式 fflush(NULL) 冲洗所有输出流是标准做法。

HFT 关联

  • 控制面日志调优：setvbuf 调大缓冲 + 后台 flush，前台 fprintf 接近 memcpy 速度
  • 崩溃日志零丢失：关键流设 _IONBF，每条 fprintf 立即 syscall，崩溃也不丢
  • DPDK rte_openlog_stream：DPDK 允许把日志流换成自定义 FILE*，可对其 setvbuf 调优
  • 数据面完全绕过 stdio：数据面用 PMD + ring + 无锁队列，setvbuf 仅影响控制面
  • 多线程日志的锁开销：glibc stdio 函数内部有锁，高频多线程 fprintf 会争锁抖动；改用每线程缓冲 + 合并 flush，或 lockfree ring + 单写线程
  • mmap 日志：极致场景用 mmap 映射日志文件，memcpy 写 + msync 落盘，完全绕过 stdio

15.14

流错误函数

本节讲什么

feof / ferror / clearerr：在 fread/fgetc 短读或 EOF 之后，区分"正常到尾"与"真出错"。这层判别在长跑 HFT 进程里关系到磁盘满、坏道、网络断的自动降级——把错误当 EOF 会静默丢数据，把 EOF 当错误会卡死读循环。

要点

三个函数的语义

int feof(FILE *stream);
int ferror(FILE *stream);
void clearerr(FILE *stream);
函数 返回非 0 的条件 持续性
feof(fp) 读操作遇到 EOF 持续，直到 clearerr/rewind/fseek
ferror(fp) 读/写操作发生错误 持续，直到 clearerr/rewind（部分实现）
clearerr(fp) — 清除 EOF 和错误两个标志

注意：EOF 和错误是两个独立标志，可以同时为真（极少见）也可能都不为真。

feof 的反直觉：滞后性

feof 只有在某次读操作尝试读过 EOF 之后才返回真。在尚未尝试读之前，即使文件指针已经在末尾，feof 也返回假：

fseek(fp, 0, SEEK_END);
feof(fp);   /* 返回 0！还没尝试读过 */

int c = fgetc(fp);   /* 尝试读，失败返回 EOF */
feof(fp);   /* 现在返回 1 */

这是为什么"用 feof 作 while 前置条件"是经典错误：

/* 错误：会多读一次或漏判错误 */
while (!feof(fp)) {
    fread(&pkt, sizeof pkt, 1, fp);
    process_quote(&pkt);   /* 最后一次读失败时 pkt 是旧数据或未初始化 */
}

正确模式是用读操作的返回值作循环条件：

while (fread(&pkt, sizeof pkt, 1, fp) == 1) {
    process_quote(&pkt);
}
/* 循环退出后再用 feof/ferror 区分原因 */
if (ferror(fp)) perror("fread");
/* feof 为真表示正常读完 */

ferror 的常见触发

  • 磁盘满（fwrite 时）
  • 坏道/IO 错误
  • 网络流断开（pipe 对端关闭）
  • 文件被外部 truncate（fread 时位置失效）

ferror 不会自动清除——一次错误后，后续所有 I/O 都会立即失败返回 EOF/0，直到 clearerr。这避免在错误状态下继续操作，但也意味着重试必须显式 clearerr。

clearerr 的两个用途

  1. 重试：网络瞬时断开后 clearerr + fseek 重新定位继续读
  2. 复用 EOF 流：从 stdin 读到 EOF 后，如果想再读（比如交互式 shell 等下一个命令），clearerr(stdin) 重置
/* 交互式 REPL */
while (fgets(buf, sizeof buf, stdin)) {
    handle(buf);
}
/* EOF（Ctrl-D） */
clearerr(stdin);   /* 重置，可以继续读 */

perror 配合 ferror

size_t n = fread(buf, 1, sizeof buf, fp);
if (n < sizeof buf && ferror(fp)) {
    perror("fread quote file");   /* 输出 "fread quote file: <strerror(errno)>" */
    clearerr(fp);
    /* 降级处理或重试 */
}

perror 读 errno，而 errno 只在错误发生那一刻有效——所以必须在 ferror 判定后立即调 perror，中间不要做其他可能改 errno 的系统调用。

区分短读的决策树

fread/fgetc 返回短读或 EOF
        │
        ├─ ferror(fp) 为真?
        │     ├─ 是 → 真出错, perror + clearerr + 重试或降级
        │     └─ 否 → feof(fp) 为真?
        │             ├─ 是 → 正常到尾, 继续后续逻辑
        │             └─ 否 → 理论不该发生, 防御性 abort

文本 vs 二进制流的 EOF

二进制流读到文件物理末尾就 EOF。文本流在某些平台有特殊 EOF 标记（如 Windows 的 Ctrl-Z = 0x1A），读到这个字节就被当作 EOF——这是历史遗留，现代代码用二进制模式打开可避免。HFT 二进制 .dat 务必 "rb"/"wb" 打开。

HFT 关联

  • 磁盘满降级：ferror 触发后切换到备用盘或内存缓冲，clearerr 后重试 fwrite
  • 网络断线重连：行情源 TCP 断开后 clearerr + 重连 + fseek 对齐 seq 继续
  • 日志磁盘坏道：日志 ferror 触发告警 + 切换到 stderr 兜底，避免日志阻塞数据面
  • 回放引擎校验：fread 短读后 ferror 为真说明历史文件损坏，需校验 checksum 重定位
  • 避免把 EOF 当错误：正常读完文件时 feof 为真，不应触发告警，只记录"replay done"
  • 多线程流的错误标志：stdio 流的错误标志是流级别的，多线程共享同一 FILE* 时 ferror 状态可能被其他线程的 I/O 改变——HFT 通常每流单线程独占

15.15

临时文件

本节讲什么

tmpfile / tmpnam / mkstemp（POSIX）/ fmemopen（POSIX）/ open_memstream（POSIX）：临时中转、内存流。HFT 场景里用于抓包临时缓存、回放分片、单测 fixture、内存内日志格式化——把磁盘 IO 转成内存操作，减延迟也减磨损。

要点

tmpfile：自动删除的临时文件

FILE *tmpfile(void);
  • 以 "wb+" 模式打开一个临时文件
  • fclose 或程序正常退出时自动删除
  • 返回 NULL 失败，errno 设值
FILE *tf = tmpfile();
if (!tf) { perror("tmpfile"); exit(1); }
fwrite(batch, sizeof batch[0], n, tf);
rewind(tf);
fread(&pkt, sizeof pkt, 1, tf);   /* 读回写过的数据 */
fclose(tf);   /* 自动删除 */

用途：报文中转、单测内存式 fixture、解析中间结果暂存。

tmpfile 的实现与 TMPDIR

  • glibc：优先用 open(O_TMPFILE)（Linux 3.11+），文件根本不出现路径，unlink 在 fd 关闭时自动发生
  • 老 Linux / BSD：tmpnam 生成名字 + fopen + 立即 unlink（文件在文件系统里但路径已删，fd 关闭后空间回收）
  • 临时目录：环境变量 TMPDIR > /tmp 默认

tmpnam 为什么不安全

char *tmpnam(char *s);   /* C99 弃用警告, C23 移除 */

tmpnam 只生成一个不冲突的名字，不创建文件。从生成名字到 fopen 创建之间存在 TOCTOU 窗口——攻击者可以抢先在这个名字创建符号链接，让你的 fopen 覆盖任意文件。这就是著名的 symlink 攻击。

替代方案：

接口 安全性 平台
tmpfile 安全（库帮你原子创建） 标准 C
mkstemp 安全（原子创建 + 返回 fd） POSIX
mkostemp 安全 + 可指定 O_CLOEXEC GNU
tmpnam 不安全，禁用 C99 弃用

mkstemp：POSIX 推荐

#include <stdlib.h>
char tmpl[] = "/tmp/quote_XXXXXX";   /* XXXXXX 必须在末尾, 6 个 X */
int fd = mkstemp(tmpl);
if (fd < 0) { perror("mkstemp"); exit(1); }
/* fd 已打开, 模式 rdwr */
/* tmpl 被替换成实际文件名 */
FILE *fp = fdopen(fd, "w+");
/* ... 用完 ... */
fclose(fp);
unlink(tmpl);   /* 显式删除, 或不 unlink 留作调试 */

mkstemp 原子地创建文件并打开，绕过 TOCTOU。XXXXXX 被替换成随机后缀，文件权限 0600。

fmemopen：内存流

FILE *fmemopen(void *buf, size_t size, const char *mode);

把一块内存当文件用，所有 fprintf/fread/fseek 都在内存里操作，零 syscall：

char membuf[4096];
FILE *mf = fmemopen(membuf, sizeof membuf, "w+");
fprintf(mf, "seq=%u bid=%u ask=%u", seq, bid, ask);
fflush(mf);
size_t len = ftell(mf);
/* membuf 里已有格式化结果, 可以直接 send() */
fclose(mf);

HFT 用途：行情字段格式化到内存再 sendto，避免 snprintf 到栈缓冲再 send 的两次拷贝。注意：

  • "w" 模式下首次 fprintf 会把 buf 第一个字节设 \0，破坏原内容
  • 追加模式 "a" 在内存流上语义微妙，需测试
  • fclose 不会 free buf（buf 由调用者提供）

open_memstream：动态增长的内存流

FILE *open_memstream(char **bufp, size_t *sizep);
  • 库自动 malloc 缓冲区，按需 realloc 扩容
  • fclose 后 *bufp 指向结果，调用者负责 free
  • 适合不知道结果多长的场景（如递归格式化嵌套结构）
char *buf; size_t len;
FILE *ms = open_memstream(&buf, &len);
fprintf(ms, "order={");
fprintf(ms, "id=%u,", id);
fprintf(ms, "price=%lu", price);
fprintf(ms, "}");
fclose(ms);
/* buf 现在是 "order={id=...,price=...}", len 是长度 */
send(sock, buf, len, 0);
free(buf);

tmpfile 与异常退出

程序 exit 正常 → tmpfile 自动删除。程序 abort / _exit / SIGKILL → 临时文件可能残留（实现相关）。HFT 长跑进程要定期清理 /tmp 残留，或用 O_TMPFILE 完全避免路径。

fmemopen 的位置状态

fmemopen 流的初始位置取决于 mode："r" 在开头，"w" 在开头（且清零），"a" 在第一个 \0 处或末尾。fseek 可定位，但定位超过 size 是未定义行为。

HFT 关联

  • 抓包临时缓存：异常 burst 时把 pcap 包 dump 到 tmpfile，事后分析；正常路径用 ring buffer 不落盘
  • 回放分片：大历史文件分片到 tmpfile 并行回放，处理完自动删除
  • 内存日志格式化：fmemopen 替代 snprintf，多字段拼接更方便，避免缓冲大小估算
  • 单测 fixture：用 fmemopen 把内存当文件，测试解析函数无需真实文件 IO
  • 避免磁盘磨损：SSD/NVMe 寿命有限，临时中转用 tmpfs（/dev/shm）或 fmemopen 减少写入
  • 多核共享临时数据：tmpfile 是进程私有，跨进程共享用 shm_open + mmap（见 ch18）
  • 崩溃诊断：核心转储 + tmpfile 残留是定位偶发 bug 的重要线索，运维脚本应保留 /tmp 副本

文件操纵函数

本节讲什么

remove / rename：文件系统级别的删除与重命名。这俩看似简单，但 rename 的原子性、跨文件系统行为、与已打开 fd 的交互，是 HFT 滚动日志、配置热加载、原子切换数据文件的核心。

要点

remove

int remove(const char *path);
  • 删除文件路径（等价 unlink(2)）
  • 删除空目录（等价 rmdir(2)）
  • 成功返回 0，失败返回非 0 并设 errno
  • 不释放已打开的 fd：Linux 允许文件被 unlink 后仍可通过 fd 读写，直到所有 fd 关闭才真正释放空间

rename

int rename(const old, const char *new);

语义：

  • 原子地把 old 改名成 new
  • new 已存在 → 覆盖（POSIX，前提是同文件系统且同为文件或同为空目录）
  • 跨文件系统 → 失败，errno = EXDEV
  • 成功返回 0，失败返回非 0 并设 errno

rename 的原子性是金矿

rename 在同文件系统内是原子的——要么成功要么完全没发生，观察者永远看不到中间状态。这使它成为原子更新文件的标配模式：

/* 原子写文件的标准模式 */
FILE *fp = fopen("config.tmp", "w");
fwrite(new_config, 1, len, fp);
fflush(fp);
fsync(fileno(fp));   /* 确保落盘 */
fclose(fp);

rename("config.tmp", "config.dat");   /* 原子替换 */

这样 config.dat 永远是完整有效的——要么是旧版本要么是新版本，不会出现"写到一半"的损坏状态。HFT 配置热加载、订单簿快照切换、pcap 滚动都用这个模式。

rename 跨文件系统的 EXDEV

/tmp 通常是 tmpfs（内存），/data 是磁盘文件系统。rename("/tmp/x", "/data/x") 会失败 EXDEV。替代方案：

  1. cp 跨 FS 复制 + rename 同 FS 替换
  2. link + unlink（但 link 也要求同 FS）
  3. 数据拷贝无法原子跨 FS，必须接受"短暂窗口期"

HFT 设计上把临时文件和最终文件放同一 FS（如都在 /data/hft），避免 EXDEV。

rename 与已打开 fd

FILE *fp = fopen("log.txt", "w");
rename("log.txt", "log.1");   /* 旧路径没了, 但 fp 仍指向原文件 */
fprintf(fp, "...");           /* 写到的是 log.1, 不是新的 log.txt */

rename 不影响已打开的 fd——fd 指向 inode 而不是路径，改名不改变 inode。这意味着滚动日志必须 fclose 旧 fd 再 fopen 新路径，否则写到的还是旧文件（已改名）。

滚动日志标准模式

void rotate_log(FILE **fp, const char *path, int max_keep) {
    fclose(*fp);
    *fp = NULL;

    /* 滚动: log.4 -> log.5 (删), log.3 -> log.4, ..., log -> log.1 */
    char old[256], new[256];
    snprintf(old, sizeof old, "%s.%d", path, max_keep - 1);
    remove(old);   /* 删最老的 */
    for (int i = max_keep - 2; i >= 1; i--) {
        snprintf(old, sizeof old, "%s.%d", path, i);
        snprintf(new, sizeof new, "%s.%d", path, i + 1);
        rename(old, new);
    }
    snprintf(new, sizeof new, "%s.1", path);
    rename(path, new);   /* 当前 -> .1 */

    *fp = fopen(path, "w");   /* 新建当前 */
    if (!*fp) { perror("fopen"); /* 降级 */ }
}

tmpfile 与 remove

tmpfile 内部已经 unlink，不需要 remove。mkstemp 创建的文件不自动删除，需要显式 remove 或 unlink。

remove 与目录

remove 删空目录，非空目录返回失败 errno ENOTEMPTY。递归删目录需要自己实现或用 nftw + unlink + rmdir。HFT 场景下递归删目录通常交给运维脚本（rm -rf），不在数据面做。

stdio vs POSIX vs syscall 分层

应用层
  │
  ├─ stdio (FILE*)   fopen/fread/fwrite/fclose   ← 跨平台, 带缓冲
  │
  ├─ POSIX (fd)      open/read/write/close/lseek  ← Unix 系调用, 无缓冲
  │
  └─ syscall         openat/pread/pwrite/sendfile/ splice
                       ← Linux 专用, 零拷贝/异步

HFT 数据面常直接用最底层：

层 接口 HFT 用途
stdio fprintf/fread 控制面日志、配置
POSIX open/write/fsync 数据面日志（绕过 stdio 缓冲）
syscall writev/sendfile/splice 零拷贝转发
mmap mmap/msync 行情文件内存映射

remove/rename 与权限

  • remove 需要对父目录有写权限，不需要对文件本身有权限
  • rename 需要对 old 和 new 的父目录都有写权限
  • HFT 运维：进程以 hft 用户跑，日志目录属主设 hft，配置目录只读避免误删

rename 的目录覆盖语义

POSIX 规定 rename(dir, newdir) 当 newdir 存在且为空目录时覆盖，非空时失败。这与文件的"无条件覆盖"不同，容易踩坑。Linux 上 rename 目录还有缓存一致性陷阱（其他进程可能仍看到旧路径），生产代码倾向 renameat2 + RENAME_EXCHANGE。

HFT 关联

  • 滚动日志：rename + fopen 是标准日志轮转模式，原子切换保证日志不丢
  • 配置热加载：写 config.tmp + fsync + rename 原子替换 config.dat，进程监测 inode 变化重新加载
  • 订单簿快照：周期 dump 到 snapshot.tmp，rename 为 snapshot.dat，重启时读 snapshot.dat 必然完整
  • pcap 滚动：抓包到 trace.pcap，满大小后 rename 为 trace.<ts>.pcap，新建 trace.pcap 继续
  • 避免跨 FS rename：日志/配置/快照放同一 FS，避开 EXDEV
  • rename 后 fd 行为：运维写脚本时常踩"改名后进程仍写到旧文件"的坑，需 lsof + 信号触发重开
  • DPDK 完全绕过：数据面 0 文件操作，remove/rename 仅用于控制面运维

代码自测

题目 1： 以下代码有什么问题？

FILE *fp = fopen("data.txt", "r");
remove("data.txt");
fclose(fp);
参考答案

在 Windows 上，文件被 fopen 打开期间不能 remove——会失败。在 Unix/Linux 上可以删除（文件在 fclose 前仍然存在，只是目录项被移除）。这是跨平台差异。正确做法：先 fclose 再 remove，或者使用平台特定的 API。
