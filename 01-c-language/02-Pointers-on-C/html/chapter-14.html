14.0

`gcc -E`：只做预处理

《Pointers on C》第 14 章配套 · 预处理器实操
ch14 README · 14.4 文件包含 · 14.3 条件编译

命令含义

gcc -E main.c -o main.i
-E 只做预处理，然后停止——不编译、不汇编、不链接
输入 main.c
输出 main.i（预处理后的 C 文本）

.i 仍是可读的 C 源码，不是汇编、不是二进制。
若 #include <stdio.h>，.i 常膨胀到几千行，属正常。

等价：cpp main.c -o main.i（效果几乎一样）。

预处理做哪些事

  1. 展开宏：#define 文本替换
  2. 展开头文件：#include 内容粘贴进来（递归）
  3. 条件编译：#ifdef / #if / #else / #endif，只留有效分支
  4. 删除注释：//、/* */ 默认去掉
  5. 生成 # 行号 "文件名" 标记（方便报错 / gdb 对回源码行）

常用变体

# 预处理结果打到屏幕（不写文件）
gcc -E main.c

# 保留注释（大写 -C），方便人读
gcc -E -C main.c -o main.i

# 命令行定义宏
gcc -E -DDEBUG=1 main.c -o main.i

GCC 四阶段对照

# 1. 预处理  c → i
gcc -E main.c -o main.i
# 2. 编译    i → s（汇编）
gcc -S main.i -o main.s
# 3. 汇编    s → o
gcc -c main.s -o main.o
# 4. 链接    o → 可执行
gcc main.o -o main

平时一条 gcc main.c -o main 是把四步串起来；出宏 / include 问题时，用 -E 单独看第一阶段。

典型使用场景

  • 宏展开结果不对：看实际被替换成了什么
  • 头文件重复包含、条件编译分支走错
  • 想确认系统头最终「喂」给编译器的是什么

排查 #include 时，常与 14.4 一起用：gcc -E a.c -o a.i。

代码自测

题目 1： 以下代码经过 gcc -E 预处理后，宏展开结果是什么？

#define SQUARE(x) ((x) * (x))
int result = SQUARE(3 + 1);
参考答案

展开为 int result = ((3 + 1) * (3 + 1));，结果为 16。注意宏参数必须加括号——如果写 #define SQUARE(x) x * x，展开为 3 + 1 * 3 + 1 = 7（优先级错误）。外层括号也很重要，防止宏与其他运算符组合时出错。

14.1

预定义符号

本节讲什么

预定义符号是编译器自动提供的内置宏，无需 #define 即可使用。它们覆盖三类信息：源码位置（__FILE__/__LINE__/__func__）、平台架构（__x86_64__/__aarch64__/__linux__）、编译器与标准版本（__GNUC__/__STDC_VERSION__）。这一节把这些宏分类讲清，并给出 HFT 工程里日志、跨平台编译、特性检测的实际用法。

认知锚点：预定义宏是 C 工程的编译期反射——在标准 C 没有运行时反射的情况下，这些宏是仅有的「让代码知道自己被谁、在哪、怎么编译」的机制。HFT 跨平台部署（x86 + ARM、Linux + FreeBSD）、日志定位、特性开关全部依赖它们。

要点

源码位置宏

#define ERR_LOG(fmt, ...) \
    fprintf(stderr, "[%s:%d %s] " fmt, \
            __FILE__, __LINE__, __func__, ##__VA_ARGS__)

ERR_LOG("order rejected: id=%lu\n", order_id);
/* 输出：[order_engine.c:142 process_order] order rejected: id=12345 */
宏 类型 含义 标准
__FILE__ const char * 当前源文件名（字符串字面量） C89
__LINE__ int 当前行号 C89
__func__ const char * 当前函数名（static const char __func__[]） C99
__DATE__ const char * 编译日期 "Jul 19 2026" C89
__TIME__ const char * 编译时间 "09:24:30" C89
__BASE_FILE__ const char * 主源文件名（GCC 扩展） GCC
__COUNTER__ int 每次使用递增（GCC/Clang 扩展） GCC

__func__ 与 __FUNCTION__ 的区别：__func__ 是 C99 标准，本质是 static const char __func__[] = "name";（编译器隐式注入）。__FUNCTION__ 是 GCC 老扩展，C99 后等价于 __func__。新代码用 __func__。

__DATE__/__TIME__ 的工程陷阱：这两个宏让构建不可复现——同一份源码不同时间编译出不同二进制。HFT 对账、监管审计要求二进制可复现，必须用 -Wdate-time 警告 + 构建系统注入固定时间戳。

平台与架构宏

#if defined(__x86_64__) && defined(__linux__)
    /* x86-64 Linux：强内存模型，可省部分屏障 */
#elif defined(__aarch64__) && defined(__linux__)
    /* ARM64 Linux：弱内存模型，需显式 smp_wmb/smp_rmb */
#elif defined(__powerpc64__)
    /* POWER：弱内存模型 */
#else
    #error "unsupported platform"
#endif
宏 含义
__x86_64__、__amd64__ x86-64 架构
__i386__ 32 位 x86
__aarch64__ 64 位 ARM
__arm__ 32 位 ARM
__powerpc__、__powerpc64__ PowerPC
__riscv RISC-V
__linux__ Linux 平台
__FreeBSD__、__OpenBSD__ BSD 系列
_WIN32、_WIN64 Windows（注意无前导下划线）
__APPLE__、__MACH__ macOS

内存模型差异是 HFT 跨平台的核心痛点：x86-64 是强内存模型（TSO），写-写天然有序，可省部分屏障；ARM64/POWER 是弱内存模型，多核同步必须显式 smp_wmb()/smp_rmb()。DPDK 的 rte_smp_wmb() 在 x86 上展开为空，在 ARM64 上展开为 dmb ishst——条件编译隐藏这种差异。

编译器与版本

宏 含义
__GNUC__ GCC 主版本号
__GNUC_MINOR__ GCC 次版本号
__clang__ Clang 编译器
__clang_major__ Clang 主版本
_MSC_VER MSVC 版本（无前导下划线）
__INTEL_COMPILER Intel 编译器 
#if defined(__clang__)
    #define COMPILER "clang " __clang_version__
#elif defined(__GNUC__)
    #define COMPILER "gcc " __VERSION__
#elif defined(_MSC_VER)
    #define COMPILER "msvc " _MSC_FULL_VER
#endif

GCC 与 Clang 的关系：Clang 定义 __clang__ 同时也定义 __GNUC__（兼容 GCC 扩展）。检测顺序必须先 Clang 后 GCC，否则 Clang 被误判为 GCC。

C 标准版本

宏 值 标准
__STDC__ 1 标准 C（C89 起）
__STDC_VERSION__ 199409L C89 + AMD1
199901L C99
201112L C11
201710L C17/C18

#elif defined(__STDC_VERSION__) && __STDC_VERSION__ >= 199901L

HFT 工程通常要求 C11 起步（<stdatomic.h>、_Static_assert、_Generic），部分前沿项目用 C17/C23。DPDK 最低要求 C99，推荐 C11。

C11/C23 特性检测宏

/* C11：__has_include 检测头文件是否存在 */
#if defined(__has_include)
    #if __has_include(<stdatomic.h>)
        #include <stdatomic.h>
        #define HAVE_STDATOMIC 1
    #endif
#endif

/* C23：__has_c_attribute 检测属性 */
#if defined(__has_c_attribute)
    #if __has_c_attribute(deprecated)
        #define DEPRECATED [[deprecated]]
    #endif
#endif

__has_include 让代码自适应不同环境——某些 libc 可能没有 <stdatomic.h>，运行时检测比硬编码平台更可靠。

HFT 关联

日志定位

#define LOG_ERR(fmt, ...) \
    rte_log(RTE_LOG_ERR, RTE_LOGTYPE_APP, \
            "[%s:%d %s] " fmt, \
            __FILE__, __LINE__, __func__, ##__VA_ARGS__)

__FILE__/__LINE__/__func__ 是 HFT 日志的三件套。生产环境关 DEBUG 日志但保留 ERROR，定位问题时一行日志就能找到源码位置。注意二进制大小：__FILE__ 是字符串字面量，每个使用点都存一份文件名，大型项目可能膨胀几 MB .rodata。优化方案：编译选项 -fmacro-prefix-map 裁剪路径前缀。

跨平台内存屏障

/* DPDK rte_smp_wmb 实现（简化） */
#if defined(RTE_ARCH_X86_64)
    #define rte_smp_wmb() do { } while (0)   /* x86 强模型，空操作 */
#elif defined(RTE_ARCH_ARM_64)
    #define rte_smp_wmb() asm volatile("dmb ishst" ::: "memory")
#elif defined(RTE_ARCH_PPC_64)
    #define rte_smp_wmb() asm volatile("lwsync" ::: "memory")
#endif

同一份 C 代码在不同架构上编译出不同的屏障指令，HFT 跨平台部署的基础。写屏障用宏而非函数——屏障必须内联到调用点，函数调用会引入额外的内存访问，破坏屏障语义。

编译期平台校验

#if !defined(__x86_64__) && !defined(__aarch64__)

14.2

#define

本节讲什么

#define 是预处理器的核心指令，它在编译第一阶段做纯文本 token 替换——无类型、无作用域检查、无求值。这一节给出预处理的完整流程、两大宏类型（对象宏与函数宏）的对比，以及 #define 在 C 工程里的定位。

认知锚点：#define 不是 C 语言的语法，而是预处理器的元指令。预处理器在编译器之前运行，把 .c 文件变换成另一份 .c 文件（翻译单元），编译器再处理变换后的结果。理解这条管线，才能理解宏为什么「无类型」——预处理器根本不懂 C 类型，它只认 token。

要点

预处理流程

C 标准定义的翻译阶段共 8 个，前 4 个属于预处理器：

源文件 (.c)
   |
   | 1. 三字符替换、行续接（\ 换行）
   | 2. token 化、注释替换为空格
   | 3. #include 嵌入头文件（递归）
   | 4. #define 宏展开、#if/#ifdef 条件裁剪
   v
翻译单元 (.i，纯 C 代码)
   |
   | 5-8. 编译器接管：语法分析、代码生成
   v
目标文件 (.o)

预处理器的输出可以用 gcc -E foo.c -o foo.i 查看，是排查宏展开问题的第一手段。#include 在第 3 步把头文件文本嵌入当前文件，#define 在第 4 步做文本替换——两者都是纯文本操作，不理解 C 语义。

两大宏类型

类型 语法 示例 用途
对象宏 #define NAME 替换体 #define BURST_SIZE 32 常量、标志位、配置
函数宏 #define NAME(params) 替换体 #define MIN(x,y) ((x)<(y)?(x):(y)) 内联计算、日志、符号操作 
/* 对象宏：无参数，纯文本替换 */
#define BURST_SIZE 32
#define PKT_FLAG_ORDER (1U << 2)
#define APP_NAME "hft-engine"

/* 函数宏：带参数，参数也是文本替换 */
#define MIN(x, y)     ((x) < (y) ? (x) : (y))
#define MAX_PKT_NUM   64
#define LOG(fmt, ...) fprintf(stderr, fmt, ##__VA_ARGS__)

宏展开的规则

  1. 递归展开：宏体里若引用其他宏，继续展开。
  2. 不递归自引用：#define FOO FOO 不会无限展开，FOO 保持原样。
  3. 字符串内不展开："BURST_SIZE" 里的 BURST_SIZE 是字符串内容，不替换。
  4. 字符常量内不展开：'BURST_SIZE' 同理。
  5. 参数替换后再展开：函数宏的参数先替换到宏体，再对整体展开。
#define A 10
#define B A
#define C B

int x = C;   /* C → B → A → 10 */

#define FOO FOO + 1
int y = FOO;  /* FOO → FOO + 1，不递归自引用 */

对象宏的典型用途

/* 配置常量 */
#define MAX_ORDERS        100000
#define ORDERBOOK_DEPTH   10
#define HEARTBEAT_NS      1000000000ULL

/* 标志位掩码 */
#define FLAG_BID    (1U << 0)
#define FLAG_ASK    (1U << 1)
#define FLAG_CANCEL (1U << 2)

/* 字符串常量 */
#define PROTO_VERSION "1.3"
#define LOG_PREFIX    "[hft] "

对象宏在编译期替换为字面量，零运行开销，比 const 变量更早可用（预处理阶段）。但 const 变量有类型、有作用域、可调试——两者选型见 14.2.2。

函数宏的典型用途

/* 内联计算 */
#define MIN(x, y)  ((x) < (y) ? (x) : (y))
#define MAX(x, y)  ((x) > (y) ? (x) : (y))
#define ABS(x)     ((x) < 0 ? -(x) : (x))

/* 日志 */
#define LOG(level, fmt, ...) \
    rte_log(level, RTE_LOGTYPE_APP, fmt, ##__VA_ARGS__)

/* 符号拼接 */
#define PKT_TYPE(name) PKT_##name

/* 断言 */
#define ASSERT(cond) do { \
    if (!(cond)) { \
        fprintf(stderr, "assert failed: %s at %s:%d\n", \
                #cond, __FILE__, __LINE__); \
        abort(); \
    } \
} while (0)

函数宏的深入用法（#/##/__VA_ARGS__）见 14.2.3。

宏的作用域

#define DEBUG 1

void f(void) {
    /* DEBUG 在此可见 */
#ifdef DEBUG
    log_debug("entering f");
#endif
}

#undef DEBUG   /* 取消定义，后续不可见 */

void g(void) {
    /* DEBUG 在此不可见 */
#ifdef DEBUG
    log_debug("entering g");   /* 不编译 */
#endif
}

宏的作用域从 #define 到 #undef 或文件末，无 C 作用域概念——函数边界、块边界对宏无效。这是宏「污染全局命名空间」的根源。

HFT 关联

DPDK 标志位

/* rte_mbuf.h 中的标志位定义 */
#define PKT_RX_RSS_HASH     (1ULL << 0)
#define PKT_RX_FDIR         (1ULL << 1)
#define PKT_RX_L4_CKSUM_BAD (1ULL << 3)
#define PKT_RX_IP_CKSUM_BAD (1ULL << 4)

DPDK 用对象宏定义 64 位标志位，编译期替换为位掩码。为什么用宏不用 const？因为标志位要用于预处理条件（#if PKT_RX_RSS_HASH）和静态初始化（uint64_t mask = PKT_RX_RSS_HASH | PKT_RX_FDIR;），const 变量在预处理阶段不可用。

rte_log 可变参数宏

#define rte_log(level, type, fmt, ...) \
    do { \
        if (rte_log_get_level(type) >= level) \
            __rte_log(level, type, fmt, ##__VA_ARGS__); \
    } while (0)

##__VA_ARGS__ 是 GCC 扩展，无参时去掉前面的逗号。这让 rte_log(ERR, "ok") 和 rte_log(ERR, "x=%d", x) 都能编译。详见 14.2.3。

/* 编译时通过 -D 注入配置 */
/* gcc -DBURST_SIZE=64 -DMAX_CORES=16 ... */

#ifndef BURST_SIZE
    #define BURST_SIZE 32   /* 默认值 */
#endif

               "BURST_SIZE must be in [1, 256]");

构建系统通过 -D 注入配置，源码用 #ifndef 提供默认值。_Static_assert 在编译期校验配置合法性，优于运行期检查。

宏与 const 的边界

#define SIZE 100           /* 宏：预处理替换，无类型 */

  • C99 起 VLA：数组大小可用变量 → const 也可。

14.2

.1 宏

本节讲什么

这一节聚焦对象宏（object-like macro）——无参数的 #define，用于定义常量、标志位、字符串。核心是括号陷阱：宏是纯文本替换，不加括号会导致运算符优先级错乱，是 C 工程最经典的初级错误。

认知锚点：对象宏看起来最简单，却藏着最易踩的坑。#define SQR(x) x*x 调用 SQR(1+2) 得到 1+2*1+2 = 5 而非 9，这种 bug 在线上代码里屡见不鲜。括号是宏的护身符——整体加括号、参数各自加括号，是写宏的铁律。

要点

基本形态

#define BURST_SIZE 32
#define PKT_FLAG_ORDER (1U << 2)
#define APP_NAME "hft-engine"
#define LOG_PREFIX "[hft] "

uint16_t n = rte_eth_rx_burst(port, q, pkts, BURST_SIZE);  /* 替换为 32 */
uint32_t flags = mbuf->ol_flags & PKT_FLAG_ORDER;          /* 替换为 (1U<<2) */
printf("%s%s starting\n", LOG_PREFIX, APP_NAME);           /* 字面量替换 */

宏体可以是任意 token 序列：数字、字符串、表达式、关键字。预处理器不解析语义，只做文本替换。

括号陷阱：经典反例

#define MUL(a, b)  a * b
int x = MUL(1+2, 3+4);
/* 展开：1+2 * 3+4 = 1 + (2*3) + 4 = 11，非 (1+2)*(3+4)=21 */

运算符优先级让 * 先于 + 结合，宏体 a * b 在参数是表达式时出错。正确写法：

#define MUL(a, b)  ((a) * (b))
int x = MUL(1+2, 3+4);
/* 展开：((1+2) * (3+4)) = 21 */

两层括号：① 每个参数加括号，防止参数内部表达式与宏体运算符错位；② 整体加括号，防止宏体作为子表达式时与外部运算符错位。

整体括号的必要性

#define DOUBLE(x)  (x) * 2
int y = 3 + DOUBLE(4);
/* 展开：3 + (4) * 2 = 3 + 8 = 11，非 3 + 4*2 = 11（碰巧对） */

#define BAD(x)  (x) + 1
int z = 5 * BAD(3);
/* 展开：5 * (3) + 1 = 16，非 5 * ((3)+1) = 20 */

BAD 宏体 (x) + 1 参数有括号但整体无括号，与外部 * 结合时优先级错乱。整体加括号：

#define GOOD(x)  ((x) + 1)
int z = 5 * GOOD(3);   /* 5 * ((3)+1) = 20 */

位运算的陷阱

#define FLAG_A  1 << 0
#define FLAG_B  1 << 1

int mask = FLAG_A | FLAG_B;
/* 看似 OK，但 */

int bad = FLAG_A + FLAG_B;
/* 展开：1 << 0 + 1 << 1 = 1 << (0+1) << 1 = 1 << 1 << 1 = 2 << 1 = 4 */
/* 本意：(1<<0) + (1<<1) = 1 + 2 = 3 */

移位运算符 << 优先级低于 +，1 << 0 + 1 被解析为 1 << (0+1)。位掩码宏必须整体加括号：

#define FLAG_A  (1U << 0)
#define FLAG_B  (1U << 1)
#define FLAG_C  (1U << 2)

DPDK、内核的标志位定义全部遵守此规则。

字符串字面量宏

#define APP_NAME "hft-engine"
#define LOG_PREFIX "[hft] "
#define VERSION_STR "1.3.0"

printf("%s%s v%s\n", LOG_PREFIX, APP_NAME, VERSION_STR);
/* 输出：[hft] hft-engine v1.3.0 */

字符串字面量宏无需加括号（字符串本身就是不可分割的 token）。但若宏体是字符串拼接，仍需注意：

#define GREETING "hello, " "world"   /* 编译期自动拼接 */
const char *s = GREETING;            /* "hello, world" */

宏的三大特点

特性 说明 影响
无类型 宏是文本替换，预处理器不检查类型 类型错误要到编译期才暴露
无作用域 从 #define 到 #undef 或文件末，无视函数/块边界 全局污染，命名冲突
调试难 宏在预处理阶段已展开，gdb 看不到宏名 断点只能打在展开后的代码 
#define BURST_SIZE 32

void f(void) {
    /* BURST_SIZE 在此可见 */
    int x = BURST_SIZE;
}

/* 函数边界对宏无效 */

void g(void) {
    /* BURST_SIZE 仍可见 */
    int y = BURST_SIZE;
}

BURST_SIZE 一旦定义，整个文件都可见，无法限制在单个函数内。这是宏相比 static const 的根本缺陷。

14.2

.2 宏与函数

本节讲什么

#define 函数宏和 static inline 函数都能做到「内联展开、零调用开销」，但两者在类型检查、参数求值、作用域、调试上差异巨大。这一节给出完整的对比表和工程选型准则——这是底层 C 工程师最常做的决策之一。

认知锚点：宏不是「更快的函数」，而是完全不同的机制——预处理器文本替换 vs 编译器语义分析。混淆两者会导致类型错误漏检、副作用重复求值、调试困难。HFT 工程的硬性准则：复杂计算用 inline，日志/常量/符号拼接用宏。

要点

完整对比表

特性 #define 函数宏 static inline 函数
参数求值 可能多次（每次出现都求值） 一次（调用时求值一次）
类型检查 无（文本替换） 有（编译器检查）
类型泛型 通过 __typeof 或 _Generic 函数重载 C 不支持，需多份
作用域 全局污染（无视函数/块边界） 文件内 static，可控
调试 难断点（已展开，gdb 看不到） 可断点（函数名可见）
递归 不支持（展开后是表达式） 支持（但 inline 递归不展开）
副作用 参数多次求值风险 安全（一次求值）
适合场景 常量、日志、#/##、断言 计算、MIN/MAX、小工具
编译速度 快（预处理器简单） 慢（编译器分析）
二进制大小 内联后可能膨胀 同样可能膨胀

副作用对比

#define MIN_MACRO(x, y)  ((x) < (y) ? (x) : (y))

static inline int min_func(int a, int b)
{
    return a < b ? a : b;
}

int a = 1, b = 3;
int m1 = MIN_MACRO(a++, b);   /* a 可能自增两次（见 14.2.4） */
int m2 = min_func(a++, b);    /* a 只自增一次 */

宏的参数每次出现都求值一次，a++ 在 MIN_MACRO 里出现两次（条件判断和返回值），自增两次。min_func 的参数在调用时求值一次，传入函数后只是局部变量，自增一次。这是宏相比函数最危险的特点，详见 14.2.4。

类型检查对比

#define ABS_MACRO(x)  ((x) < 0 ? -(x) : (x))

static inline int abs_int(int x) { return x < 0 ? -x : x; }

/* 宏：无类型检查，任何类型都能用 */
double d = ABS_MACRO(-3.14);       /* 3.14，碰巧对 */
char c = ABS_MACRO('A');           /* 65，碰巧对 */
struct { int x; } s = {0};
int bad = ABS_MACRO(s);            /* 编译错误，但报错信息晦涩 */

/* 函数：类型严格 */
double d2 = abs_int(-3.14);        /* 隐式转换，警告 */
int i = abs_int(-5);               /* 正确 */

宏无类型检查，错误信息往往晦涩（展开后的代码报错，看不出原始宏）。函数有类型检查，错误信息清晰。C11 的 _Generic 可以让宏模拟泛型，但仍不如函数类型安全。

类型泛型的处理

/* 宏 + _Generic：C11 泛型 */
#define ABS(x) _Generic((x), \
    int:    abs_int, \
    long:   labs, \
    double: fabs, \
    default: abs_int \
)(x)

/* 函数：每种类型写一份 */
static inline int    abs_int(int x)      { return x < 0 ? -x : x; }
static inline long   abs_long(long x)    { return x < 0 ? -x : x; }
static inline double abs_double(double x){ return x < 0 ? -x : x; }

宏 + _Generic 是 C11 的泛型方案，但语法复杂。C++ 的模板更优雅，但 C 没有。GNU __typeof 扩展可以简化：

#define MIN(a, b) ({ \
    __typeof(a) _a = (a); \
    __typeof(b) _b = (b); \
    _a < _b ? _a : _b; \
})

这种「statement expression」写法既保留宏的泛型性，又通过局部变量保证单次求值，是 DPDK/内核的常用技巧。详见 14.2.5。

调试对比

#define DBG_VAL(x) printf("%s = %d\n", #x, (x))

static inline void dbg_val_int(const char *name, int x)
{
    printf("%s = %d\n", name, x);
}

宏在预处理阶段已展开，gdb 看不到宏名，无法在宏上打断点。static inline 函数有符号，gdb 可断点（虽然 inline 后可能没调用点）。大型项目调试时，inline 函数的可调试性优势明显。

作用域对比

/* 宏：全局污染 */
#define HELPER(x) ((x) * 2)
void f(void) { int y = HELPER(5); }
void g(void) { int z = HELPER(10); }   /* HELPER 仍可见 */

/* static inline：文件作用域 */
static inline int helper(int x) { return x * 2; }
void f(void) { int y = helper(5); }
void g(void) { int z = helper(10); }   /* helper 仅本文件可见 */

宏无视作用域，一旦定义全文件可见。static inline 限制在文件内，不污染其他翻译单元。头文件里的宏会传播到所有包含它的 .c 文件，是命名冲突的高发区。

硬性选型准则

场景 选型 理由
常量、配置 宏或 static const 预处理可用、零开销
日志、断言 宏 需要 __FILE__/__LINE__/#cond
符号拼接 ## 宏 函数做不到
可变参数 __VA_ARGS__ 宏 C 函数不支持（C23 前）
MIN/MAX/ABS static inline 或 GNU ({}) 宏 避免副作用
复杂计算 static inline 类型安全、可调试
位掩码 宏 预处理可用、静态初始化
类型泛型 宏 + _Generic 或多份函数 C 无函数重载

口诀：需要预处理器特性（#/##/__VA_ARGS__/__FILE__）→ 宏；需要类型安全/单次求值/可调试 → inline。

HFT 关联

DPDK 的 MIN/MAX 实现

/* rte_common.h */
#define RTE_MIN(a, b) ({ \
    __typeof(a) _a = (a); \
    __typeof(b) _b = (b); \
    _a < _b ? _a : _b; \
})

#define RTE_MAX(a, b) ({ \
    __typeof(a) _a = (a); \
    __typeof(b) _b = (b); \
    _a > _b ? _a : _b; \
})

DPDK 用 GNU statement expression 实现 MIN/MAX，既保留宏的泛型性，又通过局部变量保证单次求值。这是宏与函数的折中——非标准（GNU 扩展），但 DPDK/内核的工程惯例。可移植代码应用 static inline，每种类型一份。

内核的 min_t/max_t

/* linux/minmax.h */
#define min_t(type, a, b)  ({ \
    type _a = (a); \
    type _b = (b); \
    _a < _b ? _a : _b; \
})

int x = min_t(int, a++, b);   /* 显式指定类型，单次求值 */

内核用 min_t(type, ...) 显式指定类型，避免 __typeof 的某些边角问题（例如 a 和 b 类型不同时的警告）。

日志宏 vs 日志函数

/* 宏：捕获源码位置 */
#define LOG_ERR(fmt, ...) \
    log_impl(RTE_LOG_ERR, __FILE__, __LINE__, __func__, fmt, ##__VA_ARGS__)

/* 函数：无法捕获调用点位置 */
void log_err(const char *fmt, ...);   /* 不知道是谁调的 */

日志必须用宏——__FILE__/__LINE__/__func__ 只能在宏展开时捕获调用点位置，函数里取到的是日志函数自己的位置。这是宏不可替代的场景。

断言宏

#define ASSERT(cond) do { \
    if (__builtin_expect(!(cond), 0)) { \
        fprintf(stderr, "ASSERT failed: %s at %s:%d in %s\n", \
                #cond, __FILE__, __LINE__, __func__); \
        abort(); \
    } \
} while (0)

14.2

.3 带参数的宏

本节讲什么

函数宏的高级用法：# 字符串化、## 标记拼接、__VA_ARGS__ 可变参数、do { } while(0) 多语句宏。这四样是宏区别于函数的核心能力——它们让宏能在预处理阶段做「符号操作」，是 C 工程实现代码生成、日志框架、断言机制的基础。

认知锚点：# 和 ## 是预处理器独有的运算符，函数完全做不到。__FILE__/__LINE__ 只能在宏里捕获调用点。do {} while(0) 是多语句宏的语法包装。理解这四样，才能读懂内核/DPDK 的宏密集代码，才能写出自己的日志/断言/代码生成宏。

要点

函数宏基础

#define MIN(x, y)  ((x) < (y) ? (x) : (y))
#define MAX(x, y)  ((x) > (y) ? (x) : (y))
#define ABS(x)     ((x) < 0 ? -(x) : (x))

int m = MIN(3, 5);   /* 展开：((3) < (5) ? (3) : (5)) = 3 */

函数宏的参数是文本替换——参数 x 在宏体中每次出现都被替换成调用时的实参文本。括号规则见 14.2.1。

# 字符串化

#define STR(x)      #x
#define LOG_VAR(name) printf(#name " = %d\n", name)

int seq = 42;
LOG_VAR(seq);
/* 展开：printf("seq" " = %d\n", seq); */
/* 输出：seq = 42 */

const char *s = STR(hello world);
/* s = "hello world" */

# 紧贴参数名，把参数的文本形式转成字符串字面量。字符串字面量相邻自动拼接，所以 "seq" " = %d\n" 等价 "seq = %d\n"。

字符串化中的转义

#define STR(x) #x

const char *s1 = STR("quoted");   /* s1 = "\"quoted\"" */
const char *s2 = STR(a\b);        /* s2 = "a\\b" */

# 会自动转义字符串内的 " 和 \。这让 STR("quoted") 的结果是字面量 "\"quoted\""（包含转义引号的字符串）。

## 标记拼接

#define CONCAT(a, b)   a##b
#define PKT_TYPE(name) PKT_##name

uint32_t t = PKT_TYPE(QUOTE);
/* 展开：PKT_QUOTE */

int xy = CONCAT(he, llo);
/* 展开：hello（一个标识符） */

## 把两个 token 拼成一个新 token。必须是合法的 token——## 拼出的结果必须是标识符、数字、或运算符，否则编译错误。

X-Macro 技巧

X-Macro 是 C 代码生成的经典技巧：一份 .def 文件定义数据，多次包含并用不同宏展开生成枚举、字符串表、函数声明等。新增类型只需改 .def 一处，所有生成的代码自动同步。内核、DPDK 大量使用这种模式。

__VA_ARGS__ 可变参数

#define LOG(fmt, ...)  printf("[LOG] " fmt, __VA_ARGS__)

LOG("x=%d\n", x);    /* printf("[LOG] " "x=%d\n", x); */
LOG("ok\n");         /* printf("[LOG] " "ok\n", ); <-- 末尾逗号错误！ */

C99 的 __VA_ARGS__ 代表所有可变参数。但无参时末尾逗号导致编译错误。解决方案：

/* GCC 扩展：## 去掉前面的逗号 */
#define LOG(fmt, ...)  printf("[LOG] " fmt, ##__VA_ARGS__)

LOG("ok\n");         /* printf("[LOG] " "ok\n"); <-- 逗号被去掉 */
LOG("x=%d\n", x);    /* printf("[LOG] " "x=%d\n", x); */

##__VA_ARGS__ 是 GCC 扩展，__VA_ARGS__ 为空时 ## 会吃掉前面的逗号。C20/C23 引入 __VA_OPT__ 提供标准方案：

/* C23 标准 */
#define LOG(fmt, ...)  printf("[LOG] " fmt __VA_OPT__(,) __VA_ARGS__)

LOG("ok\n");         /* printf("[LOG] " "ok\n"); <-- __VA_OPT__() 为空 */
LOG("x=%d\n", x);    /* printf("[LOG] " "x=%d\n", x); <-- __VA_OPT__(,) 为逗号 */

__VA_OPT__(x) 在 __VA_ARGS__ 非空时展开为 x，为空时展开为空。这是标准化的逗号消除方案。

命名参数 vs __VA_ARGS__

/* C99：可变参数只能用 __VA_ARGS__ */
#define LOG(level, fmt, ...) log(level, fmt, ##__VA_ARGS__)

/* GNU 扩展：可命名可变参数 */
#define LOG(level, fmt, args...) log(level, fmt, ##args)

GNU 扩展允许给可变参数命名（args...），可读性更好但非标准。可移植代码用 __VA_ARGS__。

do { } while(0) 多语句宏

#define SWAP(a, b) do { \
    typeof(a) _tmp = (a); \
    (a) = (b); \
    (b) = _tmp; \
} while (0)

if (cond)
    SWAP(x, y);     /* 单语句位置正确 */
else
    other();

/* 若宏体是 { ... }（无 do-while） */
#define BAD_SWAP(a, b) { typeof(a) _t = (a); (a) = (b); (b) = _t; }

if (cond)
    BAD_SWAP(x, y);  /* 末尾分号导致 else 无 if 配对，编译错误 */
else
    other();

do { ... } while(0) 是多语句宏的标准包装：

  1. 单语句语义：do-while 循环整体是一个语句，可放在任何单语句位置（if/for 体）。
  2. 分号自然：调用方写 SWAP(x, y); 末尾分号自然结束语句。
  3. 作用域隔离：宏内局部变量（_tmp）在 do-while 块内，不污染外部。

替代方案 { } 的问题是末尾分号会制造空语句，与 if-else 配对出错。

宏内 goto 与 return

#define TRY(cond, label) \
    do { if (!(cond)) goto label; } while (0)

int f(void) {
    void *a = malloc(100);
    TRY(a, fail_a);

    void *b = malloc(200);
    TRY(b, fail_b);

    /* ... */
    return 0;
fail_b:
    free(a);
fail_a:
    return -1;
}

宏内 goto 是错误处理的常见模式，但宏内 return 是坏实践——隐藏控制流，调用方不知道宏会 return。goto 至少跳到函数内的可见 label。

14.2

.4 宏参数的副作用

本节讲什么

函数宏的参数在宏体中每出现一次就会被求值一次。如果实参含有副作用（自增、自减、函数调用、赋值），多次求值会导致非预期行为——变量被改多次、函数被调多次。这是宏相比 static inline 函数最危险的特性，是 C 工程线上逻辑异常的高频主因。

认知锚点：MIN(a++, b) 这种调用在宏里 a 可能自增两次，在 inline 函数里只自增一次。这种差异不是「编译器优化差异」，而是宏与函数的语义根本不同。HFT 工程的硬性准则：带副作用的表达式不要传给宏，复杂逻辑改用 static inline。

要点

经典反例：MIN(a++, b)

#define MIN(x, y)  ((x) < (y) ? (x) : (y))

int a = 1, b = 3;
int m = MIN(a++, b);

宏展开：

int m = ((a++) < (b) ? (a++) : (b));

求值过程：

  1. (a++) < (b)：先比较 a（值为 1）与 b（值为 3），结果 1 < 3 为真。a++ 后缀自增，a 变成 2。
  2. 条件为真，执行 (a++)：返回 a（当前值 2），a++ 再次后缀自增，a 变成 3。
  3. m = 2，a = 3。

预期：m = 1（用自增前的 a 比较），a = 2（自增一次）。 实际：m = 2，a = 3（自增两次）。

副作用的分类

副作用形式 例子 宏里的风险
自增/自减 a++、++a、a-- 多次求值导致多次自增
函数调用 f(x) 多次调用，可能有副作用
赋值 x = 5、x += 1 多次赋值
复合表达式 x++, y++ 多次求值
I/O 操作 getchar() 多次读取
volatile 读 *hw_reg 多次硬件访问 
#define MAX(x, y)  ((x) > (y) ? (x) : (y))

/* 函数调用副作用 */
int m = MAX(rand(), rand());   /* rand() 可能调两次，结果不可预测 */

/* volatile 副作用 */
volatile uint32_t *counter = HW_COUNTER;
int latest = MAX(*counter, threshold);   /* *counter 可能读两次，值不一致 */

/* I/O 副作用 */
int c = MAX(getchar(), 'a');   /* getchar() 可能调两次，吞掉一个字符 */

三条防御措施

1. 复杂逻辑改用 static inline

static inline int min_int(int a, int b)
{
    return a < b ? a : b;
}

int a = 1, b = 3;
int m = min_int(a++, b);   /* a 只自增一次，m = 1，a = 2 */

static inline 函数的参数在调用时求值一次，传入函数后是局部变量，不存在重复求值。这是最安全的方案。

2. 宏内参数只引用一次

/* 危险：x 引用两次 */
#define SQR(x)  ((x) * (x))
int y = SQR(a++);   /* a 自增两次 */

/* 安全：x 只引用一次 */
#define INCR(x)  ((x) + 1)
int z = INCR(a++);  /* a 只自增一次 */

但「只引用一次」难以保证——MIN/MAX 必须引用两次（条件和返回值）。这种宏只能用 inline 函数或 GNU ({}) 扩展。

3. 调用处不用带副作用的表达式

/* 危险：a++ 有副作用 */
int m = MIN(a++, b);

/* 安全：先求值，再传宏 */
int tmp = a++;
int m = MIN(tmp, b);

调用方自律——不把带副作用的表达式传给宏。但这依赖开发者自觉，大型项目难免有人疏忽。

GNU statement expression：两全其美

#define MIN(a, b) ({ \
    __typeof(a) _a = (a); \
    __typeof(b) _b = (b); \
    _a < _b ? _a : _b; \
})

int x = 1, y = 3;
int m = MIN(x++, y);
/* 展开：
   ({ __typeof(x++) _a = (x++); __typeof(y) _b = (y); _a < _b ? _a : _b; })
   _a = (x++) 求值一次，x 变 2，_a = 1
   _b = (y) 求值一次，_b = 3
   比较 _a < _b，返回 _a = 1
   结果：m = 1，x = 2（自增一次）
*/

GNU ({}) 扩展让宏在块内声明局部变量保存参数值，参数只求值一次，同时保留宏的泛型性。DPDK RTE_MIN、内核 min() 都用这种写法。代价是非标准（GCC/Clang 扩展，MSVC 不支持）。

C23 的 constexpr 与泛型

C23 引入 constexpr、_Generic 增强、__VA_OPT__ 等，部分缓解宏的问题，但副作用问题仍需 GNU ({}) 或 inline 函数解决。标准 C 没有等价于 ({}) 的语法。

        .qty   = (qty), \
    }; \
    o; \
})

struct order o = MAKE_ORDER('B', NEXT_SEQ());
/* NEXT_SEQ() 调一次，OK（宏内只引用一次） */

#define BAD_MAX(x, y) ((x) > (y) ? (x) : (y))
uint64_t s = BAD_MAX(NEXT_SEQ(), last_seq);
/* NEXT_SEQ() 可能调两次，seq_counter 跳号 */

HFT 订单序号必须连续严格递增，NEXT_SEQ() 被宏调两次导致序号跳号，撮合对账失败。这种 bug 在测试环境很难复现（只在特定分支才触发），上线后才暴露。

数据面热点的宏使用

/* DPDK 报文计数宏 */
#define PKT_INC(stat)  __sync_fetch_and_add(&(stat), 1)

/* 危险：宏内若多次引用 stat，原子操作多次 */
#define BAD(stat, val) ({ \
    if ((stat) > (val)) (stat) = (val); \
    (stat); \
})

uint64_t s = BAD(PKT_INC(rx_count), threshold);
/* PKT_INC 调多次，rx_count 乱跳 */

原子操作宏如果被通用宏嵌套引用，会触发多次原子操作，性能和正确性都受影响。原子操作必须用 inline 函数封装，不要让宏嵌套。

硬件寄存器访问

volatile uint32_t *hw_counter = (void *)0xFE000000;

/* 危险：*hw_counter 读两次，值可能不一致 */
uint32_t c = MAX(*hw_counter, last_count);

/* 安全：先读到局部变量 */
uint32_t cur = *hw_counter;
uint32_t c = MAX(cur, last_count);

硬件寄存器每次读都可能返回不同值（计数器、状态寄存器）。宏里多次读会得到不一致的值，逻辑错误。volatile 读必须先存局部变量再传宏。

日志宏的安全性

#define LOG(level, fmt, ...) \
    do { \
        if (log_level >= (level)) \
            log_impl((level), fmt, ##__VA_ARGS__); \
    } while (0)

/* __VA_ARGS__ 只在 log_impl 调用处出现一次，安全 */
LOG(INFO, "seq=%lu\n", NEXT_SEQ());
/* NEXT_SEQ() 只调一次 */

日志宏的 __VA_ARGS__ 只在宏体出现一次，副作用安全。这是宏可以放心用副作用的少数场景。

14.2

.5 宏参数的命名约定

本节讲什么

宏的命名规范是 C 工程的「软法律」——不强制但必须遵守。这一节给出宏名、参数、前缀的命名约定，并深入介绍 GNU statement expression（DPDK/内核 MIN/MAX 的实现基础）和 #undef 的工程用法。

认知锚点：宏命名不像函数有作用域保护，一旦定义全文件可见。全大写命名 + 工程前缀是隔离宏命名空间的唯一手段。读 DPDK/内核源码时，RTE_MIN、min_t、container_of 这类宏的写法是工业级范例。

要点

命名规范

约定 说明 示例
宏名全大写 与变量/函数区分 BURST_SIZE、RTE_LOG
参数加括号使用 防优先级错乱 (x) 非 x
避免短名作全局宏 防与字段/局部变量冲突 不用 i、n、p 作全局宏
工程前缀 防命名空间污染 RTE_、PKT_、LINUX_
下划线分隔 全大写下划线分词 MAX_PKT_BURST 
/* 内核/DPDK 风格 */
#define RTE_MIN(a, b)  ({ ... })
#define RTE_MAX(a, b)  ({ ... })
#define RTE_DIV_ROUND_UP(n, d)  (((n) + (d) - 1) / (d))
#define RTE_ALIGN(val, align)   (((val) + (align) - 1) & ~((align) - 1))

全大写约定的重要性

#define max(a, b)  ((a) > (b) ? (a) : (b))   /* 小写，危险 */

struct stats {
    int max;        /* 字段名 max，与宏冲突 */
};

struct stats s = { .max = 100 };
int x = s.max;     /* 展开为 s.((max)>(max)?(max):(max))，编译错误 */

宏名小写会与变量、字段名冲突。全大写是隔离手段——变量/字段用小写或驼峰，宏用全大写，互不干扰。

工程前缀的必要性

/* 无前缀，全局污染 */
#define MIN(a, b)  ...
#define MAX(a, b)  ...

/* 有前缀，隔离命名空间 */
#define RTE_MIN(a, b)  ...
#define RTE_MAX(a, b)  ...
#define PKT_MIN(a, b)  ...

无前缀的 MIN/MAX 是头文件冲突的高发区——多个库都定义 MIN，包含时互相覆盖。工程前缀（RTE_、PKT_、LINUX_）让宏隶属于特定命名空间，避免冲突。

参数命名

/* 短参数名：宏体内可读性差 */
#define MUL(a, b)  ((a) * (b))

/* 描述性参数名：可读性好 */
#define MULTIPLY(factor, multiplicand)  ((factor) * (multiplicand))

宏参数命名权衡：短名（a、b）简洁但宏体内含义不清；长名（factor、multipand）清晰但调用时看不出参数顺序。工业惯例：短参数名 + 宏名语义化（RTE_MIN(a, b) 的 a/b 不重要，MIN 已说明语义）。

避免与字段冲突的参数名

/* 危险：参数名 x 与结构体字段 x 冲突 */
#define GET_X(s)  ((s)->x)

struct point { int x, y; };
int px = GET_X(p);   /* OK，但若字段改名则宏失效 */

/* 安全：参数名用 _ 前缀 */
#define GET_FIELD(s, field)  ((s)->field)
int px = GET_FIELD(p, x);

宏参数名用 _ 前缀或描述性名字，避免与字段名混淆。

GNU Statement Expression

#define RTE_MIN(a, b) ({ \
    __typeof(a) _a = (a); \
    __typeof(b) _b = (b); \
    _a < _b ? _a : _b; \
})

这是 DPDK/内核 MIN/MAX 的标准实现。语法拆解：

元素 含义
({ ... }) GNU statement expression，块内语句，最后一条表达式作为整个块的值
__typeof(a) 取 a 的类型（GNU 扩展）
_a = (a) 把参数求值一次存入局部变量
_a < _b ? _a : _b 比较，返回最小值

优势：

  1. 单次求值：参数 (a) 只在 _a = (a) 处求值一次，后续用 _a，避免副作用（见 14.2.4）。
  2. 类型泛型：__typeof(a) 自动取参数类型，int/long/double/指针都能用。
  3. 可作表达式：({}) 块有值，可放任何表达式位置（int m = RTE_MIN(x, y);）。

代价：非标准（GCC/Clang 扩展，MSVC 不支持）。可移植代码用 static inline 每种类型一份。

statement expression 的边界

#define BAD() ({ int x = 1; int y = 2; })   /* 无返回值表达式，结果未定义 */

#define GOOD() ({ int x = 1; int y = 2; x + y; })   /* 返回 3 */

({}) 块的最后一条表达式（不是语句）作为整个块的值。x + y; 是表达式语句，值是 x + y。若最后是声明或无表达式，结果未定义。

min_t / max_t：显式类型

/* linux/minmax.h */
#define min_t(type, a, b)  ({ \
    type _a = (a); \
    type _b = (b); \
    _a < _b ? _a : _b; \
})

int x = min_t(int, a, b);
long y = min_t(long, c, d);

显式指定类型，避免 __typeof 在 a 和 b 类型不同时的问题：

/* 危险：a 和 b 类型不同，比较时隐式转换 */
int a = 5;
long b = 10;
auto m = min(a, b);   /* _a 是 int，_b 是 long，比较时 int 转 long */

/* 安全：显式指定类型 */
long m = min_t(long, a, b);   /* _a 和 _b 都是 long */

#undef 的工程用法

#define TEMP_DEBUG 1

/* ... 使用 TEMP_DEBUG ... */

#undef TEMP_DEBUG   /* 取消定义，后续不可见 */

#undef 取消宏定义，让宏只在需要的代码段可见。工程场景：

  1. 局部宏：只在某段代码用的宏，用完 #undef 避免泄漏。
  2. X-Macro 重定义：同一份 .def 文件用不同宏展开，每次包含后 #undef（见 14.2.3）。
  3. 重定义前取消：标准要求重定义前先 #undef（虽然允许相同重定义）。
/* X-Macro 的 #undef 模式 */
#define DEFINE_EVENT(name, val, str)  EVT_##name = val,
enum events {
    #include "events.def"
    EVT_NUM
};
#undef DEFINE_EVENT

#define DEFINE_EVENT(name, val, str)  [val] = str,
static const char *const evt_str[] = {
    #include "events.def"
};
#undef DEFINE_EVENT

container_of 宏

/* linux/kernel.h */
#define container_of(ptr, type, member) ({ \
    void *__mptr = (void *)(ptr); \
    _Static_assert( \
        __builtin_types_compatible_p(typeof(ptr), typeof(&((type *)0)->member)) || \
        __builtin_types_compatible_p(typeof(ptr), void *), \
        "container_of: type mismatch"); \
    ((type *)(__mptr - offsetof(type, member))); \
})

/* DPDK 简化版 */
#define container_of(ptr, type, member) \
    ((type *)((char *)(ptr) - offsetof(type, member)))

container_of 是内核/DPDK 最经典的宏：

  1. 接收「成员指针 ptr、外层类型 type、成员名 member」。
  2. 用 offsetof(type, member) 算成员在外层结构中的偏移。
  3. ptr - 偏移 得到外层结构起始地址。
  4. 强转回 type *。

详见 ch12.3.2 与 ch13。

14.3

条件编译

本节讲什么

条件编译是预处理器的「编译期分支」——#if/#ifdef/#ifndef/#elif/#else/#endif 让代码块根据宏定义或常量表达式选择性编译。这一节给出条件编译的三大用途（调试开关、跨平台、特性裁剪）、指令对照表，以及 HFT 工程里线上关日志、跨架构编译、特性开关的实际形态。

认知锚点：条件编译不是「可选的预处理器特性」，而是 C 工程的基础设施。Linux 内核、DPDK、glibc 全部依赖条件编译适配不同架构、不同配置、不同编译器。理解条件编译，才能理解为什么同一份 C 代码能在 x86、ARM、PowerPC 上跑——靠的就是这套机制。

要点

指令对照表

指令 作用 示例
#if EXPR 表达式非 0 则编译 #if BURST_SIZE > 16
#ifdef NAME 宏已定义则编译 #ifdef DEBUG
#ifndef NAME 宏未定义则编译 #ifndef HEADER_H
#elif EXPR 否则若 #elif defined(__arm__)
#else 否则 #else
#endif 结束条件块 #endif

#ifdef vs #if defined

/* 等价 */
#ifdef DEBUG
    ...
#endif

#if defined(DEBUG)
    ...
#endif

/* 多条件时 #if defined 更清晰 */
#if defined(DEBUG) && defined(VERBOSE)
    ...
#endif

/* ifdef 不支持逻辑组合 */
#ifdef DEBUG && VERBOSE   /* 语法错误 */

#if defined(NAME) 是通用形式，支持 &&、||、! 逻辑组合。#ifdef NAME 是单宏判断的简写。复杂条件用 #if defined，单宏判断用 #ifdef。

#if 的常量表达式

#if BURST_SIZE > 0 && BURST_SIZE <= 256
    /* 合法范围 */
#elif BURST_SIZE > 256
    #error "BURST_SIZE too large"
#endif

#if 的表达式必须是预处理常量表达式：

  • 宏替换后是整型常量表达式。
  • defined(NAME) 判断宏是否定义。
  • 不能用 sizeof、不能调用函数（C99 起部分放宽）。
  • 未定义的宏在表达式中视为 0。
#if UNKNOWN_MACRO    /* UNKNOWN_MACRO 未定义，视为 0，条件为假 */
    ...
#endif

调试开关：HFT 线上关日志

/* debug.h */
#if HFT_DEBUG_LEVEL >= 2
    #define DBG_TRACE(fmt, ...)  log_trace(fmt, ##__VA_ARGS__)
#elif HFT_DEBUG_LEVEL >= 1
    #define DBG_TRACE(fmt, ...)  ((void)0)
#else
    #define DBG_TRACE(fmt, ...)  ((void)0)
#endif

/* 调用 */
DBG_TRACE("seq=%lu processing\n", seq);

关闭时零开销：((void)0) 展开为空操作，编译器完全删除调用。汇编中无 log_trace 调用，无参数准备，无分支判断。HFT 数据面要求生产环境日志零开销，条件编译是唯一手段。

编译期日志级别

# 测试环境

# 生产环境

构建系统通过 -D 注入日志级别，同一份源码编译出不同行为的二进制。生产环境关掉 TRACE/DEBUG 日志，保留 ERROR。

#elif defined(__aarch64__) && defined(__linux__)
    /* ARM64 Linux */
    #define HFT_ARCH_ARM64 1
    #define HFT_CACHE_LINE 64
    #define HFT_MEMBAR()    asm volatile("dmb ish" ::: "memory")  /* ARM 弱模型 */
#elif defined(__powerpc64__)
    /* PowerPC 64 */
    #define HFT_ARCH_PPC64 1
    #define HFT_CACHE_LINE 128
    #define HFT_MEMBAR()    asm volatile("lwsync" ::: "memory")
#else
    #error "unsupported platform"
#endif

条件编译隐藏平台差异：调用方用 HFT_MEMBAR()，不关心是 x86 空操作还是 ARM dmb ish。这是 DPDK rte_smp_wmb、内核 smp_wmb 的设计原理。

14.4

文件包含

本节讲什么

#include 把头文件文本嵌入当前翻译单元，是预处理器第一阶段的核心操作。这一节给出 #include 的展开流程、两种形式（<...> 系统头 vs "..." 本地头）的搜索规则、嵌套深度的工程控制，以及 HFT 工程里头文件依赖管理的实践。

认知锚点：#include 不是「导入模块」，而是纯文本拼接——预处理器把头文件内容原样插入到 #include 处，然后递归展开。理解这一点，才能理解 include guard 为什么必要、为什么循环包含会无限递归、为什么头文件互相 include 会让编译变慢。

要点

预处理展开流程

源文件 a.c:
  #include "b.h"
  #include <stdio.h>
  int main(void) { ... }

预处理步骤:
  1. 读取 a.c
  2. 遇到 #include "b.h"，找到 b.h，把内容插入 a.c 的 #include 位置
  3. 递归展开 b.h 里的 #include
  4. 遇到 #include <stdio.h>，从系统路径找到 stdio.h，插入
  5. 递归展开 stdio.h 里的 #include
  6. 所有 #include 处理完后，得到完整的翻译单元（.i 文件）
  7. 编译器接管，分析这个展开后的 .i 文件

#include 是递归的——头文件里的 #include 也会被展开，直到所有依赖都拼入。gcc -E a.c -o a.i 可以查看展开后的结果，是排查包含问题的第一手段。

两种形式

#include <stdio.h>          /* 系统头：从系统路径搜索 */
#include "my_header.h"      /* 本地头：先搜当前目录，再搜系统路径 */
#include "../common/util.h" /* 本地头：相对路径 */
#include "/opt/dpdk/include/rte_mbuf.h"  /* 绝对路径（不推荐） */
形式 搜索顺序 典型用途
<...> 系统标准路径 + -I 指定的路径 libc、内核 UAPI、第三方库
"..." 当前文件所在目录 → -I 路径 → 系统路径 工程私有头文件 
# 查看系统 include 路径
gcc -E -Wp,-v - < /dev/null

# 追加搜索路径
gcc -I/opt/dpdk/include -I./src/include ...

<...> 和 "..." 的搜索顺序由实现定义，但 POSIX 规定 "..." 必须先搜当前目录。某些编译器（MSVC）行为略有不同。

Include Guard（强制）

/* quote_proto.h */
#ifndef QUOTE_PROTO_H
#define QUOTE_PROTO_H

/* 头文件内容 */
struct quote_msg {
    uint32_t type;
    uint64_t seq;
    /* ... */
};

#endif /* QUOTE_PROTO_H */

include guard 防止重复包含：

  1. 第一次包含：QUOTE_PROTO_H 未定义，进入 #ifndef，定义 QUOTE_PROTO_H，处理内容。
  2. 第二次包含：QUOTE_PROTO_H 已定义，跳过整个 #ifndef 块。

没有 guard 会怎样：

/* 没有 guard 的 bad.h */
struct foo { int x; };

/* a.c */
#include "bad.h"
#include "bad.h"   /* struct foo 重复定义，编译错误 */

C 标准不允许同一翻译单元内重复定义结构体。include guard 是唯一可靠的防护手段。

Guard 命名规范

#ifndef MODULE_PATH_FILE_H
#define MODULE_PATH_FILE_H
/* ... */
#endif
  • 唯一性：guard 名必须全局唯一，否则不同头文件互相屏蔽。用「项目_模块_文件」前缀保证唯一。
  • 全大写：与宏命名一致。
  • 末尾注释：#endif /* QUOTE_PROTO_H */ 标注配对的 guard 名，可读性。

#pragma once

#pragma once

/* 头文件内容 */
struct foo { int x; };

#pragma once 是编译器扩展（GCC/Clang/MSVC 全部支持），告诉编译器「本文件只包含一次」。相比 include guard：

维度 include guard #pragma once
标准 C 标准 编译器扩展（主流全支持）
性能 预处理器仍要打开文件读 #ifndef 编译器基于文件路径跳过，更快
可靠性 guard 名冲突会出错 基于文件路径，无冲突
可移植性 完全可移植 主流编译器支持，部分嵌入式不支持
跨符号链接 同一文件不同路径会包含两次 基于路径，可能漏包含

DPDK/内核用 include guard（可移植性优先），现代项目用 #pragma once（简洁高效）。混用也可，但通常选其一。

嵌套深度控制

/* a.h -> b.h -> c.h -> d.h ... 深度递增 */

过度嵌套的 include 会导致：

  1. 编译变慢：每个 .c 文件展开后可能包含数万行头文件代码。
  2. 依赖复杂：改一个头文件触发大量重编译。
  3. 循环包含：A 包含 B，B 包含 A，无限递归（guard 阻断但导致内容不可见）。

前置声明减少依赖

/* bad: 包含完整定义 */
#include "order.h"
#include "quote.h"

struct order_engine {
    struct order *orders;     /* 只需要指针 */
    struct quote *quotes;     /* 只需要指针 */
};

/* good: 前置声明 */
struct order;    /* 前置声明，不需要完整定义 */
struct quote;

struct order_engine {
    struct order *orders;     /* 指针只需前置声明 */
    struct quote *quotes;
};

指针和引用只需前置声明——不需要知道结构体大小和布局。前置声明把依赖从「需要完整定义」降级为「只需名字」，大幅减少 include 传播。

PIMPL 模式

/* order_engine.h */
struct order_engine;   /* 不透明指针，实现细节隐藏 */

struct order_engine *order_engine_create(void);
void order_engine_destroy(struct order_engine *e);
int order_engine_submit(struct order_engine *e, const struct order *o);

/* order_engine.c */
#include "order_engine.h"
#include "order.h"
#include "quote.h"

struct order_engine {   /* 完整定义只在 .c 里 */
    struct order *orders;
    struct quote *quotes;
};

PIMPL（Pointer to IMPLementation）：头文件只暴露不透明指针 struct order_engine *，完整定义在 .c 文件里。用户代码只 #include "order_engine.h"，不需要 order.h/quote.h。依赖完全隔离，改内部结构不触发用户重编译。详见 ch10 结构体。

头文件的职责

/* good: 头文件只放声明 */
/* quote_proto.h */
#ifndef QUOTE_PROTO_H
#define QUOTE_PROTO_H

#include <stdint.h>     /* 依赖的标准头 */

struct quote_msg {      /* 结构体定义（需要完整定义） */
    uint32_t type;
    uint64_t seq;
};

/* 函数声明 */
int quote_parse(const uint8_t *buf, size_t len, struct quote_msg *out);

#endif

/* bad: 头文件放实现 */
/* bad.h */
static inline int helper(int x) {  /* static inline 可以，但大段实现不该在头文件 */
    /* 大段实现 */
}

头文件职责：

  • 类型定义：结构体、枚举、typedef。
  • 函数声明：原型，不含实现。
  • 宏定义：常量、标志位、内联小函数。
  • static inline 函数：小工具函数，可以放头文件。
  • extern 声明：全局变量声明（定义在 .c）。

头文件不放：函数实现（非 inline）、变量定义、static 变量定义。

HFT 项目头文件依赖复杂，前置声明 + PIMPL 是控制编译时间的标准手段。改一个内部结构不触发所有文件重编译。

头文件的 include 顺序

/* 推荐顺序 */
#include "对应本 .c 的头文件"   /* 1. 自身头文件，验证自包含 */
#include <标准库头>              /* 2. C 标准库 */
#include <系统头>                /* 3. POSIX/系统 */
#include <第三方库头>            /* 4. DPDK/内核 */
#include "工程内其他头"          /* 5. 工程内部 */

自身头文件第一：验证头文件自包含（不依赖 .c 里其他 include）。Google C++ Style 推荐此顺序，HFT 工程同样适用。

编译时间优化

# 并行编译

循环包含的陷阱

/* a.h */
#include "b.h"
struct a { struct b *bp; };

/* b.h */
#include "a.h"
struct b { struct a *ap; };

a.c 包含 a.h → 展开 b.h → b.h 又 include a.h → guard 阻断（A_H 已定义）→ b.h 继续，struct a 未定义 → struct b { struct a *ap; } 编译错误。

解决方案：前置声明替代 include。

/* a.h */
struct b;            /* 前置声明 */
struct a { struct b *bp; };

/* b.h */
struct a;            /* 前置声明 */
struct b { struct a *ap; };

指针只需前置声明，不需要完整定义，打破循环。

本节讲什么

#include <header.h> 用尖括号包含系统头文件和第三方库头。这一节给出 <...> 的搜索路径机制、头文件与库链接的关系、HFT 工程里 DPDK/内核 UAPI 头文件的实际部署形态。

认知锚点：<...> 不是「特殊的 include」，而是搜索路径不同的 include。理解搜索路径，才能理解为什么 #include <rte_mbuf.h> 需要 -I/opt/dpdk/include，才能排查「找不到头文件」的编译错误。

要点

<...> 的搜索路径

#include <stdio.h>          /* C 标准库 */
#include <stdint.h>         /* C99 标准整数类型 */
#include <pthread.h>        /* POSIX 线程 */
#include <linux/if_ether.h> /* 内核 UAPI */
#include <rte_mbuf.h>       /* DPDK 第三方库 */

搜索顺序（典型 Linux/GCC）：

  1. -I 指定的路径（命令行 -I/opt/dpdk/include）
  2. 系统标准路径：/usr/include、/usr/local/include、/usr/include/x86_64-linux-gnu/
  3. 编译器内置路径：/usr/lib/gcc/x86_64-linux-gnu/13/include/
# 查看系统 include 路径
gcc -E -Wp,-v - < /dev/null
# 输出类似：
# #include "..." search starts here:
# #include <...> search starts here:
#  /usr/include
#  /usr/local/include
#  /usr/lib/gcc/...

# 查找特定头文件位置
gcc -M foo.c | grep rte_mbuf   # 显示依赖路径
locate rte_mbuf.h
dpkg -L libdpdk-dev | grep rte_mbuf.h

<...> vs "..." 的本质区别

#include <stdio.h>      /* 系统头：不搜当前目录 */
#include "my_header.h"  /* 本地头：先搜当前目录 */
维度 <...> "..."
当前目录 不搜 先搜
-I 路径 搜 搜（当前目录之后）
系统路径 搜 搜（最后）
语义暗示 「这是外部库/系统头」 「这是工程内部头」
工程实践 libc、DPDK、第三方库 工程私有头文件

语义暗示是工程约定——<...> 告诉读者「这是外部依赖」，"..." 告诉读者「这是工程内部」。虽然技术上 <...> 也能搜当前目录（用 -I. 配合），但破坏约定。

系统标准库头文件

#include <stdio.h>      /* I/O：printf、fopen */
#include <stdlib.h>     /* 通用：malloc、exit、atoi */
#include <string.h>     /* 字符串：memcpy、strcpy */
#include <stdint.h>     /* C99 整数类型：uint32_t、int64_t */
#include <stdbool.h>    /* C99 bool 类型 */
#include <stddef.h>     /* size_t、ptrdiff_t、NULL */
#include <errno.h>      /* 错误码 */
#include <math.h>       /* 数学函数 */

这些头文件提供 C 标准库的声明，链接时自动连接 libc（-lc 通常默认）。位置在 /usr/include/，由 libc-dev 包提供。

POSIX 头文件

#include <pthread.h>    /* 线程：pthread_create */
#include <unistd.h>     /* UNIX 标准：read、write、fork */
#include <fcntl.h>      /* 文件控制：open、O_RDONLY */
#include <sys/socket.h> /* 套接字 */
#include <netinet/in.h> /* 网络地址 */
#include <arpa/inet.h>  /* inet_pton、htonl */
#include <sys/mman.h>   /* 内存映射：mmap */
#include <sys/epoll.h>  /* epoll */

POSIX 头文件在 /usr/include/ 和 /usr/include/x86_64-linux-gnu/，由 glibc-dev / libc-dev 提供。<sys/*.h> 是系统相关的 POSIX 头。

内核 UAPI 头文件

#include <linux/if_ether.h>   /* ETH_P_ALL、ethhdr */
#include <linux/ip.h>         /* iphdr */
#include <linux/tcp.h>        /* tcphdr */
#include <linux/udp.h>        /* udphdr */
#include <linux/if_packet.h>  /* sockaddr_ll */
#include <linux/version.h>    /* LINUX_VERSION_CODE */

内核 UAPI（Userspace API）头文件在 /usr/include/linux/，提供内核对用户态的稳定接口。HFT 报文解析直接用这些头，不需要手写协议结构。这些头由 linux-libc-dev 包提供，与内核版本对应。

/* 使用内核 UAPI 头解析报文 */
#include <linux/if_ether.h>
#include <linux/ip.h>

const uint8_t *pkt = rte_pktmbuf_mtod(mbuf, const uint8_t *);
const struct ethhdr *eth = (const struct ethhdr *)pkt;
const struct iphdr *ip = (const struct iphdr *)(pkt + sizeof(*eth));

if (eth->h_proto == htons(ETH_P_IP)) {
    /* 处理 IP 报文 */
}

第三方库头文件

#include <rte_mbuf.h>       /* DPDK mbuf */
#include <rte_ethdev.h>     /* DPDK 以太网设备 */
#include <rte_mempool.h>    /* DPDK 内存池 */
#include <jansson.h>        /* JSON 解析 */
#include <openssl/sha.h>    /* SHA 哈希 */

第三方库头文件通过 -I 指定搜索路径：

gcc -I/opt/dpdk/include -I/usr/local/include -c foo.c

或用 pkg-config 自动获取路径：

gcc $(pkg-config --cflags libdpdk) -c foo.c
gcc $(pkg-config --cflags jansson) -c foo.c

头文件 vs 库链接

/* 头文件：声明 */
#include <rte_mbuf.h>
struct rte_mbuf *m = rte_pktmbuf_alloc(pool);

头文件提供声明，库链接提供实现：

阶段 工具 作用
预处理 cpp 把 #include 展开，得到完整翻译单元
编译 gcc -c 翻译单元 → 目标文件（.o），只检查声明
链接 ld 把目标文件 + 库（.a/.so）合并成可执行文件 
# 编译阶段：只需要头文件
gcc -I/opt/dpdk/include -c foo.c -o foo.o

# 链接阶段：需要库文件
gcc foo.o -lrte_mbuf -lrte_eal -o foo
# 或用 pkg-config
gcc foo.o $(pkg-config --libs libdpdk) -o foo

「找不到头文件」是编译期错误，「找不到符号」是链接期错误。前者用 -I 解决，后者用 -L + -l 解决。

静态库 vs 动态库

# 静态库：libfoo.a，链接时把代码拷贝进可执行文件
gcc foo.o -lfoo -o foo     # 链接 libfoo.a

# 动态库：libfoo.so，运行时动态加载
gcc foo.o -lfoo -o foo     # 优先链接 .so，找不到则 .a
ldd foo                    # 查看运行时依赖的 .so
维度 静态库 .a 动态库 .so
链接方式 代码拷贝进可执行 运行时加载
可执行文件大小 大 小
启动速度 快 稍慢（需加载 .so）
升级 需重新链接 替换 .so 即可
HFT 偏好 静态库（避免运行时解析开销） 管理面可用

HFT 数据面倾向静态链接——避免动态库的 PLT/GOT 间接跳转开销，减少启动依赖。DPDK 通常静态链接到应用里。

内核 UAPI 直接使用

#include <linux/if_ether.h>
#include <linux/ip.h>
#include <linux/tcp.h>

{
    const uint8_t *data = rte_pktmbuf_mtod(m, const uint8_t *);

}

HFT 报文解析直接用内核 UAPI 头定义的结构，避免手写协议结构出错。ETH_HLEN、IPPROTO_TCP 等常量也来自这些头。

系统头版本的编译期检查

#include <linux/version.h>

#if LINUX_VERSION_CODE < KERNEL_VERSION(5, 4, 0)
    #error "Linux kernel >= 5.4 required"
#endif

#include <linux/if_ether.h>
/* 使用 5.4+ 才有的特性 */

跨内核版本部署时，编译期检查版本号，避免运行时崩溃。

DPDK 版本检查

#include <rte_version.h>

#if RTE_VERSION < RTE_VERSION_NUM(21, 11, 0, 0)
    #error "DPDK >= 21.11 required"
#endif

#include <rte_mbuf.h>
/* 使用 21.11+ 的 API */

DPDK 不同版本 API 有变化，编译期检查版本号保证兼容性。

头文件与 ABI 稳定性

/* 内核 UAPI 头文件承诺 ABI 稳定：用户态代码不会因内核升级而失效 */
#include <linux/if_ether.h>   /* 结构布局跨内核版本稳定 */

/* DPDK 头文件不承诺 ABI 稳定：版本升级可能破坏二进制兼容 */
#include <rte_mbuf.h>          /* 结构布局可能随版本变化 */

内核 UAPI 是 ABI 稳定的——结构布局、常量值跨版本不变，用户态代码不需重编译。DPDK 不承诺 ABI 稳定——升级 DPDK 版本通常需要重新编译应用。HFT 升级 DPDK 时必须全量重编译。

UB 边界

情形 后果 防护
找不到头文件 编译错误 检查 -I 路径、安装依赖包
头文件版本不匹配 编译通过但行为异常 编译期版本检查
头文件声明与库实现不一致 链接错误或运行时崩溃 头库版本对齐
系统头被本地头覆盖 非预期行为 不要在工程内建与系统同名的头
多版本库共存 链接到错误版本 显式 -I 和 -L 路径

头文件覆盖陷阱

/* 工程内建了 string.h（错误做法） */

/* foo.c */

14.4

.2 本地文件

本节讲什么

#include "local.h" 用双引号包含工程内部头文件。这一节给出 "..." 的搜索规则、include guard 与 #pragma once 的实战对比、前置声明与 PIMPL 减少依赖的工程实践，以及 HFT 工程里头文件组织规范。

认知锚点：本地头文件是工程内部的「契约」——它定义模块对外暴露的接口。头文件设计直接决定编译时间、依赖传播、API 稳定性。好的头文件设计让改动局部化，坏的设计让一次改动触发全量重编译。

要点

"..." 的搜索规则

#include "quote_proto.h"       /* 先搜当前目录 */
#include "../common/util.h"    /* 相对路径 */
#include "strategy/alpha.h"    /* 子目录 */

搜索顺序（POSIX）：

  1. 当前文件所在目录：#include "x.h" 的 .c 文件所在目录。
  2. -I 指定的路径：命令行 -I./src/include。
  3. 系统标准路径：/usr/include 等（最后才搜）。
# 追加搜索路径
gcc -I./src -I./include -I./common -c foo.c

# 当前目录优先意味着：
# 在 src/foo.c 里 #include "foo.h" 会先找 src/foo.h
# 而不是 ./foo.h

本地头文件的组织规范

my_hft_project/
├── include/              # 对外公开头文件
│   ├── hft_common.h
│   ├── hft_proto.h
│   └── hft_strategy.h
├── src/
│   ├── proto/
│   │   ├── proto.c       # 实现
│   │   └── proto_internal.h  # 模块内部头
│   ├── strategy/
│   │   ├── alpha.c
│   │   └── alpha.h       # 模块私有头
│   └── app/
│       └── main.c
└── Makefile
头文件类型 位置 用途
公开头 include/ 对外 API，其他模块可包含
模块私有头 src/module/ 模块内部，不对外
内部头 src/module/*_internal.h 实现细节，不对外

Include Guard 实战

/* quote_proto.h */
#ifndef HFT_QUOTE_PROTO_H
#define HFT_QUOTE_PROTO_H

#include <stdint.h>

/* 类型定义 */
struct quote_msg {
    uint32_t type;
    uint64_t seq;
    uint32_t price;
    uint32_t qty;
};

/* 函数声明 */
int quote_parse(const uint8_t *buf, size_t len, struct quote_msg *out);
size_t quote_serialize(const struct quote_msg *q, uint8_t *buf, size_t cap);

/* 宏定义 */
#define QUOTE_TYPE_HEARTBEAT  0
#define QUOTE_TYPE_TRADE      1
#define QUOTE_TYPE_DEPTH      2

#endif /* HFT_QUOTE_PROTO_H */

Guard 命名规范：

  • HFT_QUOTE_PROTO_H：项目前缀 HFT_ + 模块 QUOTE_ + 文件 PROTO_ + 后缀 _H。
  • 全大写，下划线分词，保证全局唯一。
  • #endif 后注释配对 guard 名，可读性。

#pragma once 实战

/* strategy_alpha.h */
#pragma once

#include "hft_strategy.h"

struct strategy_alpha {
    struct strategy base;   /* 继承基类 */
    uint32_t threshold;
};

int strategy_alpha_init(struct strategy_alpha *s);

#pragma once 更简洁，主流编译器全支持。现代项目推荐 #pragma once，但 DPDK/内核等可移植性要求极高的项目仍用 include guard。

Guard vs #pragma once 的可靠性边界

/* 场景：同一头文件通过不同路径包含 */
/* /home/user/proj/include/foo.h */
/* /home/user/proj/./include/foo.h */  /* 符号链接或软链接 */

/* include guard：基于宏名，两次都进入 ifndef（除非宏已定义） */
/* #pragma once：基于文件路径，可能识别为两个不同文件，包含两次 */

#pragma once 基于文件路径（inode），如果同一文件通过符号链接或不同路径被包含，可能误判为两个文件。include guard 基于宏名，不受路径影响。但这种场景罕见，现代编译器对 #pragma once 的处理已很健壮。

前置声明减少依赖

/* order_engine.h - bad: 包含完整定义 */
#include "order.h"       /* 传播 order.h 的所有依赖 */
#include "quote.h"       /* 传播 quote.h 的所有依赖 */

struct order_engine {
    struct order *orders;    /* 只需要指针 */
    struct quote *quotes;    /* 只需要指针 */
};

/* order_engine.h - good: 前置声明 */
struct order;            /* 前置声明，不需要完整定义 */
struct quote;

struct order_engine {
    struct order *orders;    /* 指针只需前置声明 */
    struct quote *quotes;
};

前置声明的规则：

使用形式 是否需要完整定义
struct T *（指针） 否，前置声明即可
struct T &（C++ 引用） 否（C 无引用）
struct T（值传递） 是，需要完整定义
sizeof(T) 是
访问成员 t->field 是

前置声明把依赖从「需要完整定义」降级为「只需名字」，大幅减少 include 传播。

PIMPL 模式

/* order_engine.h - 对外头文件 */
#pragma once

struct order_engine;   /* 不透明指针 */

struct order_engine *order_engine_create(void);
void order_engine_destroy(struct order_engine *e);
int order_engine_submit(struct order_engine *e, const struct order *o);

/* order_engine.c - 实现 */
#include "order_engine.h"
#include "order.h"       /* 完整定义只在 .c 里 */
#include "quote.h"

struct order_engine {   /* 完整定义 */
    struct order *orders;
    struct quote *quotes;
    size_t count;
};

struct order_engine *order_engine_create(void) {
    return calloc(1, sizeof(struct order_engine));
}

void order_engine_destroy(struct order_engine *e) {
    if (e) {
        free(e->orders);
        free(e->quotes);
        free(e);
    }
}

PIMPL 的好处：

  1. 依赖隔离：用户代码不需要 order.h/quote.h。
  2. ABI 稳定：改 struct order_engine 内部布局不破坏 ABI，用户不需重编译。
  3. 编译加速：改 order.h 只触发 order_engine.c 重编译，不波及用户。
  4. 封装：实现细节完全隐藏。

代价：① 每次访问需间接（e->orders 要先取 e 再取 orders，多一次访存）；② 必须堆分配（栈分配需要完整定义）。HFT 数据面在 ABI 稳定性要求高的管理面用 PIMPL，数据面热路径不用。

头文件的依赖最小化

/* strategy_alpha.h */
#pragma once

#include "hft_strategy.h"   /* 必需：基类完整定义 */
#include <stdint.h>          /* 必需：uint32_t */

/* 不必要的 include 应删除 */
/* #include "order.h" */     /* 若只需 struct order *，改用前置声明 */
/* #include <stdio.h> */     /* 若头文件未用 FILE*，删除 */

struct order;                /* 前置声明 */
struct strategy_alpha {
    struct strategy base;
    struct order *pending;   /* 指针，前置声明足够 */
    uint32_t threshold;
};

头文件依赖最小化原则：

  1. 头文件只 include 必需的头。
  2. 能用前置声明的，不用 include。
  3. 实现需要的头放 .c 里。
  4. 定期检查头文件依赖，删除多余 include。

工具：gcc -M foo.c 输出依赖关系，include-what-you-use 工具自动检测多余 include。

头文件的自包含

/* quote_proto.h 应该自包含：包含它不需要其他前置条件 */
#pragma once

#include <stdint.h>   /* 自己需要的依赖自己包含 */
#include <stddef.h>

struct quote_msg { ... };

自包含原则：头文件包含它自己用到的所有依赖，用户只需 #include "quote_proto.h" 就能使用，不需要先包含其他头。验证方法：.c 文件第一个 include 自身头文件，若编译通过则自包含。

HFT 关联

DPDK 头文件结构

/* rte_mbuf.h 是自包含的 */
#ifndef _RTE_MBUF_H_
#define _RTE_MBUF_H_

#include <stdint.h>
#include <rte_common.h>
#include <rte_mempool.h>
#include <rte_prefetch.h>

/* 完整定义 + 内联函数 + 宏 */

#endif

DPDK 头文件全部自包含，用户只需 #include <rte_mbuf.h>，不需要手动包含依赖。这是工业级库的标准。

内核头文件的分层

include/linux/        # 内核内部 + 对用户态的 UAPI
include/uapi/linux/   # 仅对用户态的稳定接口
include/uapi/asm-generic/  # 架构无关的 UAPI
arch/x86/include/uapi/     # x86 特定的 UAPI

内核把「内核内部头」和「对用户态的 UAPI」分离，UAPI 承诺 ABI 稳定。这种分层是大型项目的典范。

HFT 项目的头文件策略

/* 公开 API：放 include/，用 PIMPL 隐藏实现 */
/* include/hft_orderbook.h */
#pragma once

struct orderbook;

struct orderbook *orderbook_create(size_t levels);
void orderbook_destroy(struct orderbook *ob);
int orderbook_add(struct orderbook *ob, const struct order *o);

/* 内部实现：放 src/orderbook/，不对外 */
/* src/orderbook/orderbook_internal.h */
#pragma once

#include "hft_orderbook.h"
#include "hft_proto.h"

struct orderbook {   /* 完整定义 */
    struct level *levels;
    size_t num_levels;
    /* ... */
};

HFT 项目把公开 API（不透明指针 + 函数声明）与内部实现（完整结构定义）分离，改内部实现不触发用户重编译，ABI 稳定。

static 变量在头文件的陷阱

/* bad.h */
static int counter = 0;   /* 每个 .c 包含后各有一份 */

/* a.c */
#include "bad.h"
counter++;   /* a.c 的 counter = 1 */

/* b.c */
#include "bad.h"
counter++;   /* b.c 的 counter = 1，与 a.c 无关 */

static 变量在头文件里会导致每个包含它的 .c 文件各有一份独立副本，不是共享的全局变量。正确做法：

/* good.h */
extern int counter;   /* 声明 */

/* good.c */
#include "good.h"
int counter = 0;      /* 定义 */

extern 声明在头文件，定义在 .c 文件，所有 .c 共享同一份。

static inline 函数可以放头文件

/* util.h */
static inline int min_int(int a, int b) {
    return a < b ? a : b;
}

static inline 函数与 static 变量不同——inline 提示编译器内联展开，不生成函数地址，每个 .c 包含后只是把定义可见，不会有多份副本（除非没被内联且取了地址）。static inline 是头文件放小工具函数的标准做法。

14.5

其他指令

本节讲什么

除了 #define、#include、#if 这些核心指令，预处理器还有一组补充指令：#error（编译期阻断）、#pragma（编译器扩展）、#line（重写行号）、#undef（取消宏）。这一节给出它们的用法和 HFT 工程里的实际场景。

认知锚点：这些指令看似边缘，却是 C 工程的「安全阀」与「调节器」。#error 让非法配置在编译期失败，优于运行期崩溃；#pragma 控制编译器行为（对齐、警告、优化）；#line 是代码生成器的工具；#undef 是宏作用域控制。理解它们，才能写出健壮的工程代码。

要点

#error — 编译期阻断

#if BURST_SIZE > 256
    #error "BURST_SIZE cannot exceed 256"
#endif

#if !defined(__x86_64__) && !defined(__aarch64__)
    #error "HFT dataplane requires x86-64 or ARM64"
#endif

#if RTE_VERSION < RTE_VERSION_NUM(21, 11, 0, 0)
    #error "DPDK >= 21.11 required, please upgrade"
#endif

#error 在预处理阶段打印消息并失败，编译立即停止。用途：

  1. 配置校验：非法配置编译期失败，优于运行期崩溃。
  2. 平台检测：不支持的架构直接拒绝编译。
  3. 版本检查：依赖库版本不够时拒绝编译。
  4. 特性冲突：互斥的特性同时开启时报错。

#error 是 HFT 工程的第一道防线——把运行期可能出现的错误前移到编译期，节省调试时间。

#warning — 编译期警告

#if defined(__GNUC__) && __GNUC__ < 8
    #warning "GCC < 8 may generate suboptimal code, consider upgrading"
#endif

#warning 是 GCC 扩展（非标准），打印警告但继续编译。用于提示性信息，不阻断编译。C23 引入标准 #warning。

#pragma — 编译器扩展

#pragma 是预处理器指令，但具体语义由编译器定义。主流用途：

用法 作用 示例
#pragma once 头文件保护 见 14.4.2
#pragma pack(push, n) 结构体对齐控制 #pragma pack(push, 1)
#pragma pack(pop) 恢复对齐
#pragma GCC diagnostic 局部警告控制 #pragma GCC diagnostic ignored "-Wxxx"
#pragma GCC push_options 编译选项压栈
#pragma GCC pop_options 恢复编译选项
#pragma GCC optimize 函数级优化控制 #pragma GCC optimize("O3")
#pragma GCC target 函数级目标平台 #pragma GCC target("avx2")
#pragma message 编译期打印消息 #pragma message("building for x86")

#pragma pack — 结构体对齐

#pragma pack(push, 1)   /* 保存当前对齐，设为 1 字节 */
struct eth_hdr {
    uint8_t  dst[6];
    uint8_t  src[6];
    uint16_t type;
};
#pragma pack(pop)        /* 恢复对齐 */

/* sizeof(struct eth_hdr) = 14，无 padding */

#pragma pack(1) 让结构体按 1 字节对齐，无 padding。用于网络协议头（必须按线上格式精确布局）。push/pop 配对保证对齐设置不泄漏到其他结构。

详见 ch10 packed 结构体的权衡。

#pragma GCC diagnostic — 局部警告控制

#pragma GCC diagnostic push
#pragma GCC diagnostic ignored "-Wunused-variable"
    int unused = 42;   /* 警告被抑制 */
#pragma GCC diagnostic pop

局部抑制警告：只在必要位置关闭特定警告，用完立即恢复。比全局 -Wno-xxx 更精确。

#pragma GCC diagnostic push
#pragma GCC diagnostic error "-Wformat"   /* 把格式警告升级为错误 */
    printf("%d\n", "hello");   /* 编译错误 */
#pragma GCC diagnostic pop

把特定警告升级为错误，强制代码质量。

#pragma GCC optimize — 函数级优化

#pragma GCC optimize("O3", "unroll-loops", "fast-math")
static void hot_path_function(struct rte_mbuf *m) {
    /* 强制 O3 + 循环展开 + 快速数学 */
}

函数级优化控制，让热点函数用更激进的优化，其他函数保持默认。HFT 数据面优化常用。

#pragma GCC target — 函数级指令集

#pragma GCC target("avx2,fma")
void vector_add(float *a, float *b, float *c, int n) {
    /* 此函数可用 AVX2 + FMA 指令 */
    for (int i = 0; i < n; i++)
        c[i] = a[i] + b[i];
}

让特定函数使用 AVX2/FMA 等指令集，其他函数不依赖这些指令。用于运行时 CPU 不一定支持新指令集的场景。

#line — 重写行号与文件名

#line 100 "generated_code.c"
/* 此后的 __LINE__ 从 100 开始，__FILE__ 是 "generated_code.c" */

#line 重写 __LINE__ 和 __FILE__，主要用于代码生成器——让生成的代码报错时指向源模板，而非生成结果。

/* 代码生成器示例 */
/* template: 行 50 是用户写的 */
/* generated: 行 200 是生成的 */
#line 50 "template.dt"
int generated_func(void) { return 0; }
/* 报错时显示 template.dt:50，而非 generated.c:200 */

HFT 工程极少用 #line，除非有代码生成器（如协议解析器生成器）。日常代码不要用——会干扰调试。

#undef — 取消宏定义

#define TEMP_DEBUG 1

/* ... 使用 TEMP_DEBUG ... */

#undef TEMP_DEBUG   /* 取消定义，后续不可见 */

#ifdef TEMP_DEBUG
    /* 此块不编译 */
#endif

#undef 的用途：

  1. 局部宏：只在某段代码用的宏，用完 #undef 避免泄漏。
  2. X-Macro 重定义：同一份 .def 文件用不同宏展开，每次包含后 #undef（见 14.2.3）。
  3. 重定义前取消：标准要求重定义前先 #undef（虽然允许相同重定义）。
  4. 屏蔽系统宏：某些库定义的宏与工程冲突，#undef 后重定义。
/* X-Macro 的 #undef 模式 */
#define DEFINE_EVT(name, val, str)  EVT_##name = val,
enum events {
    #include "events.def"
    EVT_NUM
};
#undef DEFINE_EVT

#define DEFINE_EVT(name, val, str)  [val] = str,
static const char *const evt_str[] = {
    #include "events.def"
};
#undef DEFINE_EVT

#、##、#pragma 的预处理顺序

/* 预处理阶段顺序：
   1. 三字符替换、行续接
   2. token 化、注释替换
   3. #include 嵌入
   4. #define 宏展开、# 操作符、## 操作符
   5. #if/#ifdef 条件裁剪
   6. #error 触发（若条件满足）
   7. #pragma 处理（编译器阶段）
   8. #line 重写行号
*/

#pragma 实际上在编译器阶段处理（不是预处理器），但语法上是预处理指令。这意味着 #pragma 受 #if 条件控制：

#if defined(__GNUC__)
    #pragma GCC optimize("O3")
#endif

_Pragma — #pragma 的宏形式

/* C99 _Pragma 操作符，等价 #pragma */
_Pragma("pack(push, 1)")
struct packed_hdr { ... };
_Pragma("pack(pop)")

/* 可在宏内使用 */
#define PACKED_STRUCT(name, body) \
    _Pragma("pack(push, 1)") \
    struct name body; \
    _Pragma("pack(pop)")

PACKED_STRUCT(eth_hdr, {
    uint8_t  dst[6];
    uint8_t  src[6];
    uint16_t type;
});

_Pragma 是 #pragma 的字符串形式，可在宏内使用。#pragma 不能在宏内（# 是预处理指令，宏体里只能用 # 字符串化）。C99 标准化。

#pragma pack 的跨平台差异

/* GCC/Clang */
#pragma pack(push, 1)
struct foo { ... };
#pragma pack(pop)

/* MSVC */
#pragma pack(push, 1)
struct foo { ... };
#pragma pack(pop)
/* 语法相同，但某些选项不同 */

/* C11 标准方案（推荐） */
#include <stdalign.h>
struct __attribute__((packed)) foo { ... };   /* GCC 专用 */

#pragma pack 是事实标准但非 C 标准。跨平台用 __attribute__((packed))（GCC/Clang）和 #pragma pack（MSVC）条件编译，或用 C11 <stdalign.h> 的 alignas（但对齐到 1 字节需要 alignas(1)，标准支持有限）。

#pragma pack 忘 pop 的后果

#pragma pack(push, 1)
struct packed_hdr { ... };
/* 忘记 #pragma pack(pop) */

struct normal_struct {
    uint32_t a;   /* 本应 4 对齐，但被 pack(1) 影响，1 对齐 */
    uint64_t b;   /* 1 对齐，性能下降，部分平台未对齐访问 UB */
};

#pragma pack 的影响持续到下一个 #pragma pack 或文件末。push/pop 必须严格配对，否则后续结构意外被压缩。

题目 1： 以下宏有什么问题？如何修复？

#define MAX(a, b) a > b ? a : b
int x = MAX(1, 2) + 3;
参考答案

展开为 1 > 2 ? 1 : 2 + 3，由于运算符优先级，实际解析为 1 > 2 ? 1 : (2 + 3) = 5（而不是预期的 max(1,2)+3 = 5）。碰巧结果相同，但如果 MAX(2, 1) + 3 展开为 2 > 1 ? 2 : 1 + 3 = 2（错误，应为 5）。修复：#define MAX(a, b) ((a) > (b) ? (a) : (b))。
