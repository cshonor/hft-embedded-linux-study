18.2

C 和汇编语言的接口

本节讲什么

C 与汇编的边界：System V x86-64 ABI 调用约定、内联汇编 __asm__、静态链接 vs 动态链接、符号解析。这是 C 程序员理解"代码到底怎么跑"的关键一节——HFT 的极致优化、内核互调、DPDK PMD 都需要懂 ABI。

要点

System V x86-64 ABI 调用约定

项 规则
整数/指针参数 rdi, rsi, rdx, rcx, r8, r9（顺序）
浮点参数 xmm0-xmm7
多余参数 从右向左压栈
返回值（整数/指针） rax（及 rdx 用于 128 位）
返回值（浮点） xmm0
被调保存寄存器 rbx, rbp, r12, r13, r14, r15
调用者保存 rax, rcx, rdx, rsi, rdi, r8-r11, xmm0-xmm15
栈对齐 call 前 rsp 16 字节对齐
红区 叶子函数可用 rsp-128 到 rsp-8

这是 Linux/macOS 通用约定。Windows x64 ABI 不同（前 4 参数用 rcx/rdx/r8/r9，shadow space 32 字节）。

ABI 参数传递实例

int func(int a, int b, int c, int d, int e, int f, int g);
/*        rdi  rsi  rdx  rcx  r8   r9   栈 */

编译后：

func:
    mov    DWORD PTR [rsp+8],  ecx   ; g 在栈上 (调用者压的)
    ; rdi=a, rsi=b, rdx=c, rcx=d, r8=e, r9=f
    ...
    ret

内联汇编

static inline uint64_t rdtsc(void) {
    unsigned hi, lo;
    __asm__ __volatile__("rdtsc" : "=a"(lo), "=d"(hi));
    return ((uint64_t)hi << 32) | lo;
}

__asm__ 基本格式：

__asm__ (
    "汇编指令"
    : 输出操作数     (C 变量绑定)
    : 输入操作数     (C 变量绑定)
    : clobber 列表   (被修改的寄存器)
);
  • "=a"：绑定到 rax，= 表示输出
  • "a"：输入绑定到 rax
  • "d"：绑定到 rdx
  • __volatile__：禁止编译器重排或删除

内联汇编典型用途

1. 特殊指令

/* CPU 暂停, 节能 */
static inline void cpu_relax(void) {
    __asm__ __volatile__("pause" ::: "memory");
}

/* 内存屏障 */
static inline void mfence(void) {
    __asm__ __volatile__("mfence" ::: "memory");
}

/* 读取 TSC */
static inline uint64_t rdtscp(uint32_t *cpu) {
    uint32_t a, d, c;
    __asm__ __volatile__("rdtscp" : "=a"(a), "=d"(d), "=c"(c));
    if (cpu) *cpu = c;
    return ((uint64_t)d << 32) | a;
}

2. 原子操作

static inline int atomic_cas(int *p, int old, int new) {
    int result;
    __asm__ __volatile__(
        "lock cmpxchg %3, %1"
        : "=a"(result), "+m"(*p)
        : "a"(old), "r"(new)
        : "memory");
    return result == old;   /* 成功返回 1 */
}

但实际工程用 __atomic_* 内建函数或 <stdatomic.h>（C11），不手写汇编。

3. 系统调用

/* 直接 syscall, 绕过 glibc */
static inline long sys_write(int fd, const void *buf, size_t n) {
    long ret;
    __asm__ __volatile__(
        "syscall"
        : "=a"(ret)
        : "a"(1),           /* SYS_write = 1 */
          "D"(fd),          /* rdi */
          "S"(buf),         /* rsi */
          "d"(n)            /* rdx */
        : "rcx", "r11", "memory");
    return ret;
}

绕过 glibc 减少开销，HFT 极致场景用。但可移植性差，通常不值得。

静态链接 vs 动态链接

维度 静态链接 -static 动态链接（默认）
可执行文件 大（库打进 ELF） 小
启动速度 快（无 ld.so 解析） 慢（解析 .so）
内存占用 每进程一份库 共享一份物理页
升级 需重新编译 换 .so 即可
依赖 无 .so 依赖 需 .so 在系统
HFT 数据面 首选 次选 
gcc -static -o app app.c         # 静态
gcc -o app app.c                  # 动态（默认）
ldd ./app                         # 查看动态依赖

动态链接的符号解析

/* dlopen 运行时加载 .so */
#include <dlfcn.h>
void *handle = dlopen("./plugin.so", RTLD_NOW);
if (!handle) { fprintf(stderr, "%s\n", dlerror()); exit(1); }

typedef int (*func_t)(int);
func_t func = (func_t)dlsym(handle, "plugin_func");
if (!func) { fprintf(stderr, "%s\n", dlerror()); exit(1); }

int result = func(42);

dlclose(handle);
  • dlopen：加载 .so 到进程地址空间
  • dlsym：按符号名找函数/变量地址
  • dlclose：卸载（引用计数为 0 时）

HFT 策略热加载用此机制——运行时加载新策略 .so，dlsym 获取函数指针切换。

符号版本化

/* .so 版本化符号 */
__asm__(".symver old_func_v1,func@v1");
__asm__(".symver new_func_v2,func@@v2");   /* @@ 默认版本 */

int old_func_v1(int x) { return x; }
int new_func_v2(int x) { return x * 2; }

glibc 用版本化符号实现向后兼容——新版本库仍能链接旧二进制。__symver 指令控制符号版本。

链接错误

undefined reference（链接时）

/* main.c */
int main() { return undefined_func(); }

$ gcc main.c
/tmp/cc...: undefined reference to `undefined_func'

链接器找不到符号定义。原因：

  • 忘了 #include 对应头
  • 忘了链接库 -l<lib>
  • 函数名拼写错误（C++ 还有 name mangling）

undefined symbol（运行时）

$ ./app
./app: symbol lookup error: ./app: undefined symbol: foo

动态链接程序运行时找不到符号。原因：

  • .so 版本不匹配
  • 编译时 .so 有符号，运行时 .so 被替换
  • LD_LIBRARY_PATH 指向旧库

nm / objdump 观察符号

nm ./app                    # 符号表
nm -D ./app                 # 动态符号
objdump -T ./libplugin.so   # 动态符号表
objdump -d ./app | head     # 反汇编
readelf -s ./app            # 符号表
  • T：代码段定义的函数（外部链接）
  • t：代码段定义的函数（static，内部链接）
  • D：数据段定义的变量
  • U：未定义符号（需链接时找）
  • W：弱符号

弱符号 weak

__attribute__((weak)) void optional_hook(void) { /* 默认空实现 */ }

/* 其他 .c 可覆盖 */
void optional_hook(void) { /* 实际实现 */ }

弱符号允许"可选覆盖"——若其他地方定义了同名强符号，用强符号；否则用弱符号。库设计插件钩子用此机制。

C 与汇编互调的注意事项

  • ABI 一致：C 和汇编必须遵守同一调用约定（System V x86-64）
  • 寄存器保存：汇编函数若用被调保存寄存器，必须 prologue 保存、epilogue 恢复
  • 栈对齐：汇编函数入口 rsp 必须是 16k - 8（call 后）
  • DF 标志：调用方需保证方向标志位 DF=0（字符串指令方向），C ABI 要求
  • 名字修饰：C 函数名无修饰，C++ 有 name mangling，互调用需 extern "C"

HFT 关联

  • ABI 一致：DPDK PMD、内核模块、C 代码互调必须遵守同一 ABI
  • 静态链接：HFT 数据面用 -static 避免 ld.so 动态解析，启动快且确定
  • dlopen 策略热加载：运行时加载新策略 .so，dlsym 获取函数指针
  • 内联汇编：rdtsc/pause/mfence 等特殊指令用内联汇编封装
  • 绕过 glibc：极致场景直接 syscall，但可移植性差，通常不值得
  • 符号可见性：-fvisibility=hidden 减小 .so，加快动态链接
  • 弱符号插件钩子：HFT 框架用弱符号定义默认空钩子，策略模块覆盖
  • PGO + LTO：-flto 跨文件内联，PGO 引导热路径优化

代码自测

题目 1： 以下内联汇编实现了什么功能？

static inline int atomic_add(int *ptr, int val) {
    int result;
    __asm__ __volatile__(
        "lock xadd %0, %1"
        : "+r"(val), "+m"(*ptr)
        : : "memory");
    return val;
}
参考答案

实现了原子加法。lock xadd 是 x86 的原子交换加法指令——它原子地执行 ptr += val 并返回 ptr 的旧值。lock 前缀保证多核下的原子性。volatile 防止编译器优化掉这条指令。"memory" clobber 告诉编译器内存可能被修改。这是无锁编程的基础原语。

18.3

运行时效率

本节讲什么

HFT 性能收官：堆的底层（brk/mmap）、cache line 与伪共享、hugepage 与 TLB、信号与崩溃诊断、调试工具（gdb/valgrind/perf）、用户态与内核的内存对照。这是全书的工程收尾——把前面所有章节的知识点串成一套可落地的 HFT 性能与稳定性方法论。

要点

堆的底层：brk 与 mmap

malloc 不是系统调用，是 glibc 的用户态封装。底层两种机制：

机制 用途 特点
brk/sbrk 小块连续分配 调整堆顶，连续扩展
mmap(MAP_ANONYMOUS) 大块、独立映射 独立映射，可单独 munmap

glibc malloc 策略：

  • 小块（通常 < 128KB）：用 brk 扩展堆顶，分配在连续堆区
  • 大块（>= 128KB）：用 mmap 独立映射，free 时 munmap 归还内核
  • free 后内存不立即归还内核，留在 malloc 自由链表供下次复用

DPDK 的内存策略：rte_malloc / hugepage

/* DPDK 用 mmap 映射 hugepage, 完全绕过 brk */
void *buf = rte_malloc("my_buf", 4096, 64);   /* 64 字节对齐 */
rte_free(buf);
  • rte_malloc 底层是 hugepage mmap，物理连续
  • 预分配所有 hugepage 启动时映射，运行时不再 mmap
  • 每 lcore 独立 free_list，无锁分配
  • 避免 brk 碎片和 glibc malloc 全局锁

Cache Line 与伪共享

现代 CPU cache line 通常 64 字节。多线程改同 cache line 不同变量 → 互相使对方 cache line 失效 → 性能崩塌（伪共享）：

/* 伪共享: 两个计数器在同一 cache line */
struct {
    int counter_a;   /* 线程 A 改 */
    int counter_b;   /* 线程 B 改 */
} stats;             /* 两个 int 在同一 64 字节 cache line */

线程 A 改 counter_a 使整个 cache line 失效，线程 B 的 counter_b 也被迫重新加载。反之亦然。两个本应独立的变量互相干扰。

伪共享的解决

1. padding 强制不同 cache line

struct Stats {
    int counter_a;
    char pad_a[60];   /* 填充到 64 字节 */
    int counter_b;
    char pad_b[60];
};

2. alignas / attribute((aligned))

struct AlignedCounter {
    int value;
} __attribute__((aligned(64)));

struct AlignedCounter counters[NTHREADS];   /* 每个元素独占 cache line */

3. per-thread 独立变量

/* 每线程独立计数器, 避免 cache line 争用 */
__thread int my_counter;   /* 线程局部存储 */
my_counter++;
/* 汇总时才读所有线程的 my_counter */

DPDK per-lcore 变量用此模式——RTE_PER_LCORE 宏定义每核独立变量。

Hugepage 与 TLB

TLB（Translation Lookaside Buffer）缓存虚拟地址→物理地址映射。TLB miss 极慢（数百周期）。

页大小 1GB 内存需要 TLB 条目数
4KB 262144 页 远超 TLB 容量
2MB 512 页 适配 L1 TLB
1GB 1 页 极少 TLB 条目

hugepage 大幅减少 TLB miss，HFT 数据面强制使用：

# 预留 hugepage
echo 1024 > /proc/sys/vm/nr_hugepages   # 1024 个 2MB 页 = 2GB

# 挂载 hugetlbfs
mkdir /mnt/huge
mount -t hugetlbfs nodev /mnt/huge

# DPDK 启动用 hugepage
./dpdk-app -c 0xf -n 4 -- -huge

NUMA 与内存亲和

多路 CPU 系统，每 CPU 有本地内存。跨 NUMA 访问内存慢 2-3x：

/* NUMA 感知分配 */
#include <numa.h>
void *buf = numa_alloc_onnode(size, 0);   /* 分配在 NUMA 节点 0 */
numa_free(buf, size);

DPDK rte_malloc 接受 socket_id 参数，分配在指定 NUMA 节点。HFT 数据线程绑核 + 内存本地分配，避免跨 NUMA。

信号与崩溃诊断

信号 常见原因 诊断
SIGSEGV 空指针、栈/堆越界、写 .rodata gdb + core dump
SIGABRT double free、abort、assert ASan + gdb
SIGBUS 非对齐访问（ARM）、mmap 文件被 truncate 对齐检查
SIGFPE 整数除零、INT_MIN 取负 代码审查
SIGKILL OOM killer dmesg + /var/log

core dump 配置

ulimit -c unlimited          # 当前 shell
echo "* soft core unlimited" >> /etc/security/limits.conf   # 永久
echo "/var/core/core.%e.%p.%t" > /proc/sys/kernel/core_pattern

崩溃后 gdb ./app /var/core/core.app.12345.1234567890 分析：

(gdb) bt              # 栈回溯
(gdb) bt full         # 栈回溯 + 局部变量
(gdb) info registers  # 寄存器
(gdb) x/10i $rip      # 崩溃点附近反汇编

调试工具

gdb

  • 断点、单步、观察点
  • backtrace 栈回溯
  • info locals/info args 局部变量
  • print expr 求值
  • watch var 变量变化断点

valgrind

valgrind --leak-check=full --show-leak-kinds=all ./app
  • 拦截 malloc/free，查泄漏、越界、悬垂
  • 慢 10-50x，不适合生产
  • HFT 开发期用，上线前清零

AddressSanitizer (ASan)

gcc -fsanitize=address -g -o app app.c
./app
  • 编译期插桩，查越界、use-after-free、double free
  • 慢 2x，内存 2-3x
  • 比 valgrind 快，适合开发期
  • HFT CI 必跑 ASan

perf

perf record -g ./app          # 采样
perf report                   # 分析
perf stat ./app               # 计数器
perf top                      # 实时
  • CPU 热点分析
  • cache miss、TLB miss、分支预测失败
  • HFT 性能剖析主力

strace / ltrace

strace -c ./app               # 统计 syscall
ltrace -c ./app               # 统计库调用
  • 查 syscall 频次、参数、耗时
  • HFT 数据面应零 syscall，strace 验证

用户态与内核对照

用户态 内核 用途
brk 堆 kmalloc/vmalloc 小块分配
mmap vmalloc/__get_free_pages 大块映射
hugepage hugetlb 大页
malloc/free kfree 释放
pthread_mutex spinlock/mutex 同步
pthread_create kthread_create 线程
stdio FILE* file/inode 文件抽象

HFT 数据面绕过所有用户态抽象，直接操作硬件（PMD + DMA + hugepage），类似内核性能模型。

性能优化的层次

层次 优化 收益
算法 O(n²) → O(n log n) 100x
数据结构 链表 → 数组 10x
cache 伪共享修复 5x
内存 hugepage 2x
编译 -O3 + PGO + LTO 1.5x
微优化 inline + restrict 1.2x

从上往下收益递减。先优化算法和数据结构，再 cache 和内存，最后才是编译选项和微优化。

HFT 关联

  • rte_malloc + hugepage：DPDK 绕过 brk/malloc，物理连续大页，无锁 per-lcore
  • 伪共享修复：per-lcore 变量、cache line 对齐，DPDK RTE_PER_LCORE
  • NUMA 亲和：数据线程绑核 + 内存本地分配，避免跨 NUMA
  • core dump + gdb：崩溃诊断标配，ulimit -c unlimited + core_pattern
  • ASan CI：开发期必跑，查越界/UAF/double free
  • perf 剖析：CPU 热点、cache miss、TLB miss 分析
  • strace 零 syscall 验证：数据面应零 syscall，strace 统计验证

  • 静态链接：数据面 -static 避免 ld.so 解析，启动确定
  • 绕过用户态抽象：PMD + DMA + hugepage，类似内核性能模型

代码自测

题目 1： 以下两种实现在性能上有什么差异？为什么？

// A: 普通调用
for (int i = 0; i < n; i++)
    sum += arr[i];
// B: 循环展开
for (int i = 0; i < n; i += 4) {
    sum += arr[i];
    sum += arr[i+1];
    sum += arr[i+2];
    sum += arr[i+3];
}
参考答案

B 通过循环展开减少分支预测失败和循环开销。每次迭代处理 4 个元素，循环次数减少为 1/4。但现代编译器（GCC -O2）会自动做循环展开，手写展开通常没有额外收益且降低可读性。过度展开还可能损害指令缓存。注意 B 需要处理 n 不是 4 的倍数的情况。

18.4

ELF 文件回顾：sections vs segments

ELF 是什么

ELF（Executable and Linkable Format）是 Linux 上的标准二进制格式。编译器输出的 .o、链接器输出的可执行文件、共享库 .so、内核模块 .ko、内核镜像 vmlinux 都是 ELF 文件。

两种视角：Linking View vs Execution View

ELF 文件有两组头部，对应两种视角：

视角 头部 看到什么 谁用
Linking View（链接视角） Section Header Table（节头表） .text .data .bss .symtab .rel.text 等 编译器、链接器、nm、readelf -S
Execution View（执行视角） Program Header Table（程序头表） LOAD 段、INTERP、DYNAMIC 等 内核加载器、动态链接器 
  Linking View                Execution View
  ┌──────────┐                ┌──────────┐
  │ Section  │                │ Segment  │
  │ Header   │                │ Header   │
  │ Table    │                │ Table    │
  └────┬─────┘                └────┬─────┘
       │                           │
  ┌────▼───────────────────────────▼────┐
  │                                      │
  │  .text  ──────────────────►  LOAD(R) │
  │  .rodata ─────────────────►  LOAD(R) │
  │  .data  ──────────────────►  LOAD(RW)│
  │  .bss   ── (无文件内容) ───►  LOAD(RW)│
  │  .symtab (不加载)                   │
  │  .rel.text (不加载)                 │
  │                                      │
  └──────────────────────────────────────┘
  • Section（节）：给链接器看的，"代码放这、数据放那、符号表在这"
  • Segment（段）：给加载器看的，"这块读进来映射到 R，那块映射到 RW"

一个 LOAD segment 可以包含多个 section（.text + .rodata 合并到一个 R 段）。

常见 Section

Section 内容 加载?
.text 机器指令 是（R）
.rodata 只读数据（字符串常量、const 全局） 是（R）
.data 已初始化全局变量 是（RW）
.bss 未初始化全局变量（文件中不占空间） 是（RW，清零）
.symtab 符号表（编译时，可 strip） 否
.dynsym 动态符号表（运行时导出/导入） 是
.strtab 符号名字符串 否
.debug_* DWARF 调试信息（-g） 否
.rel.text .text 的重定位项 链接后清除
.note.* 备注（build-id 等） 视情况

ELF 文件类型

readelf -h a.out | grep Type
Type 含义 示例
REL 可重定位文件 .o、.ko
EXEC 可执行文件 /bin/ls
DYN 共享库 / PIE 可执行 libdpdk.so、现代 a.out
CORE core dump core

用 readelf 快速验证

# ELF 头：类型、架构、入口地址
readelf -h a.out

# 所有节（Linking View）
readelf -S a.out

# 所有程序段（Execution View）
readelf -l a.out

18.5

nm：符号表查看

nm 是什么

nm 列出 ELF 文件中的符号表（.symtab 或 .dynsym），告诉你这个文件定义了哪些函数/变量、引用了哪些外部符号。是最常用的二进制排查工具。

基本用法

nm vmlinux                 # 内核符号
nm libdpdk.so              # DPDK 共享库符号
nm mod.ko                  # 内核模块符号
nm main.o                  # 目标文件符号

常用选项

选项 作用
-D 读动态符号表 .dynsym（看 .so 真正导出的 API）
-g 只看全局符号（排除 static 局部）
-u 只看未定义符号（U 类型，这个模块依赖哪些外部函数）
--defined-only 只看已定义符号
-n 按地址排序
-S 显示符号大小
-C demangle C++ 符号名（_Z3fooi → foo(int)）

符号类型标记

nm 输出每行三列：地址 类型 符号名

标记 大小写 含义
T / t 大写=全局，小写=static 代码段（.text）函数
D / d 已初始化全局数据（.data）
B / b BSS 段，未初始化全局变量
R / r 只读数据（.rodata）
U 无大小写区分 未定义——需要外部链接提供
W / w 弱符号（weak symbol）
A 绝对地址（不可重定位）
N 调试符号
- stabs/debug 格式

大写 = 全局（external linkage），小写 = 局部（static）

示例输出

$ nm main.o
                 U printf       # 未定义：需要链接 libc
0000000000000000 T main         # 全局函数：main，地址 0（可重定位）
0000000000000000 t helper       # static 函数：helper
0000000000000000 D global_var   # 全局变量：已初始化
0000000000000004 B uninit_var   # 全局变量：未初始化（BSS）

DPDK 场景

# 看 DPDK 共享库对外暴露哪些 API
nm -D librte_eth.so | grep rte_
00000000000012a0 T rte_eth_dev_configure
00000000000015f0 T rte_eth_rx_queue_setup
00000000000018b0 T rte_eth_tx_queue_setup
0000000000001c20 T rte_eth_dev_start
                 U rte_log       # 依赖外部的 rte_log
  • T 行是这个库导出的 API
  • U 行是这个库依赖的外部符号

内核模块场景

# 看内核模块依赖哪些内核函数
$ nm -u mydriver.ko
                 U printk
                 U platform_driver_register
                 U devm_ioremap
                 U __register_chrdev

这些 U 符号就是模块加载时内核需要帮忙解析的。如果内核没导出其中某个符号，insmod 就会报 undefined symbol。

# 确认内核是否导出了某个符号
$ nm vmlinux | grep platform_driver_register
ffffffffc0001234 T platform_driver_register
# 或
$ cat /proc/kallsyms | grep platform_driver_register

nm vs readelf -s

nm readelf -s
输出 简洁（地址 类型 名字） 详细（绑定、可见性、size、ndx 等）
默认读 .symtab .symtab
动态符号 需要 -D readelf --dyn-syms
易读性 更好 更全但更冗长

日常排查用 nm，需要详细信息用 readelf -s。

思考题

我拿到一个 DPDK 的 .so，想看它对外提供哪些 API，用什么命令？

nm -D librte_eth.so | grep rte_。-D 看动态符号表（.dynsym），这是运行时真正导出的符号；不加 -D 只看 .symtab（编译时符号表，可能被 strip 掉）。

内核模块 ko 加载报错 "undefined symbol"，用哪个工具静态先确认这个 U 符号存在？

nm -u mod.ko。-u 只列未定义符号（类型 U），直接确认模块依赖哪些外部符号。再用 nm vmlinux | grep <symbol> 确认内核是否导出了它。

readelf：ELF 结构全景

readelf 是什么

readelf 解析 ELF 文件的所有结构信息——头部、节、段、符号、重定位、动态段。它是不依赖 BFD 库的纯解析器，输出最原始准确。

readelf 看结构，objdump 看内容（特别是反汇编）。

常用选项速查

选项 看什么 示例
-h ELF 头（类型、架构、入口地址） readelf -h a.out
-S 节头表（所有 sections） readelf -S a.out
-l 程序头表（所有 segments） readelf -l a.out
-s 符号表（等价 nm 但更详细） readelf -s a.out
--dyn-syms 动态符号表 .dynsym readelf --dyn-syms libdpdk.so
-r 重定位信息 readelf -r mod.ko
-d 动态段（依赖哪些 .so） readelf -d libdpdk.so
-a 所有信息（汇总） readelf -a a.out
-x .text 十六进制 dump 指定节 readelf -x .rodata a.out

思考题

objdump 和 readelf 都能看符号、重定位，核心区别是什么？

18.7

objdump：反汇编与段内容

objdump 是什么

objdump 基于 BFD 库，能反汇编（机器码→汇编）、显示段内容、查看重定位。当你需要看编译器把 C 变成了什么汇编指令时，这是唯一手段。

常用选项

选项 作用 示例
-d 反汇编 .text 代码段 objdump -d a.out
-S 源码 + 汇编混排（需 -g） objdump -S a.out
-D 反汇编所有段（含数据段） objdump -D a.out
-t 符号表（类似 nm） objdump -t mod.ko
-r 重定位信息 objdump -r mod.ko
-h 段头摘要 objdump -h a.out
-j .text 只看指定段 objdump -d -j .text a.out
-M intel Intel 语法（默认 AT&T） objdump -d -M intel a.out
--disassemble=func 只反汇编指定函数 objdump -d --disassemble=main a.out

反汇编示例

基本反汇编

$ objdump -d -M intel a.out
0000000000001145 <main>:
    1145:   push   rbp
    1146:   mov    rbp,rsp
    1149:   sub    rsp,0x10
    114d:   mov    DWORD PTR [rbp-0x4],0x0     ; int i = 0
    1154:   mov    eax,0x0
    1159:   leave
    115a:   ret

源码 + 汇编混排（-S）

$ gcc -g -O0 -c main.c -o main.o
$ objdump -S main.o

main():
/var/tmp/main.c:5
    1145:   push   rbp
    1146:   mov    rbp,rsp
/var/tmp/main.c:6
    1149:   mov    DWORD PTR [rbp-4], 42       ; int x = 42;
/var/tmp/main.c:7
    1150:   mov    eax, 0
    1155:   pop    rbp
    1156:   ret

-S 需要 -g 编译。没有调试信息就只能用 -d（纯汇编，无源码对照）。

只看指定函数

$ objdump -d --disassemble=rte_eth_dev_configure librte_eth.so

不用翻整个 .so，直接看目标函数的汇编。

为什么要反汇编

1. inline 内联函数——源码看不到单独函数

static inline int add(int a, int b) { return a + b; }
int calc(int x, int y) { return add(x, y) * 2; }

$ objdump -d calc.o
0000000000000000 <calc>:
   0:   lea    eax,[rdi+rsi]    ; add() 被内联了，没有 call 指令
   3:   add    eax,eax          ; * 2
   5:   ret

源码里 add 是独立函数，但编译器内联后它消失了——只有反汇编能看到真实指令。

2. 没有调试符号的二进制

# GDB 看不到源码（没 -g）
$ gdb ./stripped_app
(gdb) list                      ← No symbol table

# objdump 照样能反汇编
$ objdump -d ./stripped_app     ← 有汇编

3. 分析编译器优化行为

# 对比 -O0 vs -O2 的汇编
gcc -O0 -S main.c -o main_O0.s
gcc -O2 -S main.c -o main_O2.s
diff main_O0.s main_O2.s

# 或直接对比目标文件
gcc -O0 -c main.c -o main_O0.o
gcc -O2 -c main.c -o main_O2.o
objdump -d main_O0.o > asm_O0.txt
objdump -d main_O2.o > asm_O2.txt

4. 定位崩溃——没有源码时

# 崩溃地址 0x555555555155
$ objdump -d a.out | grep -B5 "5555:"

AT&T vs Intel 语法

AT&T（默认） Intel（-M intel）
源/目顺序 mov src, dst mov dst, src
寄存器前缀 %eax eax
立即数前缀 $42 42
内存大小 movl mov DWORD PTR 
# AT&T（GDB/objdump 默认）
mov    $0x2a,-0x4(%rbp)

# Intel
mov    DWORD PTR [rbp-0x4],0x2a

内核源码用 AT&T，建议熟悉 AT&T 语法。Intel 语法更易读，个人偏好。

实战案例：DPDK 与内核模块

案例 1：DPDK librte_eth*.so 导出 API 排查

场景

你的 DPDK 应用链接报错 undefined reference to rte_eth_dev_configure，需要确认这个 API 在哪个 .so 里导出。

步骤

# 1. 找到 DPDK 库目录
ls /usr/lib/x86_64-linux-gnu/librte_eth*.so

# 2. 看动态符号表，确认 API 是否导出
nm -D librte_eth.so | grep rte_eth_dev_configure
00000000000012a0 T rte_eth_dev_configure    ← T = 已定义全局函数

# 3. 如果找不到，看是不是名字变了
nm -D librte_eth.so | grep rte_eth
# DPDK 22+ 改名为 rte_ethdev_*，老代码用 rte_eth_* 可能对不上

# 4. 看这个库依赖哪些其他库
readelf -d librte_eth.so | grep NEEDED
  NEEDED libnuma.so.1
  NEEDED librte_eal.so.22
  NEEDED librte_mbuf.so.22
  NEEDED librte_mempool.so.22

# 5. 如果 API 不在这个 .so，搜索所有 DPDK 库
for f in /usr/lib/x86_64-linux-gnu/librte_*.so; do
  echo "=== $(basename $f) ==="
  nm -D "$f" | grep rte_eth_dev_configure
done

# 6. 找到后看反汇编确认实现
objdump -d --disassemble=rte_eth_dev_configure librte_eth.so

常见问题

问题 原因 排查
undefined reference 链接时找不到符号 nm -D *.so \| grep <symbol>
符号在 .so 里但运行时报找不到 LD_LIBRARY_PATH 没设 ldd ./app 看依赖
DPDK 版本升级后 API 消失 版本不兼容 nm -D 对比新旧 .so

案例 2：内核模块 .ko 排查

场景

insmod mydriver.ko 报错：Unknown symbol: platform_driver_register。

步骤

# 1. 看模块依赖哪些外部符号（U 类型）
nm -u mydriver.ko
                 U printk
                 U platform_driver_register
                 U devm_ioremap
                 U __register_chrdev

# 2. 确认内核是否导出了这个符号
# 方法 a：看 /proc/kallsyms（运行中的内核）
cat /proc/kallsyms | grep platform_driver_register
ffffffffc0001234 T platform_driver_register

# 方法 b：看 vmlinux 符号表
nm vmlinux | grep platform_driver_register
ffffffffc0001234 T platform_driver_register

# 3. 如果符号存在但模块还是加载不了，可能是 EXPORT_SYMBOL 问题
# 看内核是否用 EXPORT_SYMBOL 导出了它
grep "platform_driver_register" /lib/modules/$(uname -r)/build/Module.symvers
# 如果 Module.symvers 里没有，说明内核配置没开或者没 EXPORT

# 4. 看模块的重定位项，确认需要修补哪些地址
readelf -r mydriver.ko
Relocation section '.rela.text' at offset 0x3a8 contains 8 entries:
  Offset          Info           Type           Sym. Name
0000000000030  000a00000002 R_X86_64_PC32     printk - 4
0000000000040  000500000002 R_X86_64_PC32     platform_driver_register - 4
  ↑ call 指令的目标地址，insmod 时内核填入实际地址

# 5. 看模块的 ELF 头，确认架构匹配
readelf -h mydriver.ko
  Class:      ELF64
  Machine:    Advanced Micro Devices X86-64    ← 和内核架构一致？

常见问题

问题 原因 排查
Unknown symbol 内核没导出该符号 nm -u + /proc/kallsyms
架构不匹配 交叉编译用错工具链 readelf -h 看 Machine
版本不兼容 内核 ABI 变化 对比 Module.symvers
Invalid module format vermagic 不匹配 modinfo mydriver.ko

案例 3：分析 inline 函数的真实汇编

场景

DPDK 热路径里 rte_eth_rx_burst 是 inline 函数，性能分析需要看实际生成的汇编。

# DPDK 头文件里的 inline 函数会被展开到调用处
# 源码看不到独立函数体，只有反汇编能看到

# 方法 1：反汇编整个 .o
objdump -dS rte_ethdev.o | grep -A30 "rte_eth_rx_burst"

# 方法 2：如果链接进了 .so
objdump -d --disassemble=rte_eth_rx_burst librte_eth.so
# 如果是 inline 的，这个命令可能找不到——函数被内联到调用者里了

# 方法 3：反汇编调用者，看 inline 展开
objdump -dS librte_eth.so | grep -B5 -A20 "call.*rte_eth_rx_burst"
# 如果被完全内联，连 call 都没有——直接是内联展开的指令

案例 4：对比 -O0 vs -O2 编译输出

# 编译两个版本
gcc -g -O0 -c main.c -o main_O0.o
gcc -g -O2 -c main.c -o main_O2.o

# 对比段大小
size main_O0.o main_O2.o
   text    data     bss     dec     hex filename
     85       0       0      85      55 main_O0.o
     42       0       0      42      2a main_O2.o    ← -O2 更小

# 对比汇编
objdump -dS main_O0.o > asm_O0.txt
objdump -dS main_O2.o > asm_O2.txt
diff asm_O0.txt asm_O2.txt

# 常见 -O2 优化：
# - 循环展开
# - 常量折叠（x*2 → x<<1）
# - 尾调用优化（tail call → jmp，不留栈帧）
# - 死代码消除
# - 函数内联

18.9

静态分析 vs 动态调试边界

核心区分

静态分析 动态调试
工具 nm / readelf / objdump GDB / crash / strace / ltrace
程序状态 不用跑，分析磁盘上的文件 运行中 / 暂停 / core dump
能看到 符号、ELF 结构、重定位、汇编 内存值、寄存器、栈、变量、断点
看不到 运行时变量的值、动态分配的地址 —
适用场景 链接错误、符号缺失、编译器优化分析、逆向 运行时崩溃、逻辑错误、数据竞争

什么时候用静态分析

1. 链接错误

# 报 undefined reference to foo
nm -D libxxx.so | grep foo        # foo 在不在这个 .so？
nm -u main.o                      # main.o 期望的 foo 是什么名字？

2. 符号缺失 / 导出问题

# 内核模块加载报 Unknown symbol
nm -u mod.ko                      # 模块依赖什么？
cat /proc/kallsyms | grep <sym>   # 内核导出了吗？

3. 分析编译器优化

# -O2 后行为变了，看编译器做了什么
objdump -dS main_O2.o             # 源码+汇编混排

4. 没有源码的逆向

# 只有二进制，没有源码
objdump -d ./unknown_binary       # 反汇编看逻辑
readelf -s ./unknown_binary       # 看符号表找线索
nm -D ./unknown_binary            # 看动态符号，推测功能

5. ELF 布局 / 段大小

size a.out                        # text/data/bss 多大？
readelf -S a.out                  # 所有节的大小和地址
readelf -l a.out                  # 运行时 LOAD 段布局

什么时候用动态调试（GDB）

1. 运行时崩溃

gdb ./app core
(gdb) bt                          # 崩溃栈
(gdb) print *ptr                  # 崩溃时指针的值

静态分析看不到运行时的值——变量在运行时才被写入内存。

2. 逻辑错误（不崩溃但行为不对）

gdb ./app
(gdb) break loop_body
(gdb) run
(gdb) next
(gdb) print count                 # 看循环变量实际值

3. 多线程竞争

gdb -p <PID>
(gdb) info threads                # 所有线程在干什么
(gdb) thread apply all bt         # 全栈快照

4. 内存泄漏 / 踩内存

gdb ./app
(gdb) watch *corrupted_addr       # 谁在写这块内存？
# 或用 Valgrind（更适合内存泄漏）

完整排错工作流

遇到问题
    │
    ├── 编译/链接时错误 ──→ 静态分析
    │   ├── undefined reference → nm -D / nm -u
    │   ├── 符号冲突 → nm -g / readelf -s
    │   └── 段大小异常 → size / readelf -S
    │
    ├── 运行时崩溃 ──→ 动态调试 + 静态辅助
    │   ├── core dump → gdb bt
    │   ├── 崩溃地址 → objdump -d 找函数
    │   └── 依赖缺失 → readelf -d / ldd
    │
    ├── 运行时行为异常 ──→ 动态调试
    │   ├── 逻辑错误 → gdb 断点 + print
    │   ├── 多线程 → gdb info threads
    │   └── 性能问题 → perf（不是 gdb 强项）
    │
    └── 只有二进制 ──→ 静态分析
        ├── 反汇编 → objdump -d
        ├── 符号表 → nm / readelf -s
        └── 依赖 → readelf -d

思考题

静态工具能不能看到程序运行时变量的值？为什么？

不能。nm/readelf/objdump 分析的是磁盘上的 ELF 文件，变量值在进程运行时才写入内存。静态工具只能看到变量的地址/偏移和初始值（.data 段里），看不到运行时被修改后的值。运行时值要看 GDB（print var）或 /proc/<pid>/mem。
