## 要点

```c
typedef struct {
    const char *dli_fname;
    void       *dli_fbase;
    const char *dli_sname;
    void       *dli_saddr;
} Dl_info;
```

地址 → 所属库/符号；用于日志、简易栈信息。

## 代码示例

```c
#include <stdio.h>
#include <dlfcn.h>
#include <string.h>
/* dladdr: 通过函数地址反查所在的库和符号 */
/* 需要编译为 position-independent: gcc -fPIC -pie -o demo demo.c -ldl */

int my_function(void) { return 42; }

int main(void) {
    /* dladdr: 已知函数地址, 查找所在库信息 */
    Dl_info info;

    /* 查询 libc 的 printf */
    if (dladdr((void *)printf, &info)) {
        printf("printf:\n");
        printf("  library:  %s\n", info.dli_fname);
        printf("  base:     %p\n", info.dli_fbase);
        printf("  nearest:  %s\n", info.dli_sname ? info.dli_sname : "(unknown)");
        printf("  addr:     %p\n", info.dli_saddr);
    }

    /* 查询自己的函数 */
    if (dladdr((void *)my_function, &info)) {
        printf("\nmy_function:\n");
        printf("  library:  %s\n", info.dli_fname);
        printf("  nearest:  %s\n", info.dli_sname ? info.dli_sname : "(unknown)");
    }

    /* 查询 main 本身 */
    if (dladdr((void *)main, &info)) {
        printf("\nmain:\n");
        printf("  library:  %s\n", info.dli_fname);
    }
    return 0;
}
```
