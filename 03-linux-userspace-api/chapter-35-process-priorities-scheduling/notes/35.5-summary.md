## 代码示例

```c
#define _GNU_SOURCE
#include <sched.h>
#include <stdio.h>
int main(void) {
    /* CPU 亲和性: 绑定进程到特定 CPU */
    cpu_set_t set;
    CPU_ZERO(&set);
    CPU_SET(0, &set);  /* 绑定到 CPU 0 */

    if (sched_setaffinity(0, sizeof(set), &set) == 0) {
        printf("进程绑定到 CPU 0\n");
    }

    /* 查询当前亲和性 */
    CPU_ZERO(&set);
    sched_getaffinity(0, sizeof(set), &set);
    printf("可运行 CPU 数: %d\n", CPU_COUNT(&set));
    return 0;
}
```
