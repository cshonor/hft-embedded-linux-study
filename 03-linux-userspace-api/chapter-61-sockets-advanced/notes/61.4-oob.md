## 代码示例

```c

/* 带外数据 (Out-of-Band Data) 传输 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/wait.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9991) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 1);

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_INET, SOCK_STREAM, 0);
        struct sockaddr_in sa = { .sin_family = AF_INET, .sin_port = htons(9991) };
        inet_pton(AF_INET, "127.0.0.1", &sa.sin_addr);
        connect(cfd, (struct sockaddr *)&sa, sizeof(sa));

        /* 发送普通数据 */
        send(cfd, "normal", 7, 0);
        /* 发送带外数据：MSG_OOB */
        send(cfd, "!", 2, MSG_OOB);
        printf("client: sent normal + OOB data\n");
        close(cfd);
        _exit(0);
    }

    int afd = accept(sfd, NULL, NULL);

    /* 接收带外数据：MSG_OOB */
    char oobbuf[2];
    ssize_t n = recv(afd, oobbuf, 1, MSG_OOB);
    if (n > 0) {
        printf("server: OOB data = '%c'\n", oobbuf[0]);
    } else {
        printf("server: OOB not available yet\n");
    }

    /* 接收普通数据 */
    char buf[16];
    n = recv(afd, buf, sizeof(buf)-1, 0);
    if (n > 0) { buf[n]='\0'; printf("server: normal = '%s'\n", buf); }

    wait(NULL);
    close(afd);
    close(sfd);
    return 0;
}

```
