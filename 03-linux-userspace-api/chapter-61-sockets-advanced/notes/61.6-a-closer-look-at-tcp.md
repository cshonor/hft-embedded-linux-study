## 要点

| TCP | 可部分读写；须循环发满；对端关 → recv=0；RST → `ECONNRESET` |
| UDP | **整报**；缓冲过小 → **余部丢弃**，不分段 |

| 路线 | 要点 |

## 代码示例

```c

/* 传递文件描述符：SCM_RIGHTS 辅助数据 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/wait.h>

int main(void) {
    int sv[2];
    socketpair(AF_UNIX, SOCK_STREAM, 0, sv);

    pid_t pid = fork();
    if (pid == 0) {
        close(sv[0]);
        /* 子进程打开一个文件，通过 socket 传给父进程 */
        int fd = open("/tmp/ch61_passfd.txt", O_RDWR | O_CREAT | O_TRUNC, 0644);
        write(fd, "passed fd content", 18);

        struct msghdr msg;
        struct iovec iov;
        char buf[] = "fd";
        iov.iov_base = buf; iov.iov_len = 2;
        msg.msg_iov = &iov; msg.msg_iovlen = 1;

        char cmsgbuf[CMSG_SPACE(sizeof(int))];
        msg.msg_control = cmsgbuf;
        msg.msg_controllen = sizeof(cmsgbuf);

        struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
        cmsg->cmsg_level = SOL_SOCKET;
        cmsg->cmsg_type = SCM_RIGHTS;
        cmsg->cmsg_len = CMSG_LEN(sizeof(int));
        *(int *)CMSG_DATA(cmsg) = fd;

        sendmsg(sv[1], &msg, 0);
        printf("child: sent fd=%d\n", fd);
        close(fd);
        close(sv[1]);
        _exit(0);
    }

    close(sv[1]);
    /* 父进程接收文件描述符 */
    struct msghdr msg;
    struct iovec iov;
    char buf[2];
    iov.iov_base = buf; iov.iov_len = 2;
    msg.msg_iov = &iov; msg.msg_iovlen = 1;
    msg.msg_control = NULL; msg.msg_controllen = 0;

    char cmsgbuf[CMSG_SPACE(sizeof(int))];
    msg.msg_control = cmsgbuf;
    msg.msg_controllen = sizeof(cmsgbuf);

    recvmsg(sv[0], &msg, 0);

    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    if (cmsg && cmsg->cmsg_type == SCM_RIGHTS) {
        int recv_fd = *(int *)CMSG_DATA(cmsg);
        printf("parent: received fd=%d\n", recv_fd);

        /* 用收到的 fd 读取文件 */
        lseek(recv_fd, 0, SEEK_SET);
        char fbuf[64];
        int n = read(recv_fd, fbuf, sizeof(fbuf)-1);
        if (n > 0) { fbuf[n]='\0'; printf("parent: file content='%s'\n", fbuf); }
        close(recv_fd);
    }

    wait(NULL);
    close(sv[0]);
    return 0;
}

```
