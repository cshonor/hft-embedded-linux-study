# 19.8 Exercise

> 本章：[TLPI 第 19 章 — Monitoring File Events](./README.md)

## 练习题

### 练习 1：目录监控器

编写程序，监控指定目录，打印所有文件创建、修改、删除事件。

**要求**：
- 支持递归监控子目录（手动遍历添加 watch）
- 新创建的子目录也自动添加监控
- 使用 `epoll` 管理信号（如 Ctrl-C 退出）

### 练习 2：配置文件热加载

编写程序，监控配置文件变化，文件修改后自动重新加载配置并应用。

**要求**：
- 使用 `IN_MODIFY` 事件
- 防抖处理（编辑器可能多次保存，合并短时间内的多次事件）
- 打印重新加载的时间戳

## 参考答案

### 练习 1 参考代码

```c
#include <sys/inotify.h>
#include <dirent.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define EVENT_BUF_LEN (16 * (sizeof(struct inotify_event) + NAME_MAX + 1))
#define MAX_WATCHES 1024

int ifd;
int watches[MAX_WATCHES];
char paths[MAX_WATCHES][PATH_MAX];
int watch_count = 0;

void add_watch(const char *path) {
    uint32_t mask = IN_CREATE | IN_DELETE | IN_MODIFY
                  | IN_MOVED_FROM | IN_MOVED_TO | IN_DELETE_SELF;
    int wd = inotify_add_watch(ifd, path, mask);
    if (wd >= 0 && watch_count < MAX_WATCHES) {
        watches[watch_count] = wd;
        strncpy(paths[watch_count], path, PATH_MAX);
        watch_count++;
    }
}

void watch_recursive(const char *dir) {
    add_watch(dir);
    DIR *d = opendir(dir);
    if (!d) return;
    struct dirent *de;
    while ((de = readdir(d)) != NULL) {
        if (de->d_type == DT_DIR && strcmp(de->d_name, ".") != 0
            && strcmp(de->d_name, "..") != 0) {
            char sub[PATH_MAX];
            snprintf(sub, sizeof(sub), "%s/%s", dir, de->d_name);
            watch_recursive(sub);
        }
    }
    closedir(d);
}

int main(int argc, char *argv[]) {
    if (argc != 2) { fprintf(stderr, "usage: %s <dir>\n", argv[0]); return 1; }

    ifd = inotify_init1(IN_CLOEXEC);
    watch_recursive(argv[1]);
    printf("watching %d directories\n", watch_count);

    char buf[EVENT_BUF_LEN] __attribute__((aligned(8)));
    while (1) {
        ssize_t n = read(ifd, buf, sizeof(buf));
        if (n <= 0) break;
        char *p = buf;
        while (p < buf + n) {
            struct inotify_event *ev = (struct inotify_event *)p;
            if (ev->mask & IN_CREATE && ev->len > 0) {
                /* 新目录自动添加 watch */
                char full[PATH_MAX];
                for (int i = 0; i < watch_count; i++)
                    if (watches[i] == ev->wd) { snprintf(full, sizeof(full), "%s/%s", paths[i], ev->name); break; }
                add_watch(full);
            }
            printf("wd=%d mask=0x%x %s\n", ev->wd, ev->mask,
                   ev->len > 0 ? ev->name : "");
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    close(ifd);
    return 0;
}
```

### 练习 2 参考代码

```c
#include <sys/inotify.h>
#include <unistd.h>
#include <time.h>
#include <stdio.h>
#include <string.h>

volatile int config_changed = 0;
time_t last_event_time = 0;

void load_config(const char *path) {
    printf("[%ld] config reloaded: %s\n", (long)time(NULL), path);
    /* 实际加载配置逻辑 */
}

int main(int argc, char *argv[]) {
    const char *config = argc > 1 ? argv[1] : "app.conf";

    int ifd = inotify_init1(IN_CLOEXEC);
    inotify_add_watch(ifd, config, IN_MODIFY | IN_CLOSE_WRITE);

    char buf[4096] __attribute__((aligned(8)));
    while (1) {
        ssize_t n = read(ifd, buf, sizeof(buf));
        if (n <= 0) break;
        char *p = buf;
        while (p < buf + n) {
            struct inotify_event *ev = (struct inotify_event *)p;
            if (ev->mask & (IN_MODIFY | IN_CLOSE_WRITE)) {
                time_t now = time(NULL);
                /* 防抖：3 秒内的事件合并 */
                if (now - last_event_time >= 3) {
                    load_config(config);
                }
                last_event_time = now;
            }
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    close(ifd);
    return 0;
}
```

## HFT / 嵌入式关联

- **策略热加载**：HFT 策略参数修改后无需重启服务，inotify 触发重新加载
- **订单文件**：文件订单到达后 inotify 触发处理
- **嵌入式**：日志轮转检测（`IN_MOVE_SELF` 触发重打开日志）
