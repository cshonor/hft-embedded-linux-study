## 代码示例

```c

/* mremap 扩展/收缩映射（Linux 特有） */
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    size_t old_size = 4096;
    size_t new_size = 8192;

    /* 初始匿名映射 */
    char *m = mmap(NULL, old_size, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (m == MAP_FAILED) { perror("mmap"); exit(1); }
    strcpy(m, "original data");
    printf("old mapping: %p, size=%zu\n", (void*)m, old_size);

    /* 扩展到 new_size */
    char *m2 = mremap(m, old_size, new_size, MREMAP_MAYMOVE);
    if (m2 == MAP_FAILED) { perror("mremap"); exit(1); }
    printf("new mapping: %p, size=%zu (may have moved)\n", (void*)m2, new_size);
    printf("data preserved: %s\n", m2);

    munmap(m2, new_size);
    return 0;
}

```
