## 代码示例

```c

/* inet_pton / inet_ntop 地址转换函数 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>

int main(void) {
    /* inet_pton: presentation -> network (字符串 → 二进制) */
    struct in_addr addr4;
    if (inet_pton(AF_INET, "192.168.1.100", &addr4) == 1) {
        printf("inet_pton IPv4: 192.168.1.100 -> 0x%08x\n",
               ntohl(addr4.s_addr));
    }

    /* inet_ntop: network -> presentation (二进制 → 字符串) */
    char str[INET_ADDRSTRLEN];
    inet_ntop(AF_INET, &addr4, str, sizeof(str));
    printf("inet_ntop IPv4: 0x%08x -> %s\n", ntohl(addr4.s_addr), str);

    /* IPv6 支持 */
    struct in6_addr addr6;
    if (inet_pton(AF_INET6, "::1", &addr6) == 1) {
        char str6[INET6_ADDRSTRLEN];
        inet_ntop(AF_INET6, &addr6, str6, sizeof(str6));
        printf("inet_pton/ntop IPv6: ::1 -> %s\n", str6);
    }

    /* 无效地址测试 */
    if (inet_pton(AF_INET, "256.1.1.1", &addr4) != 1) {
        printf("inet_pton: '256.1.1.1' is invalid\n");
    }
    return 0;
}

```
