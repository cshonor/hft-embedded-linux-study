## 代码示例

```c

/* 字节序：主机字节序 vs 网络字节序 */
#include <stdio.h>
#include <stdlib.h>
#include <arpa/inet.h>

int main(void) {
    /* 检测主机字节序 */
    unsigned int x = 0x01020304;
    char *p = (char *)&x;
    if (p[0] == 0x01) {
        printf("Host byte order: Big-endian\n");
    } else {
        printf("Host byte order: Little-endian\n");
    }

    /* htons/htonl: host to network (short/long) */
    unsigned short h_port = 8080;
    unsigned short n_port = htons(h_port);
    printf("port %u -> htons -> %u\n", h_port, n_port);

    /* ntohs/ntohl: network to host */
    printf("port %u -> ntohs -> %u (round-trip)\n", n_port, ntohs(n_port));

    /* 直接用 inet_addr（已过时，推荐 inet_pton） */
    in_addr_t addr = inet_addr("127.0.0.1");
    printf("inet_addr('127.0.0.1') = 0x%x (already network order)\n",
           ntohl(addr));

    return 0;
}

```
