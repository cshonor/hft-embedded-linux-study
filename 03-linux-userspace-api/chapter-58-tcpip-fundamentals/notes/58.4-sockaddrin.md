## 本节讲什么

struct sockaddr_in { sa_family_t    sin_family;  /* AF_INET */ in_port_t      sin_port;    /* 网络序 */

---

## 要点

```c
struct sockaddr_in {
    sa_family_t    sin_family;  /* AF_INET */
    in_port_t      sin_port;    /* 网络序 */
    struct in_addr sin_addr;    /* s_addr 网络序 */
    unsigned char  sin_zero[8]; /* 置 0 */
};
```

传给 bind/connect 时转为 `struct sockaddr *`。

## 代码示例

```c

/* struct sockaddr_in 结构详解 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>
#include <netinet/in.h>
#include <sys/socket.h>

int main(void) {
    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));

    addr.sin_family = AF_INET;       /* 地址族 */
    addr.sin_port = htons(8080);     /* 端口（网络字节序） */
    inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);  /* IP 地址 */

    printf("sizeof(sockaddr_in) = %zu\n", sizeof(addr));
    printf("sin_family = %d\n", addr.sin_family);
    printf("sin_port   = %u (network=%u, host=%u)\n",
           addr.sin_port, addr.sin_port, ntohs(addr.sin_port));
    printf("sin_addr   = 0x%x\n", ntohl(addr.sin_addr.s_addr));

    /* sockaddr_in 可安全转型为 sockaddr */
    struct sockaddr *gen = (struct sockaddr *)&addr;
    printf("cast to sockaddr*: sa_family=%d\n", gen->sa_family);

    /* sockaddr_storage: 足够大，兼容 IPv4/IPv6 */
    struct sockaddr_storage ss;
    memset(&ss, 0, sizeof(ss));
    printf("sizeof(sockaddr_storage) = %zu (enough for IPv6)\n", sizeof(ss));
    return 0;
}

```
