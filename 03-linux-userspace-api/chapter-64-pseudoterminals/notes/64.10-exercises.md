### 练习 1：PTY 基本使用

**题目**：编写程序通过 PTY 运行 `ls -la /` 并捕获输出。

<details>
<summary>参考答案</summary>

```c
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>
#include <sys/wait.h>

int main() {
    int masterfd = posix_openpt(O_RDWR | O_NOCTTY);
    grantpt(masterfd);
    unlockpt(masterfd);
    char *slavename = ptsname(masterfd);

    pid_t pid = fork();
    if (pid == 0) {
        close(masterfd);
        setsid();
        int slavefd = open(slavename, O_RDWR);
        dup2(slavefd, STDIN_FILENO);
        dup2(slavefd, STDOUT_FILENO);
        dup2(slavefd, STDERR_FILENO);
        close(slavefd);
        execlp("ls", "ls", "-la", "/", NULL);
        _exit(1);
    }

    /* 父进程读取输出 */
    char buf[4096];
    ssize_t n;
    while ((n = read(masterfd, buf, sizeof(buf))) > 0)
        write(STDOUT_FILENO, buf, n);

    waitpid(pid, NULL, 0);
    close(masterfd);
    return 0;
}
```

</details>

### 练习 2：PTY 终端属性

**题目**：修改练习 1，使子进程以 80x24 的终端大小运行，并设置原始模式。

<details>
<summary>参考答案</summary>

```c
#include <sys/ioctl.h>
#include <termios.h>

/* 在 fork 前设置从设备属性 */
void setup_pty(int masterfd, const char *slavename) {
    /* 打开从设备设置属性（之后关闭） */
    int slavefd = open(slavename, O_RDWR);

    /* 设置窗口大小 80x24 */
    struct winsize ws = { .ws_row = 24, .ws_col = 80 };
    ioctl(slavefd, TIOCSWINSZ, &ws);

    /* 设置原始模式 */
    struct termios t;
    tcgetattr(slavefd, &t);
    cfmakeraw(&t);
    tcsetattr(slavefd, TCSANOW, &t);

    close(slavefd);
}

int main() {
    int masterfd = posix_openpt(O_RDWR | O_NOCTTY);
    grantpt(masterfd);
    unlockpt(masterfd);
    char *slavename = ptsname(masterfd);

    /* 在 fork 前设置属性 */
    setup_pty(masterfd, slavename);

    pid_t pid = fork();
    if (pid == 0) {
        close(masterfd);
        setsid();
        int slavefd = open(slavename, O_RDWR);
        dup2(slavefd, 0); dup2(slavefd, 1); dup2(slavefd, 2);
        close(slavefd);
        execlp("stty", "stty", "size", NULL);  /* 打印终端大小 */
        _exit(1);
    }

    char buf[4096];
    ssize_t n;
    while ((n = read(masterfd, buf, sizeof(buf))) > 0)
        write(STDOUT_FILENO, buf, n);
    waitpid(pid, NULL, 0);
    close(masterfd);
    return 0;
}
```

</details>

### 练习 3：PTY 远程终端

**题目**：设计一个通过 TCP socket 转发 PTY 数据的远程终端服务端框架。

<details>
<summary>参考答案</summary>

```c
/* 框架思路（伪代码） */

/* 1. 接受 TCP 连接 */
int clientfd = accept(listenfd, NULL, NULL);

/* 2. 创建 PTY */
int masterfd = posix_openpt(O_RDWR | O_NOCTTY);
grantpt(masterfd);
unlockpt(masterfd);
char *slavename = ptsname(masterfd);

/* 3. fork 子进程运行 shell */
pid_t pid = fork();
if (pid == 0) {
    close(masterfd);
    close(clientfd);
    setsid();

    int slavefd = open(slavename, O_RDWR);
    /* 同步终端属性（从 socket 接收） */
    dup2(slavefd, 0); dup2(slavefd, 1); dup2(slavefd, 2);
    close(slavefd);
    execlp("bash", "bash", NULL);
    _exit(1);
}

/* 4. 父进程：双向数据转发 */
/* 使用 epoll 监听 clientfd 和 masterfd */
int epfd = epoll_create1(0);
struct epoll_event ev;

ev.events = EPOLLIN; ev.data.fd = clientfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, clientfd, &ev);
ev.data.fd = masterfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, masterfd, &ev);

char buf[4096];
while (1) {
    struct epoll_event events[2];
    int n = epoll_wait(epfd, events, 2, -1);
    for (int i = 0; i < n; i++) {
        int fd = events[i].data.fd;
        int len = read(fd, buf, sizeof(buf));
        if (len <= 0) goto done;
        int dest = (fd == clientfd) ? masterfd : clientfd;
        write(dest, buf, len);
    }
}
done:
close(clientfd);
close(masterfd);
```

</details>
