导读

章节目标

pipe/mkfifo；读写阻塞与 EOF/SIGPIPE；PIPE_BUF 原子写；popen 风险；stdio 缓冲；FIFO open 配对；局限与选型。

思考题要点

  1. 不关写端 → 读不到 EOF，永久阻塞。
  2. 无读端写 → SIGPIPE/EPIPE；可忽略 SIGPIPE 并查 EPIPE。
  3. PIPE_BUF≠总容量。
  4. 消息 ≤PIPE_BUF 或外层加锁/协议。
  5. FIFO O_WRONLY|O_NONBLOCK → ENXIO。
  6. popen 走 shell；改自行 exec。

背诵卡

# 要点
1 fd[0] 读 / fd[1] 写；关无用端
2 写端全关 → read EOF；读端全关 → SIGPIPE
3 PIPE_BUF 原子写；容量另计
4 FIFO 路径入口；数据不落盘
5 FIFO open 读写配对；ENXIO
6 popen 有注入；管道上注意块缓冲

参考

  • Kerrisk · TLPI Ch44（非 Ch18）
  • man 2 pipe · man 3 mkfifo · man 3 popen · man 7 pipe

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <string.h>
#include <sys/wait.h>

/* Ch44 管道与 FIFO — pipe + mkfifo 命名管道。
 * 演示匿名管道(父子通信) + FIFO(无关进程通信)。
 * 编译: gcc -o ch44_demo ch44_demo.c */

#define FIFO_PATH "/tmp/ch44_fifo"

int main(void) {
    /* === 匿名管道 === */
    int pipefd[2];
    pipe(pipefd);

    pid_t pid = fork();
    if (pid == 0) {
        close(pipefd[0]);
        write(pipefd[1], "pipe msg", 8);
        close(pipefd[1]);
        _exit(0);
    }
    close(pipefd[1]);
    char buf[64];
    read(pipefd[0], buf, 8);
    buf[8] = '\0';
    printf("Pipe received: %s\n", buf);
    waitpid(pid, NULL, 0);

    /* === FIFO (命名管道) === */
    mkfifo(FIFO_PATH, 0644);

    pid = fork();
    if (pid == 0) {
        /* 子进程: 写 FIFO */
        int fd = open(FIFO_PATH, O_WRONLY);
        write(fd, "fifo msg", 8);
        close(fd);
        _exit(0);
    }

    /* 父进程: 读 FIFO */
    int fd = open(FIFO_PATH, O_RDONLY);
    read(fd, buf, 8);
    buf[8] = '\0';
    printf("FIFO received: %s\n", buf);
    close(fd);
    waitpid(pid, NULL, 0);

    unlink(FIFO_PATH);
    return 0;
}

44.1

Pipe vs FIFO

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：本章首页 - 下一节：44.2 pipe() 范式

本节讲什么

FIFO 文件名 ≠ 磁盘数据；缓冲在内核。 ---

要点

匿名管道 FIFO
创建 pipe(fd[2]) mkfifo / mkfifo(3)
名称 无 路径（ls 类型 p）
进程 仅亲缘（继承 fd） 任意本地（路径 + 权限）
打开 创建即有两端 fd 读写双方互相等待（默认可阻塞）
IO 半双工字节流 同左
持久 进程持久（全 close 毁） 节点可留；缓冲仍进程/内核语义

FIFO 文件名 ≠ 磁盘数据；缓冲在内核。

HFT / 嵌入式关联

路线 要点
嵌入式 匿名 pipe 只限 fork 亲缘进程；FIFO 有路径+权限，供无关进程（采集→处理）通信
HFT 管道是字节流经内核缓冲有拷贝，热路径共享数据用 shm；FIFO 用于低频控制/日志通道

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/stat.h>
#include <fcntl.h>
/* pipe (匿名) vs FIFO (命名): 区别与使用场景 */
int main(void) {
    /* pipe: 匿名管道, 只能有亲缘进程使用 */
    int pfd[2];
    pipe(pfd);
    printf("pipe created: read=%d, write=%d\n", pfd[0], pfd[1]);

    /* FIFO: 命名管道, 任何进程都能打开 */
    const char *fifopath = "/tmp/ch44_fifo_demo";
    mkfifo(fifopath, 0644);
    printf("FIFO created at %s\n", fifopath);

    /* FIFO 本质是内核缓冲区, 有路径名 */
    /* 读端阻塞直到有写端打开, 反之亦然 */
    if (fork() == 0) {
        int fd = open(fifopath, O_WRONLY);
        write(fd, "via FIFO", 8);
        close(fd);
        _exit(0);
    }
    int fd = open(fifopath, O_RDONLY);
    char buf[16];
    read(fd, buf, 8);
    buf[8] = '\0';
    printf("FIFO received: %s\n", buf);
    close(fd);
    wait(NULL);
    unlink(fifopath);
    return 0;
}

44.2

`pipe()` 范式

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.1 Pipe vs FIFO - 下一节：44.3 阻塞读写规则

本节讲什么

int fd[2]; pipe(fd); / fd[0] 读, fd[1] 写 / fork();

要点

int fd[2];
pipe(fd);          /* fd[0] 读, fd[1] 写 */
fork();
/* 父写子读：父 close(fd[0]); 子 close(fd[1]); */

双向 → 两根管道。通信完 close。

Demo：code/pipe_demo.c

HFT / 嵌入式关联

路线 要点
嵌入式 fd[0]读/fd[1]写，fork 后各关一端即单向；双向要两根管道，通信完 close
HFT 管道数据经内核缓冲有拷贝，仅作低频命令通道；热路径改 shm/mmap 零拷贝

代码示例

#include <stdio.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>
/* pipe() 范式: fd[0]读, fd[1]写, fork后各自关闭一端 */
int main(void) {
    int fd[2];
    if (pipe(fd) < 0) { perror("pipe"); return 1; }

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 关闭读端, 只写 */
        close(fd[0]);
        const char *msg = "hello pipe";
        write(fd[1], msg, strlen(msg) + 1);
        close(fd[1]);
        _exit(0);
    }

    /* 父进程: 关闭写端, 只读 */
    close(fd[1]);
    char buf[64];
    ssize_t n = read(fd[0], buf, sizeof(buf));
    printf("Parent received %zd bytes: %s\n", n, buf);
    close(fd[0]);
    waitpid(pid, NULL, 0);
    return 0;
}

44.3

阻塞读写规则

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.2 pipe() 范式 - 下一节：44.4 PIPE_BUF vs 容量

本节讲什么

read write ---

要点

read

条件 行为
有数据 返回已读长度
空且仍有写端 阻塞
空且写端全关 0（EOF）

write

条件 行为
有空间 写入
缓冲满 阻塞
读端全关 SIGPIPE（默认同亡）；忽略则 -1/EPIPE

HFT / 嵌入式关联

路线 要点
嵌入式 写端全关 read 返回 0(EOF) 作优雅退出信号；读端全关写触发 SIGPIPE 须屏蔽/忽略
HFT 阻塞 read 会卡死热路径，必须 O_NONBLOCK；SIGPIPE 默认杀进程，须忽略防误杀

代码示例

#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/wait.h>
/* 管道阻塞规则: 读空管道阻塞, 写满管道阻塞 */
int main(void) {
    int fd[2];
    pipe(fd);

    /* 默认阻塞: read 空管道会阻塞直到有数据 */
    /* O_NONBLOCK: 非阻塞模式 */
    fcntl(fd[0], F_SETFL, O_NONBLOCK);

    char buf[64];
    ssize_t n = read(fd[0], buf, sizeof(buf));
    if (n < 0)
        perror("non-blocking read (EAGAIN expected)");

    /* 恢复阻塞模式 */
    fcntl(fd[0], F_SETFL, 0);

    /* 写满后也会阻塞: PIPE_BUF 通常是 4096 或 65536 */
    long pipe_buf = fpathconf(fd[1], _PC_PIPE_BUF);
    printf("PIPE_BUF = %ld\n", pipe_buf);

    /* 测试: 关闭所有写端后, read 返回 0 (EOF) */
    close(fd[1]);
    n = read(fd[0], buf, sizeof(buf));
    printf("After write end closed: read returned %zd (EOF)\n", n);
    close(fd[0]);
    return 0;
}

44.4

`PIPE_BUF` vs 容量

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.3 阻塞读写规则 - 下一节：44.5 –44.6 O_NONBLOCK · popen

本节讲什么

> PIPE_BUF 不保证原子。 ---

要点
PIPE_BUF Linux 常 4096；≤ 此长度的 write 原子（多写者不交错）
Pipe capacity 总缓冲（Linux 常约 64KB）；满则阻塞

> PIPE_BUF 不保证原子。

HFT / 嵌入式关联

路线 要点
嵌入式 ≤PIPE_BUF 的 write 原子，多进程写日志/状态不会交错
HFT 多生产者写同一管道用 ≤PIPE_BUF 保证原子；超长消息自加锁或改用 mq

代码示例

#include <stdio.h>
#include <unistd.h>
#include <string.h>
/* PIPE_BUF: 原子写入保证 */
int main(void) {
    long pipe_buf = fpathconf(STDOUT_FILENO, _PC_PIPE_BUF);
    printf("PIPE_BUF = %ld bytes\n", pipe_buf);
    printf("Writes <= PIPE_BUF are atomic\n");
    printf("Writes > PIPE_BUF may be interleaved\n");

    /* 多进程同时写管道: 小于 PIPE_BUF 的写不会交错 */
    int fd[2];
    pipe(fd);

    for (int i = 0; i < 3; i++) {
        if (fork() == 0) {
            close(fd[0]);
            char msg[64];
            int len = snprintf(msg, sizeof(msg), "child %d\n", i);
            /* 如果 len <= PIPE_BUF, 这个 write 是原子的 */
            write(fd[1], msg, len);
            close(fd[1]);
            _exit(0);
        }
    }

    close(fd[1]);
    char buf[256];
    ssize_t n;
    while ((n = read(fd[0], buf, sizeof(buf))) > 0)
        write(STDOUT_FILENO, buf, n);
    close(fd[0]);

    while (wait(NULL) > 0);
    return 0;
}

44.5

–44.6 `O_NONBLOCK` · `popen`

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.4 PIPE_BUF vs 容量 - 下一节：44.6 管道 + stdio

本节讲什么

非阻塞：空读/写满 → -1/EAGAIN（细节见 man 7 pipe）。 FILE popen(const char command, const char type); / "r"|"w" / int pclose(FILE stream);

要点

非阻塞：空读/写满 → -1/EAGAIN（细节见 man 7 pipe）。

FILE *popen(const char *command, const char *type);  /* "r"|"w" */
int pclose(FILE *stream);

内部 fork+pipe+shell → 注入风险；单向；难控 fd 标志。替代：自管 pipe+fork+exec（无 shell）。

HFT / 嵌入式关联

路线 要点
嵌入式 O_NONBLOCK 空读/满写返回 EAGAIN，配 epoll 做非阻塞管道
HFT popen 内部 fork+shell 有注入风险+开销，热路径用自管 pipe+fork+exec（无 shell）

代码示例

#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
/* popen: 管道 + fork + exec 的封装 */
int main(void) {
    /* popen: 执行命令并获取其输出 */
    FILE *fp = popen("ls -l /tmp", "r");
    if (!fp) { perror("popen"); return 1; }

    char line[256];
    while (fgets(line, sizeof(line), fp))
        printf("  %s", line);
    pclose(fp);

    /* popen("w") 可以向命令的 stdin 写 */
    fp = popen("wc -l", "w");
    if (fp) {
        for (int i = 0; i < 10; i++)
            fprintf(fp, "line %d\n", i);
        pclose(fp);
    }

    /* O_NONBLOCK 管道: 非阻塞读写 */
    int fd[2];
    pipe(fd);
    fcntl(fd[0], F_SETFL, O_NONBLOCK);
    char buf[64];
    ssize_t n = read(fd[0], buf, sizeof(buf));
    printf("\nNon-blocking read: n=%zd (EAGAIN if empty)\n", n);
    close(fd[0]); close(fd[1]);
    return 0;
}

44.6

管道 + stdio

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.5 –44.6 O_NONBLOCK · popen - 下一节：44.7 –44.8 FIFO

本节讲什么

管道上 stdout 多为块缓冲 → printf 不立刻进管道。 fflush / setvbuf。 ---

要点

管道上 stdout 多为块缓冲 → printf 不立刻进管道。
fflush / setvbuf。

HFT / 嵌入式关联

路线 要点
嵌入式 管道上 stdout 块缓冲，printf 不立刻进管道，需 fflush/setvbuf
HFT stdio 缓冲致延迟不可控，热路径用 write() 直写 fd 绕开 stdio 缓冲

代码示例

#include <stdio.h>
#include <unistd.h>
/* 管道 + stdio: fdopen 将 fd 包装为 FILE* */
int main(void) {
    int fd[2];
    pipe(fd);

    if (fork() == 0) {
        close(fd[0]);
        /* fdopen: 将 fd 包装为 FILE*, 可以用 fprintf */
        FILE *fp = fdopen(fd[1], "w");
        fprintf(fp, "line 1 via fprintf\n");
        fprintf(fp, "line 2 via fprintf\n");
        fflush(fp);  /* 必须 flush, 否则数据在 stdio buffer 中 */
        fclose(fp);
        _exit(0);
    }

    close(fd[1]);
    FILE *fp = fdopen(fd[0], "r");
    char line[128];
    while (fgets(line, sizeof(line), fp))
        printf("read: %s", line);
    fclose(fp);
    wait(NULL);
    return 0;
}

44.7

–44.8 FIFO

本章：TLPI 第 44 章 — Pipes and FIFOs - 上一节：44.6 管道 + stdio - 下一节：44.9 局限

本节讲什么

mkfifo(path, mode); open(path, O_RDONLY); / 阻塞到有写端 / open(path, O_WRONLY); / 阻塞到有读端 /

要点

mkfifo(path, mode);
open(path, O_RDONLY);   /* 阻塞到有写端 */
open(path, O_WRONLY);   /* 阻塞到有读端 */

O_NONBLOCK：O_RDONLY 常立刻成功；O_WRONLY 无读端 → ENXIO。
勿依赖 O_RDWR 开 FIFO（可移植性差）。

简单 C/S：一服务端读 FIFO，多客户端写 — 并发写仍可能交织（≤PIPE_BUF 可原子）。

Demo：code/fifo_demo.c + README 双进程用法。

HFT / 嵌入式关联

路线 要点
嵌入式 open O_WRONLY 无读端 ENXIO；一服务端读+多客户端写，≤PIPE_BUF 可原子
HFT FIFO 有路径名跨无关进程，但字节流无边界+多客户端交织，仅低频控制

代码示例

#include <stdio.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/wait.h>
#include <string.h>
/* FIFO (命名管道): 无关进程间通信 */
#define FIFO_PATH "/tmp/ch44_fifo_demo2"

int main(void) {
    /* mkfifo: 创建 FIFO 特殊文件 */
    mkfifo(FIFO_PATH, 0644);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 写端 */
        int fd = open(FIFO_PATH, O_WRONLY);
        write(fd, "hello FIFO", 10);
        close(fd);
        _exit(0);
    }

    /* 父进程: 读端 (open 会阻塞直到有写端) */
    int fd = open(FIFO_PATH, O_RDONLY);
    char buf[64];
    ssize_t n = read(fd, buf, sizeof(buf));
    buf[n] = '\0';
    printf("Received: %s\n", buf);
    close(fd);

    waitpid(pid, NULL, 0);
    unlink(FIFO_PATH);  /* 删除 FIFO 文件 */
    return 0;
}

44.8

A Client-Server Application Using FIFOs

本章：TLPI 第 44 章

本节讲什么

使用 FIFO 实现客户端-服务器通信：服务端创建知名 FIFO，客户端通过 FIFO 发送请求。展示了 FIFO 的单向通信特性和多客户端场景。

要点

架构

Client A ─→ ┌──────────────┐
Client B ─→ │ Server FIFO  │ ─→ Server
Client C ─→ │ (known path) │
            └──────────────┘

服务端模式

/* 1. 创建知名 FIFO */
mkfifo("/tmp/server.fifo", 0666);
/* 2. 打开读取请求 */
int rfd = open("/tmp/server.fifo", O_RDONLY);
/* 3. 持续读取多个客户端请求 */
char buf[4096];
while (read(rfd, buf, sizeof(buf)) > 0) {
    /* 处理请求 */
}

客户端模式

/* 直接写知名 FIFO */
int fd = open("/tmp/server.fifo", O_WRONLY);
write(fd, request, len);
close(fd);

限制

限制 说明
单向 FIFO 只能单向传输
无消息边界 多个客户端写交错
无回复通道 服务端需要额外 FIFO 回复
阻塞 open O_RDONLY 等待 writer

双向通信方案

服务端 FIFO (/tmp/srv.fifo)  ← 客户端写请求
客户端 FIFO (/tmp/cli-PID.fifo) ← 服务端写回复

/* 客户端：先创建自己的回复 FIFO */
char myfifo[64];
sprintf(myfifo, "/tmp/cli-%d.fifo", getpid());
mkfifo(myfifo, 0666);
/* 请求中包含回复 FIFO 路径 */
snprintf(request, sizeof(request), "%s %s", myfifo, query);
write(srv_fd, request, len);
/* 从自己的 FIFO 读回复 */
int rfd = open(myfifo, O_RDONLY);
read(rfd, reply, sizeof(reply));
unlink(myfifo);

HFT / 嵌入式关联

  • 简单 IPC 场景：配置进程向监控进程发送命令
  • 嵌入式 shell 脚本与 C 程序通信
  • 注意：FIFO 无消息边界，不适合高频交易消息

代码示例

#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>

#define SERVER_FIFO "/tmp/hft_server.fifo"

void run_server(void) {
    mkfifo(SERVER_FIFO, 0666);
    int fd = open(SERVER_FIFO, O_RDONLY);
    /* 保持 writer 防止 EOF */
    int dummy = open(SERVER_FIFO, O_WRONLY | O_NONBLOCK);

    char buf[4096];
    ssize_t n;
    while ((n = read(fd, buf, sizeof(buf))) > 0) {
        buf[n] = '\0';
        printf("Request: %s", buf);
    }
    close(fd);
    close(dummy);
    unlink(SERVER_FIFO);
}

void run_client(const char *msg) {
    int fd = open(SERVER_FIFO, O_WRONLY);
    write(fd, msg, strlen(msg));
    close(fd);
}

要点

半双工；无消息边界；仅本机；匿名仅亲缘；无优先级。复杂并发 → socket / mq / shm+sync。

HFT / 嵌入式关联

路线 要点
嵌入式 半双工/无边界/仅本机/匿名仅亲缘/无优先级，复杂并发要 socket/mq/shm
HFT 字节流无消息边界+内核拷贝，实时数据用 shm+信号量或 AF_UNIX

代码示例

#include <stdio.h>
#include <unistd.h>
/* 管道与 FIFO 的局限性 */
int main(void) {
    /* 1. pipe 只能用于有亲缘关系的进程 (fork 继承 fd) */
    /* 2. FIFO 可用于无关进程, 但只在本机 */
    /* 3. 管道是字节流, 无消息边界 */
    /* 4. 管道缓冲区有限 (通常 64KB) */

    int fd[2];
    pipe(fd);

    /* 字节流: 两次 write 可能被一次 read 读出 */
    write(fd[1], "AAA", 3);
    write(fd[1], "BBB", 3);

    char buf[16];
    ssize_t n = read(fd[0], buf, sizeof(buf));
    buf[n] = '\0';
    printf("Read %zd bytes: %s (no message boundary!)\n", n, buf);

    /* 查看管道缓冲区大小 */
    long buf_size = fpathconf(fd[0], _PC_PIPE_BUF);
    printf("PIPE_BUF (atomic write): %ld\n", buf_size);

    /* 管道容量: 通常 64KB (可查询) */
    int pipe_size = fcntl(fd[0], F_GETPIPE_SZ);
    printf("Pipe capacity: %d bytes\n", pipe_size);

    close(fd[0]); close(fd[1]);
    printf("\nLimitations: byte stream, no boundary, same machine only\n");
    return 0;
}

44.10

Semantics of read() and write() on Pipes and FIFOs

本章：TLPI 第 44 章

本节讲什么

管道和 FIFO 上 read()/write() 的特殊语义：原子性、PIPE_BUF 限制、EOF 行为、SIGPIPE 信号。

要点

PIPE_BUF 原子性

写入量 行为
≤ PIPE_BUF (通常 4096) 原子写入，不与其他进程交错
> PIPE_BUF 非原子，可能被交错

EOF 语义

情况 read 返回
有数据 实际字节数
无数据，所有 writer 已关闭 0 (EOF)
无数据，有 writer 打开 阻塞

write 阻塞

情况 write 行为
管道有空间 立即返回
管道满 阻塞（无 reader 时 SIGPIPE）
无 reader SIGPIPE 信号

SIGPIPE 处理

/* 方法1：忽略信号 */
signal(SIGPIPE, SIG_IGN);
/* 方法2：send + MSG_NOSIGNAL */
send(fd, buf, len, MSG_NOSIGNAL);
/* 方法3：检查 errno */
if (write(fd, buf, len) == -1 && errno == EPIPE) { /* reader gone */ }

非阻塞模式

条件 read 返回 write 返回
无数据/满 -1 EAGAIN -1 EAGAIN
有数据/空间 数据 写入数

管道容量

# 查看系统 PIPE_BUF
cat /proc/sys/fs/pipe-max-size   # 最大管道大小
getconf PIPE_BUF                  # 原子写入限制
参数 典型值 说明
PIPE_BUF 4096 原子写入最大字节数
管道总容量 65536 默认管道缓冲区
最大管道 1048576 可设最大值

HFT / 嵌式关联

  • 交易系统管道消息 ≤ PIPE_BUF 确保原子性
  • 信号处理：忽略 SIGPIPE 避免被杀
  • 嵌入式管道容量有限，需流量控制

代码示例

#include <unistd.h>
#include <signal.h>
#include <stdio.h>
#include <string.h>
#include <errno.h>

int safe_write_pipe(int fd, const char *msg, size_t len) {
    /* 忽略 SIGPIPE */
    signal(SIGPIPE, SIG_IGN);

    if (len > 4096) {
        fprintf(stderr, "Message > PIPE_BUF, not atomic\n");
    }
    ssize_t n = write(fd, msg, len);
    if (n < 0 && errno == EPIPE) {
        fprintf(stderr, "Reader closed pipe\n");
        return -1;
    }
    return n;
}

Exercises

本章：TLPI 第 44 章

本节讲什么

第 44 章的练习题与参考答案。

要点

练习 1：管道实现 popen

题目：使用 pipe + fork + exec 实现 popen() 功能：执行命令并获取其输出。

参考答案 
#include <unistd.h>
#include <stdio.h>
#include <fcntl.h>

FILE *my_popen(const char *cmd, const char *type) {
    int pfd[2];
    pipe2(pfd, O_CLOEXEC);

    pid_t pid = fork();
    if (pid == 0) {
        if (*type == 'r') {
            close(pfd[0]);
            dup2(pfd[1], STDOUT_FILENO);
        } else {
            close(pfd[1]);
            dup2(pfd[0], STDIN_FILENO);
        }
        execl("/bin/sh", "sh", "-c", cmd, NULL);
        _exit(127);
    }
    if (*type == 'r') {
        close(pfd[1]);
        return fdopen(pfd[0], "r");
    } else {
        close(pfd[0]);
        return fdopen(pfd[1], "w");
    }
}

练习 2：PIPE_BUF 原子性

题目：两个进程同时向同一 FIFO 写入 > PIPE_BUF 的数据，观察数据交错现象。

参考答案

当写入量 > PIPE_BUF（通常 4096）时，write 不保证原子性。两个进程的写操作可能交替执行，导致输出数据交错。解决方法：(1) 每条消息 ≤ PIPE_BUF 确保原子性；(2) 使用消息队列（有边界保证）；(3) 自定义协议（长度前缀+完整消息）；(4) 使用文件锁序列化写入。
