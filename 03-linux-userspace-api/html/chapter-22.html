代码示例

#include <stdio.h>
#include <signal.h>
#include <unistd.h>
#include <string.h>

/* Ch22 高级信号 — sigprocmask/sigpending/sigsuspend。
 * 演示信号阻塞 + 待决信号 + 可靠等待。
 * 编译: gcc -o ch22_demo ch22_demo.c */

void sigint_after_unblock(int sig) {
    const char *m = "Caught SIGINT after unblock!\n";
    write(STDOUT_FILENO, m, strlen(m));
    _exit(0);
}

int main(void) {
    sigset_t block_set, old_set;
    sigemptyset(&block_set);
    sigaddset(&block_set, SIGINT);
    sigprocmask(SIG_BLOCK, &block_set, &old_set);
    printf("SIGINT blocked for 5 seconds. Try Ctrl+C...\n");

    /* 在阻塞期间，信号变为"待决"但不递送 */
    sleep(5);

    /* 检查待决信号 */
    sigset_t pending;
    sigpending(&pending);
    if (sigismember(&pending, SIGINT))
        printf("SIGINT is pending (was sent while blocked)\n");
    else
        printf("No SIGINT pending\n");

    /* 解除阻塞: 待决的 SIGINT 立即递送 */
    printf("Unblocking SIGINT...\n");
    /* 先注册处理器，否则解除阻塞后默认终止 */
    signal(SIGINT, sigint_after_unblock);

    sigprocmask(SIG_SETMASK, &old_set, NULL);

    /* 如果之前有待决 SIGINT，此处会立即被中断 */
    pause();
    return 0;
}

22.1

`pause()`

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：本章首页 - 下一节：22.2 sigsuspend()（核心）

本节讲什么

int pause(void); / 永远 -1 / EINTR：等到某信号被捕获并跑完 handler / 竞态： if (!flag) pause(); — 检查与休眠之间信号已到并置位 → 永久睡死。 pause 不能原子地「放行某信号 + 休眠」。

要点

int pause(void);   /* 永远 -1 / EINTR：等到某信号被捕获并跑完 handler */

竞态： if (!flag) pause(); — 检查与休眠之间信号已到并置位 → 永久睡死。
pause 不能原子地「放行某信号 + 休眠」。

HFT / 嵌入式关联

路线 要点
嵌入式 if(!flag) pause() 存在「检查与休眠之间信号已到」竞态会永久睡死，嵌入式 watchdog 进程尤其危险；正式代码用 sigsuspend
HFT pause 是阻塞休眠，HFT 关键路径绝不能用它等事件（醒来延迟不可控）；事件通知走 epoll/eventfd 轮询，pause 只出现在非实时管理线程

代码示例

#include <signal.h>
#include <unistd.h>
#include <stdio.h>
void handler(int sig) { printf("唤醒\n"); }
int main(void) {
    struct sigaction sa = {0};
    sa.sa_handler = handler;
    sigaction(SIGUSR1, &sa, NULL);
    printf("pause 等待信号...\n");
    pause();  /* 阻塞直到收到信号 */
    printf("pause 返回\n");
    return 0;
}

22.2

`sigsuspend()`（核心）

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.1 pause() - 下一节：22.3 同步等待：sigwait 族

本节讲什么

int sigsuspend(const sigset_t mask); 原子步骤： 1. 将线程掩码换成* mask

要点

int sigsuspend(const sigset_t *mask);

原子步骤：

  1. 将线程掩码换成 mask
  2. 休眠等信号
  3. 捕获并跑完 handler 后，恢复调用前掩码
  4. 返回 -1 / EINTR

安全范式：

/* 1) 先 SIG_BLOCK 目标信号 */
/* 2) wait_mask = 当前掩码，但去掉要等的信号 */
while (!got_signal)
    sigsuspend(&wait_mask);
sigprocmask + pause sigsuspend
改掩码+睡 非原子 原子

Demo：code/sigsuspend_wait.c

HFT / 嵌入式关联

路线 要点
嵌入式 sigsuspend 原子「放行目标信号 + 休眠」修掉 pause 竞态，是嵌入式等待异步信号的正确原语；配合「先 BLOCK、wait_mask 去掉目标」范式
HFT 虽原子但仍阻塞休眠，HFT 主循环不用它；仅当信号是唯一唤醒源的管理线程可用，否则混用 epoll+signalfd 更可控

代码示例

#include <stdio.h>
#include <signal.h>
#include <unistd.h>
/* sigsuspend: 原子地"解除阻塞+等待信号" */
static volatile sig_atomic_t got = 0;

void handler(int sig) { got = 1; }

int main(void) {
    signal(SIGUSR1, handler);

    sigset_t mask, old;
    sigemptyset(&mask);
    sigaddset(&mask, SIGUSR1);
    sigprocmask(SIG_BLOCK, &mask, &old);

    printf("SIGUSR1 blocked. Sending to self...\n");
    raise(SIGUSR1);  /* 变为待决 */

    /* 错误方式: 先解除阻塞再 pause 有竞态
     * sigprocmask(SIG_UNBLOCK, &mask, NULL);
     * pause();  // 信号可能在 pause 之前递送!
     */

    /* 正确方式: sigsuspend 原子操作 */
    sigset_t empty;
    sigemptyset(&empty);
    sigsuspend(&empty);  /* 原子: 解除所有阻塞 + 等待 */

    printf("got = %d\n", got);
    sigprocmask(SIG_SETMASK, &old, NULL);
    return 0;
}

22.3

同步等待：`sigwait` 族

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.2 sigsuspend()（核心） - 下一节：22.4 实时信号与 sigqueue

本节讲什么

不跑 handler，从 pending 里同步取出信号。 int sigwait(const sigset_t set, int sig); int sigwaitinfo(const sigset_t set, siginfo_t info);

要点

不跑 handler，从 pending 里同步取出信号。

int sigwait(const sigset_t *set, int *sig);
int sigwaitinfo(const sigset_t *set, siginfo_t *info);
int sigtimedwait(const sigset_t *set, siginfo_t *info,
                 const struct timespec *timeout);

多线程推荐：

  1. 关心的信号在所有线程上先阻塞
  2. 专用线程循环 sigwaitinfo
  3. 处理逻辑可 malloc/printf/加锁

前提：目标信号必须已阻塞，否则可能先异步进 handler。

Demo：code/sigwaitinfo_loop.c

HFT / 嵌入式关联

路线 要点
嵌入式 sigwaitinfo 把信号转成同步队列消费，处理逻辑可随意 malloc/加锁、无 async-safe 限制，是嵌入式多线程处理信号的推荐模型
HFT 这是 HFT 处理信号的首选——专用线程 sigwaitinfo 消费信号，主循环零信号中断、零 async-safe 约束；代价是信号须先阻塞，避免误进 handler

代码示例

#include <stdio.h>
#include <signal.h>
#include <unistd.h>
#include <sys/wait.h>
/* sigwait: 同步等待信号 (不用信号处理器) */
int main(void) {
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGUSR1);
    sigaddset(&mask, SIGUSR2);
    sigprocmask(SIG_BLOCK, &mask, NULL);  /* 必须先阻塞 */

    /* 子进程发信号 */
    pid_t pid = fork();
    if (pid == 0) {
        sleep(1);
        kill(getppid(), SIGUSR1);
        sleep(1);
        kill(getppid(), SIGUSR2);
        _exit(0);
    }

    /* sigwait: 同步等待, 不需要 handler */
    int sig;
    for (int i = 0; i < 2; i++) {
        sigwait(&mask, &sig);
        printf("sigwait returned signal %d\n", sig);
    }

    waitpid(pid, NULL, 0);
    printf("Done (no signal handler needed!)\n");
    return 0;
}

22.4

实时信号与 `sigqueue`

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.3 同步等待：sigwait 族 - 下一节：22.5 备用信号栈

本节讲什么

int sigqueue(pid_t pid, int sig, const union sigval value); 标准信号不排队；实时信号可排队，小号优先。 Demo：code/sigqueue_rt.c

要点

int sigqueue(pid_t pid, int sig, const union sigval value);
vs kill
可带 sigval（int 或 ptr） 跨进程 勿传指针（地址空间不同）
可排队 队列满则失败
接收端 SA_SIGINFO，读 si_value

标准信号不排队；实时信号可排队，小号优先。

Demo：code/sigqueue_rt.c

HFT / 嵌入式关联

路线 要点
嵌入式 sigqueue 带 int 值且排队，嵌入式进程间可传少量带编号控制消息（通道号/事件 ID）；跨进程勿传指针（地址空间不同）
HFT 实时信号虽排队带值，但递送仍是异步中断，HFT 不用它做高频数据通路；仅作低频控制信令（切模式/降级），数据走共享内存 + eventfd

代码示例

#include <signal.h>
#include <unistd.h>
#include <stdio.h>
void handler(int sig, siginfo_t *si, void *ctx) {
    printf("收到 sig=%d, 值=%d\n", sig, si->si_value.sival_int);
}
int main(void) {
    struct sigaction sa = {0};
    sa.sa_sigaction = handler;
    sa.sa_flags = SA_SIGINFO;
    sigaction(SIGUSR1, &sa, NULL);

    union sigval sv;
    sv.sival_int = 42;
    sigqueue(getpid(), SIGUSR1, sv);  /* 带数据的信号 */
    return 0;
}

22.5

备用信号栈

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.4 实时信号与 sigqueue - 下一节：22.6 prctl（Linux）

本节讲什么

int sigaltstack(const stack_t ss, stack_t old_ss); / sa_flags |= SA_ONSTACK / 默认栈耗尽时仍要跑 SIGSEGV handler → 预分配独立栈（大小常用 SIGSTKSZ）。

要点

int sigaltstack(const stack_t *ss, stack_t *old_ss);
/* sa_flags |= SA_ONSTACK */

默认栈耗尽时仍要跑 SIGSEGV handler → 预分配独立栈（大小常用 SIGSTKSZ）。

HFT / 嵌入式关联

路线 要点
嵌入式 栈溢出（SIGSEGV）时默认栈已不可用，备用栈让 handler 仍能跑起来记录现场，嵌入式排障内存越界/栈耗尽时关键
HFT 栈溢出会瞬间打死交易进程，SA_ONSTACK 的 SIGSEGV handler 能在崩溃前留现场（写 core/日志），是 HFT 事后定责的兜底手段；代价是预分配一块固定内存

代码示例

#include <signal.h>
#include <stdio.h>
void handler(int sig) {
    char buf[64];
    int n = snprintf(buf, sizeof(buf), "alt stack handler sig=%d\n", sig);
    write(STDOUT_FILENO, buf, n);
}
int main(void) {
    /* 分配备用信号栈 (栈溢出时仍能执行 handler) */
    stack_t ss;
    ss.ss_sp = malloc(SIGSTKSZ);
    ss.ss_size = SIGSTKSZ;
    ss.ss_flags = 0;
    sigaltstack(&ss, NULL);

    struct sigaction sa = {0};
    sa.sa_handler = handler;
    sa.sa_flags = SA_ONSTACK;  /* 在备用栈上执行 */
    sigaction(SIGSEGV, &sa, NULL);

    printf("备用信号栈已设置 (SA_ONSTACK)\n");
    free(ss.ss_sp);
    return 0;
}

22.6

`prctl`（Linux）

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.5 备用信号栈 - 下一节：22.7 EINTR 再强调

本节讲什么

prctl(PR_SET_PDEATHSIG, sig); / 父死则本进程收 sig / 感知直接父进程死亡；常见于 worker 跟随父退出。 ---

要点

prctl(PR_SET_PDEATHSIG, sig);  /* 父死则本进程收 sig */

感知直接父进程死亡；常见于 worker 跟随父退出。

HFT / 嵌入式关联

路线 要点
嵌入式 PR_SET_PDEATHSIG 让 worker 在父进程死后自动收信号退出，是嵌入式守护进程防止孤儿残留的简洁做法
HFT 策略 worker 绑定父进程生命周期，父崩了 worker 跟着退，避免「半死」交易进程继续下单；注意它感知的是直接父进程，中间层转手要重设

代码示例

#include <sys/prctl.h>
#include <unistd.h>
#include <stdio.h>
int main(void) {
    /* PR_SET_PDEATHSIG: 父进程退出时收到信号 */
    prctl(PR_SET_PDEATHSIG, SIGTERM);
    printf("父退出时本进程将收到 SIGTERM\n");

    /* PR_SET_NAME: 设置线程名 */
    prctl(PR_SET_NAME, "my_process", 0, 0, 0);
    char name[16] = {0};
    prctl(PR_GET_NAME, name);
    printf("线程名: %s\n", name);
    return 0;
}

22.7

`EINTR` 再强调

本章：TLPI 第 22 章 — Signals: Advanced Features - 上一节：22.6 prctl（Linux） - 下一节：22.8 三大等待对比

本节讲什么

不靠 SA_RESTART 包打天下；connect/accept/sleep/epoll_wait 等行为因调用而异 → 显式重试。 ---

要点

不靠 SA_RESTART 包打天下；connect/accept/sleep/epoll_wait 等行为因调用而异 → 显式重试。

HFT / 嵌入式关联

路线 要点
嵌入式 connect/accept/sleep/epoll_wait 的 EINTR 行为因调用而异，嵌入式代码不能统一假定 SA_RESTART，逐调用显式重试才稳妥
HFT epoll_wait 被信号打断返回 EINTR 若不重试会漏掉 I/O 事件，直接造成行情/成交延迟；HFT 对 epoll_wait 必须显式重试，这是延迟敏感系统的常见坑

代码示例

#include <signal.h>
#include <unistd.h>
#include <stdio.h>
void handler(int sig) { printf("handler\n"); }
int main(void) {
    struct sigaction sa = {0};
    sa.sa_handler = handler;
    sa.sa_flags = 0;  /* 无 SA_RESTART */
    sigaction(SIGUSR1, &sa, NULL);

    printf("sleep 5 秒...\n");
    unsigned remain = sleep(5);
    printf("sleep 剩余 %u 秒 (被信号中断)\n", remain);

    /* 处理 EINTR 的标准模式:
     * while (read(fd, buf, n) < 0 && errno == EINTR) continue;
     */
    return 0;
}

方式 模型 评价
pause 异步 + 睡 有竞态，正式代码少用
sigsuspend 异步 + 原子睡 修竞态；仍受 async-safe 约束
sigwaitinfo 同步取 pending 无 async 限制；信号须先阻塞

HFT / 嵌入式关联

路线 要点
嵌入式 三者决定 handler 模型——pause 有竞态、sigsuspend 修竞态但受 async-safe 约束、sigwaitinfo 转同步最省心，嵌入式按「是否多线程」选
HFT 结论明确：HFT 一律 sigwaitinfo（同步消费、零 async-safe 限制、主循环无中断）；pause/sigsuspend 的异步+休眠模型只适合非实时管理线程

代码示例

#include <sys/wait.h>
#include <unistd.h>
#include <stdio.h>
int main(void) {
    pid_t pid = fork();
    if (pid == 0) { _exit(7); }

    int status;
    /* wait: 阻塞等任意子进程 */
    wait(&status);
    if (WIFEXITED(status))
        printf("wait: 子进程退出码=%d\n", WEXITSTATUS(status));

    pid = fork();
    if (pid == 0) { _exit(3); }
    /* waitpid: 可指定 PID, 可非阻塞 */
    waitpid(pid, &status, 0);
    printf("waitpid: 退出码=%d\n", WEXITSTATUS(status));

    /* waitid: 最灵活, 可指定哪些状态变化 */
    printf("\nwait: 阻塞等任意子进程\n");
    printf("waitpid: 指定 PID, WNOHANG 非阻塞\n");
    printf("waitid: 最灵活, 指定状态类型\n");
    return 0;
}

22.9

使用掩码等待信号：sigsuspend()

本章：TLPI 第 22 章 — 高级信号处理 - 上一节：22.8 系统调用中断与 SA_RESTART - 下一节：22.10 同步等待信号

本节讲什么

sigsuspend() — 原子地"解除信号阻塞 + 等待信号"，消除 pause() 的竞态。

要点

核心问题：pause() 的竞态

/* 竞态: check 和 pause 之间信号可能到达 */
if (ready == 0)
    pause();  /* 如果信号在 if 之后、pause 之前到达 → 永远阻塞 */

sigsuspend 解决方案

int sigsuspend(const sigset_t *mask);
  • 原子操作：将进程信号掩码替换为 mask，然后等待信号
  • 信号处理函数返回后，掩码恢复为调用前的值
  • 总是返回 -1，errno = EINTR

正确的临界区等待模式

sigset_t mask, oldmask;
sigemptyset(&mask);
sigaddset(&mask, SIGINT);

/* 1. 阻塞 SIGINT */
sigprocmask(SIG_BLOCK, &mask, &oldmask);

/* 2. 临界区 — SIGINT 被阻塞 */
while (!ready) {
    /* 3. 原子地: 解除 SIGINT 阻塞 + 等待 */
    sigsuspend(&oldmask);  /* oldmask 不含 SIGINT */
}

/* 4. 恢复原掩码 */
sigprocmask(SIG_SETMASK, &oldmask, NULL);

与 pause 的对比

pause() sigsuspend()
原子性 无（check + wait 分离） 有（unblock + wait 原子）
竞态 有 无
生产使用 不推荐 推荐

HFT / 嵌入式关联

路线 要点
嵌入式 sigsuspend 是信号同步等待的标准做法，无竞态
HFT 用 sigsuspend 等待配置重载信号（SIGHUP），保证不丢失事件

代码示例

#include <stdio.h>
#include <signal.h>
#include <unistd.h>
#include <string.h>
/* sigsuspend: 原子等待 SIGINT, 无竞态 */
volatile sig_atomic_t got_sigint = 0;
void handler(int sig) { got_sigint = 1; }

int main(void) {
    struct sigaction sa;
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = handler;
    sigaction(SIGINT, &sa, NULL);

    sigset_t intmask, oldmask;
    sigemptyset(&intmask);
    sigaddset(&intmask, SIGINT);

    /* 阻塞 SIGINT */
    sigprocmask(SIG_BLOCK, &intmask, &oldmask);

    printf("waiting for SIGINT (race-free)...\n");
    while (!got_sigint) {
        /* oldmask 不含 SIGINT → 原子解除阻塞 + 等待 */
        sigsuspend(&oldmask);
    }
    printf("got SIGINT\n");

    sigprocmask(SIG_SETMASK, &oldmask, NULL);
    return 0;
}

22.10

同步等待信号

本章：TLPI 第 22 章 — 高级信号处理 - 上一节：22.9 sigsuspend - 下一节：22.11 通过文件描述符获取信号

本节讲什么

sigwaitinfo() / sigtimedwait() — 同步等待信号，不跑 handler，直接在调用线程中接收。

要点

API

int sigwaitinfo(const sigset_t *set, siginfo_t *info);  /* 无限等待 */
int sigtimedwait(const sigset_t *set, siginfo_t *info,
                 const struct timespec *timeout);        /* 带超时 */

核心行为

  • 等待 set 中的任意信号
  • 信号被 消费（不会触发 handler）
  • info 返回信号详情（si_signo, si_pid, si_uid, si_value）
  • sigwaitinfo 成功返回信号编号，失败返回 -1

与 sigsuspend 的对比

sigsuspend sigwaitinfo
模式 异步（handler） 同步（直接返回）
handler 需要 不需要
siginfo 不提供 提供
适用 必须跑 handler 纯事件等待

典型用法

sigset_t set;
sigemptyset(&set);
sigaddset(&set, SIGUSR1);
sigaddset(&set, SIGUSR2);

/* 阻塞这些信号（sigwaitinfo 要求信号被阻塞） */
sigprocmask(SIG_BLOCK, &set, NULL);

siginfo_t info;
int sig = sigwaitinfo(&set, &info);
printf("got signal %d from pid %d\n", sig, info.si_pid);

注意事项

  • 调用前必须阻塞目标信号，否则行为未定义
  • 多线程中只有一个线程会收到信号
  • sigtimedwait 超时返回 -1，errno = EAGAIN

HFT / 嵌入式关联

路线 要点
嵌入式 sigwaitinfo 避免 handler 的异步干扰，简化逻辑
HFT 专用信号线程用 sigwaitinfo 收信号，主线程零中断

代码示例

#include <stdio.h>
#include <signal.h>
#include <string.h>
#include <time.h>
/* sigwaitinfo / sigtimedwait: 同步等待信号 */
int main(void) {
    sigset_t set;
    sigemptyset(&set);
    sigaddset(&set, SIGUSR1);
    sigaddset(&set, SIGUSR2);

    /* 必须先阻塞 */
    sigprocmask(SIG_BLOCK, &set, NULL);

    /* 自发信号 */
    raise(SIGUSR1);

    /* sigwaitinfo: 无限等待 */
    siginfo_t info;
    int sig = sigwaitinfo(&set, &info);
    printf("sigwaitinfo got: signal=%d, pid=%d\n", sig, info.si_pid);

    /* sigtimedwait: 带超时 */
    raise(SIGUSR2);
    struct timespec ts = { .tv_sec = 1, .tv_nsec = 0 };
    sig = sigtimedwait(&set, &info, &ts);
    if (sig > 0) {
        printf("sigtimedwait got: signal=%d\n", sig);
    } else {
        printf("sigtimedwait timeout or error\n");
    }
    return 0;
}

22.11

通过文件描述符获取信号

本章：TLPI 第 22 章 — 高级信号处理 - 上一节：22.10 同步等待信号 - 下一节：22.12 进程间信号通信

本节讲什么

signalfd() — Linux 特有，将信号转换为可从文件描述符读取的事件，融入 epoll/poll 事件循环。

要点

API

int signalfd(int fd, const sigset_t *mask, int flags);
/* fd=-1 创建新 fd; fd=已有 fd 则更新 mask */
/* flags: SFD_NONBLOCK, SFD_CLOEXEC */

核心行为

  1. 创建 signalfd 前必须阻塞目标信号（否则信号会触发 handler）
  2. 信号到达后，read(fd, &siginfo, sizeof(siginfo)) 读取 signalfd_siginfo 结构
  3. 可融入 epoll/poll，实现统一事件循环

signalfd_siginfo 关键字段

字段 含义
ssi_signo 信号编号
ssi_code 来源
ssi_pid 发送进程 PID
ssi_uid 发送进程 UID
ssi_int / ssi_ptr sigqueue 携带的数据

优势

  • 信号变成普通 I/O 事件，可融入 epoll
  • 无 handler，无线程切换，纯同步处理
  • 多个信号共用一个 fd

限制

  • Linux 专有（非 POSIX）
  • 信号必须先阻塞
  • 标准信号仍不排队（实时信号可排队）

HFT / 嵌入式关联

路线 要点
嵌入式 signalfd + epoll 统一事件循环，简化架构
HFT 信号融入 epoll，避免 handler 中断热路径；SIGHUP/SIGTERM 优雅处理

代码示例

#include <stdio.h>
#include <signal.h>
#include <sys/signalfd.h>
#include <unistd.h>
#include <string.h>
/* signalfd: 通过文件描述符读取信号 */
int main(void) {
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGINT);
    sigaddset(&mask, SIGUSR1);

    /* 必须先阻塞信号 */
    sigprocmask(SIG_BLOCK, &mask, NULL);

    /* 创建 signalfd */
    int sfd = signalfd(-1, &mask, 0);
    if (sfd < 0) { perror("signalfd"); return 1; }

    /* 自发信号 */
    raise(SIGUSR1);

    /* 读取信号事件 */
    struct signalfd_siginfo fdsi;
    ssize_t n = read(sfd, &fdsi, sizeof(fdsi));
    if (n == sizeof(fdsi)) {
        printf("got signal %d from pid %d\n",
               fdsi.ssi_signo, fdsi.ssi_pid);
    }

    close(sfd);
    return 0;
}

22.12

进程间信号通信

本章：TLPI 第 22 章 — 高级信号处理 - 上一节：22.11 通过文件描述符获取信号 - 下一节：22.13 早期信号 API

本节讲什么

用信号在进程间传递数据：sigqueue() 发送 + SA_SIGINFO 接收。

要点

sigqueue API

int sigqueue(pid_t pid, int sig, const union sigval value);
  • union sigval 可携带 int 或 void*
  • 需配合 SA_SIGINFO 的 sa_sigaction 接收
  • 实时信号可排队，标准信号不排队

数据传递

/* 发送方 */
union sigval sv;
sv.sival_int = 42;
sigqueue(pid, SIGRTMIN, sv);

/* 接收方 (SA_SIGINFO handler) */
void handler(int sig, siginfo_t *si, void *ctx) {
    printf("got data: %d\n", si->si_value.sival_int);
}

实时信号 vs 标准信号

标准信号 (1-31) 实时信号 (SIGRTMIN-SIGRTMAX)
排队 不排队 排队（保证顺序）
数据 不保证携带 携带 sigval
多次发送 只传递一次 每次都传递
编号 固定 可用范围 SIGRTMIN+0..N

限制

  • SIGRTMIN/SIGRTMAX 在 glibc 中预留了部分编号给 pthread 库
  • 可用范围通常从 SIGRTMIN+2 开始
  • 信号数据量很小（一个 int 或一个指针）

替代方案对比

方案 数据量 顺序保证 性能
sigqueue 1 int/ptr 实时信号有 中（内核态）
管道 任意 有 中（拷贝）
共享内存 任意 需自同步 高（零拷贝）
eventfd 1 uint64 有 高

HFT / 嵌入式关联

路线 要点
嵌入式 sigqueue 携带少量数据适合简单通知，但管道更通用
HFT 不用信号传数据——延迟不可控，用共享内存 + eventfd

代码示例

#include <stdio.h>
#include <signal.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>
/* sigqueue + SA_SIGINFO: 进程间带数据信号通信 */
void handler(int sig, siginfo_t *si, void *ctx) {
    printf("child got signal %d, data=%d, from pid=%d\n",
           sig, si->si_value.sival_int, si->si_pid);
}

int main(void) {
    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 注册 handler, 等待信号 */
        struct sigaction sa;
        memset(&sa, 0, sizeof(sa));
        sa.sa_sigaction = handler;
        sa.sa_flags = SA_SIGINFO;
        sigaction(SIGRTMIN, &sa, NULL);

        printf("child waiting (pid=%d)...\n", getpid());
        sleep(2);
        _exit(0);
    }

    /* 父进程: 发送带数据的实时信号 */
    sleep(1);
    union sigval sv;
    sv.sival_int = 99;
    sigqueue(pid, SIGRTMIN, sv);
    printf("parent sent SIGRTMIN with data=99\n");

    wait(NULL);
    return 0;
}

22.13

早期信号 API（System V 和 BSD）

本章：TLPI 第 22 章 — 高级信号处理 - 上一节：22.12 进程间信号通信 - 下一节：22.14 总结

本节讲什么

signal()、sigset()、sighold()、sigrelse()、sigpause() — 遗留 API，了解即可，生产代码用 sigaction。

要点

signal() 的问题

行为 BSD System V
handler 触发后 保持 重置为 SIG_DFL
系统调用中断 自动重启 返回 EINTR

→ 同一代码在 BSD 和 SysV 上行为不同，不可移植

遗留 API 一览

函数 来源 替代
signal(sig, handler) BSD/SysV sigaction
sigset(sig, handler) SysV sigaction
sighold(sig) SysV sigprocmask(SIG_BLOCK)
sigrelse(sig) SysV sigprocmask(SIG_UNBLOCK)
sigignore(sig) SysV sigaction(SIG_IGN)
sigpause(sig) BSD sigsuspend
sigvec BSD sigaction
sigmask BSD sigaddset + sigemptyset

结论

  • 所有遗留 API 都不要用，统一用 POSIX sigaction + sigprocmask + sigsuspend
  • 阅读老代码时能理解即可

HFT / 嵌入式关联

路线 要点
嵌入式 遗留嵌入式代码可能用 signal()，迁移时替换为 sigaction
HFT 无关联，新项目不应出现这些 API

代码示例

#include <stdio.h>
#include <signal.h>
/* 遗留 API 演示 — 不要在生产代码中使用 */
int main(void) {
    /* signal(): 行为不确定 (BSD vs SysV) */
    /* 在 Linux/glibc 上, signal() 默认是 BSD 语义 (不会重置) */
    /* 但不可移植! */
    signal(SIGINT, SIG_IGN);
    printf("SIGINT ignored (via legacy signal())\n");

    /* 正确做法: sigaction */
    struct sigaction sa;
    sa.sa_handler = SIG_IGN;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    sigaction(SIGINT, &sa, NULL);
    printf("SIGINT ignored (via sigaction)\n");

    return 0;
}

代码自测

1. sigsuspend 如何消除 pause 的竞态？

sigsuspend 将"解除信号阻塞"和"等待信号"合并为一个原子操作。这样信号不可能在解除阻塞和进入等待之间到达并丢失。

2. signalfd 的前置条件是什么？

目标信号必须先被 sigprocmask 阻塞，否则信号会触发 handler 而非到达 signalfd。创建后通过 read 读取 signalfd_siginfo 结构。

22.15

练习题

本章：TLPI 第 22 章 — 高级信号处理

本节讲什么

原书第 22 章练习题要点。

1. 用 sigsuspend 实现一个原子的 "等待 SIGUSR1 或 SIGUSR2" 逻辑 
sigset_t mask, oldmask;
sigemptyset(&mask);
sigaddset(&mask, SIGUSR1);
sigaddset(&mask, SIGUSR2);
sigprocmask(SIG_BLOCK, &mask, &oldmask);

/* oldmask 不含 SIGUSR1/SIGUSR2 → 原子解除 + 等待 */
sigsuspend(&oldmask);

sigprocmask(SIG_SETMASK, &oldmask, NULL);
2. sigwaitinfo 和 sigsuspend 有什么本质区别？

sigsuspend 是异步模式：信号到达后触发 handler，handler 返回后 sigsuspend 返回。sigwaitinfo 是同步模式：不触发 handler，信号直接被消费，sigwaitinfo 返回信号编号和 siginfo。sigwaitinfo 不需要注册 handler。

3. 为什么 signalfd 要求信号先被阻塞？

如果不阻塞，信号到达时内核会触发 handler（如果有注册）或执行默认动作（如终止进程）。阻塞后信号进入 pending 队列，signalfd 的 read 能从中取出。这保证了信号只被一种机制消费。
