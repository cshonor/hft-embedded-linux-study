章节目标

mprotect · mlock* · mincore · madvise；页对齐约束；实时锁定与 advice 语义。

对比速记

调用 作用 权限
mprotect 改 r/w/x 通常不需 root
mlock* 禁 swap ulimit / 特权
mincore 驻留快照 否
madvise 访问提示 否

陷阱

  1. 未页对齐 → EINVAL
  2. mprotect 越权 → EACCES
  3. mlock 非引用计数叠加
  4. mincore 不可靠同步
  5. MADV_DONTNEED + PRIVATE 丢数据（Linux）
  6. MCL_FUTURE 耗尽锁定额

背诵卡

# 要点
1 四件套：protect / lock / mincore / advise
2 页对齐；JIT 写完再 RX
3 mlock 抗 swap；ulimit；exec 解锁
4 mincore 仅快照
5 madvise 非强制；DONTNEED 移植坑
6 大量 mlock 挤内存

代码示例

#include <stdio.h>
#include <sys/mman.h>
#include <unistd.h>
#include <string.h>

/* Ch50 虚拟内存 — mprotect/madvise/mincore。
 * 演示修改内存保护属性 + 内存使用建议。
 * 编译: gcc -o ch50_demo ch50_demo.c */

int main(void) {
    long pagesize = sysconf(_SC_PAGESIZE);
    printf("Page size: %ld bytes\n", pagesize);

    /* 分配页对齐的内存 */
    char *mem = mmap(NULL, pagesize,
                     PROT_READ | PROT_WRITE,
                     MAP_PRIVATE | MAP_ANONYMOUS,
                     -1, 0);
    if (mem == MAP_FAILED) { perror("mmap"); return 1; }

    /* 写入数据 */
    strcpy(mem, "writable data");
    printf("Wrote: %s\n", mem);

    /* mprotect: 改为只读 */
    if (mprotect(mem, pagesize, PROT_READ) == 0) {
        printf("Memory is now read-only\n");
        printf("Reading still works: %s\n", mem);
        /* 以下写入会触发 SIGSEGV:
         * mem[0] = 'X';  -- 段错误!
         */
        printf("Writing would cause SIGSEGV now\n");
    }

    /* mprotect: 改为不可读写 */
    mprotect(mem, pagesize, PROT_NONE);
    printf("Memory is now inaccessible (PROT_NONE)\n");
    /* 读或写都会触发 SIGSEGV */

    /* madvise: 给内核内存使用提示 */
    char *mem2 = mmap(NULL, pagesize * 4,
                      PROT_READ | PROT_WRITE,
                      MAP_PRIVATE | MAP_ANONYMOUS,
                      -1, 0);
    madvise(mem2, pagesize * 4, MADV_SEQUENTIAL);  /* 顺序访问 */
    printf("madvise(MADV_SEQUENTIAL): hint for sequential access\n");

    munmap(mem, pagesize);
    munmap(mem2, pagesize * 4);
    return 0;
}

50.2

`mprotect`

本章：TLPI 第 50 章 — Virtual Memory Operations - 上一节：50.1 总览 - 下一节：50.3 mlock / mlockall

本节讲什么

int mprotect(void addr, size_t length, int prot); PROT_NONE|READ|WRITE|EXEC。违例 → SIGSEGV*。 不能超过后备权限（只读文件映射 → 不可 PROT_WRITE → EACCES）。

要点

int mprotect(void *addr, size_t length, int prot);

PROT_NONE|READ|WRITE|EXEC。违例 → SIGSEGV。
不能超过后备权限（只读文件映射 → 不可 PROT_WRITE → EACCES）。
JIT：W 写码 → 改成 R|X；现代常强制 W^X（不可同时 W+X）。

代码示例


/* mprotect 修改内存保护属性 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <signal.h>
#include <setjmp.h>
#include <sys/mman.h>
#include <unistd.h>

static jmp_buf jmpbuf;
static void handler(int sig) { longjmp(jmpbuf, 1); }

int main(void) {
    size_t len = 4096;
    char *p = mmap(NULL, len, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) { perror("mmap"); exit(1); }
    strcpy(p, "hello");

    /* 改为只读 */
    if (mprotect(p, len, PROT_READ) < 0) { perror("mprotect"); exit(1); }
    printf("mprotect -> PROT_READ OK\n");

    /* 尝试写入会触发 SIGSEGV */
    signal(SIGSEGV, handler);
    if (setjmp(jmpbuf) == 0) {
        p[0] = 'X';  /* 这里会触发 SIGSEGV */
    } else {
        printf("caught SIGSEGV: write to read-only page\n");
    }

    /* 改回可写 */
    if (mprotect(p, len, PROT_READ | PROT_WRITE) < 0) { perror("mprotect 2"); exit(1); }
    p[0] = 'X';
    printf("mprotect -> PROT_READ|PROT_WRITE OK, wrote 'X'\n");

    munmap(p, len);
    return 0;
}

50.3

`mlock` / `mlockall`

本章：TLPI 第 50 章 — Virtual Memory Operations - 上一节：50.2 mprotect - 下一节：50.4 mincore

本节讲什么

int mlock(const void addr, size_t length); int munlock(const void addr, size_t length); int mlockall(int flags); / MCL_CURRENT | MCL_FUTURE /

要点

int mlock(const void *addr, size_t length);
int munlock(const void *addr, size_t length);
int mlockall(int flags);   /* MCL_CURRENT | MCL_FUTURE */
int munlockall(void);

禁 swap，减缺页 IO 延迟（实时 / HFT）。
- 用户有 ulimit -l 上限；超限失败
- 同区多次 mlock，一次 munlock 即解（非叠加计数）
- fork 继承锁；exec 全解锁
- MCL_FUTURE：后续分配也锁 → 易耗尽限额
慎用：挤占 RAM → 他进程 OOM。

路线 要点
嵌入式 mlock / mlockall 在嵌入式系统的应用场景

代码示例


/* mlock / munlock 锁定内存页，防止被换出到 swap */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/resource.h>
#include <unistd.h>

int main(void) {
    size_t len = 4096;
    char *p = mmap(NULL, len, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) { perror("mmap"); exit(1); }

    /* mlock 锁定这一页，确保不换出（HFT 延迟敏感场景常用） */
    if (mlock(p, len) < 0) {
        perror("mlock (need CAP_IPC_LOCK or RLIMIT_MEMLOCK)");
    } else {
        printf("mlock: %zu bytes locked in RAM\n", len);
    }

    strcpy(p, "locked memory");
    printf("data: %s\n", p);

    /* munlock 解锁 */
    munlock(p, len);
    printf("munlock: unlocked\n");

    /* mlockall 锁定进程所有页 */
    if (mlockall(MCL_CURRENT | MCL_FUTURE) < 0) {
        perror("mlockall");
    } else {
        printf("mlockall: all current+future pages locked\n");
        munlockall();
    }

    munmap(p, len);
    return 0;
}

代码示例


/* mincore 查询页是否驻留在物理内存 */
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    long page_size = sysconf(_SC_PAGESIZE);
    int num_pages = 4;
    size_t len = (size_t)page_size * num_pages;

    char *p = mmap(NULL, len, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) { perror("mmap"); exit(1); }

    /* 触摸前两页使其驻留 */
    p[0] = 'a';
    p[page_size] = 'b';

    /* mincore 返回每页驻留状态 */
    unsigned char vec[num_pages];
    if (mincore(p, len, vec) < 0) { perror("mincore"); exit(1); }

    for (int i = 0; i < num_pages; i++) {
        printf("page %d: %s\n", i, (vec[i] & 1) ? "resident" : "not resident");
    }

    munmap(p, len);
    return 0;
}

要点

int madvise(void *addr, size_t length, int advice);

内核可忽略。常用：

advice 含义
MADV_NORMAL 默认
MADV_RANDOM 少预读
MADV_SEQUENTIAL 激进预读、用过可早收
MADV_WILLNEED 提前装入
MADV_DONTNEED Linux PRIVATE：丢本地修改；SHARED 多为丢页缓存语义

代码示例


/* madvise 给内核内存使用提示 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    size_t len = 4096 * 4;
    char *p = mmap(NULL, len, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) { perror("mmap"); exit(1); }

    /* MADV_WILLNEED：预期近期访问，内核可预读 */
    madvise(p, len, MADV_WILLNEED);
    printf("MADV_WILLNEED: hint to prefetch\n");

    /* 写入数据 */
    memset(p, 'X', len);

    /* MADV_DONTNEED：近期不再访问，内核可回收页 */
    madvise(p, len, MADV_DONTNEED);
    printf("MADV_DONTNEED: pages can be reclaimed\n");
    printf("after MADV_DONTNEED, content: %d (zeroed if reclaimed)\n", p[0]);

    /* MADV_SEQUENTIAL：顺序访问模式 */
    madvise(p, len, MADV_SEQUENTIAL);
    printf("MADV_SEQUENTIAL: sequential access hint\n");

    munmap(p, len);
    return 0;
}

练习题

练习 1：mmap 文件映射

编写程序，使用 mmap() 将文件映射到内存，通过指针读写文件内容，最后 munmap() 解除映射。

练习 2：mprotect 改变内存保护

分配一页内存（mmap），先用 PROT_READ 映射，然后尝试写入（应该 SIGSEGV），再用 mprotect 改为 PROT_READ|PROT_WRITE，写入成功。

练习 3：mlockall 延迟测试

编写程序，对比 mlockall 前后访问大数组（1GB）的延迟分布。

参考答案

练习 1 参考代码

#include "tlpi_hdr.h"
#include <sys/mman.h>
#include <fcntl.h>

int main(int argc, char *argv[]) {
    if (argc != 2) usageErr("usage: %s <file>\n", argv[0]);

    int fd = open(argv[1], O_RDWR);
    if (fd == -1) errExit("open");

    off_t len = lseek(fd, 0, SEEK_END);

    char *map = mmap(NULL, len, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (map == MAP_FAILED) errExit("mmap");

    /* 通过指针读写 */
    printf("first 10 bytes: %.10s\n", map);
    map[0] = 'X';   /* 写入会同步到文件（MAP_SHARED） */

    munmap(map, len);
    close(fd);
    return 0;
}

练习 2 参考代码

#include "tlpi_hdr.h"
#include <sys/mman.h>
#include <signal.h>
#include <setjmp.h>

static sigjmp_buf env;

void handler(int sig) {
    siglongjmp(env, 1);
}

int main(void) {
    signal(SIGSEGV, handler);

    char *p = mmap(NULL, 4096, PROT_READ, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    if (p == MAP_FAILED) errExit("mmap");

    /* 尝试写入 → SIGSEGV */
    if (sigsetjmp(env, 1) == 0) {
        p[0] = 'X';
        printf("write succeeded (unexpected)\n");
    } else {
        printf("write failed with SIGSEGV (expected)\n");
    }

    /* 改为可读写 */
    if (mprotect(p, 4096, PROT_READ | PROT_WRITE) == -1)
        errExit("mprotect");

    p[0] = 'X';  /* 现在成功了 */
    printf("write after mprotect: %c\n", p[0]);

    munmap(p, 4096);
    return 0;
}

练习 3 答案

#include <sys/mman.h>
#include <time.h>
#include <stdio.h>
#include <string.h>

#define SIZE (1UL << 30)  /* 1GB */
#define ITERATIONS 1000

void bench_access(char *buf, const char *label) {
    struct timespec start, end;
    long long max_ns = 0, min_ns = 1LL<<62, total_ns = 0;

    for (int i = 0; i < ITERATIONS; i++) {
        size_t offset = (size_t)rand() * 4096 % SIZE;
        clock_gettime(CLOCK_MONOTONIC, &start);
        buf[offset] = 'x';  /* 触发页错误如果未映射 */
        clock_gettime(CLOCK_MONOTONIC, &end);

        long long ns = (end.tv_sec - start.tv_sec) * 1000000000LL
                     + (end.tv_nsec - start.tv_nsec);
        if (ns > max_ns) max_ns = ns;
        if (ns < min_ns) min_ns = ns;
        total_ns += ns;
    }
    printf("%s: avg=%lldns min=%lldns max=%lldns\n",
           label, total_ns/ITERATIONS, min_ns, max_ns);
}

int main(void) {
    char *buf = mmap(NULL, SIZE, PROT_READ|PROT_WRITE,
                     MAP_PRIVATE|MAP_ANONYMOUS, -1, 0);

    /* 未 mlockall：访问可能触发缺页 */
    bench_access(buf, "before mlockall");

    mlockall(MCL_CURRENT | MCL_FUTURE);
    bench_access(buf, "after mlockall");

    munmap(buf, SIZE);
    return 0;
}

预期：mlockall 前 max 延迟可能达毫秒级（硬页错误），mlockall 后 max 延迟降到微秒级。

HFT / 嵌入式关联

  • mlockall：HFT 必须锁定所有内存防止换页
  • mmap：HFT 用文件映射共享行情数据
  • Huge Pages：mmap(MAP_HUGETLB) 减少 TLB miss
  • 嵌入式：无 MMU 系统不支持 mprotect
