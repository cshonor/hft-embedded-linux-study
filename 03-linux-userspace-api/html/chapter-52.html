章节目标

mq_open/send/receive/close/unlink；属性；优先级；mq_notify；epoll；vs SysV mq。

vs System V mq

SysV POSIX
句柄 非 fd fd / epoll
筛选 mtype 仅最高优先级
非阻塞 每次 IPC_NOWAIT 描述符 O_NONBLOCK
删除 IPC_RMID unlink+引用计数
通知 / 超时 弱 notify + timed*

新事件驱动优先 POSIX；要按类型挑消息仍看 SysV / 自建协议。

陷阱

  1. receive 缓冲 < msgsize → 失败
  2. notify 一次性 + 仅空→首条
  3. notify ∥ epoll 竞争
  4. timed 用绝对时间
  5. 名不能 /a/b
  6. setattr 改不了容量

背诵卡

# 要点
1 优先级收；无 type 筛选
2 mqd_t 可 epoll（Linux）
3 unlink / 全 close 销毁
4 notify：空→首条；一次失效
5 maxmsg/msgsize 创建固定
6 常 -lrt；timed 绝对时钟

代码示例

#include <stdio.h>
#include <mqueue.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

/* Ch52 POSIX 消息队列 — mq_open/mq_send/mq_receive/mq_notify。
 * 演示父子进程通过 POSIX 消息队列通信。
 * 编译: gcc -o ch52_demo ch52_demo.c -lrt */

#define MQ_NAME "/ch52_demo"
#define MSG_SIZE 64

int main(void) {
    struct mq_attr attr = {
        .mq_flags = 0,
        .mq_maxmsg = 10,
        .mq_msgsize = MSG_SIZE,
        .mq_curmsgs = 0
    };

    mqd_t mqd = mq_open(MQ_NAME, O_CREAT | O_RDWR, 0644, &attr);
    if (mqd < 0) { perror("mq_open"); return 1; }

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 发送消息 */
        unsigned prio;
        for (int i = 1; i <= 3; i++) {
            char msg[MSG_SIZE];
            snprintf(msg, MSG_SIZE, "Message %d", i);
            mq_send(mqd, msg, strlen(msg) + 1, i);  /* 优先级 = i */
            printf("Child: sent (prio=%d) %s\n", i, msg);
        }
        _exit(0);
    }

    sleep(1);

    /* 父进程: 接收消息 (高优先级先出) */
    for (int i = 0; i < 3; i++) {
        char buf[MSG_SIZE];
        unsigned prio;
        ssize_t n = mq_receive(mqd, buf, MSG_SIZE, &prio);
        if (n > 0)
            printf("Parent: received (prio=%u) %s\n", prio, buf);
    }

    waitpid(pid, NULL, 0);
    mq_close(mqd);
    mq_unlink(MQ_NAME);
    return 0;
}

代码示例


/* POSIX 消息队列基本概念：mq_open + 属性查询 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>

int main(void) {
    /* 创建消息队列，设置属性 */
    struct mq_attr attr = {
        .mq_flags = 0,        /* 0 = 阻塞模式 */
        .mq_maxmsg = 10,      /* 最大消息数 */
        .mq_msgsize = 256,    /* 单条消息最大字节数 */
        .mq_curmsgs = 0       /* 当前消息数（只读） */
    };

    mqd_t mq = mq_open("/concept_demo", O_CREAT | O_RDWR, 0644, &attr);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }
    printf("MQ created: /concept_demo\n");

    /* 查询当前属性 */
    struct mq_attr cur;
    mq_getattr(mq, &cur);
    printf("  flags=%ld, maxmsg=%ld, msgsize=%ld, curmsgs=%ld\n",
           cur.mq_flags, cur.mq_maxmsg, cur.mq_msgsize, cur.mq_curmsgs);

    /* 设置非阻塞模式 */
    mq_setattr(mq, &(struct mq_attr){ .mq_flags = O_NONBLOCK }, NULL);
    mq_getattr(mq, &cur);
    printf("after setattr: flags=%ld (O_NONBLOCK=%d)\n",
           cur.mq_flags, O_NONBLOCK);

    mq_close(mq);
    mq_unlink("/concept_demo");
    return 0;
}

代码示例


/* mq_send / mq_receive 完整 API：优先级排序 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>

int main(void) {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 64 };
    mqd_t mq = mq_open("/api_demo", O_CREAT | O_RDWR, 0644, &attr);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }

    /* 发送多条不同优先级的消息 */
    mq_send(mq, "low", 4, 1);
    mq_send(mq, "high", 5, 10);
    mq_send(mq, "mid", 4, 5);
    printf("sent 3 messages (prio 1, 10, 5)\n");

    /* 接收：高优先级先出 */
    char buf[64];
    unsigned int prio;
    ssize_t n;
    while ((n = mq_receive(mq, buf, sizeof(buf), &prio)) >= 0) {
        buf[n] = '\0';
        printf("received: '%s' (prio=%u)\n", buf, prio);
    }
    /* 输出顺序：high(10) -> mid(5) -> low(1) */

    mq_close(mq);
    mq_unlink("/api_demo");
    return 0;
}

52.3

fork / exec

本章：TLPI 第 52 章 — POSIX Message Queues - 上一节：52.2 API 要点 - 下一节：52.7 mq_notify（难点）

本节讲什么

要点
fork 复制 mqd_t；不继承 mq_notify
exec / 退出 自动 mq_close；清除 notify

HFT / 嵌入式关联

路线 要点
嵌入式 fork 复制 mqd_t 不继承 notify；exec 自动 close
HFT exec 后队列对象持久、描述符丢失，跨 exec 需重新 mq_open

代码示例


/* POSIX MQ 在 fork/exec 后的继承行为 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 64 };
    mqd_t mq = mq_open("/inherit_demo", O_CREAT | O_RDWR, 0644, &attr);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }

    /* 父进程发送消息 */
    mq_send(mq, "from parent", 12, 1);

    /* fork：子进程继承 mqd_t 描述符（同一队列引用） */
    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程可以用继承的 mq 接收 */
        char buf[64];
        unsigned int prio;
        ssize_t n = mq_receive(mq, buf, sizeof(buf), &prio);
        if (n >= 0) {
            buf[n] = '\0';
            printf("child received: '%s' (prio=%u)\n", buf, prio);
        }
        /* 注意：mqd_t 在 exec 后不保留（不是 fd） */
        _exit(0);
    }
    wait(NULL);

    /* exec 后描述符丢失，但队列对象持久存在 */
    printf("fork: mqd_t inherited; exec: mqd_t NOT inherited\n");
    printf("queue object persists until mq_unlink or reboot\n");

    mq_close(mq);
    mq_unlink("/inherit_demo");
    return 0;
}

52.4

Message Queue Attributes

本章：TLPI 第 52 章

本节讲什么

POSIX 消息队列的属性系统：mq_attr 结构、创建时设置属性、运行时查询属性，以及各属性对消息传递行为的影响。

要点

struct mq_attr

#include <mqueue.h>
struct mq_attr {
    long mq_flags;      /* 标志（0 或 O_NONBLOCK） */
    long mq_maxmsg;     /* 最大消息数 */
    long mq_msgsize;    /* 单条消息最大字节数 */
    long mq_curmsgs;    /* 当前消息数（只读） */
    /* 以下 Linux 特有 */
    long mq_sendwait;   /* 等待发送的进程数（只读） */
    long mq_recvwait;   /* 等待接收的进程数（只读） */
};

属性说明

属性 创建时可设 运行时可改 说明
mq_flags 否（总是0） 是（O_NONBLOCK） 阻塞/非阻塞
mq_maxmsg 是 否 队列容量（消息数）
mq_msgsize 是 否 单消息上限
mq_curmsgs 否 否 当前消息数（只读）

创建时设置

struct mq_attr attr = {
    .mq_flags = 0,
    .mq_maxmsg = 100,     /* 最多100条 */
    .mq_msgsize = 4096,   /* 每条最大4KB */
};
mqd_t mqd = mq_open("/myqueue", O_CREAT | O_RDWR, 0644, &attr);

查询属性

struct mq_attr attr;
mq_getattr(mqd, &attr);
printf("maxmsg=%ld msgsize=%ld curmsgs=%ld\n",
       attr.mq_maxmsg, attr.mq_msgsize, attr.mq_curmsgs);

修改非阻塞标志

/* 设为非阻塞 */
struct mq_attr attr;
mq_getattr(mqd, &attr);
attr.mq_flags |= O_NONBLOCK;
mq_setattr(mqd, &attr, NULL);

/* 恢复阻塞 */
attr.mq_flags &= ~O_NONBLOCK;
mq_setattr(mqd, &attr, NULL);

mq_setattr 范围

属性 mq_setattr 可改
mq_flags 是（仅 O_NONBLOCK）
mq_maxmsg 否
mq_msgsize 否
mq_curmsgs 否

vs SysV MQ 限制对比

特性 SysV MQ POSIX MQ
消息大小 系统 MSGMAX 限制 创建时自定
队列大小 系统 MSGMNB 限制 创建时自定
调整 msgctl(IPC_SET) 仅 flags
非阻塞 IPC_NOWAIT 标志 O_NONBLOCK 属性

HFT / 嵌入式关联

  • HFT 消息队列配置大消息容量（行情快照 > 8KB）
  • 嵌入式根据内存预算精确设置 mq_maxmsg 和 mq_msgsize
  • 非阻塞模式用于不阻塞交易主循环的日志队列

代码示例

#include <mqueue.h>
#include <stdio.h>
#include <string.h>

int main() {
    /* 创建消息队列 */
    struct mq_attr attr = {
        .mq_maxmsg = 10,
        .mq_msgsize = 8192,
    };
    mqd_t mqd = mq_open("/hft_test", O_CREAT | O_RDWR, 0644, &attr);
    if (mqd == (mqd_t)-1) { perror("mq_open"); return 1; }

    /* 查询属性 */
    struct mq_attr cur;
    mq_getattr(mqd, &cur);
    printf("Flags:     %ld\n", cur.mq_flags);
    printf("Max msg:   %ld\n", cur.mq_maxmsg);
    printf("Msg size:  %ld\n", cur.mq_msgsize);
    printf("Current:   %ld\n", cur.mq_curmsgs);

    /* 发送消息 */
    char msg[8192] = "Hello POSIX MQ";
    mq_send(mqd, msg, strlen(msg), 1);

    /* 再查属性（curmsgs 变化） */
    mq_getattr(mqd, &cur);
    printf("After send, current: %ld\n", cur.mq_curmsgs);

    /* 设置非阻塞 */
    cur.mq_flags |= O_NONBLOCK;
    mq_setattr(mqd, &cur, NULL);

    mq_close(mqd);
    mq_unlink("/hft_test");
    return 0;
}

代码自测

mq_maxmsg 和 mq_msgsize 在创建后能否修改？

不能。mq_maxmsg 和 mq_msgsize 只能在 mq_open(O_CREAT) 创建时设置，之后不可修改。mq_setattr 只能修改 mq_flags（O_NONBLOCK）。如果需要不同容量，只能创建新队列。这是设计限制：内核在创建时分配固定大小的缓冲区，修改需要重新分配，不安全。

52.5

Exchanging Messages

本章：TLPI 第 52 章

本节讲什么

POSIX 消息队列的消息发送和接收 API：mq_send()/mq_receive() 和带超时的 mq_timedsend()/mq_timedreceive()。支持消息优先级。

要点

核心 API

#include <mqueue.h>
int mq_send(mqd_t mqd, const char *msg, size_t len,
            unsigned int prio);
ssize_t mq_receive(mqd_t mqd, char *msg, size_t len,
                   unsigned int *prio);

int mq_timedsend(mqd_t mqd, const char *msg, size_t len,
                  unsigned int prio, const struct timespec *abstime);
ssize_t mq_timedreceive(mqd_t mqd, char *msg, size_t len,
                          unsigned int *prio, const struct timespec *abstime);

消息优先级

prio 值 行为
0 普通消息，FIFO 顺序
> 0 高优先级，prio 大的先出
同 prio FIFO 顺序

接收顺序：按 prio 降序排列，相同 prio 的 FIFO。

发送行为

条件 行为
队列未满 立即发送，返回 0
队列满 + 阻塞 阻塞直到有空间
队列满 + O_NONBLOCK 返回 -1, errno=EAGAIN
队列满 + timedsend 超时返回 -1, errno=ETIMEDOUT

接收行为

条件 行为
有消息 立即接收最高优先级消息
无消息 + 阻塞 阻塞直到消息到达
无消息 + O_NONBLOCK 返回 -1, errno=EAGAIN
无消息 + timedreceive 超时返回 -1, errno=ETIMEDOUT

超时参数

/* 绝对时间（CLOCK_REALTIME） */
struct timespec ts;
clock_gettime(CLOCK_REALTIME, &ts);
ts.tv_sec += 5;  /* 5秒后超时 */
mq_timedreceive(mqd, buf, bufsize, &prio, &ts);

消息大小限制

限制 来源
单条消息大小 mq_msgsize（创建时设置）
接收缓冲区 必须 ≥ mq_msgsize，否则返回 -1 EMSGSIZE
系统上限 /proc/sys/fs/mqueue/msg_max（消息数）

vs SysV MQ

特性 SysV MQ POSIX MQ
筛选 mtype 匹配 不可筛选（按优先级取）
优先级 mtype 近似 真正的 prio 参数
超时 不支持 mq_timedsend/recv
数据拷贝 2次（用户→内核→用户） 2次

HFT / 嵌入式关联

  • HFT 订单队列：高优先级订单先处理
  • 超时控制：交易消息超时返回错误，不无限等待
  • 行情分发：按优先级处理重要行情更新
  • 嵌入式消息缓冲：精确控制队列容量

代码示例

#include <mqueue.h>
#include <time.h>
#include <string.h>
#include <stdio.h>

#define MAX_MSG 4096

/* 发送带优先级的消息 */
int send_prio(mqd_t mqd, const char *data, size_t len, int prio) {
    return mq_send(mqd, data, len, prio);
}

/* 带超时接收消息 */
int recv_timeout(mqd_t mqd, char *buf, int *prio, int sec) {
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);
    ts.tv_sec += sec;

    unsigned int p;
    ssize_t n = mq_timedreceive(mqd, buf, MAX_MSG, &p, &ts);
    if (n < 0) return -1;
    if (prio) *prio = p;
    return (int)n;
}

int main() {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = MAX_MSG };
    mqd_t mqd = mq_open("/hft_orders", O_CREAT | O_RDWR, 0644, &attr);

    /* 发送不同优先级消息 */
    send_prio(mqd, "normal order", 12, 0);
    send_prio(mqd, "urgent order", 12, 10);
    send_prio(mqd, "cancel order", 12, 5);

    /* 接收（按优先级排序） */
    char buf[MAX_MSG];
    int prio;
    while (recv_timeout(mqd, buf, &prio, 5) > 0) {
        printf("[prio=%d] %s\n", prio, buf);
    }
    /* 输出:
       [prio=10] urgent order
       [prio=5]  cancel order
       [prio=0]  normal order
    */

    mq_close(mqd);
    mq_unlink("/hft_orders");
    return 0;
}

代码自测

mq_timedreceive 的超时参数为什么是绝对时间而非相对时间？

绝对时间（struct timespec 是从 epoch 的绝对时刻）避免了相对时间的一个问题：如果被信号中断后重试，相对时间需要重新计算已过时间。绝对时间只需重新传入原始的 abstime，因为超时时刻是固定的。这和 pthread_cond_timedwait 的设计一致。

52.6

Message Notification

本章：TLPI 第 52 章

本节讲什么

POSIX 消息队列的异步通知机制：当队列从空变为非空时通知进程。支持两种通知方式：信号（SIGEV_SIGNAL）和线程（SIGEV_THREAD）。

要点

mq_notify() 函数

#include <mqueue.h>
int mq_notify(mqd_t mqd, const struct sigevent *sevp);
// 返回：0 成功，-1 失败

struct sigevent

struct sigevent {
    int    sigev_notify;           /* 通知方式 */
    int    sigev_signo;            /* 信号编号 */
    union sigval sigev_value;     /* 附加数据 */
    void  *sigev_notify_function; /* 线程回调函数 */
    void  *sigev_notify_attribute; /* 线程属性 */
    pid_t  sigev_notify_thread_id;  /* 目标线程 ID */
};

三种通知方式

方式 常量 效果
信号通知 SIGEV_SIGNAL 发送 sigev_signo 信号
线程通知 SIGEV_THREAD 创建新线程调用回调
无通知 SIGEV_NONE 取消通知
定向信号 SIGEV_THREAD_ID 发送到特定线程（Linux 扩展）

通知触发条件

队列空 → 收到消息 → 队列变非空 → 触发通知
条件 是否通知
空队列收到消息 是
非空队列收到消息 否
接收消息后队列变空 → 再收到 是
多次空→非空 仅通知一次

单次通知规则

1. 注册通知
2. 队列空 → 收到消息 → 通知触发 → 通知取消
3. 需要再次通知 → 重新注册 mq_notify
4. 如果消息仍积压 → 不再通知

信号通知模式

/* 注册信号通知 */
struct sigevent sev = {0};
sev.sigev_notify = SIGEV_SIGNAL;
sev.sigev_signo = SIGUSR1;
sev.sigev_value.sival_ptr = (void *)mqd;  /* 附加数据 */
mq_notify(mqd, &sev);

/* 信号处理函数中必须重新注册 */
static void handler(int sig, siginfo_t *si, void *uc) {
    mqd_t mqd = (mqd_t)si->si_value.sival_ptr;
    /* 消耗消息 */
    mq_receive(mqd, buf, size, &prio);
    /* 重新注册通知 */
    mq_notify(mqd, &sev);
}

线程通知模式

void notify_thread(union sigval sv) {
    mqd_t mqd = (mqd_t)sv.sival_ptr;
    mq_receive(mqd, buf, size, &prio);
    /* 重新注册 */
    struct sigevent sev = {0};
    sev.sigev_notify = SIGEV_THREAD;
    sev.sigev_value.sival_ptr = (void *)mqd;
    sev.sigev_notify_function = notify_thread;
    mq_notify(mqd, &sev);
}

/* 初始注册 */
struct sigevent sev = {0};
sev.sigev_notify = SIGEV_THREAD;
sev.sigev_value.sival_ptr = (void *)mqd;
sev.sigev_notify_function = notify_thread;
mq_notify(mqd, &sev);

通知与 epoll 结合 (Linux)

Linux 上 POSIX MQ 的 fd 可用于 epoll（通过扩展机制）：

mqd_t mqd = mq_open(...);
int fd = (int)mqd;  /* Linux: mqd_t 可用作 fd */
/* epoll_ctl(epfd, EPOLL_CTL_ADD, (int)mqd, &ev); */
/* 注意：可移植性不保证 */

HFT / 嵌入式关联

  • HFT 行情队列：线程通知模式处理行情更新，不阻塞主循环
  • 订单队列监控：信号通知在订单到达时唤醒处理线程
  • 嵌入式事件驱动：消息通知替代轮询，降低 CPU 占用
  • 注意单次通知规则：通知后需重新注册

代码示例

#include <mqueue.h>
#include <signal.h>
#include <string.h>
#include <stdio.h>
#include <unistd.h>

static mqd_t g_mqd;

/* 线程回调 */
static void on_message(union sigval sv) {
    char buf[4096];
    unsigned prio;
    ssize_t n;

    /* 消耗所有积压消息 */
    while ((n = mq_receive(g_mqd, buf, sizeof(buf), &prio)) >= 0) {
        printf("Received [prio=%u]: %.*s\n", prio, (int)n, buf);
    }

    /* 重新注册通知 */
    struct sigevent sev = {0};
    sev.sigev_notify = SIGEV_THREAD;
    sev.sigev_value.sival_ptr = (void *)g_mqd;
    sev.sigev_notify_function = on_message;
    mq_notify(g_mqd, &sev);
}

int main() {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 4096 };
    g_mqd = mq_open("/hft_notify", O_CREAT | O_RDWR, 0644, &attr);

    /* 注册线程通知 */
    struct sigevent sev = {0};
    sev.sigev_notify = SIGEV_THREAD;
    sev.sigev_value.sival_ptr = (void *)g_mqd;
    sev.sigev_notify_function = on_message;
    mq_notify(g_mqd, &sev);

    /* 主线程做其他工作 */
    /* 从其他进程发送消息会触发 on_thread 回调 */
    pause();

    mq_close(g_mqd);
    mq_unlink("/hft_notify");
    return 0;
}

代码自测

为什么 mq_notify 通知只触发一次？如何实现持续通知？

POSIX 设计为单次通知以避免通知风暴：如果队列持续收到消息，反复通知会淹没进程。实现持续通知的方法：在每次通知回调中处理完消息后，重新调用 mq_notify 注册下一次通知。这是标准模式：(1) 收到通知 → (2) 消耗所有积压消息 → (3) 重新注册 mq_notify → (4) 队列再次空→非空时触发下一次通知。

52.7

`mq_notify`（难点）

本章：TLPI 第 52 章 — POSIX Message Queues - 上一节：52.3 fork / exec - 下一节：52.8 epoll（Linux）

本节讲什么

空队列收到第一条消息时通知一次。 1. 每队列同时只一个注册者（新盖旧） 2. 触发后自动注销 → 须再 mq_notify

要点

空队列收到第一条消息时通知一次。

  1. 每队列同时只一个注册者（新盖旧）
  2. 触发后自动注销 → 须再 mq_notify
  3. 队列非空时再发不触发

模式：SIGEV_SIGNAL / SIGEV_THREAD / SIGEV_NONE。
标准环：注册 → 等信号 → 立刻重注册 → 循环 mq_receive 抽干。

HFT / 嵌入式关联

路线 要点
嵌入式 空队列收首条消息通知一次，触发后自动注销须重注册；SIGEV_SIGNAL/THREAD
HFT 异步到达通知避免轮询；标准环注册→等信号→立刻重注册→mq_receive 抽干

代码示例


/* mq_notify 异步通知：消息到达时发信号 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <signal.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>
#include <unistd.h>

static mqd_t g_mq;
static void notify_handler(int sig) {
    char buf[64];
    unsigned int prio;
    ssize_t n = mq_receive(g_mq, buf, sizeof(buf), &prio);
    if (n >= 0) {
        buf[n] = '\0';
        printf("notify: received '%s' (prio=%u)\n", buf, prio);
    }
}

int main(void) {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 64 };
    g_mq = mq_open("/notify_demo", O_CREAT | O_RDWR | O_NONBLOCK, 0644, &attr);
    if (g_mq == (mqd_t)-1) { perror("mq_open"); exit(1); }

    /* 注册异步通知 */
    struct sigevent sev = {
        .sigev_notify = SIGEV_SIGNAL,
        .sigev_signo = SIGUSR1,
    };
    if (mq_notify(g_mq, &sev) < 0) { perror("mq_notify"); exit(1); }
    signal(SIGUSR1, notify_handler);
    printf("notify registered, waiting for message...\n");

    /* fork 子进程发消息 */
    pid_t pid = fork();
    if (pid == 0) {
        sleep(1);  /* 等父进程注册完毕 */
        mqd_t child_mq = mq_open("/notify_demo", O_WRONLY);
        mq_send(child_mq, "async msg", 10, 1);
        mq_close(child_mq);
        _exit(0);
    }
    wait(NULL);

    /* 等信号处理完成 */
    sleep(1);
    printf("done\n");

    mq_close(g_mq);
    mq_unlink("/notify_demo");
    return 0;
}

/* POSIX MQ 配合 epoll（Linux 扩展：mqd_t 可作为 fd） */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>
#include <sys/epoll.h>
#include <unistd.h>

int main(void) {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 64 };
    mqd_t mq = mq_open("/epoll_demo", O_CREAT | O_RDWR | O_NONBLOCK, 0644, &attr);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }

    /* Linux 上 mqd_t 本身就是 fd，可直接加入 epoll */

    struct epoll_event ev = { .events = EPOLLIN, .data.fd = (int)mq };
    epoll_ctl(epfd, EPOLL_CTL_ADD, (int)mq, &ev);
    printf("MQ fd added to epoll\n");

    /* fork 子进程发消息 */
    pid_t pid = fork();
    if (pid == 0) {
        sleep(1);
        mqd_t cm = mq_open("/epoll_demo", O_WRONLY);
        mq_send(cm, "epoll msg", 10, 1);
        mq_close(cm);
        _exit(0);
    }

    /* 等待 epoll 通知 */
    struct epoll_event events[4];
    int n = epoll_wait(epfd, events, 4, 5000);
    if (n > 0) {
        printf("epoll returned %d event(s)\n", n);
        char buf[64];
        unsigned int prio;
        ssize_t len = mq_receive(mq, buf, sizeof(buf), &prio);
        if (len >= 0) {
            buf[len] = '\0';
            printf("received: '%s' (prio=%u)\n", buf, prio);
        }
    }

    wait(NULL);
    close(epfd);
    mq_close(mq);
    mq_unlink("/epoll_demo");
    return 0;
}

练习 1：优先级订单队列

题目：实现一个订单处理队列，高优先级订单先处理。模拟 3 个不同优先级的订单发送和接收。

参考答案 
#include <mqueue.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    int order_id;
    int quantity;
    char symbol[8];
} order_t;

int main() {
    struct mq_attr attr = { .mq_maxmsg = 100, .mq_msgsize = sizeof(order_t) };
    mqd_t mqd = mq_open("/orders", O_CREAT | O_RDWR, 0644, &attr);

    /* 发送不同优先级订单 */
    order_t o1 = { .order_id=1, .quantity=100, .symbol="AAPL" };
    order_t o2 = { .order_id=2, .quantity=200, .symbol="GOOG" };
    order_t o3 = { .order_id=3, .quantity=50,  .symbol="TSLA" };

    mq_send(mqd, (char*)&o1, sizeof(o1), 0);   /* 普通 */
    mq_send(mqd, (char*)&o2, sizeof(o2), 10);  /* 紧急 */
    mq_send(mqd, (char*)&o3, sizeof(o3), 5);   /* 高 */

    /* 接收：按优先级排序 */
    order_t recv;
    unsigned prio;
    while (mq_receive(mqd, (char*)&recv, sizeof(recv), &prio) != -1) {
        printf("[pri=%u] ID=%d %s x%d\n",
               prio, recv.order_id, recv.symbol, recv.quantity);
    }
    /* 输出:
       [pri=10] ID=2 GOOG x200    ← 紧急先出
       [pri=5]  ID=3 TSLA x50    ← 高优先级
       [pri=0]  ID=1 AAPL x100   ← 普通
    */

    mq_close(mqd);
    mq_unlink("/orders");
    return 0;
}

    }

    /* 重新注册 */
    struct sigevent sev = {0};
    sev.sigev_notify = SIGEV_THREAD;
    sev.sigev_value.sival_ptr = (void *)g_mqd;
    sev.sigev_notify_function = on_msg;
    mq_notify(g_mqd, &sev);
}

int main() {
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 4096 };
    g_mqd = mq_open("/notify_test", O_CREAT | O_RDWR, 0644, &attr);

    struct sigevent sev = {0};
    sev.sigev_notify = SIGEV_THREAD;
    sev.sigev_value.sival_ptr = (void *)g_mqd;
    sev.sigev_notify_function = on_msg;
    mq_notify(g_mqd, &sev);

    /* 主线程可以做其他工作 */
    pause();  /* 等待 */

    mq_close(g_mqd);
    mq_unlink("/notify_test");
    return 0;
}

信号通知 vs 线程通知：

方面 SIGEV_SIGNAL SIGEV_THREAD
回调环境 信号处理函数（限制多） 普通线程（可自由操作）
异步安全 只能调用异步安全函数 可调用任意函数
响应延迟 低（信号中断主线程） 需创建线程，稍高
复杂度 需管理信号掩码 线程管理简单
适用 简单唤醒 复杂处理逻辑

HFT 推荐 SIGEV_THREAD：可以在回调中直接做复杂操作（如解析行情、更新状态），不受信号安全限制。
