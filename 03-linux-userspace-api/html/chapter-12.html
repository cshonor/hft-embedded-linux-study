/proc 速查（背诵）

要什么 看哪里
内核版本 uname 或 /proc/version
内存 /proc/meminfo（或谨慎用 sysinfo）
CPU /proc/cpuinfo、get_nprocs
负载 /proc/loadavg
本进程状态 /proc/self/status
本进程映射 /proc/self/maps
本进程 fd /proc/self/fd/

代码示例

#include <stdio.h>
#include <sys/utsname.h>
#include <sys/sysinfo.h>
#include <unistd.h>

/* Ch12 系统与进程信息 — uname/sysinfo + /proc 文件系统。
 * 编译: gcc -o ch12_demo ch12_demo.c */

int main(void) {
    /* uname: 内核/系统信息 */
    struct utsname uts;
    if (uname(&uts) == 0) {
        printf("sysname:  %s\n", uts.sysname);
        printf("nodename: %s\n", uts.nodename);
        printf("release:  %s\n", uts.release);
        printf("version:  %s\n", uts.version);
        printf("machine:  %s\n", uts.machine);
    }

    /* sysinfo: 内存/uptime/负载 */
    struct sysinfo info;
    if (sysinfo(&info) == 0) {
        printf("\nuptime:      %ld sec\n", info.uptime);
        printf("total RAM:   %lu MB\n", info.totalram / 1024 / 1024);
        printf("free RAM:    %lu MB\n", info.freeram / 1024 / 1024);
        printf("load avg:    %.2f %.2f %.2f\n",
               info.loads[0] / 65536.0,
               info.loads[1] / 65536.0,
               info.loads[2] / 65536.0);
    }

    printf("\n/proc/self/status shows per-process info (try: cat /proc/self/status)\n");
    return 0;
}

`/proc` — Linux 核心特色（非 POSIX）

本章：TLPI 第 12 章 — System and Process Information - 上一节：12.1 uname() — POSIX - 下一节：12.3 读 /proc 编程要点

本节讲什么

虚拟文件系统：不占磁盘，内容由内核实时生成。 mount | grep proc Demo：code/proc_self_status.c · code/mini_ps.c

要点

虚拟文件系统：不占磁盘，内容由内核实时生成。
mount | grep proc

类 路径 内容
全局 /proc/xxx 整机信息
进程 /proc/[pid]/ 每进程一份；/proc/self = 当前进程

常用全局文件

路径 用途
/proc/meminfo MemTotal / MemFree / Buffers / Cached…
/proc/cpuinfo CPU 型号、核、缓存
/proc/version 内核版本字符串
/proc/loadavg 1/5/15 负载、运行中任务、最近 PID
/proc/sys/ 可调内核参数（sysctl）
/proc/sys/kernel/pid_max PID 上限
/proc/sys/fs/file-max 系统级文件句柄上限

/proc/[pid]/ 关键项

路径 用途
cmdline 启动命令；参数以 \0 分隔
status 易读：UID/GID、VmRSS、VmSize、线程数
stat 紧凑数字字段（ps 数据源之一）
statm 内存页数概览
maps 虚拟地址区域（库 / 堆 / 栈 / mmap）
fd/ 已打开 fd 的符号链接
cwd / exe 工作目录 / 可执行文件（符号链接）
environ 环境变量；\0 分隔

Demo：code/proc_self_status.c · code/mini_ps.c

代码示例

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

/* /proc 文件系统: 内核信息的文本接口 */
int main(void) {
    char buf[4096];

    /* /proc/version: 内核版本 */
    int fd = open("/proc/version", O_RDONLY);
    if (fd >= 0) {
        ssize_t n = read(fd, buf, sizeof(buf) - 1);
        if (n > 0) { buf[n] = '\0'; printf("内核: %s", buf); }
        close(fd);
    }

    /* /proc/loadavg: 系统负载 */
    fd = open("/proc/loadavg", O_RDONLY);
    if (fd >= 0) {
        ssize_t n = read(fd, buf, sizeof(buf) - 1);
        if (n > 0) { buf[n] = '\0'; printf("负载: %s", buf); }
        close(fd);
    }

    /* /proc/self/status: 当前进程信息 */
    fd = open("/proc/self/status", O_RDONLY);
    if (fd >= 0) {
        ssize_t n = read(fd, buf, sizeof(buf) - 1);
        if (n > 0) { buf[n] = '\0'; printf("进程状态:\n%s", buf); }
        close(fd);
    }
    return 0;
}

12.3 读 /proc 编程要点

本章：TLPI 第 12 章 — System and Process Information - 上一节：12.2 /proc — Linux 核心特色（非 POSIX） - 下一节：12.4 sysctl() 系统调用

本节讲什么

  1. 内容动态变，多次读可能不一致 2. 行格式 / 字段顺序无铁律，跨内核要容错 3. 权限：未必能读别人进程的部分文件

要点

  1. 内容动态变，多次读可能不一致
  2. 行格式 / 字段顺序无铁律，跨内核要容错
  3. 权限：未必能读别人进程的部分文件
  4. 按文本行解析，勿当巨型二进制一次吞
  5. 进程会退出 → 打开时可能已 ENOENT
  6. 不要 mmap /proc 文件（非普通磁盘文件）

HFT / 嵌入式关联

路线 要点
嵌入式 读 /proc 编程要点 在嵌入式系统的应用场景
HFT 读 /proc 编程要点 对低延迟和性能的影响

代码示例

#include <stdio.h>
#include <stdlib.h>

/* /proc/self/ 下的关键文件 */
int main(void) {
    char buf[4096];

    /* /proc/self/maps: 进程内存映射 */
    FILE *fp = fopen("/proc/self/maps", "r");
    if (fp) {
        printf("=== 内存映射 (前5行) ===\n");
        for (int i = 0; i < 5 && fgets(buf, sizeof(buf), fp); i++)
            printf("%s", buf);
        fclose(fp);
    }

    /* /proc/self/stat: 进程统计信息 */
    fp = fopen("/proc/self/stat", "r");
    if (fp) {
        printf("\n进程 stat: %s", fgets(buf, sizeof(buf), fp) ?: "");
        fclose(fp);
    }

    /* /proc/self/cmdline: 命令行参数 (null 分隔) */
    fp = fopen("/proc/self/cmdline", "r");
    if (fp) {
        printf("\n命令行: ");
        int c;
        while ((c = fgetc(fp)) != EOF)
            putchar(c == '\0' ? ' ' : c);
        putchar('\n');
        fclose(fp);
    }
    return 0;
}

int main(void) {

12.7 get_nprocs() — GNU

#include <sys/sysinfo.h>
#include <stdio.h>

int main(void) {
    /* get_nprocs: 可用 CPU 数 (GNU 扩展) */
    int nprocs = get_nprocs();
    int nprocs_conf = get_nprocs_conf();

    printf("配置的 CPU 数: %d\n", nprocs_conf);
    printf("可用的 CPU 数: %d\n", nprocs);

    long configured = sysconf(_SC_NPROCESSORS_CONF);
    printf("sysconf: online=%ld, configured=%ld\n", online, configured);
    return 0;
}

代码示例

#include <stdio.h>
#include <sys/utsname.h>
#include <sys/sysinfo.h>
#include <unistd.h>

/* 系统信息三来源:
 * 1. uname() → 内核名称/版本/架构
 * 2. /proc/ → 详细运行时信息 (maps, stat, status...)
 * 3. sysinfo() → 内存/负载/进程数概览
 */
int main(void) {
    struct utsname uts;
    uname(&uts);
    printf("uname: %s %s\n", uts.sysname, uts.machine);

    printf("sysconf OPEN_MAX=%ld, CPU=%ld\n",
           sysconf(_SC_OPEN_MAX),
           sysconf(_SC_NPROCESSORS_ONLN));

    struct sysinfo si;
    sysinfo(&si);
    printf("sysinfo: uptime=%lds, procs=%d\n", si.uptime, si.procs);
    return 0;
}

代码自测

1. /proc/self 是什么？和 /proc/getpid() 有什么区别？

/proc/self 是一个特殊的符号链接，始终指向当前进程的 /proc/[pid] 目录。它和 /proc/getpid() 指向同一目录，但 /proc/self 不需要先调用 getpid()，且在多线程程序中始终指向当前线程组。实际上 /proc/self 是一个内核生成的虚拟符号链接。

2. /proc/[pid]/maps 每行包含哪些信息？

格式：address perms offset dev inode pathname

00400000-00401000 r-xp 00000000 fd:00 12345  /usr/bin/myapp
00600000-00601000 r--p 00000000 fd:00 12345  /usr/bin/myapp
00601000-00602000 rw-p 00001000 fd:00 12345  /usr/bin/myapp
7f1234000000-7f1234021000 r-xp 00000000 fd:00 67890  /usr/lib/libc.so.6
  • address：虚拟地址范围
  • perms：r/w/x/p（p=private, s=shared）
  • offset：文件内偏移
  • dev：设备号
  • inode：inode 号
  • pathname：映射文件（或 [heap]/[stack]/[anon]）
3. sysinfo() 和读取 /proc/meminfo 哪个更好？

各有适用场景： - sysinfo()：一次系统调用获取结构化数据，开销小，适合程序内部监控 - /proc/meminfo：信息更详细（包含 Cached、Buffers、Slab 等细分项），适合需要精细分析 - HFT 监控通常用 sysinfo() 减少文件 I/O 开销；运维分析用 /proc/meminfo

练习 1 参考代码

#include "tlpi_hdr.h"
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
    unsigned long start, end;
    char perms[5];
    char path[256];
} map_entry_t;

int read_maps(pid_t pid) {
    char path[64];
    snprintf(path, sizeof(path), "/proc/%d/maps", pid);

    FILE *f = fopen(path, "r");
    if (!f) { perror("fopen"); return -1; }

    char line[512];
    int n = 0;
    while (fgets(line, sizeof(line), f)) {
        map_entry_t e;
        char offset[32], dev[32];
        long inode;

        /* 格式: start-end perms offset dev inode path */
        sscanf(line, "%lx-%lx %4s %s %s %ld %255[^\n]",
               &e.start, &e.end, e.perms, offset, dev, &inode, e.path);

        printf("%012lx-%012lx %s %s\n",
               e.start, e.end, e.perms,
               e.path[0] ? e.path : "[anon]");
        n++;
    }
    fclose(f);
    printf("total: %d mappings\n", n);
    return 0;
}

int main(int argc, char *argv[]) {
    if (argc != 2) usageErr("usage: %s <pid>\n", argv[0]);
    return read_maps(atoi(argv[1]));
}

#include <stdio.h>

int main(void) {
    FILE *f = fopen("/proc/self/status", "r");
    if (!f) { perror("fopen"); return 1; }

    char line[256];

            strncmp(line, "Threads:", 8) == 0 ||
            strncmp(line, "State:", 6) == 0) {
            printf("%s", line);
        }
    }
    fclose(f);
    return 0;
}
