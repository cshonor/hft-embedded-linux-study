代码示例

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
#include <string.h>
#include <signal.h>

/* Ch33 线程深入 — pthread_attr/线程栈/信号掩码。
 * 演示自定义线程属性: 栈大小 + 分离状态 + 信号掩码。
 * 编译: gcc -o ch33_demo ch33_demo.c -lpthread */

void *worker(void *arg) {
    int id = *(int *)arg;
    printf("Thread %d: running (stack custom sized)\n", id);
    /* 检查线程栈位置 */
    int local;
    printf("Thread %d: stack near %p\n", id, (void *)&local);
    return NULL;
}

int main(void) {
    pthread_attr_t attr;
    pthread_attr_init(&attr);

    /* 设置分离状态: 线程结束自动回收，不需 join */
    pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);

    /* 设置栈大小: 默认 8MB，改为 2MB */
    size_t stacksize;
    pthread_attr_getstacksize(&attr, &stacksize);
    printf("Default stack size: %zu bytes\n", stacksize);
    pthread_attr_setstacksize(&attr, 2 * 1024 * 1024);

    /* 设置线程不继承父进程的信号掩码 */
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGINT);
    pthread_attr_setsigmask_np(&attr, &mask);

    /* 创建线程 */
    int id = 42;
    pthread_t tid;
    pthread_create(&tid, &attr, worker, &id);

    pthread_attr_destroy(&attr);

    /* 分离状态线程不能 join，用 sleep 等待 */
    sleep(1);
    printf("Main: detached thread finished\n");
    return 0;
}

33.1

线程栈

本章：TLPI 第 33 章 — Threads: Further Details - 上一节：本章首页 - 下一节：33.2 线程与信号（重难点）

本节讲什么

每线程私有栈；默认大小受 RLIMIT_STACK 等影响。 pthread_attr_setstacksize / getstacksize pthread_attr_setstack / 自备栈，少用 /

要点

每线程私有栈；默认大小受 RLIMIT_STACK 等影响。

pthread_attr_setstacksize / getstacksize
pthread_attr_setstack          /* 自备栈，少用 */
sysconf(_SC_THREAD_STACK_MIN)

栈溢出 → SIGSEGV；海量线程吃地址空间 → 可调小栈；深递归 → 加大。

HFT / 嵌入式关联

路线 要点
嵌入式 每线程私有栈，默认大小受 RLIMIT_STACK；栈溢出 SIGSEGV
HFT 海量线程调小栈省地址空间；深递归加大防溢出；栈大小是内存预算一部分

代码示例

#include <pthread.h>
#include <stdio.h>
void *worker(void *arg) {
    char buf[1024];  /* 使用栈空间 */
    printf("栈变量地址: %p\n", (void *)buf);
    return NULL;
}
int main(void) {
    pthread_attr_t attr;
    pthread_attr_init(&attr);
    /* 设置栈大小 */
    pthread_attr_setstacksize(&attr, 256 * 1024);  /* 256KB */
    pthread_t tid;
    pthread_create(&tid, &attr, worker, NULL);
    pthread_join(tid, NULL);
    pthread_attr_destroy(&attr);
    printf("默认栈大小通常 8MB, HFT 可缩小减少开销\n");
    return 0;
}

33.2

线程与信号（重难点）

本章：TLPI 第 33 章 — Threads: Further Details - 上一节：33.1 线程栈 - 下一节：33.3 进程控制

本节讲什么

int pthread_sigmask(int how, const sigset_t set, sigset_t oldset); int pthread_kill(pthread_t thread, int sig); / 仅同进程内 / int sigwait(const sigset_t set, int sig); / 同步取信号 /

要点

进程级（共享） 线程级（私有）
sigaction 处置 / 默认动作 信号掩码（pthread_sigmask）
进程 pending（标准信号不排队） 线程 pending；递送目标线程 
int pthread_sigmask(int how, const sigset_t *set, sigset_t *oldset);
int pthread_kill(pthread_t thread, int sig);   /* 仅同进程内 */
int sigwait(const sigset_t *set, int *sig);    /* 同步取信号 */
投递
异步信号（SIGINT 等） 任选未阻塞该信号的线程
pthread_kill 指定线程
handler 跑在收到信号的那条线程上

推荐： 创建业务线程前主线程阻塞关心信号 → 专用线程 sigwait → 业务线程不装异步 handler。

Demo：code/thread_sigwait.c

HFT / 嵌入式关联

路线 要点
嵌入式 信号处置进程级共享，掩码线程级私有
HFT 主线程先阻塞→专用线程 sigwait→业务线程不装 handler，避免异步信号毛刺

代码示例

#include <pthread.h>
#include <signal.h>
#include <stdio.h>
void handler(int sig) { printf("线程 %lu 收到信号 %d\n", (unsigned long)pthread_self(), sig); }
void *worker(void *arg) {
    signal(SIGUSR1, handler);
    pause();
    return NULL;
}
int main(void) {
    pthread_t tid;
    pthread_create(&tid, NULL, worker, NULL);
    sleep(1);
    /* pthread_kill: 给特定线程发信号 */
    pthread_kill(tid, SIGUSR1);
    pthread_join(tid, NULL);
    return 0;
}

进程控制

本章：TLPI 第 33 章 — Threads: Further Details - 上一节：33.2 线程与信号（重难点） - 下一节：33.4 实现模型

本节讲什么

pthread_atfork(prepare, parent, child); ---

要点

调用 多线程行为
fork 子进程只留调用线程；锁状态危险 → 立刻 exec；pthread_atfork 仅缓解
exec 只留调用线程；handler→DFL
exit / main return 整进程全线程死
只退主线程 主线程 pthread_exit 
pthread_atfork(prepare, parent, child);

HFT / 嵌入式关联

路线 要点
嵌入式 fork/exec 只留调用线程，exit 全线程死
HFT 多线程进程 fork 后立刻 exec；exit 会杀整进程（含其它线程）

代码示例

#include <pthread.h>
#include <unistd.h>
#include <stdio.h>
void *worker(void *arg) {
    printf("线程 PID=%d\n", getpid());
    /* fork 在多线程中只复制调用线程 */
    pid_t pid = fork();
    if (pid == 0) { printf("子进程: 只有 fork 的线程\n"); _exit(0); }
    wait(NULL);
    return NULL;
}
int main(void) {
    pthread_t tid;
    pthread_create(&tid, NULL, worker, NULL);
    pthread_join(tid, NULL);
    return 0;
}

实现模型

本章：TLPI 第 33 章 — Threads: Further Details - 上一节：33.3 进程控制 - 下一节：33.5 LinuxThreads vs NPTL

本节讲什么

要点

模型 要点
M:1 用户级；快；难多核；阻塞拖死全体
1:1 Linux NPTL；每 pthread ≈ 一内核任务；可多核

路线 要点
嵌入式 M:1 快但难多核；Linux NPTL 是 1:1

#include <pthread.h>

/* 线程模型对比:

 * M:N: 混合
 * Linux 用 1:1 (NPTL), 简单高效
 */
int main(void) {
    printf("Linux 线程模型: 1:1 (NPTL)\n");
    printf("每个 pthread → 一个 clone(CLONE_VM) 的内核线程\n");
    return 0;
}

33.5

LinuxThreads vs NPTL

本章：TLPI 第 33 章 — Threads: Further Details - 上一节：33.4 实现模型 - 下一节：33.6 高级同步（简介）

本节讲什么

勿混：pthread_t ≠ 内核 TID（gettid）。 ---

要点

LinuxThreads（废） NPTL（现行）
PID 每线程一 PID（非 POSIX） 同进程 TGID 相同；独立 TID
信号 怪异 POSIX 语义
同步 — 常基于 futex

勿混：pthread_t ≠ 内核 TID（gettid）。

HFT / 嵌入式关联

路线 要点
嵌入式 NPTL 现行，同进程共享 TGID、独立 TID
HFT NPTL 同步基于 futex（无竞争不陷内核）；pthread_t ≠ TID

代码示例

#include <pthread.h>
#include <stdio.h>
int main(void) {
    /* LinuxThreads (旧) vs NPTL (新):
     * LinuxThreads: 每个线程有不同 PID, 信号处理有问题
     * NPTL (2.6+): 同进程线程共享 PID, 符合 POSIX
     */
    pthread_t tid;
    pthread_create(&tid, NULL, (void *(*)(void *))sleep, (void *)0);
    printf("PID=%d (NPTL: 线程共享 PID)\n", getpid());
    pthread_join(tid, NULL);
    return 0;
}

本节讲什么

要点

原语 用途
pthread_rwlock_t 读共享、写独占

代码示例

#include <pthread.h>
#include <stdio.h>
/* 线程同步方式:
 * 1. mutex: 互斥锁
 * 2. cond: 条件变量
 * 3. rwlock: 读写锁
 * 4. semaphore: 信号量
 * 5. barrier: 屏障
 */
pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
int ready = 0;
void *worker(void *arg) {
    pthread_mutex_lock(&mtx);
    while (!ready) pthread_cond_wait(&cond, &mtx);
    printf("条件满足\n");
    pthread_mutex_unlock(&mtx);
    return NULL;
}
int main(void) {
    pthread_t tid;
    pthread_create(&tid, NULL, worker, NULL);
    sleep(1);
    pthread_mutex_lock(&mtx);
    ready = 1;
    pthread_cond_signal(&cond);
    pthread_mutex_unlock(&mtx);
    pthread_join(tid, NULL);
    return 0;
}

代码自测

1. 为什么多线程程序中 fork 后只应调用 async-signal-safe 函数？

fork 只复制调用线程，其他线程消失。如果消失的线程持有 malloc 锁或 stdio 锁，子进程继承这些锁的锁定状态但无持有者。子进程调用 malloc 或 stdio 函数会死锁。async-signal-safe 函数不使用锁，是安全的。直到 exec 替换地址空间后才能安全调用其他函数。

2. 如何让信号只由一个指定线程处理？ 
/* 所有线程屏蔽目标信号 */
sigset_t set;
sigemptyset(&set);
sigaddset(&set, SIGINT);
pthread_sigmask(SIG_BLOCK, &set, NULL);

/* 信号处理线程用 sigwait */
void *signal_thread(void *arg) {
    int sig;
    sigwait(&set, &sig);
    printf("received signal %d\n", sig);
    return NULL;
}

这种模式避免信号处理器中调用非异步安全函数的问题。

3. SCHED_FIFO 和 SCHED_RR 的区别？
  • SCHED_FIFO：同优先级线程按 FIFO 顺序运行，除非主动让出（sched_yield）或被更高优先级抢占，否则会一直运行
  • SCHED_RR：同优先级线程轮转，每个线程运行一个时间片（默认 100ms）后被抢占，轮到下一个同优先级线程

HFT 通常用 SCHED_FIFO（策略线程独占 CPU 核心，不被同优先级线程打断）。

练习 1 答案

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

void *thread_a(void *arg) {
    pthread_mutex_lock(&mutex);
    sleep(5);  /* 持有锁 5 秒 */
    pthread_mutex_unlock(&mutex);
    return NULL;
}

void *thread_b(void *arg) {
    sleep(1);  /* 等 thread_a 持有锁 */
    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程：只复制了 thread_b，thread_a 消失 */
        /* mutex 仍然锁定，但没有持有者 */
        pthread_mutex_lock(&mutex);  /* 死锁！ */
        printf("child got lock\n");
        pthread_mutex_unlock(&mutex);
        _exit(0);
    }
    return NULL;
}

子进程死锁——mutex 从父进程继承的锁定状态，但没有线程持有它。pthread_mutex_lock 会永远阻塞。

解决：用 pthread_atfork() 注册 handler 在子进程中重新初始化 mutex，或避免在多线程中 fork。

练习 2 参考代码

#include <pthread.h>
#include <signal.h>
#include <stdatomic.h>
#include <stdio.h>

static atomic_int should_exit = 0;
static sigset_t signal_set;

void *signal_thread(void *arg) {
    int sig;
    while (1) {
        sigwait(&signal_set, &sig);
        printf("signal %d received\n", sig);
        if (sig == SIGINT || sig == SIGTERM) {
            atomic_store(&should_exit, 1);
            break;
        }
    }
    return NULL;
}

int main(void) {
    /* 屏蔽信号（所有线程继承） */
    sigemptyset(&signal_set);
    sigaddset(&signal_set, SIGINT);
    sigaddset(&signal_set, SIGTERM);
    pthread_sigmask(SIG_BLOCK, &signal_set, NULL);

    /* 启动信号线程 */
    pthread_t sig_tid;
    pthread_create(&sig_tid, NULL, signal_thread, NULL);

    /* 工作循环 */
    while (!atomic_load(&should_exit)) {
        /* 工作 */
    }

    pthread_join(sig_tid, NULL);
    printf("exiting gracefully\n");
    return 0;
}

练习 3 答案

SCHED_FIFO 线程优先级高于 SCHED_OTHER，会独占 CPU 直到主动让出。SCHED_OTHER 线程会被饥饿（得不到 CPU）。
