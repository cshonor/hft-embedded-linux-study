#include <stdio.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <string.h>
#include <poll.h>
#include <signal.h>
#include <sys/wait.h>

/* Ch61 套接字高级 — select/poll/epoll 多路复用。
 * 演示 poll 同时监控多个套接字。
 * 编译: gcc -o ch61_demo ch61_demo.c */

#define PORT 9997

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {
        .sin_family = AF_INET,
        .sin_addr.s_addr = htonl(INADDR_LOOPBACK),
        .sin_port = htons(PORT)
    };
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    /* fork 2 个客户端 */
    for (int i = 0; i < 2; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            int cfd = socket(AF_INET, SOCK_STREAM, 0);
            connect(cfd, (struct sockaddr *)&addr, sizeof(addr));
            char msg[32];
            snprintf(msg, sizeof(msg), "hello %d", i);
            sleep(i + 1);  /* 错开发送时间 */
            write(cfd, msg, strlen(msg) + 1);
            char buf[64];
            read(cfd, buf, sizeof(buf));
            printf("Client %d: got '%s'\n", i, buf);
            close(cfd);
            _exit(0);
        }
    }

    /* 服务器: 用 poll 监控监听套接字 + 已连接套接字 */
    struct pollfd fds[16];
    int nfds = 1;
    fds[0].fd = sfd;
    fds[0].events = POLLIN;

    printf("Server: polling for connections...\n");

    for (int served = 0; served < 2; ) {
        poll(fds, nfds, 10000);  /* 10秒超时 */

        for (int i = 0; i < nfds; i++) {
            if (!(fds[i].revents & POLLIN))
                continue;

            if (i == 0) {
                /* 监听套接字可读 -> 新连接 */
                int cfd = accept(sfd, NULL, NULL);
                fds[nfds].fd = cfd;
                fds[nfds].events = POLLIN;
                nfds++;
                printf("Server: accepted new client (fd=%d)\n", cfd);
            } else {
                /* 已连接套接字可读 -> 数据到达 */
                char buf[64];
                int n = read(fds[i].fd, buf, sizeof(buf));
                if (n > 0) {
                    printf("Server: received '%s' from fd=%d\n",
                           buf, fds[i].fd);
                    write(fds[i].fd, "ok", 3);
                    served++;
                }
                close(fds[i].fd);
                fds[i].fd = -1;  /* poll 忽略 fd=-1 */
            }
        }
    }

    while (wait(NULL) > 0)
        ;
    close(sfd);
    return 0;
}

61.1

API

本章：TLPI 第 61 章 — Sockets: Advanced Topics - 上一节：本章首页 - 下一节：61.2 SOL_SOCKET（必考）

本节讲什么

getsockopt(fd, level, optname, optval, &optlen); setsockopt(fd, level, optname, optval, optlen); level：SOL_SOCKET · IPPROTO_TCP · IPPROTO_IP/IPV6。

要点

getsockopt(fd, level, optname, optval, &optlen);
setsockopt(fd, level, optname, optval, optlen);

level：SOL_SOCKET · IPPROTO_TCP · IPPROTO_IP/IPV6。
部分选项须 bind/connect 前设置。

HFT / 嵌入式关联

路线 要点
嵌入式 getsockopt/setsockopt 读写 socket 选项，在 bind/connect 前设置（SO_REUSEADDR 须 listen 前）；level 区分 SOL_SOCKET/IPPROTO_TCP
HFT 选项须在连接建立前设好（TCP_NODELAY、SO_RCVBUF 等），启动期一次性配置，热路径零 setsockopt 系统调用

代码示例


/* getsockopt / setsockopt API 用法 */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/socket.h>

int main(void) {
    int fd = socket(AF_INET, SOCK_STREAM, 0);

    /* getsockopt: 读取当前选项值 */
    int optval = 0;
    socklen_t optlen = sizeof(optval);

    getsockopt(fd, SOL_SOCKET, SO_TYPE, &optval, &optlen);
    printf("SO_TYPE: %d (SOCK_STREAM=%d)\n", optval, SOCK_STREAM);

    getsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &optval, &optlen);
    printf("SO_REUSEADDR: %d (default)\n", optval);

    /* setsockopt: 设置选项 */
    optval = 1;
    setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &optval, sizeof(optval));
    printf("set SO_REUSEADDR = 1\n");

    getsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &optval, &optlen);
    printf("SO_REUSEADDR now: %d\n", optval);

    /* SO_ERROR: 获取异步错误 */
    int err = 0;
    optlen = sizeof(err);
    getsockopt(fd, SOL_SOCKET, SO_ERROR, &err, &optlen);
    printf("SO_ERROR: %d\n", err);

    close(fd);
    return 0;
}

61.2

SOL_SOCKET（必考）

本章：TLPI 第 61 章 — Sockets: Advanced Topics - 上一节：61.1 API - 下一节：61.3 TCP 选项

本节讲什么

要点

选项 要点
SO_REUSEADDR TIME_WAIT 端口可再 bind；listen 前开
SO_REUSEPORT 多进程同 IP:port，内核负载分担
SO_RCVBUF/SNDBUF 建议值，内核对齐；影响窗口
SO_LINGER close 行为；linger=0→RST，慎用
SO_KEEPALIVE 死连接探测；业务心跳更可控
SO_ERROR 取异步错（如非阻塞 connect）后清零
SO_RCVTIMEO/SNDTIMEO 超时；主要作用于 recv/send/…（对 read/write 因实现而异）
SO_BROADCAST UDP 广播

HFT / 嵌入式关联

路线 要点
嵌入式 SO_REUSEADDR 重启快绑、SO_REUSEPORT 多进程同端口负载分担；SO_RCVBUF/SNDBUF 影响窗口，按带宽调
HFT SO_RCVBUF/SNDBUF 决定 TCP 窗口，大报文吞吐要调大；SO_LINGER=0 发 RST 慎用（丢数据）；SO_REUSEPORT 多进程收行情分散负载；SIGPIPE 用 MSG_NOSIGNAL 规避

常见陷阱

  1. SIGPIPE
send(fd, buf, n, 0);  /* 对端关闭 → SIGPIPE */
send(fd, buf, n, MSG_NOSIGNAL);  /* 安全 */

向已关闭连接的 socket 写数据触发 SIGPIPE。用 MSG_NOSIGNAL 或忽略 SIGPIPE。

  1. TIME_WAIT
bind(fd, (struct sockaddr*)&addr, len);
/* EADDRINUSE if port in TIME_WAIT */

服务端重启时端口可能还在 TIME_WAIT。用 SO_REUSEADDR 允许绑定。

  1. accept 返回值未检查
int cfd = accept(lfd, NULL, NULL);
/* cfd 可能 -1 (EMFILE/ENFILE) */

accept 可能因 fd 耗尽返回 -1。不检查直接用会导致后续操作失败。

自测题

题目 1（选择）： listen() 的第二个参数 backlog 表示什么？ A. 最大连接数 B. 等待 accept 的队列长度 C. 缓冲区大小 D. 超时秒数

参考答案

B。backlog 是已完成连接但尚未 accept 的队列长度。超过后新连接可能被拒绝。

题目 2（判断）： bind 可以绑定任意端口号。

参考答案

错误。1-1023 是特权端口，需要 root 权限。已占用的端口会返回 EADDRINUSE。

题目 3（踩坑）： 以下代码有什么问题？

int sfd = socket(AF_INET, SOCK_STREAM, 0);
connect(sfd, (struct sockaddr*)&addr, sizeof(addr));
send(sfd, "data", 4, 0);
参考答案

没有检查 connect 返回值。connect 可能失败（连接被拒绝、超时等），不检查直接 send 会导致错误。

题目 4（应用）： 写代码创建 TCP 服务端 socket 并监听。

参考答案

int opt = 1;
setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

代码示例


/* SOL_SOCKET 级别选项 */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/socket.h>

int main(void) {
    int fd = socket(AF_INET, SOCK_STREAM, 0);

    /* SO_REUSEADDR: 地址重用 */
    int opt = 1;
    setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    /* SO_KEEPALIVE: TCP 保活 */
    setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE, &opt, sizeof(opt));

    /* SO_BROADCAST: 允许广播 */
    setsockopt(fd, SOL_SOCKET, SO_BROADCAST, &opt, sizeof(opt));

    /* SO_RCVBUF / SO_SNDBUF: 缓冲区大小 */
    int buf = 65536;
    setsockopt(fd, SOL_SOCKET, SO_RCVBUF, &buf, sizeof(buf));
    setsockopt(fd, SOL_SOCKET, SO_SNDBUF, &buf, sizeof(buf));

    /* 验证设置 */
    int val; socklen_t len = sizeof(val);
    getsockopt(fd, SOL_SOCKET, SO_RCVBUF, &val, &len);
    printf("SO_RCVBUF: %d (requested %d, kernel may double it)\n", val, buf);

    getsockopt(fd, SOL_SOCKET, SO_SNDBUF, &val, &len);
    printf("SO_SNDBUF: %d\n", val);

    /* SO_RCVTIMEO / SO_SNDTIMEO: 超时 */
    struct timeval tv = { .tv_sec = 5, .tv_usec = 0 };
    setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
    printf("SO_RCVTIMEO: 5s\n");

    close(fd);
    return 0;
}

61.3

TCP 选项

本章：TLPI 第 61 章 — Sockets: Advanced Topics - 上一节：61.2 SOL_SOCKET（必考） - 下一节：61.4 OOB

本节讲什么

要点

| TCP_NODELAY | 关 Nagle → 小包立刻发（交互/低延迟） | | TCP_KEEPIDLE 等 | 配 SO_KEEPALIVE |

HFT / 嵌入式关联

路线 要点
嵌入式 TCP_NODELAY 关 Nagle 让小包立刻发，交互/低延迟协议必开；TCP_KEEPIDLE 等配 SO_KEEPALIVE 探测死连接
HFT TCP_NODELAY 是 HFT 铁律——Nagle 会攒够 MSS 才发，让订单/行情小包延迟毫秒级；所有 TCP 连接必设 TCP_NODELAY=1

代码示例


/* IPPROTO_TCP 级别选项 */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/tcp.h>

int main(void) {
    int fd = socket(AF_INET, SOCK_STREAM, 0);

    /* TCP_NODELAY: 禁用 Nagle 算法 */
    /* Nagle: 攒够 MSS 才发，减少小包；禁用后立即发送（HFT 必用） */
    int opt = 1;
    setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &opt, sizeof(opt));
    printf("TCP_NODELAY: 1 (Nagle disabled, send immediately)\n");

    /* TCP_MAXSEG: 最大段大小 */
    int mss = 0; socklen_t len = sizeof(mss);
    getsockopt(fd, IPPROTO_TCP, TCP_MAXSEG, &mss, &len);
    printf("TCP_MAXSEG (MSS): %d\n", mss);

    /* TCP_KEEPIDLE: 保活探测间隔 */
    opt = 60;
    setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE, &opt, sizeof(opt));
    printf("TCP_KEEPIDLE: %d seconds\n", opt);

    /* TCP_KEEPINTVL: 保活探测重试间隔 */
    opt = 10;
    setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL, &opt, sizeof(opt));
    printf("TCP_KEEPINTVL: %d seconds\n", opt);

    /* TCP_KEEPCNT: 保活探测最大重试次数 */
    opt = 3;
    setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT, &opt, sizeof(opt));
    printf("TCP_KEEPCNT: %d retries\n", opt);

    close(fd);
    return 0;
}

61.4

OOB

本章：TLPI 第 61 章 — Sockets: Advanced Topics - 上一节：61.3 TCP 选项 - 下一节：61.5 sendmsg / recvmsg

本节讲什么

MSG_OOB；TCP 仅 1 字节紧急数据；SIGURG 或 recv(..., MSG_OOB)。工程少用，自定协议更好。 ---

要点

MSG_OOB；TCP 仅 1 字节紧急数据；SIGURG 或 recv(..., MSG_OOB)。工程少用，自定协议更好。

HFT / 嵌入式关联

路线 要点
嵌入式 TCP OOB 仅 1 字节紧急数据，工程少用，嵌入式自定协议（带优先级字段）比 OOB 更可控
HFT OOB 只有 1 字节且语义混乱，HFT 用自定协议（消息头带紧急标志/优先级）替代；OOB 的 SIGURG 异步信号反而引入中断毛刺，禁用

代码示例


/* 带外数据 (Out-of-Band Data) 传输 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/wait.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9991) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 1);

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_INET, SOCK_STREAM, 0);
        struct sockaddr_in sa = { .sin_family = AF_INET, .sin_port = htons(9991) };
        inet_pton(AF_INET, "127.0.0.1", &sa.sin_addr);
        connect(cfd, (struct sockaddr *)&sa, sizeof(sa));

        /* 发送普通数据 */
        send(cfd, "normal", 7, 0);
        /* 发送带外数据：MSG_OOB */
        send(cfd, "!", 2, MSG_OOB);
        printf("client: sent normal + OOB data\n");
        close(cfd);
        _exit(0);
    }

    int afd = accept(sfd, NULL, NULL);

    /* 接收带外数据：MSG_OOB */
    char oobbuf[2];
    ssize_t n = recv(afd, oobbuf, 1, MSG_OOB);
    if (n > 0) {
        printf("server: OOB data = '%c'\n", oobbuf[0]);
    } else {
        printf("server: OOB not available yet\n");
    }

    /* 接收普通数据 */
    char buf[16];
    n = recv(afd, buf, sizeof(buf)-1, 0);
    if (n > 0) { buf[n]='\0'; printf("server: normal = '%s'\n", buf); }

    wait(NULL);
    close(afd);
    close(sfd);
    return 0;
}

61.5

`sendmsg` / `recvmsg`

本章：TLPI 第 61 章 — Sockets: Advanced Topics - 上一节：61.4 OOB - 下一节：61.6 短读写

本节讲什么

最强收发：msg_iov 分散聚集；msg_control 辅助数据 → UNIX 传 fd / 凭证。普通 send 做不到。 ---

要点

最强收发：msg_iov 分散聚集；msg_control 辅助数据 → UNIX 传 fd / 凭证。普通 send 做不到。

HFT / 嵌入式关联

路线 要点
嵌入式 sendmsg/recvmsg 分散聚集 I/O + SCM_RIGHTS 传 fd，进程间零拷贝传 fd（如 epoll fd、事件 fd）省去重建
HFT recvmsg 分散聚集一次收多段（报文头+负载）省拷贝；SCM_RIGHTS 传 fd 让风控进程把事件 fd 交给策略进程，进程间优雅共享 I/O 对象

代码示例


/* sendmsg / recvmsg：分散/聚集 I/O + 辅助数据 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/wait.h>

int main(void) {
    int sv[2];
    socketpair(AF_UNIX, SOCK_STREAM, 0, sv);

    pid_t pid = fork();
    if (pid == 0) {
        close(sv[0]);
        /* sendmsg: 多缓冲区发送 */
        struct iovec iov[2];
        char part1[] = "hello ";
        char part2[] = "world";
        iov[0].iov_base = part1; iov[0].iov_len = 6;
        iov[1].iov_base = part2; iov[1].iov_len = 5;

        struct msghdr msg;
        memset(&msg, 0, sizeof(msg));
        msg.msg_iov = iov;
        msg.msg_iovlen = 2;

        ssize_t n = sendmsg(sv[1], &msg, 0);
        printf("sendmsg: sent %zd bytes (scatter-gather)\n", n);
        close(sv[1]);
        _exit(0);
    }

    close(sv[1]);
    /* recvmsg: 接收到一个缓冲区 */
    char buf[64];
    struct iovec iov = { .iov_base = buf, .iov_len = sizeof(buf) };
    struct msghdr msg;
    memset(&msg, 0, sizeof(msg));
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;

    ssize_t n = recvmsg(sv[0], &msg, 0);
    if (n > 0) { buf[n] = '\0'; printf("recvmsg: received '%s' (%zd bytes)\n", buf, n); }

    wait(NULL);
    close(sv[0]);
    return 0;
}

#include <string.h>

#include <sys/socket.h>
#include <sys/wait.h>

#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

int main(void) {
    /* UDP connect: 不实际建立连接，只绑定默认对端 */

    if (s6 >= 0) {

        addr6.sin6_addr = in6addr_loopback;  /* ::1 */

}

#include <netinet/tcp.h>
#include <stdio.h>

void setup_hft_socket(int sockfd) {
    int opt;

    /* 1. 禁用 Nagle：立即发送小包 */
    opt = 1;
    setsockopt(sockfd, IPPROTO_TCP, TCP_NODELAY, &opt, sizeof(opt));

    /* 2. 快速 ACK：禁用延迟确认 */
    opt = 1;
    setsockopt(sockfd, IPPROTO_TCP, TCP_QUICKACK, &opt, sizeof(opt));

    /* 3. 增大缓冲区 */
    opt = 4 * 1024 * 1024;  /* 4MB */
    setsockopt(sockfd, SOL_SOCKET, SO_RCVBUF, &opt, sizeof(opt));
    setsockopt(sockfd, SOL_SOCKET, SO_SNDBUF, &opt, sizeof(opt));

    /* 4. 地址重用 */
    opt = 1;
    setsockopt(sockfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    /* 5. TCP 保活：10秒空闲，3秒间隔，3次探测 */
    opt = 1;
    setsockopt(sockfd, SOL_SOCKET, SO_KEEPALIVE, &opt, sizeof(opt));
    opt = 10;  /* 10秒空闲后开始探测 */
    setsockopt(sockfd, IPPROTO_TCP, TCP_KEEPIDLE, &opt, sizeof(opt));
    opt = 3;   /* 探测间隔3秒 */
    setsockopt(sockfd, IPPROTO_TCP, TCP_KEEPINTVL, &opt, sizeof(opt));
    opt = 3;   /* 3次探测失败后判定断连 */
    setsockopt(sockfd, IPPROTO_TCP, TCP_KEEPCNT, &opt, sizeof(opt));
}

代码自测

TCP_NODELAY 和 TCP_CORK 有什么区别？何时该用哪个？

TCP_NODELAY 禁用 Nagle 算法，每个 write() 立即作为单独的 TCP 段发送，适合小延迟敏感的交互式协议（如 HFT 交易消息）。TCP_CORK 更进一步，缓存所有数据（包括带 PSH 的），直到主动取消 Cork 或缓冲区满才发送，适合批量数据传输（如 HTTP 响应头+body 一并发出）。HFT 交易消息通常用 TCP_NODELAY，而批量行情快照用 TCP_CORK + 显式 flush。

61.11

Inheritance of Flags and Options across accept() and Other Operations

本章：TLPI 第 61 章

本节讲什么

accept() 返回的新套接字继承了监听套接字的哪些属性：选项、标志、信号处理等。了解哪些继承哪些不继承，避免意外行为。

要点

accept() 继承规则

属性 是否继承 说明
SO_REUSEADDR 是 选项继承
SO_KEEPALIVE 是 选项继承
SO_RCVBUF/SO_SNDBUF 否 使用系统默认值
SO_OOBINLINE 是 选项继承
TCP_NODELAY 否 新连接默认 Nagle on
TCP_CORK 否 新连接默认 Cork off
SO_LINGER 否 使用系统默认
FD flags (O_NONBLOCK) 否 accept 返回阻塞 fd
SO_DOMAIN/SO_TYPE 是 协议和类型相同
signal disposition N/A 进程级，非 fd

设置继承选项的正确时机

/* 错误：在 accept 后设置每个新连接 */
int connfd = accept(lfd, NULL, NULL);
setsockopt(connfd, IPPROTO_TCP, TCP_NODELAY, &opt, sizeof(opt));
/* 高并发下每个连接都设置，开销大 */

/* 正确：在 listen 前设置监听 fd，继承到 accept */
setsockopt(lfd, IPPROTO_TCP, TCP_NODELAY, &opt, sizeof(opt));
int connfd = accept(lfd, NULL, NULL);  /* 继承 TCP_NODELAY */

不继承的选项需单独设置

/* 缓冲区大小不继承，需在每个新连接设置 */
int connfd = accept(lfd, NULL, NULL);
int bufsz = 4 * 1024 * 1024;
setsockopt(connfd, SOL_SOCKET, SO_RCVBUF, &bufsz, sizeof(bufsz));
setsockopt(connfd, SOL_SOCKET, SO_SNDBUF, &bufsz, sizeof(bufsz));

非阻塞模式继承

/* 方法1：accept 前设置监听 fd 非阻塞 */
int flags = fcntl(lfd, F_GETFL);
fcntl(lfd, F_SETFL, flags | O_NONBLOCK);
/* accept4 返回的 fd 也是非阻塞（Linux 扩展） */

/* 方法2：使用 accept4 直接设置 */
int connfd = accept4(lfd, NULL, NULL, SOCK_NONBLOCK | SOCK_CLOEXEC);

accept4 flags (Linux 2.6.28+)

flag 说明
SOCK_NONBLOCK 新 fd 非阻塞
SOCK_CLOEXEC 新 fd close-on-exec

HFT / 嵌入式关联

  • HFT 服务器在 listen 前统一设置选项，减少每连接开销
  • 使用 accept4(SOCK_NONBLOCK | SOCK_CLOEXEC) 原子设置
  • 确保所有交易连接的 TCP_NODELAY 统一

代码示例

#include <sys/socket.h>
#include <netinet/tcp.h>
#include <fcntl.h>

void setup_listen_socket(int lfd) {
    int opt = 1;

    /* 在 listen 前设置可继承选项 */
    setsockopt(lfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    setsockopt(lfd, IPPROTO_TCP, TCP_NODELAY, &opt, sizeof(opt));

    /* 注意：SO_RCVBUF/SO_SNDBUF 不继承到 accept */
    /* 需在每个新连接单独设置 */

    /* 监听 fd 设为非阻塞（accept4 也继承非阻塞） */
    int flags = fcntl(lfd, F_GETFL);
    fcntl(lfd, F_SETFL, flags | O_NONBLOCK);
}

int accept_conn(int lfd) {
    /* accept4 原子设置 NONBLOCK + CLOEXEC */
    int connfd = accept4(lfd, NULL, NULL,
                         SOCK_NONBLOCK | SOCK_CLOEXEC);

    /* 设置不继承的选项 */
    int bufsz = 4 * 1024 * 1024;
    setsockopt(connfd, SOL_SOCKET, SO_RCVBUF, &bufsz, sizeof(bufsz));
    setsockopt(connfd, SOL_SOCKET, SO_SNDBUF, &bufsz, sizeof(bufsz));

    /* 快速 ACK（不继承） */
    int opt = 1;
    setsockopt(connfd, IPPROTO_TCP, TCP_QUICKACK, &opt, sizeof(opt));

    return connfd;
}

为什么建议使用 accept4 而不是 accept + fcntl？

accept4(lfd, NULL, NULL, SOCK_NONBLOCK | SOCK_CLOEXEC) 在一次系统调用中同时完成 accept 和设置标志，避免了 accept + fcntl 的两步操作。两步操作之间存在竞争窗口：在 accept 返回和 fcntl 设置之间，fd 已经存在但未设置 CLOEXEC，如果此时其他线程执行 exec()，fd 会泄漏到新进程。accept4 是原子的，消除竞争窗口，性能也更好（一次系统调用）。

代码自测

SO_REUSEPORT 和 SO_REUSEADDR 有什么区别？
  • SO_REUSEADDR：允许 bind 处于 TIME_WAIT 状态的地址，或绑定通配地址与本机地址共存。主要解决服务器重启时端口占用问题。
  • SO_REUSEPORT：允许多个套接字绑定完全相同的地址和端口，内核将新连接在多个套接字间负载均衡。适用于多进程/多线程服务器，避免 accept 惊群。SO_REUSEPORT 是 SO_REUSEADDR 的超集功能，解决了连接分发问题。

练习 3：accept4 优势

题目：解释 accept4(lfd, NULL, NULL, SOCK_NONBLOCK | SOCK_CLOEXEC) 相比 accept + fcntl 的优势。

  1. 原子性：accept4 在一次系统调用中完成 accept 和设置标志，避免 accept + fcntl 之间的竞争窗口
