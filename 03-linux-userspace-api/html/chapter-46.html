导读

章节目标

msgget/msgsnd/msgrcv/msgctl；msgtyp 三规则；msqid_ds 与限额；与 pipe 对比；局限。

思考题要点

  1. msgtyp 0 / >0 / <0（上表）。
  2. EINTR；SA_RESTART 无效。
  3. mq RMID 立即标记 vs shm 等 detach。
  4. MSG_NOERROR vs E2BIG。
  5. id≠fd。
  6. 崩溃 + 内核持久 → ipcrm/IPC_RMID。
  7. 0 长度消息合法。

背诵卡

# 要点
1 消息边界 + mtype>0
2 msgtyp：0 / 等于 / ≤|n| 最小
3 msgsz 不含 mtype；正文可 0
4 中断 → EINTR，无 RESTART
5 非 fd；内核持久；IPC_RMID
6 新项目优先 POSIX mq / UDS

代码示例

#include <stdio.h>
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

/* Ch46 SysV 消息队列 — msgget/msgsnd/msgrcv。
 * 演示父子进程通过消息队列通信。
 * 编译: gcc -o ch46_demo ch46_demo.c */

typedef struct {
    long mtype;           /* 消息类型 (必须 > 0) */
    char mtext[64];       /* 消息数据 */
} msgbuf_t;

int main(void) {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    if (msqid < 0) { perror("msgget"); return 1; }

    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid == 0) {
        /* 子进程: 发送消息 */
        msgbuf_t msg;
        msg.mtype = 1;  /* 类型 1 */
        strcpy(msg.mtext, "Hello from child!");
        msgsnd(msqid, &msg, strlen(msg.mtext) + 1, 0);
        printf("Child: sent message\n");

        msg.mtype = 2;  /* 类型 2 */
        strcpy(msg.mtext, "Second message");
        msgsnd(msqid, &msg, strlen(msg.mtext) + 1, 0);
        printf("Child: sent second message\n");
        _exit(0);
    }

    /* 父进程: 接收消息 */
    sleep(1);

    msgbuf_t msg;

    /* 接收类型 1 的消息 */
    msgrcv(msqid, &msg, sizeof(msg.mtext), 1, 0);
    printf("Parent: received type %ld: %s\n", msg.mtype, msg.mtext);

    /* 接收任意类型的消息 */
    msgrcv(msqid, &msg, sizeof(msg.mtext), 0, 0);
    printf("Parent: received type %ld: %s\n", msg.mtype, msg.mtext);

    waitpid(pid, NULL, 0);
    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

46.1

特性（相对 Pipe/FIFO）

本章：TLPI 第 46 章 — System V Message Queues - 上一节：本章首页 - 下一节：46.2 API

本节讲什么

数据在内核缓冲，不落盘。 ---

要点

Pipe/FIFO SysV mq
模型 字节流 整条消息
筛选 无 mtype
句柄 fd → epoll msqid → 不能
持久 进程持久 内核持久

数据在内核缓冲，不落盘。

代码示例

#include <stdio.h>
#include <sys/msg.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
/* 消息队列 vs 管道: 有消息边界 + 类型过滤 */
int main(void) {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);

    /* 消息队列: 每条消息有类型, 可以按类型过滤接收 */
    struct { long mtype; char mtext[64]; } msg;

    /* 发送不同类型的消息 */
    msg.mtype = 1; strcpy(msg.mtext, "urgent"); msgsnd(msqid, &msg, 7, 0);
    msg.mtype = 2; strcpy(msg.mtext, "normal"); msgsnd(msqid, &msg, 7, 0);
    msg.mtype = 3; strcpy(msg.mtext, "low");    msgsnd(msqid, &msg, 3, 0);

    /* 管道是字节流, 没有边界; 消息队列有边界 */
    printf("Messages sent with types 1, 2, 3\n");

    /* 按类型接收: 只收类型 2 */
    msgrcv(msqid, &msg, sizeof(msg.mtext), 2, 0);
    printf("Received type 2: %s\n", msg.mtext);

    /* 接收所有类型 (msgtyp=0: 按顺序) */
    msgrcv(msqid, &msg, sizeof(msg.mtext), 0, 0);
    printf("Received (FIFO): type=%ld text=%s\n", msg.mtype, msg.mtext);

    /* 接收类型 <= 3 的最大类型 */
    msgrcv(msqid, &msg, sizeof(msg.mtext), 3, 0);
    printf("Received type 3: %s\n", msg.mtext);

    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

46.2

API

本章：TLPI 第 46 章 — System V Message Queues - 上一节：46.1 特性（相对 Pipe/FIFO） - 下一节：46.3 –46.5 msqid_ds · 限额 · 运维

本节讲什么

复用 Ch45：IPC_CREAT|IPC_EXCL + mode。 msgctl：IPC_STAT / IPC_SET / IPC_RMID（立即标记删；残留消息丢弃；持有 id 者可继续用完）。 ⚠️ shm 的 IPC_RMID 是延迟到全部 shmdt；mq/sem 是立即标记。

要点

msgget / msgctl

复用 Ch45：IPC_CREAT|IPC_EXCL + mode。
msgctl：IPC_STAT / IPC_SET / IPC_RMID（立即标记删；残留消息丢弃；持有 id 者可继续用完）。
⚠️ shm 的 IPC_RMID 是延迟到全部 shmdt；mq/sem 是立即标记。

msgsnd

struct msgbuf { long mtype; /* >0 */ char mtext[]; };
int msgsnd(int msqid, const void *msgp, size_t msgsz, int msgflg);
  • msgsz = 正文长度（不含 mtype）；可为 0
  • 0 阻塞满队列；IPC_NOWAIT → EAGAIN
  • 信号中断 → EINTR（无 SA_RESTART）

msgrcv（重点）

ssize_t msgrcv(int msqid, void *msgp, size_t msgsz, long msgtyp, int msgflg);
msgtyp 行为
0 队首（FIFO）
>0 第一条 类型等于 msgtyp
<0 类型 ≤ |msgtyp| 中 最小类型 的那条

例：队列类型 [100,300,200,400]，msgtyp=-300 → 先 100，再 200，再 300；跳过 400。

标志
IPC_NOWAIT 无匹配 → ENOMSG
MSG_NOERROR 过长则截断；否则 E2BIG

返回：正文字节数。

代码示例

#include <stdio.h>
#include <sys/msg.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
/* 消息队列 API: msgget/msgsnd/msgrcv/msgctl */
typedef struct {
    long mtype;
    char mtext[64];
} message_t;

int main(void) {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    printf("msgget: id=%d\n", msqid);

    /* msgsnd: 发送消息
     * 参数: msqid, &msg, msgsz(不含mtype), msgflg
     * msgflg: IPC_NOWAIT (非阻塞), MSG_NOERROR (截断而非报错)
     */
    message_t msg = {.mtype = 1};
    strcpy(msg.mtext, "Hello, message queue!");
    size_t msgsz = strlen(msg.mtext) + 1;
    if (msgsnd(msqid, &msg, msgsz, 0) < 0)
        perror("msgsnd");
    printf("Sent: type=%ld text='%s' size=%zu\n",
           msg.mtype, msg.mtext, msgsz);

    /* msgrcv: 接收消息
     * 参数: msqid, &msg, maxmsgsz, msgtyp, msgflg
     * msgtyp=0: 接收第一条 (FIFO)
     * msgtyp>0: 接收指定类型
     * msgtyp<0: 接收类型 <= |msgtyp| 的最小类型
     */
    message_t recv;
    ssize_t n = msgrcv(msqid, &recv, sizeof(recv.mtext), 0, 0);
    if (n >= 0)
        printf("Received: type=%ld text='%s' size=%zd\n",
               recv.mtype, recv.mtext, n);

    /* msgctl: 控制 */
    struct msqid_ds ds;
    msgctl(msqid, IPC_STAT, &ds);
    printf("Queue: msgs=%lu, bytes=%lu\n",
           (unsigned long)ds.msg_qnum, (unsigned long)ds.msg_cbytes);

    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

代码示例

#include <stdio.h>
#include <sys/msg.h>
#include <time.h>
/* struct msqid_ds: 消息队列属性 */
int main(void) {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);

    /* 发送几条消息 */
    struct { long mtype; char mtext[32]; } msg = {1, "test"};
    msgsnd(msqid, &msg, 5, 0);
    msgsnd(msqid, &msg, 5, 0);

    /* IPC_STAT: 获取队列属性 */
    struct msqid_ds ds;
    msgctl(msqid, IPC_STAT, &ds);

    printf("msqid_ds fields:\n");
    printf("  msg_perm: uid=%u gid=%u mode=%04o\n",
           ds.msg_perm.uid, ds.msg_perm.gid, ds.msg_perm.mode & 0777);
    printf("  msg_stime: %s", ctime(&ds.msg_stime));  /* 最后发送时间 */
    printf("  msg_rtime: %s", ctime(&ds.msg_rtime));  /* 最后接收时间 */
    printf("  msg_ctime: %s", ctime(&ds.msg_ctime));  /* 最后修改时间 */
    printf("  msg_cbytes: %lu (current bytes)\n", (unsigned long)ds.msg_cbytes);
    printf("  msg_qnum: %lu (messages in queue)\n", (unsigned long)ds.msg_qnum);
    printf("  msg_qbytes: %lu (max bytes)\n", (unsigned long)ds.msg_qbytes);
    printf("  msg_lspid: %d (last sender)\n", ds.msg_lspid);
    printf("  msg_lrpid: %d (last receiver)\n", ds.msg_lrpid);

    /* IPC_SET: 修改队列属性 (如最大字节数) */
    ds.msg_qbytes = 2048;
    msgctl(msqid, IPC_SET, &ds);
    printf("\nAfter IPC_SET: msg_qbytes=%lu\n",
           (unsigned long)ds.msg_qbytes);

    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

46.4

Message Queue Associated Data Structure

本章：TLPI 第 46 章

本节讲什么

System V 消息队列关联的 msqid_ds 结构，包含权限、时间戳和队列统计信息。通过 msgctl(IPC_STAT) 获取。

要点

struct msqid_ds

struct msqid_ds {
    struct ipc_perm msg_perm;   /* 权限 */
    time_t         msg_stime;   /* 最后 msgsnd 时间 */
    time_t         msg_rtime;   /* 最后 msgrcv 时间 */
    time_t         msg_ctime;   /* 最后 msgctl 修改时间 */
    unsigned long  msg_cbytes;  /* 当前队列字节数 */
    unsigned long  msg_qnum;   /* 当前消息数 */
    unsigned long  msg_qbytes;  /* 最大字节限制 */
    pid_t          msg_lspid;   /* 最后发送者 PID */
    pid_t          msg_lrpid;   /* 最后接收者 PID */
};

字段说明

字段 类型 说明
msg_perm ipc_perm 权限和属主
msg_stime time_t 最后 msgsnd 时间
msg_rtime time_t 最后 msgrcv 时间
msg_ctime time_t 最后 msgctl 修改时间
msg_cbytes unsigned long 当前队列大小（字节）
msg_qnum unsigned long 当前消息数
msg_qbytes unsigned long 队列最大字节限制
msg_lspid pid_t 最后发送进程 PID
msg_lrpid pid_t 最后接收进程 PID

监控指标

指标 计算方式 用途
队列利用率 msg_cbytes / msg_qbytes 容量监控
消息积压 msg_qnum 消费者是否跟上
最后活跃 max(stime, rtime) 队列是否被使用
发送者/接收者 lspid/lrpid 审计

查看消息队列

ipcs -q
# ------ Message Queues --------
# key msqid owner perms used-bytes messages
# 0xabcd1234 0 hft 660 2048 10

HFT / 嵌入式关联

  • 监控消息积压：msg_qnum 过高表示消费者跟不上
  • 队列容量告警：msg_cbytes 接近 msg_qbytes
  • 嵌入式系统调试：ipcs -q 检查消息队列泄漏

代码示例

#include <sys/msg.h>
#include <stdio.h>
#include <time.h>

void monitor_msgqueue(int msqid) {
    struct msqid_ds ds;
    if (msgctl(msqid, IPC_STAT, &ds) == -1) {
        perror("msgctl");
        return;
    }

    time_t now = time(NULL);
    double usage = (double)ds.msg_cbytes / ds.msg_qbytes * 100;

    printf("=== Message Queue %d ===\n", msqid);
    printf("Key:            0x%x\n", ds.msg_perm.__key);
    printf("Owner:          %d:%d\n", ds.msg_perm.uid, ds.msg_perm.gid);
    printf("Permissions:   %04o\n", ds.msg_perm.mode);
    printf("Messages:       %lu\n", ds.msg_qnum);
    printf("Bytes:          %lu / %lu (%.1f%%)\n",
           ds.msg_cbytes, ds.msg_qbytes, usage);
    printf("Last snd:       %s", ctime(&ds.msg_stime));
    printf("Last rcv:       %s", ctime(&ds.msg_rtime));
    printf("Last snd PID:   %d\n", ds.msg_lspid);
    printf("Last rcv PID:   %d\n", ds.msg_lrpid);

    if (ds.msg_qnum > 1000)
        printf("WARNING: %lu messages backlog!\n", ds.msg_qnum);
    if (usage > 80)
        printf("WARNING: queue %.1f%% full!\n", usage);
}

代码自测

msg_qbytes 为 0 意味着什么？

msg_qbytes 是队列最大字节限制，默认值由系统参数 MSGMNB 决定（通常 16384 字节）。如果设置为 0，队列不能存储任何消息，msgsnd 会立即返回 EAGAIN（队列满）。可以通过 msgctl(IPC_SET) 修改此限制（需要适当权限或 CAP_SYS_RESOURCE）。

46.5

Message Queue Limits

本章：TLPI 第 46 章

本节讲什么

System V 消息队列的系统级限制：队列大小、消息大小、总消息数等。这些限制影响队列容量和性能。

要点

系统限制参数

参数 文件 典型值 说明
MSGMAX /proc/sys/kernel/msgmax 8192 单条消息最大字节
MSGMNB /proc/sys/kernel/msgmnb 16384 队列最大字节
MSGMNI /proc/sys/kernel/msgmni 32000 系统最大队列数

队列级 vs 系统级限制

级别 限制 调整方式
队列级 msg_qbytes (默认=MSGMNB) msgctl(IPC_SET)
系统级 MSGMAX/MSGMNB/MSGMNI /proc/sys/kernel/msg*
硬编码 内核编译时上限 重新编译内核

调整限制

# 查看当前限制
cat /proc/sys/kernel/msgmax   # 单条消息最大字节
cat /proc/sys/kernel/msgmnb   # 队列最大字节
cat /proc/sys/kernel/msgmni   # 最大队列数

# 临时调整（重启失效）
echo 65536 > /proc/sys/kernel/msgmax
echo 65536 > /proc/sys/kernel/msgmnb

# 永久调整
echo "kernel.msgmax=65536" >> /etc/sysctl.conf
echo "kernel.msgmnb=65536" >> /etc/sysctl.conf
sysctl -p

队列级限制调整

struct msqid_ds ds;
msgctl(msqid, IPC_STAT, &ds);
ds.msg_qbytes = 65536;  /* 增大队列容量 */
msgctl(msqid, IPC_SET, &ds);

限制影响

限制 满时行为
msg_qbytes msgsnd 阻塞或 EAGAIN
MSGMAX msgsnd 返回 EINVAL
MSGMNI msgget 返回 ENOSPC

常见错误

errno 原因
EAGAIN 队列满且 IPC_NOWAIT
EINVAL 消息超 MSGMAX 或 mtype<0
ENOSPC 系统队列数超限

HFT / 嵌入式关联

  • HFT 行情消息可能超默认 MSGMAX，需调大
  • 嵌入式系统资源有限，控制队列数和大小
  • 监控 msg_qbytes 使用率，避免队列满阻塞

msgsnd 返回 EAGAIN 和 EINVAL 分别表示什么？
  • EAGAIN：队列已满（msg_cbytes + 新消息大小 > msg_qbytes），且设置了 IPC_NOWAIT 标志。如果不设置 IPC_NOWAIT，会阻塞直到队列有空间。

Client-Server Programming with Message Queues

本章：TLPI 第 46 章

本节讲什么

使用 System V 消息队列实现客户端-服务器通信模式，利用 mtype 字段实现多客户端复用和服务端回复路由。

要点

mtype 路由模式

mtype = 1   → 服务端请求通道（所有客户端发到此处）
mtype = PID → 客户端回复通道（服务端用客户端 PID 作为 mtype 回复）

Client A (PID=1001)             Server
  |                                |
  | msgsnd(mtype=1, request)      |
  |------------------------------→|
  |                                |
  |                    msgrcv(mtype=1001)
  |←------------------------------|
  | msgsnd(mtype=1, request)      |
  |------------------------------→|

服务端模式

/* 服务端：接收所有 mtype=1 的请求 */
struct {
    long mtype;
    char mtext[256];
} msg;

while (1) {
    /* 只接收 mtype=1 的消息 */
    if (msgrcv(msqid, &msg, sizeof(msg.mtext), 1, 0) == -1) {
        perror("msgrcv");
        continue;
    }

    /* 处理请求... */

    /* 用客户端 PID 作为回复 mtype */
    pid_t client_pid = msg.mtext[0]; /* 约定：请求首字节为 PID */
    msg.mtype = client_pid;

    /* 发送回复 */
    msgsnd(msqid, &msg, reply_len, 0);
}

客户端模式

/* 客户端 */
struct {
    long mtype;
    char mtext[256];
} msg;

pid_t my_pid = getpid();

/* 发送请求：mtype=1（服务端通道） */
msg.mtype = 1;
msg.mtext[0] = (char)my_pid;  /* 约定：首字节为 PID */
/* ... 填充请求数据 ... */
msgsnd(msqid, &msg, request_len, 0);

/* 接收回复：mtype=我的PID */
if (msgrcv(msqid, &msg, sizeof(msg.mtext), my_pid, 0) == -1) {
    perror("msgrcv");
}
/* msg.mtext 包含回复 */

并发客户端

Client A (PID=1001): snd(mtype=1) → rcv(mtype=1001)
Client B (PID=1002): snd(mtype=1) → rcv(mtype=1002)
Server: rcv(mtype=1) → 处理 → snd(mtype=客户端PID)

每个客户端用自己的 PID 作为回复 mtype，互不干扰。

替代方案：多队列

方案 优点 缺点
单队列+mtype 共享资源，管理简单 单点瓶颈
多队列（每客户端一个） 隔离性好 资源消耗大
请求/回复双队列 方向清晰 需2个队列

HFT / 嵌入式关联

  • HFT 多策略进程向风控进程发送审批请求
  • 嵌入式多传感器向控制进程发送数据
  • 注意：SysV MQ 性能不如 POSIX MQ 或 socket，HFT 中逐渐被替代

代码示例

#include <sys/msg.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    long mtype;
    pid_t client_pid;
    int request_id;
    char data[256];
} request_msg;

typedef struct {
    long mtype;
    int status;
    char data[256];
} reply_msg;

/* 服务端 */
void server_loop(int msqid) {
    request_msg req;
    reply_msg rep;

    while (1) {
        if (msgrcv(msqid, &req, sizeof(req) - sizeof(long),
                   1, 0) == -1)
            continue;

        /* 处理请求 */
        rep.mtype = req.client_pid;  /* 回复到客户端 */
        rep.status = process_request(req.request_id, req.data);
        snprintf(rep.data, sizeof(rep.data), "OK: %d", req.request_id);

        msgsnd(msqid, &rep, sizeof(rep) - sizeof(long), 0);
    }
}

/* 客户端 */
int client_request(int msqid, int req_id, const char *input) {
    request_msg req;
    reply_msg rep;

    req.mtype = 1;                /* 服务端通道 */
    req.client_pid = getpid();    /* 回复路由用 */
    req.request_id = req_id;
    strncpy(req.data, input, sizeof(req.data) - 1);

    if (msgsnd(msqid, &req, sizeof(req) - sizeof(long), 0) == -1)
        return -1;

    /* 等待回复：mtype = 我的 PID */
    if (msgrcv(msqid, &rep, sizeof(rep) - sizeof(long),
               getpid(), 0) == -1)
        return -1;

    return rep.status;
}

代码自测

为什么用 PID 作为回复 mtype？有什么风险？

PID 唯一标识进程，天然适合作为回复路由 key。风险：(1) PID 可能被复用——旧客户端退出后 PID 被新进程使用，旧回复可能被新进程错误接收；(2) PID 取模后可能冲突——mtype 是 long，PID 通常在合理范围内，冲突概率低但非零；(3) 超过 PID_MAX 的值不可用作 mtype。

A File Server Application Using Message Queues

本章：TLPI 第 46 章

本节讲什么

使用 SysV 消息队列实现的文件服务器示例：客户端发送文件请求，服务端读取文件内容并通过消息队列返回。展示了请求-回复模式和大数据分块传输。

要点

架构设计

┌─────────┐     msgget(key)      ┌──────────┐
│ Client A │ ──msgsnd(mtype=1)──→│          │
│          │ ←msgrcv(mtype=PID)─│  Server  │
├─────────┤                     │          │
│ Client B │ ──msgsnd(mtype=1)──→│  读取文件 │
│          │ ←msgrcv(mtype=PID)─│  分块回复  │
└─────────┘                     └──────────┘

消息格式

/* 请求消息 */
typedef struct {
    long mtype;        /* = 1（服务端通道） */
    pid_t client_pid;  /* 客户端 PID（回复路由） */
    int  cmd;          /* 命令：GET/PUT/LIST */
    char path[256];    /* 文件路径 */
} request_t;

/* 回复消息 */
typedef struct {
    long mtype;        /* = 客户端 PID */
    int  seq;          /* 分块序号 */
    int  total;        /* 总块数 */
    int  len;          /* 本块数据长度 */
    char data[4096];   /* 文件数据 */
} reply_t;

大文件分块传输

请求：GET /path/to/file (size=100KB, 每块4KB → 25块)

回复序列：
  seq=0  total=25  len=4096  data=...
  seq=1  total=25  len=4096  data=...
  ...
  seq=24 total=25  len=1440  data=...  (最后一块)
  seq=-1 total=25  len=0     (EOF标记)

流程

步骤 客户端 服务端
1 msgsnd(请求, mtype=1) msgrcv(mtype=1) 等待
2 open(path), 读取文件
3 msgrcv(mtype=PID) 等待 msgsnd(分块回复, mtype=PID) 循环
4 组装数据 close(fd)

异常处理

异常 处理
文件不存在 回复 error 状态
客户端中途退出 服务端 msgsnd 得到 EAGAIN/超时
队列满 阻塞等待或返回错误

HFT / 嵌入式关联

  • 配置文件分发的进程间通信模型
  • 嵌入式设备远程文件读取（替代 NFS）
  • 注意：大数据传输 SysV MQ 不如 socket 高效，分块开销大

代码示例

#include <sys/msg.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    long mtype;
    pid_t client_pid;
    int  cmd;  /* 0=GET */
    char path[256];
} request_t;

typedef struct {
    long mtype;
    int  seq;
    int  total;
    int  len;
    char data[4096];
} reply_t;

#define CMD_GET 0

/* 服务端处理文件请求 */
void file_server(int msqid) {
    request_t req;
    reply_t rep;

    while (1) {
        if (msgrcv(msqid, &req, sizeof(req) - sizeof(long),
                   1, 0) == -1)
            continue;

        if (req.cmd == CMD_GET) {
            int fd = open(req.path, O_RDONLY);
            if (fd == -1) {
                /* 错误回复 */
                rep.mtype = req.client_pid;
                rep.seq = -1; rep.total = 0; rep.len = 0;
                msgsnd(msqid, &rep, sizeof(rep) - sizeof(long), 0);
                continue;
            }

            /* 计算总块数 */
            off_t fsize = lseek(fd, 0, SEEK_END);
            lseek(fd, 0, SEEK_SET);
            int total = (fsize + sizeof(rep.data) - 1) / sizeof(rep.data);
            if (total == 0) total = 1;

            /* 分块发送 */
            int seq = 0;
            ssize_t n;
            while ((n = read(fd, rep.data, sizeof(rep.data))) > 0) {
                rep.mtype = req.client_pid;
                rep.seq = seq++;
                rep.total = total;
                rep.len = n;
                msgsnd(msqid, &rep,
                       sizeof(rep) - sizeof(long), 0);
            }

            /* EOF 标记 */
            rep.mtype = req.client_pid;
            rep.seq = -1; rep.len = 0;
            msgsnd(msqid, &rep, sizeof(rep) - sizeof(long), 0);
            close(fd);
        }
    }
}

/* 客户端获取文件 */
int fetch_file(int msqid, const char *path, int out_fd) {
    request_t req = { .mtype = 1, .client_pid = getpid(),
                      .cmd = CMD_GET };
    strncpy(req.path, path, sizeof(req.path) - 1);

    if (msgsnd(msqid, &req, sizeof(req) - sizeof(long), 0) == -1)
        return -1;

    reply_t rep;
    do {
        if (msgrcv(msqid, &rep, sizeof(rep) - sizeof(long),
                   getpid(), 0) == -1)
            return -1;

        if (rep.seq == -1) break;  /* EOF */
        if (rep.len > 0)
            write(out_fd, rep.data, rep.len);
    } while (1);

    return 0;
}

代码自测

SysV 消息队列做文件传输的主要瓶颈是什么？
  1. 分块开销：每条消息受 MSGMAX 限制（通常 8KB），大文件需多次 msgsnd/msgrcv，每次系统调用开销
  2. 数据拷贝：用户空间 → 内核 → 用户空间，两次拷贝（不如 mmap 或 sendfile 零拷贝）
  3. 流量控制：队列满会阻塞，没有 TCP 的流控机制
  4. 无连接管理：客户端崩溃时服务端可能等待回复队列空间
  5. 单点瓶颈：所有客户端共用一个队列

代码自测

为什么 SysV 消息队列不能用于 epoll/select 多路复用？

SysV 消息队列返回的是 IPC ID（整数），不是文件描述符。epoll 和 select 只能操作文件描述符。Linux 不提供将 IPC ID 转换为 fd 的接口（不像 POSIX MQ 可以获取 fd）。因此使用 SysV MQ 时必须单独线程阻塞等待消息，无法与其他 I/O 事件统一管理。POSIX MQ 在某些 Linux 版本上支持获取 fd 用于 epoll。

SysV 消息队列的 mtype 为负值时 msgrcv 的行为是什么？

msgrcv 的 msgtyp 参数为负值时，读取队列中 mtype 小于或等于 |msgtyp| 的消息中 mtype 最小的那一条。例如 msgtyp=-3，会读取 mtype <= 3 的消息中 mtype 最小的。这可以实现"读取优先级 <= N 的最高优先级消息"的语义。注意 mtype 始终为正（0 不可用），负 msgtyp 的绝对值是上限。

练习 1：消息队列优先级

题目：使用 mtype 实现优先级消息队列。优先级 1 为最高，5 为最低。发送多条不同优先级消息，验证接收顺序。

参考答案 
#include <sys/msg.h>
#include <string.h>
#include <stdio.h>

typedef struct { long mtype; char text[64]; } msg_t;

int main() {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);

    /* 按非优先级顺序发送 */
    msg_t m;
    m.mtype = 3; strcpy(m.text, "normal");
    msgsnd(msqid, &m, sizeof(m.text), 0);

    m.mtype = 1; strcpy(m.text, "urgent!");
    msgsnd(msqid, &m, sizeof(m.text), 0);

    m.mtype = 5; strcpy(m.text, "low");
    msgsnd(msqid, &m, sizeof(m.text), 0);

    m.mtype = 1; strcpy(m.text, "urgent2");
    msgsnd(msqid, &m, sizeof(m.text), 0);

    /* 用负 mtype 读取：mtype <= N 的最小 mtype */
    /* msgrcv(msgtyp=-1) → mtype <= 1 的最小 → 读取 mtype=1 */
    while (msgrcv(msqid, &m, sizeof(m.text), -1, IPC_NOWAIT) != -1) {
        printf("pri=%ld: %s\n", m.mtype, m.text);
    }
    /* 输出:
       pri=1: urgent!
       pri=1: urgent2
       pri=3: normal
       pri=5: low
    */

    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

练习 2：安全发送大文件

题目：消息队列 MSGMAX 限制为 8192，如何发送 100KB 的文件？设计分块协议。

参考答案 
/* 协议：
 * 请求消息: mtype=1, { client_pid, cmd=GET, path }
 * 回复消息块: mtype=client_pid, { seq, total, len, data[7000] }
 * EOF 标记: seq=-1, len=0
 */

#define CHUNK_SIZE 7000  /* < MSGMAX-8192 留余量 */

typedef struct {
    long mtype;        /* 1=请求, client_pid=回复 */
    pid_t client_pid;
    int  cmd;
    char path[256];
} req_t;

typedef struct {
    long mtype;        /* client_pid */
    int  seq;
    int  total;
    int  len;
    char data[CHUNK_SIZE];
} chunk_t;

void send_file(int msqid, pid_t client_pid, const char *path) {
    int fd = open(path, O_RDONLY);
    if (fd < 0) {
        /* 发送错误块 */
        chunk_t err = { .mtype = client_pid, .seq = -1, .len = 0 };
        msgsnd(msqid, &err, sizeof(err) - sizeof(long) - sizeof(err.data), 0);
        return;
    }

    off_t size = lseek(fd, 0, SEEK_END);
    lseek(fd, 0, SEEK_SET);
    int total = (size + CHUNK_SIZE - 1) / CHUNK_SIZE;

    chunk_t chunk;
    int seq = 0;
    ssize_t n;
    while ((n = read(fd, chunk.data, CHUNK_SIZE)) > 0) {
        chunk.mtype = client_pid;
        chunk.seq = seq++;
        chunk.total = total;
        chunk.len = n;
        msgsnd(msqid, &chunk,
               sizeof(chunk) - sizeof(long) - sizeof(chunk.data) + n, 0);
    }

    /* EOF 标记 */
    chunk.mtype = client_pid;
    chunk.seq = -1;
    chunk.len = 0;
    msgsnd(msqid, &chunk,
           sizeof(chunk) - sizeof(long) - sizeof(chunk.data), 0);
    close(fd);
}

练习 3：为什么不用 SysV MQ

题目：列出三个不推荐在 HFT 系统中使用 SysV 消息队列的理由，并给出替代方案。

参考答案

  1. 不支持 I/O 多路复用：SysV MQ 返回 IPC ID 而非 fd，不能用于 epoll/select。HFT 交易系统需要统一管理网络、定时器、消息等事件，无法多路复用意味着需要单独线程阻塞等待。替代：socketpair 或 POSIX MQ。

  2. 消息大小限制：MSGMAX 限制（~8KB）迫使大消息分块发送，增加系统调用次数和延迟。HFT 行情快照可能数十 KB。替代：共享内存传递大数据 + eventfd 通知。

  3. 无超时控制：标准 msgrcv 不支持超时等待，IPC_NOWAIT 只能非阻塞返回。HFT 场景需要精确控制等待时间避免死锁。替代：POSIX MQ 的 mq_timedreceive 或 socket 的 SO_RCVTIMEO。
