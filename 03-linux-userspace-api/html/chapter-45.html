导读

章节目标

三类对象共用的 key/get/ctl/操作 API；ipc_perm；get 算法；内核持久与 ipcs/ipcrm；限额；IPC_RMID 差异；与 POSIX 对比铺垫。

思考题要点

  1. IPC_PRIVATE：亲缘；ftok：无关进程约定路径。
  2. ftok：固定路径文件勿删；或固定 key + 约定文档。
  3. key≠id；id 非 fd → 无 epoll。
  4. CREAT|EXCL：独占创建。
  5. 内核持久 + 忘 RMID → 泄漏占限额。
  6. shm 的 RMID 须等 detach。

背诵卡

# 要点
1 get / ctl / 操作 三套对称 API
2 key 定位；id 操作；非 fd
3 CREAT\|EXCL 独占创建
4 内核持久 → IPC_RMID / ipcrm
5 ftok 吃 inode；文件重建 key 变
6 shm 的 RMID 等全部 shmdt

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/types.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>
#include <fcntl.h>

/* Ch45 SysV IPC 概述 — ftok 生成 key + ipcs/ipcrm 管理。
 * SysV IPC 三件套: msgget/semget/shmget，用 key 标识。
 * 编译: gcc -o ch45_demo ch45_demo.c */

int main(void) {
    /* ftok: 用文件路径 + 项目号生成 IPC key */
    /* 需要一个存在的文件 */
    int fd = open("/tmp/ch45_ftok_file", O_CREAT | O_RDWR, 0644);
    close(fd);

    key_t key = ftok("/tmp/ch45_ftok_file", 'A');
    printf("ftok key = 0x%x\n", (unsigned)key);

    /* 创建消息队列 (只是演示创建+删除) */
    int msqid = msgget(key, IPC_CREAT | 0666);
    if (msqid >= 0) {
        printf("Message queue created: id=%d\n", msqid);
        struct msqid_ds info;
        msgctl(msqid, IPC_STAT, &info);
        printf("  queue size: %lu bytes\n", (unsigned long)info.msg_cbytes);
        printf("  messages:   %lu\n", (unsigned long)info.msg_qnum);
        msgctl(msqid, IPC_RMID, NULL);  /* 删除 */
        printf("  (deleted)\n");
    }

    /* 创建信号量集 */
    int semid = semget(key, 1, IPC_CREAT | 0666);
    if (semid >= 0) {
        printf("Semaphore set created: id=%d\n", semid);
        semctl(semid, 0, IPC_RMID);
        printf("  (deleted)\n");
    }

    /* 创建共享内存 */
    int shmid = shmget(key, 4096, IPC_CREAT | 0666);
    if (shmid >= 0) {
        printf("Shared memory created: id=%d, size=4096\n", shmid);
        shmctl(shmid, IPC_RMID, NULL);
        printf("  (deleted)\n");
    }

    remove("/tmp/ch45_ftok_file");
    printf("\nUse 'ipcs' to list, 'ipcrm' to remove IPC objects\n");
    return 0;
}

45.1

三类对象 · 统一 API

本章：TLPI 第 45 章 — Introduction to System V IPC - 上一节：本章首页 - 下一节：45.2 Key（key_t）

本节讲什么

要点

操作 消息队列 信号量 共享内存 类比
创建/打开 msgget semget shmget open
控制/删 msgctl semctl shmctl fcntl
主体操作 msgsnd/msgrcv semop shmat/shmdt 读写

HFT / 嵌入式关联

路线 要点
嵌入式 get/ctl/op 三件套统一范式：msgget/semget/shmget → ctl → 主体操作
HFT SysV 对象是 id 非 fd，无法 epoll；热路径多路复用选 POSIX fd 型

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/types.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>
/* SysV IPC API 总览: get/ctl/op 三类操作 */
int main(void) {
    /* 三类 IPC 共有的模式:
     * 1. xxxget(key, ...)  — 创建/获取 IPC 对象
     * 2. xxxctl(id, cmd, ...) — 控制(查看/设置/删除)
     * 3. xxxop(id, ...)    — 操作(发送/接收/同步)
     */
    printf("SysV IPC API overview:\n");
    printf("  msgget/msgctl/msgsnd/msgrcv\n");
    printf("  semget/semctl/semop\n");
    printf("  shmget/shmctl/shmat/shmdt\n");

    /* IPC_CREAT: 创建, IPC_EXCL: 排他创建 */
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    printf("\nCreated msgq: id=%d\n", msqid);

    /* IPC_STAT: 查看属性 */
    struct msqid_ds ds;
    msgctl(msqid, IPC_STAT, &ds);
    printf("  owner uid=%u, mode=%o\n", ds.msg_perm.uid, ds.msg_perm.mode);

    /* IPC_RMID: 删除 (必须显式做, 否则泄漏!) */
    msgctl(msqid, IPC_RMID, NULL);
    printf("  deleted\n");
    return 0;
}

45.2

Key（`key_t`）

本章：TLPI 第 45 章 — Introduction to System V IPC - 上一节：45.1 三类对象 · 统一 API - 下一节：45.3 –45.4 标识符 · ipc_perm

本节讲什么

跨进程定位同一对象 ≈「文件名」。 ---

要点

跨进程定位同一对象 ≈「文件名」。

方式 说明
IPC_PRIVATE 每次 get 新建；适合亲缘进程（传 id）
硬编码整数 易冲突
ftok(path, proj_id) inode+设备+proj_id；文件重建 → key 变

HFT / 嵌入式关联

路线 要点
嵌入式 IPC_PRIVATE 每次新建适合亲缘传 id；ftok(path,proj) 跨进程定位同一对象
HFT ftok 依赖文件 inode，文件重建 key 变致进程失联，用固定路径+proj 管理

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/types.h>
#include <fcntl.h>
/* ftok: 用文件路径 + 项目号生成 IPC key */
int main(void) {
    /* 创建一个临时文件用于 ftok */
    int fd = open("/tmp/ftok_demo", O_CREAT | O_RDWR, 0644);
    close(fd);

    /* ftok: 组合文件的 inode 号 + 项目号(low 8 bits) */
    key_t k1 = ftok("/tmp/ftok_demo", 'A');  /* proj='A' */
    key_t k2 = ftok("/tmp/ftok_demo", 'B');  /* proj='B' */
    key_t k3 = ftok("/tmp/ftok_demo", 'A');  /* 相同参数 = 相同 key */

    printf("ftok('A') = 0x%x\n", (unsigned)k1);
    printf("ftok('B') = 0x%x (different proj)\n", (unsigned)k2);
    printf("ftok('A') = 0x%x (same as first)\n", (unsigned)k3);

    /* IPC_PRIVATE: 总是创建新对象 (key=0) */
    printf("IPC_PRIVATE = 0x%x\n", (unsigned)IPC_PRIVATE);

    remove("/tmp/ftok_demo");
    return 0;
}

45.3

–45.4 标识符 · `ipc_perm`

本章：TLPI 第 45 章 — Introduction to System V IPC - 上一节：45.2 Key（key_t） - 下一节：45.5 get() 算法（高频）

本节讲什么

  • key：外部约定名 - id（msgid/semid/shmid）：内核句柄；后续全用 id - 不是 fd → 不能 select/poll/epoll

要点

  • key：外部约定名
  • id（msgid/semid/shmid）：内核句柄；后续全用 id
  • 不是 fd → 不能 select/poll/epoll

struct ipc_perm：uid/gid、cuid/cgid、mode（r/w，无执行语义）。权限逻辑类似文件。

HFT / 嵌入式关联

路线 要点
嵌入式 key 是外部约定名，id 是内核句柄；id 非 fd 不能 select/poll/epoll
HFT SysV 无法纳入 epoll 主循环，多路复用选 POSIX fd；权限走 ipc_perm uid/gid

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/types.h>
#include <unistd.h>
/* struct ipc_perm: IPC 对象的权限信息 */
int main(void) {
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0644);
    if (msqid < 0) { perror("msgget"); return 1; }

    struct msqid_ds ds;
    msgctl(msqid, IPC_STAT, &ds);

    /* ipc_perm 结构 */
    struct ipc_perm *perm = &ds.msg_perm;
    printf("IPC permissions:\n");
    printf("  key:    0x%x\n", (unsigned)perm->key);
    printf("  uid:    %u (owner)\n", (unsigned)perm->uid);
    printf("  gid:    %u\n", (unsigned)perm->gid);
    printf("  cuid:   %u (creator)\n", (unsigned)perm->cuid);
    printf("  cgid:   %u\n", (unsigned)perm->cgid);
    printf("  mode:   %04o\n", perm->mode & 0777);
    printf("  seq:    %u (sequence number)\n", (unsigned)perm->seq);

    /* 修改权限: IPC_SET */
    perm->mode = 0600;
    msgctl(msqid, IPC_SET, &ds);
    printf("\nAfter IPC_SET: mode=%04o\n", perm->mode & 0777);

    msgctl(msqid, IPC_RMID, NULL);
    return 0;
}

45.4

IPC Identifiers and Client-Server Applications

本章：TLPI 第 45 章 — System V IPC Introduction

本节讲什么

介绍 SysV IPC 的标识符系统：IPC key → IPC ID 的映射，以及如何利用 key 在无亲缘关系的客户端和服务器间定位同一个 IPC 对象。

要点

IPC key 和 ID

概念 说明
IPC key (key_t) 客户端/服务器约定的标识符
IPC ID 内核分配的内部标识符
ftok() 从路径+项目号生成 key
xxxget() 用 key 创建/获取 IPC 对象，返回 ID

IPC 对象创建流程

服务器:
  key = ftok("/usr/lib/hft", 'A');  /* 生成 key */
  id = msgget(key, IPC_CREAT | 0666);  /* 创建，返回 ID */

客户端:
  key = ftok("/usr/lib/hft", 'A');  /* 相同路径 → 相同 key */
  id = msgget(key, 0);  /* 获取已有对象 */

ftok 函数

key_t ftok(const char *pathname, int proj_id);
参数 说明
pathname 必须存在且可访问的文件路径
proj_id 项目标识（仅低 8 位有效，非 0）
返回值 组合 inode 号 + dev 号 + proj_id 生成 key
失败 -1（文件不存在或不可访问）

ftok 的陷阱： - 路径文件被替换（重建）→ inode 变 → key 变 - 不同文件系统可能有相同 inode 号 → key 碰撞 - 建议用固定路径（如 /usr/lib/myapp）

IPC 约定的 key 值

特殊 key 含义
IPC_PRIVATE (= 0) 创建私有对象，仅 fork 的子进程可继承
ftok() 返回值 约定的公共 key

客户端-服务器模式

模式 1：服务器创建，客户端获取

Server: id = xxxget(key, IPC_CREAT | 0666);
Client: id = xxxget(key, 0);  /* 必须已存在 */

模式 2：IPC_PRIVATE（仅 fork 关系）

Server: id = xxxget(IPC_PRIVATE, IPC_CREAT | 0666);
Server: 将 id 写入文件 / 通过管道传给 Client
Client: 使用收到的 id

模式 3：客户端创建私有回复通道

Client: reply_id = msgget(IPC_PRIVATE, IPC_CREAT | 0600);
Client: 将 reply_id 作为消息的一部分发给 Server
Server: 用 reply_id 向客户端回复

IPC 对象的权限

struct ipc_perm {
    uid_t  cuid;   /* 创建者 UID */
    gid_t  cgid;   /* 创建者 GID */
    uid_t  uid;    /* 当前属主 UID */
    gid_t  gid;    /* 当前属组 GID */
    mode_t mode;   /* 权限位 (rwx for owner/group/other) */
};

权限检查： - 读操作需要 owner 读权限 / group 读权限 / other 读权限 / root - 写操作需要对应的写权限

HFT / 嵌入式关联

  • ftok 路径：HFT 用固定路径如 /var/run/hft 生成 key
  • IPC_PRIVATE：HFT 多进程通过共享 ID 文件传递 IPC ID
  • 权限：生产环境设 0600，仅允许服务用户访问
  • 嵌入式：SysV IPC 较少使用，通常用 POSIX IPC 或 socket
  • 替代方案：用 Unix socket + 抽象地址名替代 ftok 约定

代码示例

#include <sys/ipc.h>
#include <sys/msg.h>
#include <stdio.h>

/* 服务器端 */
int server_init(void) {
    key_t key = ftok("/var/run/hft", 'M');
    if (key == -1) { perror("ftok"); return -1; }

    int msqid = msgget(key, IPC_CREAT | 0660);
    if (msqid == -1) { perror("msgget"); return -1; }

    printf("server: key=0x%x msqid=%d\n", (int)key, msqid);
    return msqid;
}

/* 客户端 */
int client_connect(void) {
    key_t key = ftok("/var/run/hft", 'M');
    if (key == -1) { perror("ftok"); return -1; }

    int msqid = msgget(key, 0);  /* 获取已存在对象 */
    if (msqid == -1) { perror("msgget"); return -1; }

    return msqid;
}

代码自测

1. ftok 为什么可能产生相同的 key？

ftok 组合三部分：文件的设备号（dev）、inode 号、proj_id。如果两个不同路径在不同文件系统上有相同的 inode 号和设备号组合，ftok 会返回相同的 key。这在 NFS 或某些文件系统上可能发生。解决：(1) 确保使用唯一的路径文件；(2) 用不同的 proj_id 区分；(3) 考虑用 IPC_PRIVATE + 文件传递 ID。

2. IPC_PRIVATE 和 ftok 生成的 key 有什么区别？
  • IPC_PRIVATE (= 0)：创建一个私有 IPC 对象，无法被其他进程通过 key 找到。只能通过 fork 继承或 ID 文件传递。适用于有亲缘关系的进程间通信。
  • ftok() 生成的 key：非 0 值，任何知道相同路径+proj_id 的进程都能找到同一个 IPC 对象。适用于无亲缘关系的客户端-服务器。
45.5

`get()` 算法（高频）

本章：TLPI 第 45 章 — Introduction to System V IPC - 上一节：45.3 –45.4 标识符 · ipc_perm - 下一节：45.6 内核持久 · 运维

本节讲什么

msgget(key, flag)，flag = IPC_CREAT / IPC_EXCL + mode： 1. 按 key 查找 2. 不存在：无 IPC_CREAT → ENOENT；有则创建

要点

msgget(key, flag)，flag = IPC_CREAT / IPC_EXCL + mode：

  1. 按 key 查找
  2. 不存在：无 IPC_CREAT → ENOENT；有则创建
  3. 已存在：CREAT|EXCL → EEXIST；仅 CREAT → 返回已有 id

服务端常用：get(key, IPC_CREAT | IPC_EXCL | 0600)。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 IPC_CREAT|IPC_EXCL 原子排他创建，服务端首个创建者唯一，无竞态
HFT 排他创建语义防多实例抢同一对象；EEXIST 判已有实例

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>
/* xxxget: 创建/获取 IPC 对象 */
int main(void) {
    /* msgget: 消息队列 */
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    printf("msgget: id=%d\n", msqid);

    /* semget: 信号量集 (第二个参数=信号量个数) */
    int semid = semget(IPC_PRIVATE, 2, IPC_CREAT | 0666);
    printf("semget: id=%d (2 semaphores)\n", semid);

    /* shmget: 共享内存 (第二个参数=大小) */
    int shmid = shmget(IPC_PRIVATE, 4096, IPC_CREAT | 0666);
    printf("shmget: id=%d (4096 bytes)\n", shmid);

    /* IPC_CREAT | IPC_EXCL: 排他创建 (如果已存在则失败) */
    /* key = IPC_PRIVATE 总是创建新对象 */
    printf("\nIPC_CREAT: create if not exists\n");
    printf("IPC_EXCL: fail if already exists\n");
    printf("IPC_PRIVATE: always create new\n");

    /* 清理 */
    msgctl(msqid, IPC_RMID, NULL);
    semctl(semid, 0, IPC_RMID);
    shmctl(shmid, IPC_RMID, NULL);
    return 0;
}

要点

进程全退对象仍在 → *ctl(..., IPC_RMID) 或重启。

ipcs -q/-s/-m
ipcrm -q <msqid>    # 或 -M key / -S / -m 等

路线 要点
嵌入式 进程全退对象仍在，必须 IPC_RMID 或重启；ipcs/ipcrm 清理
HFT 崩溃遗留僵尸对象耗尽资源，启动前 ipcrm 清理或改用 POSIX 自动回收

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>
#include <string.h>
/* xxxop: IPC 操作 (msgsnd/msgrcv, semop, shmat/shmdt) */
int main(void) {
    /* 消息队列操作 */
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    struct { long mtype; char mtext[32]; } msg = {1, "hello"};
    msgsnd(msqid, &msg, strlen(msg.mtext)+1, 0);
    msgrcv(msqid, &msg, sizeof(msg.mtext), 0, 0);
    printf("msgq: %s\n", msg.mtext);

    /* 信号量操作 */
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    /* 初始化信号量 */
    semctl(semid, 0, SETVAL, 1);
    /* P 操作 (减1) */
    struct sembuf sop = {0, -1, 0};
    semop(semid, &sop, 1);
    printf("sem: P done, value=%d\n", semctl(semid, 0, GETVAL));
    /* V 操作 (加1) */
    sop.sem_op = 1;
    semop(semid, &sop, 1);
    printf("sem: V done, value=%d\n", semctl(semid, 0, GETVAL));

    /* 共享内存操作 */
    int shmid = shmget(IPC_PRIVATE, 4096, IPC_CREAT | 0666);
    char *shm = shmat(shmid, NULL, 0);
    strcpy(shm, "shared!");
    printf("shm: %s\n", shm);
    shmdt(shm);

    /* 清理 */
    msgctl(msqid, IPC_RMID, NULL);
    semctl(semid, 0, IPC_RMID);
    shmctl(shmid, IPC_RMID, NULL);
    return 0;
}

45.7

–45.8 缺陷与限额

本章：TLPI 第 45 章 — Introduction to System V IPC - 上一节：45.6 内核持久 · 运维 - 下一节：45.9 IPC_RMID 差异

本节讲什么

  1. 非 fd，无多路复用 2. 易泄漏，无自动回收 3. 无路径名，ls/rm 管不了

要点

  1. 非 fd，无多路复用
  2. 易泄漏，无自动回收
  3. 无路径名，ls/rm 管不了
  4. API 老旧；新项目优先 POSIX IPC

限额：/proc/sys/kernel/msg*、sem、shm*；超限常 ENOSPC。

HFT / 嵌入式关联

路线 要点
嵌入式 非 fd 无多路复用、易泄漏无自动回收、无路径名 ls/rm 管不了，API 老旧
HFT 新项目优先 POSIX IPC（fd+unlink 引用计数），SysV 仅遗留系统兼容

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <unistd.h>
/* SysV IPC 的缺陷: 泄漏/配额/无引用计数 */
int main(void) {
    /* 缺陷1: 内核持久, 进程退出不自动删除 */
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    printf("Created msgq id=%d\n", msqid);
    printf("If we forget to delete, it LEAKS until reboot!\n");

    /* 缺陷2: 系统配额限制 */
    struct msqid_ds ds;
    msgctl(msqid, IPC_STAT, &ds);
    printf("Max queue bytes: %lu\n", (unsigned long)ds.msg_qbytes);

    /* 查看系统限制 */
    printf("System limits (ipcs -l):\n");
    printf("  max queues: %ld\n", sysconf(_SC_MQ_OPEN_MAX));

    /* 缺陷3: key 冲突可能 */
    printf("\nftok may collide if files share inode low bits\n");

    /* 正确做法: 用完立即删 */
    msgctl(msqid, IPC_RMID, NULL);
    printf("Deleted (no leak)\n");
    return 0;
}

HFT / 嵌入式关联

  • shmmax 调大：HFT 共享内存行情数据可能数 GB，需调高 shmmax 和 shmall
  • msgmax：HFT 行情消息可能超过默认 8KB，调大到 64KB
  • shmmni：多策略进程共享多段，可能需调高
  • systemd LimitSHM：LimitSHM=infinity 放宽共享内存限制
  • 嵌入式：精简系统可能减少 IPC 限制或裁剪 SysV IPC 支持
  • 监控：运维脚本定期检查 ipcs 输出，发现泄漏 IPC 对象

代码示例

#include <stdio.h>
#include <sys/shm.h>

void check_shm_limits(void) {
    int max_segments = 0;
    FILE *f = fopen("/proc/sys/kernel/shmmni", "r");
    if (f) { fscanf(f, "%d", &max_segments); fclose(f); }
    printf("max shm segments: %d\n", max_segments);

    long max_size = 0;
    f = fopen("/proc/sys/kernel/shmmax", "r");
    if (f) { fscanf(f, "%ld", &max_size); fclose(f); }
    printf("max shm size: %ld bytes (%.1f GB)\n",
           max_size, (double)max_size / (1<<30));
}

int create_large_shm(key_t key, size_t size) {
    int shmid = shmget(key, size, IPC_CREAT | 0660);
    if (shmid == -1) {
        if (errno == EINVAL) {
            fprintf(stderr, "size %zu exceeds shmmax\n", size);
        } else if (errno == ENOSPC) {
            fprintf(stderr, "shm segments limit reached\n");
        }
    }
    return shmid;
}

代码自测

1. shmmax 和 shmall 的区别？
  • shmmax：单个共享内存段的最大字节数
  • shmall：系统所有共享内存段的总页数上限（乘以页大小得到字节数）

即使 shmmax 允许创建 2GB 的段，如果 shmall 限制总共享内存为 4MB，也无法创建。两者需同时满足。

2. 为什么 IPC 对象不自动回收？

SysV IPC 对象的生命周期独立于创建进程——即使创建进程退出，IPC 对象仍然存在直到：(1) 显式 xxxctl(IPC_RMID) 删除；(2) 系统重启。这是设计意图（允许进程崩溃后 IPC 对象保留），但也是常见资源泄漏来源。运维需定期 ipcs 检查并 ipcrm 清理。

`IPC_RMID` 差异

本节讲什么

要点

对象 行为
消息队列 / 信号量集 立即标记删除；持有 id 者可继续用，引用尽后释放
共享内存 仅标记；全部 shmdt 后才真正销毁

HFT / 嵌入式关联

路线 要点
嵌入式 mq/sem 立即标记删，shm 延迟到全 shmdt；持有者仍可继续用
HFT 理解 shm 延迟删除语义，避免一边 RMID 一边仍 shmat 访问

代码示例

#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>
/* IPC_RMID: 删除 IPC 对象 (必须显式调用) */
int main(void) {
    /* 创建三个 IPC 对象 */
    int msqid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    int shmid = shmget(IPC_PRIVATE, 4096, IPC_CREAT | 0666);

    printf("Created: msgq=%d, sem=%d, shm=%d\n", msqid, semid, shmid);
    printf("Use 'ipcs' to see them\n");

    /* 删除消息队列 */
    if (msgctl(msqid, IPC_RMID, NULL) == 0)
        printf("msgq %d deleted\n", msqid);

    /* 删除信号量集 */
    if (semctl(semid, 0, IPC_RMID) == 0)
        printf("sem %d deleted\n", semid);

    /* 删除共享内存 */
    if (shmctl(shmid, IPC_RMID, NULL) == 0)
        printf("shm %d deleted\n", shmid);

    /* 如果忘记删除, 重启前一直存在 */
    /* 命令行删除: ipcrm -q <id> / ipcrm -s <id> / ipcrm -m <id> */
    return 0;
}

练习 1 参考代码

#include "tlpi_hdr.h"
#include <sys/ipc.h>
#include <sys/msg.h>

int main(int argc, char *argv[]) {
    if (argc != 3)
        usageErr("usage: %s <path> <proj_id_char>\n", argv[0]);

    key_t key = ftok(argv[1], argv[2][0]);
    if (key == -1) errExit("ftok");

    printf("key = 0x%08x\n", (unsigned int)key);

    int msqid = msgget(key, IPC_CREAT | IPC_EXCL | 0660);
    if (msqid == -1) {
        if (errno == EEXIST)
            printf("queue already exists, key=0x%x\n", (unsigned int)key);
        else
            errExit("msgget");
    } else {
        printf("created msqid=%d\n", msqid);
    }
    return 0;
}

void show_msgq(int msqid) {
    struct msqid_ds ds;
    if (msgctl(msqid, IPC_STAT, &ds) == -1) return;

    printf("Message Queue (id=%d):\n", msqid);

练习 3 参考代码

/* 清理脚本（需 root 权限扫描所有对象） */
#include <stdio.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/sem.h>
#include <sys/shm.h>

/* 注意：Linux 没有 API 枚举所有 IPC 对象。
 * 实际清理使用 ipcs/ipcrm 命令。
 * 以下用 system() 调用 ipcrm */
void cleanup_ipc(uid_t target_uid) {
    char cmd[256];
    snprintf(cmd, sizeof(cmd),
        "ipcs -q | awk '$3==\"%d\" {print $2}' | xargs -r ipcrm -q", target_uid);
    system(cmd);

    snprintf(cmd, sizeof(cmd),
        "ipcs -s | awk '$3==\"%d\" {print $2}' | xargs -r ipcrm -s", target_uid);
    system(cmd);

    snprintf(cmd, sizeof(cmd),
        "ipcs -m | awk '$3==\"%d\" {print $2}' | xargs -r ipcrm -m", target_uid);
    system(cmd);
}
