章节目标

理解堆与 program break、brk/sbrk；掌握 malloc/calloc/realloc/free；分清用户态分配器与内核虚拟内存交互；虚拟地址 ≠ 立刻占用物理页。

易错清单

  1. brk/sbrk =（或封装）syscall；malloc/free = 库函数。
  2. 虚拟 ≠ 物理；抬 break 不立刻吃物理页。
  3. free 不保证还内核；别用 sbrk(0) 当「真实占用」仪表。
  4. 勿写 ptr = realloc(ptr, …) 无临时变量。
  5. malloc 返回值满足最大基本对齐。
  6. 书内 mmap 章是 Ch49，不是 Ch48。

章节链路

Ch6  地址空间（堆在哪）
  → Ch7  brk/sbrk + malloc 族 + 双路径
  → Ch49 mmap 匿名映射
  → Ch24 fork：堆随地址空间 COW

双线提示

路线
嵌入式 控泄漏；大块/常驻慎用；理解为何 RSS ≠ 所有 malloc 之和
HFT 热路径少 malloc；预分配 / 池；大块与 mmap/大页衔接 Ch49–50

背诵卡

# 要点
1 program break = 堆上界；上移只加虚拟地址
2 sbrk(0) 查边界；业务禁用 brk/sbrk
3 free 小块常进空闲链表，break 未必降
4 小块 sbrk / 大块 mmap
5 realloc 用临时指针再赋值

  • Kerrisk, The Linux Programming Interface, Chapter 7 — Memory Allocation
  • OUTLINE · Ch6 · Ch49

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>

/* Ch7 内存分配 — malloc/free vs sbrk/brk。
 * malloc 内部用 sbrk/mmap 向内核要内存；
 * 程序员只管 malloc/free，不必直接调 sbrk。
 * 编译: gcc -o ch7_demo ch7_demo.c */

int main(void) {
    /* 查看当前 program break */
    void *brk1 = sbrk(0);
    printf("initial brk = %p\n", brk1);

    /* malloc 向 libc 要内存，libc 内部可能调 sbrk */
    char *p = malloc(1024 * 1024);  /* 1 MB */
    if (!p) { perror("malloc"); return 1; }
    memset(p, 'A', 1024 * 1024);

    void *brk2 = sbrk(0);
    printf("after malloc(1MB): brk = %p (delta = %ld)\n",
           brk2, (long)(brk2 - brk1));

    /* realloc 调整大小 */
    p = realloc(p, 2 * 1024 * 1024);
    printf("realloc to 2MB: %s\n", p ? "OK" : "FAIL");

    free(p);
    printf("freed\n");
    return 0;
}

7.1

Program Break（程序断点）

本章：TLPI 第 07 章 — Memory Allocation - 上一节：本章首页 - 下一节：7.2 brk / sbrk

本节讲什么

堆在 BSS 上方；program break = 堆当前上边界。 ---

要点

堆在 BSS 上方；program break = 堆当前上边界。

动作 效果
上移断点 扩大堆，获得更多虚拟地址空间
下移断点 缩小堆，归还虚拟地址给内核

关键： 抬高断点只预留虚拟地址；首次访问时才通过缺页异常分配物理页（延迟分配）。

代码示例

#include <unistd.h>
#include <stdio.h>

int main(void) {
    /* sbrk(0) 返回当前 program break（堆上边界） */
    void *brk1 = sbrk(0);
    printf("初始 program break = %p\n", brk1);

    /* 手动抬高断点 4096 字节（分配虚拟地址） */
    void *brk2 = sbrk(4096);
    printf("sbrk(4096) 返回旧 break = %p\n", brk2);
    printf("新 break = %p (差 %ld 字节)\n", sbrk(0), (char *)sbrk(0) - (char *)brk1);

    /* 可以在抬高的区域写数据 */
    char *p = (char *)brk2;
    p[0] = 'X';
    p[4095] = 'Y';
    printf("写入成功: p[0]=%c p[4095]=%c\n", p[0], p[4095]);

    /* 下移断点，归还虚拟地址 */
    sbrk(-4096);
    printf("sbrk(-4096) 后 break = %p (回到原位)\n", sbrk(0));

    return 0;
}

7.2 brk / sbrk

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.1 Program Break（程序断点） - 下一节：7.3 malloc / free

本节讲什么

include int brk(void end_data_segment); / 断点设到绝对地址；0 成功，-1 失败 / void sbrk(intptr_t increment); / 相对移动；成功返回调整前地址 /

要点

#include <unistd.h>
int brk(void *end_data_segment);     /* 断点设到绝对地址；0 成功，-1 失败 */
void *sbrk(intptr_t increment);      /* 相对移动；成功返回调整前地址 */
规则
sbrk(0) 不移动，返回当前 break（查边界）
sbrk 成功 返回调整前的 break；失败 (void *)-1
Linux sbrk 多是 glibc 封装，底层 brk
业务代码 禁止直接 brk/sbrk；只为理解 malloc
下移过猛 勿把断点压到初始堆底以下 → 易 SIGSEGV

Demo：code/sbrk_probe.c

代码示例

#include <unistd.h>
#include <stdio.h>

int main(void) {
    void *initial = sbrk(0);
    printf("初始 break = %p\n", initial);

    /* sbrk(n): 相对移动 n 字节，返回调整前的 break */
    void *old = sbrk(1024);
    if (old == (void *)-1) { perror("sbrk"); return 1; }
    printf("sbrk(1024): old=%p, new=%p\n", old, sbrk(0));

    /* 在新分配的区域使用 */
    char *buf = (char *)old;
    buf[0] = 'A';
    buf[1023] = 'Z';
    printf("写入: buf[0]=%c, buf[1023]=%c\n", buf[0], buf[1023]);

    /* brk(addr): 设置绝对断点地址 */
    void *target = (char *)initial + 2048;
    if (brk(target) != 0) { perror("brk"); return 1; }
    printf("brk(%p) 后 break = %p\n", target, sbrk(0));

    /* 恢复到初始位置 */
    brk(initial);
    printf("恢复 break = %p\n", sbrk(0));

    /* 注意：业务代码永远不要直接用 brk/sbrk，用 malloc/free */
    return 0;
}

7.3 malloc / free

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.2 brk / sbrk - 下一节：7.4 calloc / realloc

本节讲什么

include void malloc(size_t size); / 未初始化；失败 NULL + errno / void free(void ptr);

要点

#include <stdlib.h>
void *malloc(size_t size);   /* 未初始化；失败 NULL + errno */
void free(void *ptr);

块布局

用户指针不含分配器头部；元数据（大小、链表指针等）在用户区前方。
越界写 → 破坏头部 → free 常直接崩。

free 要点

  1. ptr 必须来自 malloc/calloc/realloc；否则 UB。
  2. free(NULL) 安全、空操作。
  3. free 不一定下调 program break
    - 小块：进 glibc 空闲链表，供后续复用（少 syscall）
    - 大块连续空闲：才可能 sbrk(-n) 还虚拟内存
  4. free 不清零内容（信息残留风险）。

Demo：code/free_and_sbrk.c（Listing 7-1 精神）

代码示例

#include <stdlib.h>
#include <stdio.h>
#include <errno.h>
#include <string.h>

int main(void) {
    /* malloc 基本用法 + 错误检查 */
    int *arr = malloc(10 * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "malloc 失败: %s\n", strerror(errno));
        return 1;
    }

    /* malloc 不清零！内容不确定 */
    for (int i = 0; i < 10; i++) {
        arr[i] = i * i;
    }
    printf("arr[5] = %d\n", arr[5]);  /* 25 */

    /* realloc 改变大小 */
    int *bigger = realloc(arr, 20 * sizeof(int));
    if (bigger == NULL) {
        free(arr);  /* 原 arr 仍有效 */
        perror("realloc");
        return 1;
    }
    arr = bigger;  /* realloc 成功后旧指针已失效 */
    arr[15] = 999;
    printf("arr[15] = %d (realloc 后的新区域)\n", arr[15]);

    /* free(NULL) 安全，什么都不做 */
    free(NULL);
    printf("free(NULL) 安全\n");

    /* free 后指针变悬挂 — 置 NULL 是好习惯 */
    free(arr);
    arr = NULL;
    printf("free 完成, arr = NULL\n");

    return 0;
}

7.4 calloc / realloc

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.3 malloc / free - 下一节：7.5 malloc(0)

本节讲什么

void calloc(size_t numitems, size_t size); / 置零 / void realloc(void ptr, size_t newsize); / ❌ ptr = realloc(ptr, n); 失败时原指针丢失 → 泄漏 */

要点

void *calloc(size_t numitems, size_t size);  /* 置零 */
void *realloc(void *ptr, size_t newsize);
calloc(n,s) ≈ malloc(n*s) + 清零；适合数组
realloc 扩容 后方够 → 原地；不够 → 新块 + 拷贝 + 释放旧块（指针可能变）
realloc 缩容 截断尾部；起始指针通常不变 
/* ❌ ptr = realloc(ptr, n); 失败时原指针丢失 → 泄漏 */
void *np = realloc(ptr, newsize);
if (np == NULL) { /* ptr 仍有效 */ }
else { ptr = np; }

另：realloc(NULL, n) ≈ malloc(n)；realloc(ptr, 0) 平台不统一，尽量避免。

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
/* calloc: 分配并清零; realloc: 调整大小 */
int main(void) {
    /* calloc: 分配 10 个 int, 自动清零 */
    int *arr = calloc(10, sizeof(int));
    printf("calloc[0]=%d (zeroed)\n", arr[0]);

    /* realloc: 扩大到 20 个 int */
    arr = realloc(arr, 20 * sizeof(int));
    /* realloc 不保证新增部分清零 */
    printf("realloc[15]=%d (may be garbage)\n", arr[15]);

    /* realloc(NULL, size) 等价于 malloc */
    int *p = realloc(NULL, 100);
    printf("realloc(NULL,100) = %p\n", (void*)p);

    /* realloc(ptr, 0) 等价于 free */
    realloc(p, 0);

    free(arr);
    return 0;
}

7.5 malloc(0)

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.4 calloc / realloc - 下一节：7.6 glibc 双路径（拓展重点）

本节讲什么

POSIX 允许返回 NULL 或 可 free 的非空指针。 glibc：常返回有效指针，必须 free，不能当「失败」处理。 ---

要点

POSIX 允许返回 NULL 或 可 free 的非空指针。
glibc：常返回有效指针，必须 free，不能当「失败」处理。

代码示例

#include <stdlib.h>
#include <stdio.h>

int main(void) {
    /* malloc(0) 的行为: POSIX 允许返回 NULL 或可 free 的非空指针 */
    void *p = malloc(0);
    printf("malloc(0) = %p\n", p);

    if (p != NULL) {
        printf("glibc 返回非空指针 → 必须 free!\n");
        /* 不能解引用！大小为 0，访问是 UB */
        free(p);
        printf("已 free\n");
    } else {
        printf("返回 NULL → 不需要 free\n");
    }

    /* 实际场景: size 变量为 0 时 */
    size_t n = 0;  /* 可能来自计算结果 */
    char *buf = malloc(n);
    if (buf == NULL && n > 0) {
        perror("malloc");
        return 1;
    }
    /* 无论 NULL 还是非空，free 都安全 */
    free(buf);

    return 0;
}

7.6 glibc 双路径（拓展重点）

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.5 malloc(0) - 下一节：7.7 典型内存错误

本节讲什么

→ 另一种拿虚拟内存的主路径：书内 Ch49 Memory Mappings（勿与 SysV SHM Ch48 混淆）。 ---

要点

路径 方式 特点
小块 sbrk 扩主堆 释放后常留在空闲链表
大块（默认约 >128KB，可配） mmap(MAP_ANONYMOUS) free 时常立刻还内核；不绑在 program break 上

→ 另一种拿虚拟内存的主路径：书内 Ch49 Memory Mappings（勿与 SysV SHM Ch48 混淆）。

代码示例

#include <stdlib.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    void *brk_before = sbrk(0);

    /* 小块分配: 走 sbrk 扩主堆 */
    void *small = malloc(64);
    printf("小块(64B): %p, sbrk 变化: %s\n",
           small,
           sbrk(0) != brk_before ? "是(主堆扩展)" : "否");

    /* 大块分配: 走 mmap (默认 >128KB) */
    void *large = malloc(256 * 1024);  /* 256KB */
    printf("大块(256KB): %p\n", large);

    /* 检查大块是否在 mmap 区域（远离堆） */
    void *brk_after = sbrk(0);
    long heap_grew = (char *)brk_after - (char *)brk_before;
    printf("堆增长: %ld 字节 (大块不走 sbrk, 堆几乎没变)\n", heap_grew);

    /* free 大块时常立刻还内核 (mmap 的区域 munmap) */
    free(large);
    printf("free 大块后: 堆 break = %p (不变, mmap 区域已归还)\n", sbrk(0));

    /* free 小块后: 堆 break 通常不变 (留在空闲链表) */
    free(small);
    printf("free 小块后: 堆 break = %p (不变, 留在空闲链表)\n", sbrk(0));

    return 0;
}

7.7 典型内存错误

本章：TLPI 第 07 章 — Memory Allocation - 上一节：7.6 glibc 双路径（拓展重点） - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

  1. 泄漏：丢指针未 free 2. 双重 free：破坏空闲链表 3. 野指针 free：栈/全局/非堆指针

要点

  1. 泄漏：丢指针未 free
  2. 双重 free：破坏空闲链表
  3. 野指针 free：栈/全局/非堆指针
  4. 越界：覆盖块头
  5. 悬挂指针：free 后继续用

代码示例

#include <stdlib.h>
#include <stdio.h>

int main(void) {
    /* 1. 内存泄漏: 丢指针未 free */
    {
        int *p = malloc(sizeof(int));
        *p = 42;
        /* 作用域结束, p 消失, 但堆内存没 free → 泄漏 */
    }
    printf("1. 泄漏: 块无法回收\n");

    /* 2. 正确做法: 用完就 free */
    {
        int *p = malloc(sizeof(int));
        *p = 42;
        free(p);       /* 释放内存 */
        p = NULL;      /* 置空防悬挂 */
    }
    printf("2. 正确: free + 置 NULL\n");

    /* 3. free(NULL) 安全 */
    int *p = NULL;
    free(p);
    printf("3. free(NULL) 安全\n");

    /* 4. calloc 自动清零 (对比 malloc 不清零) */
    int *arr = calloc(5, sizeof(int));
    printf("4. calloc 清零: arr[0]=%d, arr[4]=%d\n", arr[0], arr[4]);
    free(arr);

    /* 5. 越界写破坏块头 (演示概念, 不要这样做!) */
    int *bad = malloc(4);
    /* bad[10] = 0xDEAD;  // 越界! 破坏分配器元数据, free 时崩溃 */
    free(bad);
    printf("5. 越界写 → free 时可能崩溃 (未演示)\n");

    return 0;
}

7.2

Allocating Memory on the Stack: alloca()

本章：TLPI 第 7 章

本节讲什么

alloca() 在栈上分配内存，函数返回时自动释放。

原型

void *alloca(size_t size);

特性

  • 分配在栈帧上，函数返回自动释放
  • 不需要 free，不会内存泄漏
  • 速度极快（仅调整 sp）
  • 分配大小受栈限制（默认 8MB）

vs malloc

特性 alloca malloc
分配位置 栈 堆
释放方式 函数返回自动 手动 free
速度 极快 较慢
大小限制 栈大小 虚拟地址空间

变长数组（C99）

C99 的变长数组（VLA）本质等同于 alloca，也在栈上分配。

int n = 10;
int arr[n];  // VLA，栈上分配

注意

  • alloca 不释放内存，只是调整 sp → 不要在循环中大量调用
  • 信号处理器中用 alloca 可能导致栈溢出

内存分配方式对比

堆分配 malloc/calloc/realloc 堆 free 最常用，需手动管理

malloc 家族

函数 原型 说明
malloc(size) void *malloc(size_t size) 分配 size 字节，内容未初始化

malloc 调整规则

场景 行为
realloc(NULL, n) 等价 malloc(n)
realloc(p, 0) 实现定义（C11 起未定义，glibc 行为：释放并返回 NULL）
realloc(p, smaller) 可能原地缩减，返回 p
realloc(p, bigger) 可能移动，原内存自动释放

错误与陷阱

问题 说明 预防
内存泄漏 malloc 不 free valgrind/asan
重复释放 free 同一指针两次 p = NULL 释放后置空
越界写 写超出分配边界 asan/fortify
use-after-free 释放后仍访问 悬空指针置 NULL
未检查返回值 malloc 返回 NULL 被解引用 检查 if (p == NULL)
realloc 后旧指针 移动后访问旧地址 只用返回的新指针

HFT / 嵌入式关联

  • 低延迟分配：HFT 热路径避免 malloc（不可预测延迟），预分配池/arena
  • 内存池：固定大小块分配器（obstack/自实现），O(1) 分配释放
  • Huge Pages：mmap(MAP_HUGETLB) 减少 TLB miss，HFT 关键数据结构
  • 对齐：posix_memalign 用于 SIMD（AVX-512 需 64 字节对齐）
  • 嵌入式：无 MMU 系统（uClinux）不能用 brk，只能 mmap
  • mmap 替代：大块内存用 mmap 更高效，可 MAP_LOCKED 防止 swap

代码示例

/* 安全 realloc 模式 */
void *safe_realloc(void **ptr, size_t new_size) {
    void *new_ptr = realloc(*ptr, new_size);
    if (new_ptr == NULL) {
        free(*ptr);   /* 原内存仍需释放 */
        *ptr = NULL;
        return NULL;
    }
    *ptr = new_ptr;
    return new_ptr;
}

/* 内存池简单实现 */
#define POOL_SIZE 4096
static char pool[POOL_SIZE];
static size_t pool_off = 0;

void *pool_alloc(size_t size) {
    size = (size + 15) & ~15;  /* 16 字节对齐 */
    if (pool_off + size > POOL_SIZE) return NULL;
    void *p = pool + pool_off;
    pool_off += size;
    return p;  /* 不需要 free，整池重置 */
}

代码自测

1. calloc 和 malloc + memset 的区别？
  • calloc(n, size) 内部检测 n × size 整数溢出，更安全
  • calloc 可能利用内核的页清零（mmap 返回的匿名页已被清零），避免实际 memset
  • malloc + memset 需要两次操作，且不检查溢出
  • 在 glibc 中，大块 calloc 使用 mmap，页可能已经被清零
2. realloc 移动内存后，原指针的状态？

realloc 成功且返回新地址时，原内存被自动释放。旧指针变为悬空指针，不可再使用。必须只使用返回值。如果 realloc 失败（返回 NULL），原内存仍有效，需要手动 free。

3. alloca 的优势和使用限制？

优势：栈上分配，函数返回自动释放，无需 free，不会内存泄漏。 限制：不能跨函数返回值传递；栈大小有限（默认 8MB），大块分配导致栈溢出；不可移植（非标准 C）。

练习 1 参考代码

#include <unistd.h>
#include <string.h>

typedef struct block {
    size_t size;        /* 数据区大小 */
    int free;          /* 是否空闲 */
    struct block *next; /* 空闲链表 */
} block_t;

#define META_SIZE sizeof(block_t)
static block_t *free_list = NULL;

void *my_malloc(size_t size) {
    block_t *prev = NULL, *curr = free_list;
    while (curr) {
        if (curr->free && curr->size >= size) {
            /* 分割大块 */
            if (curr->size >= size + META_SIZE + 16) {
                block_t *split = (block_t *)((char *)curr + META_SIZE + size);
                split->size = curr->size - size - META_SIZE;
                split->free = 1;
                split->next = curr->next;
                curr->next = split;
                curr->size = size;
            }
            curr->free = 0;
            return (char *)curr + META_SIZE;
        }
        prev = curr;
        curr = curr->next;
    }
    /* 没有合适块，sbrk 分配新块 */
    block_t *new_block = sbrk(0);
    size_t total = size + META_SIZE;
    if (sbrk(total) == (void *)-1) return NULL;
    new_block->size = size;
    new_block->free = 0;
    new_block->next = NULL;
    if (prev) prev->next = new_block;
    else free_list = new_block;
    return (char *)new_block + META_SIZE;
}

void my_free(void *ptr) {
    if (!ptr) return;
    block_t *b = (block_t *)((char *)ptr - META_SIZE);
    b->free = 1;
    /* 合并相邻空闲块（简化版：只检查 next） */
    if (b->next && b->next->free) {
        b->size += META_SIZE + b->next->size;
        b->next = b->next->next;
    }
}

练习 2 答案

角度 calloc malloc + memset
安全性 检查 n × size 整数溢出 不检查，n*size 可能回绕
性能 大块可能利用内核零页，跳过 memset 必须实际清零

练习 3 答案

realloc(ptr, 0) 的行为在不同标准间存在差异：

标准行为 说明
C99/C11 实现定义：可能释放 ptr 返回 NULL，也可能返回非空
glibc 释放 ptr，返回 NULL（free(ptr) + 返回 NULL）
BSD/macOS 释放 ptr，返回指向最小分配块的指针（非 NULL）

为什么未定义：realloc 的语义是"调整大小"，大小为 0 时应"释放"但返回值语义矛盾——是 NULL 还是有效指针？标准选择不规定，留给实现。

建议：永远不要 realloc(ptr, 0)，直接 free(ptr); ptr = NULL;。

HFT / 嵌入式关联

  • 自定义分配器：HFT 场景下简化 malloc 的思路可扩展为固定大小内存池
  • 零延迟路径：HFT 热路径完全避免动态分配，用栈分配或预分配池
  • 嵌入式：无 MMU 系统（uClinux）不能使用 sbrk，只能 mmap
