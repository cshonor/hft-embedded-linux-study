章节目标

理解 xattr 键值模型与四大命名空间；掌握 set/get/remove/list 系统调用；熟悉权限与大小限制；为 ACL、文件能力、安全标签打底。

16.8 速查：命名空间 · 调用族

path 不跟随链接 fd
set setxattr lsetxattr fsetxattr
get getxattr lgetxattr fgetxattr
remove removexattr lremovexattr fremovexattr
list listxattr llistxattr flistxattr
NS 一句话
user 用户元数据；受文件 r/w
trusted 仅 root
security 安全模块 / caps
system 内核；含 ACL

16.9 易错清单

  1. 键不存在：ENODATA（ENOATTR）
  2. getxattr/listxattr 先 size=0 探长度
  3. list 无 value，要逐个 get
  4. 备份丢 EA → ACL/caps 一起丢
  5. 勿假设全 FS 支持 xattr
  6. 默认 setxattr 跟随符号链接

练习

  1. 对 user.* 做增删改查 + listxattr
  2. 硬链接共享；cp 默认 vs --preserve=xattr
  3. 非 root 写 trusted.* → 拒绝
  4. （选）看目录上是否已有 system.posix_acl_*（有 ACL 时）

背诵卡

# 要点
1 键格式 namespace.name；日常用 user.*
2 ACL / caps 分别在 system.* / security.*
3 list 是 \0 分隔键表；先 size=0 再分配
4 cp/rsync 默认可能丢 EA
5 硬链接共享；处理 ENOTSUP
6 trusted 仅 root；user 看文件 r/w

代码示例

#include <stdio.h>
#include <sys/xattr.h>
#include <string.h>

/* Ch16 扩展属性 — getxattr/setxattr/listxattr。
 * xattr 是文件 inode 上附带的 key-value 元数据。
 * 编译: gcc -o ch16_demo ch16_demo.c
 * 需要 attr 包: apt install attr */

int main(void) {
    const char *path = "/tmp/ch16_test.txt";
    FILE *fp = fopen(path, "w");
    if (!fp) { perror("fopen"); return 1; }
    fprintf(fp, "test\n");
    fclose(fp);

    /* 设置扩展属性 */
    if (setxattr(path, "user.comment", "hello xattr", 11, 0) < 0)
        perror("setxattr (need user namespace + ext4/xfs)");

    /* 列出所有扩展属性 */
    char list[256];
    ssize_t llen = listxattr(path, list, sizeof(list));
    if (llen > 0) {
        printf("xattr keys:\n");
        for (ssize_t i = 0; i < llen; ) {
            printf("  %s\n", list + i);
            i += strlen(list + i) + 1;
        }
    }

    /* 读取扩展属性值 */
    char val[256];
    ssize_t vlen = getxattr(path, "user.comment", val, sizeof(val) - 1);
    if (vlen >= 0) {
        val[vlen] = '\0';
        printf("user.comment = %s\n", val);
    }

    remove(path);
    return 0;
}

16.1

概念

本章：TLPI 第 16 章 — Extended Attributes - 上一节：本章首页 - 下一节：16.2 系统调用组

本节讲什么

标准 inode 字段固定；扩展属性（EA） 附加任意 key → value（多为小元数据，非大文件内容替代）。 - 对象：普通文件、目录、链接、设备、FIFO 等（视 FS） - 支持：ext4 / XFS / Btrfs / tmpfs 等；不保证所有 FS → 处理 ENOTSUP

要点

标准 inode 字段固定；扩展属性（EA） 附加任意 key → value（多为小元数据，非大文件内容替代）。

  • 对象：普通文件、目录、链接、设备、FIFO 等（视 FS）
  • 支持：ext4 / XFS / Btrfs / tmpfs 等；不保证所有 FS → 处理 ENOTSUP
  • 符号链接：默认跟随目标；操作链接本体用 l* 系列（FS 对 symlink EA 支持常很弱）

键名：namespace.name

命名空间 谁用 权限要点
user.* 普通自定义元数据 受文件 r/w 权限约束
trusted.* 管理员私有 仅 root（EUID）
security.* SELinux、security.capability 等 多为内核/特权
system.* 内核用途；ACL 在此（system.posix_acl_*） 普通用户一般不能乱改

POSIX ACL 实质是存在 system.* 里的 EA → Ch17。

代码示例

#include <sys/xattr.h>
#include <stdio.h>

int main(void) {
    const char *path = "/tmp/xattr_test.txt";

    /* 创建测试文件 */
    FILE *fp = fopen(path, "w");
    if (fp) { fprintf(fp, "test"); fclose(fp); }

    /* 设置扩展属性 (key-value 元数据) */
    if (setxattr(path, "user.comment", "hello world", 11, 0) == 0) {
        printf("设置 user.comment = hello world\n");
    } else {
        perror("setxattr");
    }

    /* 读取扩展属性 */
    char buf[256];
    ssize_t n = getxattr(path, "user.comment", buf, sizeof(buf) - 1);
    if (n > 0) {
        buf[n] = '\0';
        printf("读取 user.comment = %s\n", buf);
    }

    /* user.* 命名空间: 普通用户可用 (需有写权限) */
    /* trusted.* 命名空间: 需 root */
    /* security.* / system.*: 内核用 */
    return 0;
}

代码示例

#include <stdio.h>
#include <sys/xattr.h>
#include <string.h>
/* xattr 系统调用族: set/get/list/remove */
int main(void) {
    const char *path = "/tmp/xattr_syscall_test.txt";
    FILE *fp = fopen(path, "w"); fprintf(fp, "x"); fclose(fp);

    /* setxattr: 设置 */
    setxattr(path, "user.author", "alice", 5, 0);
    setxattr(path, "user.priority", "high", 4, 0);

    /* listxattr: 列出所有 key */
    char list[256];
    ssize_t llen = listxattr(path, list, sizeof(list));
    printf("xattr keys (%zd bytes):\n", llen);
    for (ssize_t i = 0; i < llen; ) {
        printf("  %s\n", list + i);
        i += strlen(list + i) + 1;
    }

    /* getxattr: 读取单个 */
    char val[32];
    ssize_t vlen = getxattr(path, "user.author", val, sizeof(val));
    if (vlen > 0) { val[vlen] = '\0'; printf("user.author = %s\n", val); }

    /* removexattr: 删除 */
    removexattr(path, "user.priority");
    printf("After removexattr(user.priority):\n");
    llen = listxattr(path, list, sizeof(list));
    printf("  remaining keys: %zd bytes\n", llen);

    remove(path);
    return 0;
}

16.3

`listxattr` 格式

本章：TLPI 第 16 章 — Extended Attributes - 上一节：16.2 系统调用组 - 下一节：16.4 权限（Linux 要点）

本节讲什么

缓冲区为连续 \0 结尾的键名： user.foo\0user.bar\0 返回值 = 总字节数。范式：

要点

缓冲区为连续 \0 结尾的键名：

user.foo\0user.bar\0

返回值 = 总字节数。范式：

  1. listxattr(path, NULL, 0)（或 size=0）得长度
  2. 分配缓冲再读；getxattr(..., size=0) 同理得 value 长度

listxattr 只有键，value 需再 getxattr。

Demo：code/xattr_demo.c

代码示例

#include <sys/xattr.h>
#include <stdio.h>

int main(void) {
    const char *path = "/tmp/xattr_list_test.txt";

    /* 创建文件并设置多个 xattr */
    FILE *fp = fopen(path, "w");
    if (fp) { fprintf(fp, "test"); fclose(fp); }
    setxattr(path, "user.comment", "hello", 5, 0);
    setxattr(path, "user.author", "alice", 5, 0);
    setxattr(path, "user.version", "1", 1, 0);

    /* listxattr: 列出所有 xattr 名 (null 分隔) */
    char list[1024];
    ssize_t size = listxattr(path, list, sizeof(list));
    if (size > 0) {
        printf("扩展属性列表 (%zd bytes):\n", size);
        for (ssize_t i = 0; i < size; ) {
            printf("  %s\n", list + i);
            i += strlen(list + i) + 1;
        }
    } else {
        printf("无扩展属性\n");
    }
    return 0;
}

16.4

权限（Linux 要点）

本章：TLPI 第 16 章 — Extended Attributes - 上一节：16.3 listxattr 格式 - 下一节：16.5 限制与踩坑

本节讲什么

user.*：不是「创建者永久拥有」——换人改权限后规则仍按当前 r/w。 ---

要点

命名空间 set / remove get
user.* 文件写权限 文件读权限
trusted.* 仅 root 仅 root
security.* / system.* 多为特权/内核 视策略

user.*：不是「创建者永久拥有」——换人改权限后规则仍按当前 r/w。

代码示例

#include <sys/xattr.h>
#include <stdio.h>

int main(void) {
    /* user.* 命名空间: 需要对文件有写权限 */
    /* trusted.* 命名空间: 需要 root (CAP_SYS_ADMIN) */

    const char *path = "/etc/passwd";

    /* 尝试设置 user.* xattr 到不拥有的文件 */
    if (setxattr(path, "user.test", "x", 1, 0) != 0) {
        printf("设置 /etc/passwd 的 user.* 失败 (预期): ");
        perror("");
    }

    /* 读取不需要写权限, 但需要读权限 */
    char buf[256];
    ssize_t n = getxattr(path, "user.test", buf, sizeof(buf));
    if (n < 0) {
        printf("读取 user.test: 不存在或无权限 (预期)\n");
    }

    /* 权限规则:
     * user.* → 需要文件写权限 (设/删), 读权限 (读)
     * trusted.* → 需要 CAP_SYS_ADMIN
     */
    return 0;
}

16.5

限制与踩坑

本章：TLPI 第 16 章 — Extended Attributes - 上一节：16.4 权限（Linux 要点） - 下一节：16.6 命令行

本节讲什么

  1. 单文件 EA 总量有限（ext4 inode 内空间有限，过大则外挂块） 2. 单 value 有上限 → 只适小元数据 3. 旧挂载可能要 user_xattr；ext4 默认通常已开

要点

  1. 单文件 EA 总量有限（ext4 inode 内空间有限，过大则外挂块）
  2. 单 value 有上限 → 只适小元数据
  3. 旧挂载可能要 user_xattr；ext4 默认通常已开
  4. cp 默认不带 EA；cp --preserve=xattr、rsync -X
  5. 硬链接共享 inode → xattr 一改全见
  6. 程序须处理 ENOTSUP / EOPNOTSUPP

HFT / 嵌入式关联

路线 要点
嵌入式 限制与踩坑 在嵌入式系统的应用场景

代码示例

#include <sys/xattr.h>
#include <stdio.h>

int main(void) {
    /* xattr 大小限制:
     * - 单个值: 通常 64KB (VFS 限制)
     * - 所有 xattr 总和: 取决于文件系统 (ext4 通常一个块)
     * - key 长度: 通常 255 字节
     */

    const char *path = "/tmp/xattr_limit_test.txt";
    FILE *fp = fopen(path, "w");
    if (fp) { fclose(fp); }

    /* 测试: 设置大 value */
    char big[65536];
    memset(big, 'X', sizeof(big));

    int ret = setxattr(path, "user.big", big, sizeof(big), 0);
    if (ret != 0) {
        printf("设置 64KB xattr 失败: ");
        perror("");
        printf("(可能超出文件系统限制)\n");
    } else {
        printf("设置 64KB xattr 成功\n");
    }

    /* 查看各文件系统的限制 */
    printf("\n常见限制:\n");
    printf("ext4: 单值最大 ~64KB, 总和受块大小限制\n");
    printf("XFS:  支持更大 xattr\n");
    return 0;
}
