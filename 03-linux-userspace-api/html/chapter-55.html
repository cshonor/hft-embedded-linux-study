导读

章节目标

flock / fcntl 记录锁；绑定对象与 close/fork；劝告 vs 强制；pid 文件；死锁与对比。

陷阱

  1. fcntl：关「无关」fd 也放全锁
  2. flock：dup 后 close 放锁
  3. flock+fcntl 混用失效
  4. GETLK 非原子获取
  5. stdio：锁内外 fflush；关键路径用 read/write
  6. NFS 锁不可靠

背诵卡

# 要点
1 默认劝告；协作才有效
2 flock=OFD 整文件；fcntl=进程+区
3 fcntl：任意 close → 全锁放
4 fork 两类都不继承锁
5 勿混用 flock/fcntl
6 pid 文件用 F_WRLCK 单实例

  • Kerrisk · TLPI Ch55（非「第 14 章」误标）

代码示例

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

/* Ch55 文件锁 — flock(BSD 整文件锁) vs fcntl(POSIX 记录锁)。
 * 演示两种文件锁的使用。
 * 编译: gcc -o ch55_demo ch55_demo.c */

#define LOCK_FILE "/tmp/ch55_lock.txt"

int main(void) {
    /* 创建测试文件 */
    int fd = open(LOCK_FILE, O_RDWR | O_CREAT, 0644);
    write(fd, "test data\n", 10);

    /* === flock: BSD 锁, 锁整个文件 === */
    printf("=== flock (BSD whole-file lock) ===\n");
    if (flock(fd, LOCK_EX | LOCK_NB) == 0) {
        printf("Acquired exclusive flock\n");

        pid_t pid = fork();
        if (pid == 0) {
            int cfd = open(LOCK_FILE, O_RDWR);
            /* 子进程尝试获取锁, 会失败 (非阻塞) */
            if (flock(cfd, LOCK_EX | LOCK_NB) < 0)
                printf("Child: flock failed (already locked)\n");
            else
                printf("Child: got flock (unexpected)\n");
            close(cfd);
            _exit(0);
        }
        waitpid(pid, NULL, 0);
        flock(fd, LOCK_UN);  /* 释放 */
    }
    close(fd);

    /* === fcntl: POSIX 记录锁, 可锁文件的一部分 === */
    printf("\n=== fcntl (POSIX record lock) ===\n");
    fd = open(LOCK_FILE, O_RDWR);

    /* 锁定字节 0-9 */
    struct flock fl;
    fl.l_type = F_WRLCK;    /* 写锁 */
    fl.l_whence = SEEK_SET;
    fl.l_start = 0;
    fl.l_len = 10;           /* 锁 10 字节 */

    if (fcntl(fd, F_SETLK, &fl) == 0) {
        printf("Locked bytes 0-9 (exclusive)\n");

        pid_t pid = fork();
        if (pid == 0) {
            int cfd = open(LOCK_FILE, O_RDWR);
            struct flock cfl = fl;
            /* 尝试锁同一区域 */
            if (fcntl(cfd, F_GETLK, &cfl) == 0) {
                if (cfl.l_type == F_UNLCK)
                    printf("Child: region unlocked\n");
                else
                    printf("Child: region locked by pid %d\n",
                           (int)cfl.l_pid);
            }
            close(cfd);
            _exit(0);
        }
        waitpid(pid, NULL, 0);

        /* 解锁 */
        fl.l_type = F_UNLCK;
        fcntl(fd, F_SETLK, &fl);
        printf("Unlocked\n");
    }

    close(fd);
    remove(LOCK_FILE);
    return 0;
}

55.1

模型

本章：TLPI 第 55 章 — File Locking - 上一节：本章首页 - 下一节：55.2 flock

本节讲什么

多读单写：共享读锁可并存；排他写锁独占区域。 默认 Advisory；强制锁 Linux 特有且不推荐。 ---

要点

多读单写：共享读锁可并存；排他写锁独占区域。
默认 Advisory；强制锁 Linux 特有且不推荐。

HFT / 嵌入式关联

路线 要点
嵌入式 劝告锁需进程自觉配合，内核不强制；多进程写同一文件靠协作约定，违背约定即冲突
HFT 多进程写同一文件必须显式加锁，否则写覆盖竞态导致数据损坏，需统一锁协议

代码示例


/* 咨询锁 vs 强制锁概念演示 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>

int main(void) {
    int fd = open("/tmp/ch55_advisory.txt", O_RDWR | O_CREAT, 0644);
    write(fd, "advisory lock demo", 18);

    /* flock 是咨询锁：只有协作进程才检查 */
    if (flock(fd, LOCK_EX | LOCK_NB) < 0) {
        perror("flock");
    } else {
        printf("Advisory lock acquired (flock LOCK_EX)\n");
        printf("Another process CAN still read/write via different fd\n");
        printf("(unless it also calls flock and respects the lock)\n");
    }

    /* 咨询锁不阻止不检查锁的进程 */
    int fd2 = open("/tmp/ch55_advisory.txt", O_RDONLY);
    char buf[32];
    int n = read(fd2, buf, sizeof(buf));
    buf[n] = '\0';
    printf("Uncooperative read succeeded: '%s'\n", buf);

    flock(fd, LOCK_UN);
    close(fd);
    close(fd2);
    return 0;
}

55.2

`flock`

本章：TLPI 第 55 章 — File Locking - 上一节：55.1 模型 - 下一节：55.3 –55.4 fcntl 记录锁

本节讲什么

int flock(int fd, int operation); / LOCK_SH|EX|UN [| LOCK_NB] / ---

要点

int flock(int fd, int operation);  /* LOCK_SH|EX|UN [| LOCK_NB] */
规则
粒度 整文件
绑定 打开文件描述（OFD）
dup/fork 出的 fd 同 OFD → 共享同一 flock；任一 close 可放锁
同文件不同 open 不同 OFD → 独立锁
fork 不继承 flock
vs fcntl Linux 互不阻塞

HFT / 嵌入式关联

路线 要点
嵌入式 flock 整文件锁，绑定 OFD，fork 后子进程共享同一锁，close 该 OFD 才释放
HFT 用 flock 保护日志文件追加，避免多写者交错；整文件粒度足够且开销低于 fcntl

代码示例


/* flock 完整 API：LOCK_SH / LOCK_EX / LOCK_UN / LOCK_NB */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>
#include <sys/wait.h>

int main(void) {
    int fd = open("/tmp/ch55_flock.txt", O_RDWR | O_CREAT, 0644);

    /* 共享锁（读锁） */
    if (flock(fd, LOCK_SH | LOCK_NB) == 0) {
        printf("LOCK_SH: shared lock acquired\n");
    }

    /* 尝试排他锁（会失败，因为已有共享锁） */
    int fd2 = open("/tmp/ch55_flock.txt", O_RDWR);
    if (flock(fd2, LOCK_EX | LOCK_NB) < 0) {
        printf("LOCK_EX failed (expected): shared lock held\n");
    }

    /* 释放共享锁 */
    flock(fd, LOCK_UN);
    printf("LOCK_UN: shared lock released\n");

    /* 现在可以获取排他锁 */
    if (flock(fd2, LOCK_EX | LOCK_NB) == 0) {
        printf("LOCK_EX: exclusive lock acquired\n");
    }
    flock(fd2, LOCK_UN);

    close(fd);
    close(fd2);
    return 0;
}

55.3

–55.4 `fcntl` 记录锁

本章：TLPI 第 55 章 — File Locking - 上一节：55.2 flock - 下一节：55.5 劝告 vs 强制（Linux）

本节讲什么

struct flock { short l_type, l_whence; off_t l_start, l_len; pid_t l_pid; }; / F_SETLK / F_SETLKW / F_GETLK / - l_len=0：从 start 锁到 EOF（含后续追加）

要点

struct flock { short l_type, l_whence; off_t l_start, l_len; pid_t l_pid; };
/* F_SETLK / F_SETLKW / F_GETLK */
  • l_len=0：从 start 锁到 EOF（含后续追加）
  • F_GETLK：仅探测（TOCTOU）；冲突填 l_pid
  • 读锁需可读打开；写锁需可写打开
  • 绑定 进程 + inode；同进程多 fd 新锁覆盖同区旧锁
  • 任一 fd close → 该进程对此文件全部记录锁没了
  • fork：不继承；退出自动释放；死锁阻塞可 EDEADLK

lockf：fcntl 封装，仅排他锁，与 fcntl 互通。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 fcntl 记录锁按字节范围，进程+inode 绑定，任一 fd close 释放该进程全部记录锁
HFT F_SETLKW 阻塞会卡住热路径，用 F_SETLK 非阻塞 + 重试；注意 close 误释放全局锁的坑

代码示例


/* fcntl 记录锁（字节范围锁）完整用法 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>

int main(void) {
    int fd = open("/tmp/ch55_fcntl.txt", O_RDWR | O_CREAT, 0644);
    write(fd, "0123456789ABCDEF", 16);

    /* 写锁（排他）: 锁定字节 0-4 */
    struct flock wrlk = {
        .l_type = F_WRLCK, .l_whence = SEEK_SET,
        .l_start = 0, .l_len = 5
    };
    if (fcntl(fd, F_SETLK, &wrlk) < 0) {
        perror("F_SETLK F_WRLCK");
    } else {
        printf("Write lock: bytes 0-4\n");
    }

    /* 查询锁信息：F_GETLK */
    struct flock query = {
        .l_type = F_WRLCK, .l_whence = SEEK_SET,
        .l_start = 0, .l_len = 5
    };
    fcntl(fd, F_GETLK, &query);
    if (query.l_type == F_UNLCK) {
        printf("GETLK: bytes 0-4 unlocked\n");
    } else {
        printf("GETLK: bytes 0-4 locked (type=%d, pid=%d)\n",
               query.l_type, query.l_pid);
    }

    /* 解锁 */
    wrlk.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &wrlk);
    printf("Unlocked bytes 0-4\n");

    /* 阻塞式写锁：F_SETLKW */
    wrlk.l_type = F_WRLCK;
    printf("Calling F_SETLKW (blocks if conflicting lock)...\n");
    fcntl(fd, F_SETLKW, &wrlk);
    printf("F_SETLKW: acquired\n");

    wrlk.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &wrlk);
    close(fd);
    return 0;
}

55.4

Mandatory Locking

本章：TLPI 第 55 章

本节讲什么

强制锁（mandatory locking）是 Linux 的一种文件锁机制，通过设置 setgid 位且清除执行位来启用。与建议锁不同，强制锁会阻止其他进程的 open()/read()/write() 操作。

要点

启用强制锁

# 文件设置 setgid + 无组执行位
chmod g+s,g-x file
# ls -l 显示: -rw-r-Sr--
# ^^^ 大写 S 表示 setgid 但无执行位 = 强制锁启用

强制锁 vs 建议锁

特性 建议锁 (advisory) 强制锁 (mandatory)
获取方式 fcntl(F_SETLK) fcntl(F_SETLK)
启用条件 无 setgid+无组执行
阻止 open() 否 是
阻止 read/write 否（仅锁本身） 是
端口性 POSIX 标准 Linux 特有
推荐 是 不推荐

强制锁行为

进程 A: flock(F_SETLK, 写锁)  ← 获取写锁
进程 B: open(file, O_WRONLY) ← 被阻塞（强制锁）
进程 C: read(fd, ...)         ← 被阻塞（强制锁）
进程 A: flock(F_UNLCK)        ← 释放
进程 B/C: 继续执行

限制与问题

  1. 竞态条件：在 open 和 lock 之间可能被其他进程操作
  2. 死锁风险：多个进程互相等待锁释放
  3. 性能影响：每次 I/O 都检查锁
  4. 非标准：部分文件系统不支持
  5. 已不推荐：现代使用文件租约（lease）或 O_NONBLOCK

HFT / 嵌入式关联

  • HFT 不推荐使用强制锁（性能差，死锁风险）
  • 嵌入式场景如需强制保护，用 O_EXCL 或 flock 建议锁替代
  • 了解强制锁原理有助于调试遗留系统

代码示例

#include <fcntl.h>
#include <sys/file.h>
#include <unistd.h>
#include <stdio.h>

int main(int argc, char *argv[]) {
    int fd = open(argv[1], O_RDWR);
    /* 获取写锁（如果是强制锁文件则阻塞其他 I/O） */
    struct flock fl = {
        .l_type = F_WRLCK,
        .l_whence = SEEK_SET,
        .l_start = 0,
        .l_len = 0,  /* 0 = 到文件尾 */
    };
    fcntl(fd, F_SETLKW, &fl);
    printf("Locked %s\n", argv[1]);
    sleep(10);
    fl.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &fl);
    close(fd);
    return 0;
}

代码自测

为什么强制锁已不推荐使用？

强制锁存在多个问题：(1) open 和 lock 之间的竞态；(2) 可能导致死锁；(3) 性能开销大（每次 I/O 检查锁）；(4) 非标准，移植性差；(5) 某些文件系统不支持。现代替代方案包括文件租约（F_SETLEASE）、O_NONBLOCK + 轮询、或使用建议锁 + 应用层协调。

55.5

劝告 vs 强制（Linux）

本章：TLPI 第 55 章 — File Locking - 上一节：55.3 –55.4 fcntl 记录锁 - 下一节：55.6 –55.9 对比 · 运维 · 场景

本节讲什么

强制：挂载 -o mand + 文件 g+s,g-x。IO 被内核拦截。 TLPI：慢、有坑、不可移植 → 别用。 ---

要点

强制：挂载 -o mand + 文件 g+s,g-x。IO 被内核拦截。
TLPI：慢、有坑、不可移植 → 别用。

HFT / 嵌入式关联

路线 要点
嵌入式 Linux 强制锁慢且有坑、不可移植，实际工程几乎不用
HFT 强制锁会拖慢每次读写、引入不可控延迟，HFT 千万别用，用劝告锁 + 进程协作

代码示例


/* 锁的继承与释放：fork 和 close 的行为 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>
#include <sys/wait.h>

int main(void) {
    int fd = open("/tmp/ch55_inherit.txt", O_RDWR | O_CREAT, 0644);

    /* flock 锁在 fork 后：父子共享同一锁（不是各自一份） */
    flock(fd, LOCK_EX);
    printf("parent: flock LOCK_EX acquired\n");

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程继承 fd，但不继承 flock 锁的所有权 */
        /* 子进程可以解锁父进程的锁 */
        flock(fd, LOCK_UN);
        printf("child: unlocked parent's flock\n");
        _exit(0);
    }
    wait(NULL);

    /* fcntl 记录锁：fork 后子进程不继承锁 */
    struct flock fl = { .l_type = F_WRLCK, .l_whence = SEEK_SET,
                        .l_start = 0, .l_len = 0 };
    fcntl(fd, F_SETLK, &fl);
    printf("parent: fcntl F_WRLCK acquired\n");

    /* close(fd) 释放该 fd 持有的所有 fcntl 锁 */
    printf("close(fd) releases all fcntl locks on this fd\n");

    close(fd);
    return 0;
}

路线 要点
嵌入式 用 flock/F_WRLCK 锁文件实现单实例守护，防重复启动；/proc/locks 查看锁状态
HFT PID 文件 + F_WRLCK 排他锁保证交易进程唯一实例，崩溃后内核自动放锁可重启

/* flock vs fcntl 锁比较与死锁避免 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>

int main(void) {
    printf("=== flock vs fcntl comparison ===\n\n");

    printf("flock:\n");
    printf("  - whole-file lock (no byte ranges)\n");
    printf("  - advisory only (Linux)\n");
    printf("  - same fd: lock upgraded/downgraded\n");
    printf("  - fork: child can unlock parent's lock\n\n");

    printf("fcntl:\n");
    printf("  - byte-range record lock\n");
    printf("  - POSIX standard\n");
    printf("  - fork: child does NOT inherit locks\n");
    printf("  - close: releases all locks on that fd\n\n");

    /* 演示 fcntl 锁升级 */
    int fd = open("/tmp/ch55_cmp.txt", O_RDWR | O_CREAT, 0644);
    struct flock fl = { .l_type = F_RDLCK, .l_whence = SEEK_SET,
                        .l_start = 0, .l_len = 10 };
    fcntl(fd, F_SETLK, &fl);
    printf("fcntl: F_RDLCK (read lock) on 0-9\n");

    /* 升级为写锁 */
    fl.l_type = F_WRLCK;
    if (fcntl(fd, F_SETLK, &fl) == 0) {
        printf("fcntl: upgraded to F_WRLCK\n");
    }

    fl.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &fl);
    close(fd);
    return 0;
}

int ensure_single_instance(const char *lockfile) {
    int fd = open(lockfile, O_CREAT | O_EXCL | O_RDWR, 0600);
    if (fd < 0) {

代码自测

flock 和 fcntl 锁为什么不兼容？

Linux 早期 flock 和 fcntl 使用不同的锁机制（文件描述符级 vs 进程级），互不感知。Linux 2.0+ 做了兼容映射，但语义仍有差异：flock 锁与文件描述符关联（dup 共享），fcntl 锁与进程关联（同进程不同 fd 不互斥）。在网络文件系统（NFS）上行为也不同。混用时可能出现意外行为。

Summary

本章：TLPI 第 55 章

本节讲什么

第 55 章总结：文件锁的三种 API（fcntl/flock/lockf）、强制锁与建议锁、锁的陷阱和最佳实践。

要点

API 总览

API 粒度 类型 推荐
fcntl(F_SETLK) 字节范围 建议锁 是（标准）
flock() 整个文件 建议锁 简单场景
lockf() 字节范围 建议锁 较少用
open(O_EXCL) 文件级 存在性 单实例

fcntl 锁关键特性

  • 字节范围锁：可锁定文件的一部分
  • 进程级：同进程不同 fd 不互斥
  • fork 不继承，exec 继承
  • 锁在 close 时释放（可能产生意外释放）
  • NFS 支持（需 lockd）

锁选择决策

场景 推荐
字节范围锁 fcntl
整个文件锁 flock（简单）或 fcntl
单实例守护 open(O_EXCL)
多进程写不同部分 fcntl 范围锁
多进程写整个文件 flock 或 fcntl 全文件

陷阱

  1. close 释放所有锁（包括其他 fd 的锁）
  2. fork 不继承锁
  3. 强制锁性能差且非标准
  4. flock 和 fcntl 语义不一致
  5. NFS 上锁行为可能不同

HFT / 嵌入式关联

  • 配置文件并发写入保护：fcntl 范围锁
  • 单实例守护进程：O_EXCL 锁文件
  • 日志文件追加写入：flock 或 O_APPEND

练习 1：字节范围锁

题目：多进程并发写入同一文件的不同区域，使用 fcntl 范围锁确保各自区域互斥。

参考答案 
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <sys/wait.h>

#define REGION_SIZE 1024

void write_region(int fd, int region_idx, const char *data) {
    off_t offset = region_idx * REGION_SIZE;
    struct flock fl = {
        .l_type = F_WRLCK,
        .l_whence = SEEK_SET,
        .l_start = offset,
        .l_len = REGION_SIZE,
    };
    fcntl(fd, F_SETLKW, &fl);
    pwrite(fd, data, strlen(data), offset);
    fl.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &fl);
}

int main() {
    int fd = open("shared.dat", O_CREAT | O_RDWR, 0644);
    ftruncate(fd, 10 * REGION_SIZE);

    for (int i = 0; i < 5; i++)
        if (fork() == 0) {
            char buf[64];
            snprintf(buf, sizeof(buf), "Process %d data\n", i);
            write_region(fd, i, buf);
            _exit(0);
        }

    while (wait(NULL) > 0);
    close(fd);
    return 0;
}

练习 2：close 释放锁陷阱

题目：解释以下代码为什么不能保护文件不被其他进程修改。

int fd1 = open("file", O_RDWR);
fcntl(fd1, F_SETLK, &lock);  /* 加锁 */
int fd2 = open("file", O_RDWR);
fcntl(fd2, F_SETLK, &lock);  /* 又加锁（同进程成功） */
close(fd2);                   /* 关闭 fd2 → 所有锁被释放！ */
参考答案

fcntl 锁是进程级的，同一进程可以对同一文件多次加锁（不会阻塞自己）。但关键陷阱：关闭该文件的任一描述符会释放该进程在该文件上的所有锁。close(fd2) 会释放 fd1 和 fd2 上的所有锁，因为内核按 (pid, inode) 而非 (pid, fd) 管理锁。这在多线程程序中尤其危险——某个线程不知情地关闭一个 fd 就会释放其他线程的锁。
