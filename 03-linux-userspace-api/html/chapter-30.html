代码示例

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>

/* Ch30 线程同步 — mutex/cond/rwlock。
 * 演示互斥锁保护共享数据 + 条件变量通知。
 * 编译: gcc -o ch30_demo ch30_demo.c -lpthread */

static int counter = 0;
static pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
static int ready = 0;

void *producer(void *arg) {
    for (int i = 0; i < 5; i++) {
        pthread_mutex_lock(&lock);
        counter++;
        ready = 1;
        printf("Producer: counter=%d, signaling\n", counter);
        pthread_cond_signal(&cond);
        pthread_mutex_unlock(&lock);
        usleep(200000);
    }
    return NULL;
}

void *consumer(void *arg) {
    for (int i = 0; i < 5; i++) {
        pthread_mutex_lock(&lock);
        while (!ready)
            pthread_cond_wait(&cond, &lock);  /* 原子释放锁+等待 */
        ready = 0;
        printf("Consumer: consumed counter=%d\n", counter);
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}

int main(void) {
    pthread_t pt, ct;
    pthread_create(&pt, NULL, producer, NULL);
    pthread_create(&ct, NULL, consumer, NULL);
    pthread_join(pt, NULL);
    pthread_join(ct, NULL);
    printf("Done: counter=%d\n", counter);
    return 0;
}

30.1

互斥量 `pthread_mutex_t`

本章：TLPI 第 30 章 — Threads: Thread Synchronization - 上一节：本章首页 - 下一节：30.2 条件变量 pthread_cond_t

本节讲什么

pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER; / 静态 / pthread_mutex_init(&mtx, &attr); / 动态 / pthread_mutex_destroy(&mtx); / 须未持锁 /

要点

概念
竞态 无保护并发读写，结果依赖调度
临界区 须互斥执行的共享访问代码
mutex 同一时刻至多一线程进入临界区

初始化

pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;   /* 静态 */
pthread_mutex_init(&mtx, &attr);                   /* 动态 */
pthread_mutex_destroy(&mtx);                       /* 须未持锁 */

禁止拷贝 mutex；行为未定义。

API

pthread_mutex_lock / trylock / timedlock / unlock

trylock 拿不到 → EBUSY（正错误码，Pthreads 风格）。

类型（attr）

类型 要点
NORMAL（常见默认） 同线程再 lock → 死锁；乱 unlock → UB
RECURSIVE 可嵌套；unlock 次数须匹配
ERRORCHECK 非法操作返回错误；调试用
ADAPTIVE_NP（Linux） 短自旋再睡；短临界区

死锁

  1. 同线程对普通锁二次 lock
  2. 多锁获取顺序不一致

→ 全局统一加锁顺序。

Demo：code/thread_incr_mutex.c（对比 Ch29 thread_race）

HFT / 嵌入式关联

路线 要点
嵌入式 保护共享外设/缓冲区；优先级反转（低优先级持锁阻塞高优先级）用 PTHREAD_PRIO_INHERIT 化解
HFT HFT 关键路径尽量少锁/无锁（锁有争抢开销 + 优先级反转延迟）；短临界区用 ADAPTIVE_NP 短自旋；多锁必须全局统一加锁顺序防死锁

常见陷阱

  1. 死锁
pthread_mutex_lock(&a);
pthread_mutex_lock(&b);  /* 如果另一线程先 lock b → 死锁 */

多个 mutex 的加锁顺序不一致会导致死锁。所有线程必须按相同顺序加锁。

  1. 忘记 unlock
pthread_mutex_lock(&m);
if (error) return;  /* 没 unlock */
pthread_mutex_unlock(&m);

所有代码路径都必须 unlock。用 goto cleanup 或 C++ RAII。

  1. 优先级反转
/* 低优先级线程持有 mutex */
/* 高优先级线程等待 mutex → 被低优先级阻塞 */

低优先级线程持有锁时高优先级线程被阻塞。用 PTHREAD_PRIO_INHERIT 或避免在关键路径上加锁。

自测题

题目 1（选择）： pthread_mutex_trylock 返回 EBUSY 表示什么？ A. 系统繁忙 B. mutex 已被其他线程持有 C. mutex 不存在 D. 参数无效

参考答案

B。mutex 已被其他线程持有，trylock 不会阻塞而是立即返回 EBUSY。

题目 2（判断）： 同一个线程可以对同一个 mutex 加锁两次。

参考答案

错误（默认情况下）。普通 mutex 不可重入，同线程第二次 lock 会死锁。需要可重入 mutex 用 PTHREAD_MUTEX_RECURSIVE。

题目 3（踩坑）： 以下代码有什么问题？

pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_lock(&m);
/* 处理数据 */
if (error)
    exit(1);
pthread_mutex_unlock(&m);
参考答案

error 时直接 exit(1)，没有 unlock mutex。虽然 exit 会终止进程（mutex 随之销毁），但如果改为 return 就会死锁。应养成所有路径都 unlock 的习惯。

题目 4（应用）： 写代码用 mutex 保护一个计数器。

代码示例

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
/* pthread_mutex_t: 互斥锁 */
static int counter = 0;
static pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;

void *worker(void *arg) {
    for (int i = 0; i < 10000; i++) {
        pthread_mutex_lock(&mtx);
        counter++;  /* 临界区: 受互斥锁保护 */
        pthread_mutex_unlock(&mtx);
    }
    return NULL;
}

int main(void) {
    pthread_t threads[4];
    for (int i = 0; i < 4; i++)
        pthread_create(&threads[i], NULL, worker, NULL);
    for (int i = 0; i < 4; i++)
        pthread_join(threads[i], NULL);

    printf("counter = %d (expected 40000)\n", counter);

    /* 锁的类型: NORMAL / ERRORCHECK / RECURSIVE */
    pthread_mutex_t rec_mtx;
    pthread_mutexattr_t attr;
    pthread_mutexattr_init(&attr);
    pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);
    pthread_mutex_init(&rec_mtx, &attr);
    pthread_mutex_lock(&rec_mtx);
    pthread_mutex_lock(&rec_mtx);  /* 递归锁: 允许同线程多次加锁 */
    printf("Recursive lock: locked twice\n");
    pthread_mutex_unlock(&rec_mtx);
    pthread_mutex_unlock(&rec_mtx);
    pthread_mutex_destroy(&rec_mtx);
    pthread_mutexattr_destroy(&attr);
    return 0;
}

30.2

条件变量 `pthread_cond_t`

本章：TLPI 第 30 章 — Threads: Thread Synchronization - 上一节：30.1 互斥量 pthread_mutex_t - 下一节：30.3 职责与易错

本节讲什么

mutex = 互斥访问；cond = 等某个状态成立。cond 必须与 mutex 一起用。 pthread_cond_t cond = PTHREAD_COND_INITIALIZER; pthread_cond_wait / timedwait / signal / broadcast

要点

mutex = 互斥访问；cond = 等某个状态成立。cond 必须与 mutex 一起用。

pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
pthread_cond_wait / timedwait / signal / broadcast

wait 语义

原子：放锁 → 睡 → 醒后再拿锁再返回。避免「先 signal、后 wait」丢唤醒（在持锁改状态 + wait 的正确用法下）。

谓词必须 while

存在虚假唤醒；也可能 broadcast 后条件仍不满足：

pthread_mutex_lock(&mtx);
while (!predicate)
    pthread_cond_wait(&cond, &mtx);
/* use shared state */
pthread_mutex_unlock(&mtx);
signal 唤醒至少一个等待者
broadcast 唤醒全部（多等待者 / 不确定谁该醒）

生产者-消费者

mutex 保护缓冲与计数；空/满时 wait；变更后 signal/broadcast。

Demo：code/prod_condvar.c

HFT / 嵌入式关联

路线 要点
嵌入式 生产者-消费者（采集线程→处理线程）用 cond 通知；谓词必须 while 防虚假唤醒
HFT 行情接收→策略线程的唤醒可用 cond/信号量，但 HFT 更常用 lock-free 队列 + 忙等/自旋（cond 唤醒有系统调用开销）；务必 while 谓词防虚假唤醒

代码示例

#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
/* pthread_cond_t: 条件变量 (生产者-消费者) */
static int data_ready = 0;
static pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t cond = PTHREAD_COND_INITIALIZER;

void *consumer(void *arg) {
    pthread_mutex_lock(&mtx);
    while (!data_ready)
        pthread_cond_wait(&cond, &mtx);  /* 原子: 解锁+等待 */
    printf("Consumer: consumed data\n");
    data_ready = 0;
    pthread_mutex_unlock(&mtx);
    return NULL;
}

void *producer(void *arg) {
    sleep(1);
    pthread_mutex_lock(&mtx);
    data_ready = 1;
    printf("Producer: data ready, signaling\n");
    pthread_cond_signal(&cond);  /* 唤醒一个等待者 */
    pthread_mutex_unlock(&mtx);
    return NULL;
}

int main(void) {
    pthread_t pt, ct;
    pthread_create(&ct, NULL, consumer, NULL);
    pthread_create(&pt, NULL, producer, NULL);
    pthread_join(pt, NULL);
    pthread_join(ct, NULL);
    return 0;
}

30.3

职责与易错

本章：TLPI 第 30 章 — Threads: Thread Synchronization - 上一节：30.2 条件变量 pthread_cond_t - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

  1. wait 不持锁 / 用 if 判条件 2. 非持有者 unlock 3. 递归锁 unlock 次数不对

要点

工具 职责
mutex 保护状态读写
cond 等待状态变化
  1. wait 不持锁 / 用 if 判条件
  2. 非持有者 unlock
  3. 递归锁 unlock 次数不对
  4. 多锁无序 → 死锁
  5. 以为 cond「存信号」：wait 前乱 signal 会丢（须在持锁下改谓词再 signal）

HFT / 嵌入式关联

路线 要点
嵌入式 明确 mutex（保护状态）与 cond（等状态变化）的职责分工；持锁改谓词再 signal 防丢唤醒
HFT HFT 多线程最易踩「cond 存信号」的坑——wait 前乱 signal 会丢唤醒；关键路径保证「持锁改状态 → signal」原子性，否则延迟下偶发丢单难查

代码示例

#include <pthread.h>
#include <stdio.h>
static int count = 0;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;
void *worker(void *arg) {
    for (int i = 0; i < 100000; i++) {
        pthread_mutex_lock(&lock);
        count++;
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}
int main(void) {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_join(t1, NULL); pthread_join(t2, NULL);
    printf("count = %d (期望 200000)\n", count);
    return 0;
}

练习题

练习 1：生产者-消费者队列

用互斥锁和条件变量实现一个线程安全的固定大小队列，支持多生产者多消费者。

要求： - 队列容量固定（如 1024 个元素） - push()：队列满时阻塞 - pop()：队列空时阻塞 - 支持优雅关闭（所有消费者在队列空且关闭后退出）

练习 2：读写锁 vs 互斥锁性能对比

编写基准测试程序，对比 pthread_rwlock 和 pthread_mutex 在读多写少场景下的性能。

练习 3：避免死锁

编写程序演示两个线程各持有一把锁并请求对方锁导致死锁，然后修复为使用 trylock + 退避策略或固定锁顺序。

参考答案

练习 1 参考代码

#include <pthread.h>
#include <stdlib.h>

#define QUEUE_CAP 1024

typedef struct {
    void *buf[QUEUE_CAP];
    int head, tail, count;
    int closed;
    pthread_mutex_t lock;
    pthread_cond_t not_full;
    pthread_cond_t not_empty;
} blocking_queue_t;

void bq_init(blocking_queue_t *q) {
    q->head = q->tail = q->count = 0;
    q->closed = 0;
    pthread_mutex_init(&q->lock, NULL);
    pthread_cond_init(&q->not_full, NULL);
    pthread_cond_init(&q->not_empty, NULL);
}

void bq_push(blocking_queue_t *q, void *item) {
    pthread_mutex_lock(&q->lock);
    while (q->count == QUEUE_CAP && !q->closed)
        pthread_cond_wait(&q->not_full, &q->lock);
    if (q->closed) { pthread_mutex_unlock(&q->lock); return; }
    q->buf[q->tail] = item;
    q->tail = (q->tail + 1) % QUEUE_CAP;
    q->count++;
    pthread_cond_signal(&q->not_empty);
    pthread_mutex_unlock(&q->lock);
}

void *bq_pop(blocking_queue_t *q) {
    pthread_mutex_lock(&q->lock);
    while (q->count == 0 && !q->closed)
        pthread_cond_wait(&q->not_empty, &q->lock);
    if (q->count == 0) { pthread_mutex_unlock(&q->lock); return NULL; }
    void *item = q->buf[q->head];
    q->head = (q->head + 1) % QUEUE_CAP;
    q->count--;
    pthread_cond_signal(&q->not_full);
    pthread_mutex_unlock(&q->lock);
    return item;
}

void bq_close(blocking_queue_t *q) {
    pthread_mutex_lock(&q->lock);
    q->closed = 1;
    pthread_cond_broadcast(&q->not_empty);
    pthread_cond_broadcast(&q->not_full);
    pthread_mutex_unlock(&q->lock);
}

练习 2 答案

读写锁在读多写少场景显著优于互斥锁： - 多个读线程可以同时持有读锁 - 写锁需要等待所有读锁释放

/* 基准测试框架 */
#define N_READERS 8
#define N_WRITERS 1
#define ITERATIONS 1000000

void *reader(void *arg) {
    for (int i = 0; i < ITERATIONS; i++) {
        pthread_rwlock_rdlock(&rwlock);
        /* 读操作 */
        pthread_rwlock_unlock(&rwlock);
    }
    return NULL;
}

void *writer(void *arg) {
    for (int i = 0; i < ITERATIONS / 100; i++) {
        pthread_rwlock_wrlock(&rwlock);
        /* 写操作 */
        pthread_rwlock_unlock(&rwlock);
    }
    return NULL;
}

预期结果：8 读 1 写场景下 rwlock 比 mutex 快 3-5 倍。

练习 3 答案

死锁代码：

/* 线程 A：先 lock1 再 lock2 */
/* 线程 B：先 lock2 再 lock1 */
/* → 死锁 */

修复方案 1：固定锁顺序

/* 总是先锁 ID 小的锁 */
void transfer(account_t *from, account_t *to, int amount) {
    account_t *first = (from < to) ? from : to;
    account_t *second = (from < to) ? to : from;
    pthread_mutex_lock(&first->lock);
    pthread_mutex_lock(&second->lock);
    /* 转账 */
    pthread_mutex_unlock(&second->lock);
    pthread_mutex_unlock(&first->lock);
}

修复方案 2：trylock + 退避

while (1) {
    pthread_mutex_lock(&lock1);
    if (pthread_mutex_trylock(&lock2) == 0) break;
    pthread_mutex_unlock(&lock1);
    usleep(1000);  /* 退避 */
}

HFT / 嵌入式关联

  • 无锁队列：HFT 热路径用 SPSC（单生产者单消费者）无锁队列，避免锁开销
  • 自旋锁：HFT 低延迟场景用 pthread_spinlock（忙等，无上下文切换）
  • 读写锁：行情数据（读多写少）用 rwlock 提高并发
  • 嵌入式：RTOS 可能无 pthread，用平台特定同步原语
