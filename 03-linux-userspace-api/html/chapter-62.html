导读

章节目标

终端类型；tcgetattr/tcsetattr；规范/非规范/原始；标志与 c_cc；会话/控制终端/前台组；终端相关信号。

陷阱

  1. 改属性后未恢复
  2. 改输出用错 TCSANOW
  3. VMIN/VTIME 搞混
  4. 关 ISIG 后 Ctrl+C 无效
  5. 后台读写被暂停
  6. 对非 tty 调 tcgetattr

背诵卡

# 要点
1 ICANON=规范；关则非规范
2 VMIN/VTIME 定非规范 read
3 改完必须恢复 termios
4 Ctrl+C→前台组 SIGINT
5 后台读写 → TTIN/TTOU
6 PTY 细节见 Ch64

参考

  • Kerrisk · TLPI Ch62
  • man 3 termios · man 3 isatty · man 7 termio

代码示例

#include <stdio.h>
#include <termios.h>
#include <unistd.h>
#include <string.h>
#include <fcntl.h>

/* Ch62 终端 — termios API 控制终端属性。
 * 演示关闭回显 (密码输入) + 设置原始模式。
 * 编译: gcc -o ch62_demo ch62_demo.c */

int main(void) {
    /* 保存当前终端设置 */
    struct termios old_term, new_term;
    tcgetattr(STDIN_FILENO, &old_term);
    new_term = old_term;

    printf("=== Echo off demo (type something, it won't show) ===\n");

    /* 关闭回显 */
    new_term.c_lflag &= ~ECHO;
    tcsetattr(STDIN_FILENO, TCSANOW, &new_term);

    char password[64];
    printf("Enter password: ");
    fflush(stdout);
    if (fgets(password, sizeof(password), stdin)) {
        printf("\nYou entered: %s", password);
    }

    /* 恢复回显 */
    tcsetattr(STDIN_FILENO, TCSANOW, &old_term);

    /* === 原始模式 (无缓冲, 无回显, 无特殊字符处理) === */
    printf("\n\n=== Raw mode demo (press 'q' to quit) ===\n");

    new_term.c_lflag &= ~(ICANON | ECHO);  /* 关闭规范模式 + 回显 */
    new_term.c_cc[VMIN] = 1;               /* 至少读 1 字节 */
    new_term.c_cc[VTIME] = 0;              /* 无超时 */
    tcsetattr(STDIN_FILENO, TCSANOW, &new_term);

    /* 在原始模式下逐字符读取 */
    char c;
    while (read(STDIN_FILENO, &c, 1) == 1) {
        printf("Got char: 0x%02x ('%c')\n", (unsigned char)c,
               (c >= 32 && c < 127) ? c : '.');
        if (c == 'q') break;
    }

    /* 恢复原始终端设置 */
    tcsetattr(STDIN_FILENO, TCSANOW, &old_term);
    printf("\nTerminal restored.\n");

    /* 获取终端窗口大小 */
    struct winsize ws;
    if (ioctl(STDIN_FILENO, TIOCGWINSZ, &ws) == 0) {
        printf("Terminal size: %d rows x %d cols\n",
               ws.ws_row, ws.ws_col);
    }
    return 0;
}

62.1

类型

本章：TLPI 第 62 章 — Terminals - 上一节：本章首页 - 下一节：62.2 termios

本节讲什么

要点

类型 例
串口 /dev/ttyS0
虚拟控制台 /dev/tty1…
PTY master /dev/ptmx，slave /dev/pts/N（ssh、终端模拟器）

HFT / 嵌入式关联

路线 要点
嵌入式 串口 /dev/ttyS0 是无头嵌入式设备的调试/console 标配，没显示器时靠串口登录；虚拟控制台用于本机多终端切换
HFT 交易主机常跑 headless，stdout 重定向到日志文件而非 tty；用 isatty 判断是否走交互式缓冲/彩色输出，避免日志被 ANSI 转义污染

代码示例


/* 终端类型：交互式终端 vs 管道 vs 普通文件 */
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <termios.h>

int main(void) {
    /* isatty: 判断 fd 是否连接到终端 */
    printf("STDIN  isatty: %s\n", isatty(STDIN_FILENO) ? "yes" : "no");
    printf("STDOUT isatty: %s\n", isatty(STDOUT_FILENO) ? "yes" : "no");
    printf("STDERR isatty: %s\n", isatty(STDERR_FILENO) ? "yes" : "no");

    /* ttyname: 获取终端设备名 */
    if (isatty(STDIN_FILENO)) {
        char *name = ttyname(STDIN_FILENO);
        printf("ttyname(STDIN): %s\n", name ? name : "NULL");
    }

    /* ctermid: 获取当前终端名 */
    char term[L_ctermid];
    ctermid(term);
    printf("ctermid: %s\n", term);

    return 0;
}

62.2

`termios`

本章：TLPI 第 62 章 — Terminals - 上一节：62.1 类型 - 下一节：62.3 输入模式（高频）

本节讲什么

tcgetattr(fd, &t); tcsetattr(fd, TCSANOW|TCSADRAIN|TCSAFLUSH, &t); 字段：c_iflag / c_oflag / c_cflag / c_lflag / c_cc[]。

要点

tcgetattr(fd, &t);
tcsetattr(fd, TCSANOW|TCSADRAIN|TCSAFLUSH, &t);
何时生效
TCSANOW 立即
TCSADRAIN 输出排空后（改输出标志优先）
TCSAFLUSH 排空输出并丢弃未读输入

字段：c_iflag / c_oflag / c_cflag / c_lflag / c_cc[]。

HFT / 嵌入式关联

路线 要点
嵌入式 串口通信全靠 termios 配波特率/数据位/流控（IXON）；嵌入式串口必须先 tcgetattr 再改 c_cflag，TCSADRAIN 保证输出排空再切换
HFT 改终端属性是低频一次性操作，但 TCSANOW 立即生效可能截断在途输出；连交易所串口/专线时用 TCSAFLUSH 丢弃陈旧输入，避免读到上一段残留数据

代码示例


/* struct termios 结构体详解 */
#include <stdio.h>
#include <stdlib.h>
#include <termios.h>
#include <unistd.h>

int main(void) {
    struct termios tio;
    tcgetattr(STDIN_FILENO, &tio);

    printf("=== struct termios ===\n\n");

    printf("c_iflag (input):  0x%x\n", tio.c_iflag);
    printf("  ICANON=%d, ICRNL=%d, IXON=%d\n",
           !!(tio.c_iflag & ICRNL), !!(tio.c_iflag & ICRNL), !!(tio.c_iflag & IXON));

    printf("c_oflag (output): 0x%x\n", tio.c_oflag);
    printf("  OPOST=%d (output processing)\n", !!(tio.c_oflag & OPOST));

    printf("c_cflag (control): 0x%x\n", tio.c_cflag);
    printf("  CSIZE=%d, CSTOPB=%d\n",
           tio.c_cflag & CSIZE, !!(tio.c_cflag & CSTOPB));

    printf("c_lflag (local):   0x%x\n", tio.c_lflag);
    printf("  ICANON=%d, ECHO=%d, ISIG=%d\n",
           !!(tio.c_lflag & ICANON), !!(tio.c_lflag & ECHO), !!(tio.c_lflag & ISIG));

    printf("c_cc[VMIN]  = %d\n", tio.c_cc[VMIN]);
    printf("c_cc[VTIME] = %d\n", tio.c_cc[VTIME]);

    return 0;
}

62.3

输入模式（高频）

本章：TLPI 第 62 章 — Terminals - 上一节：62.2 termios - 下一节：62.4 –62.6 标志 · 控制字符 · 行控制

本节讲什么

VMIN/VTIME 四组合：MIN>0 等够字节；MIN=0 可轮询/短超时（TIME 单位 0.1s）。 Demo：code/（关 ECHO 读一行再恢复） ---

要点

模式
规范（ICANON） 按行；行编辑；shell 默认
非规范 关 ICANON；VMIN+VTIME 控 read
原始 Raw 非规范 + 关信号/回显/输出加工等
Cooked 口语=规范

VMIN/VTIME 四组合：MIN>0 等够字节；MIN=0 可轮询/短超时（TIME 单位 0.1s）。

Demo：code/（关 ECHO 读一行再恢复）

HFT / 嵌入式关联

路线 要点
嵌入式 非规范模式 + VMIN/VTIME 是嵌入式串口逐字节读取按键/传感器数据的标准做法，无需回车即可处理
HFT VMIN=0/VTIME=0 实现非阻塞轮询读，避免 read 阻塞拖住主循环；密码/密钥输入时关 ECHO 防回显到日志，读完立即恢复 ICANON

代码示例


/* 规范模式 (canonical) vs 原始模式 (raw) */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <termios.h>
#include <unistd.h>

int main(void) {
    struct termios oldtio, newtio;
    tcgetattr(STDIN_FILENO, &oldtio);
    newtio = oldtio;

    /* 原始模式：关闭 ICANON + ECHO，设置 VMIN=1, VTIME=0 */
    newtio.c_lflag &= ~(ICANON | ECHO);
    newtio.c_cc[VMIN] = 1;   /* 至少读 1 字节 */
    newtio.c_cc[VTIME] = 0;  /* 无超时 */
    tcsetattr(STDIN_FILENO, TCSANOW, &newtio);

    printf("Raw mode: press any key (no Enter needed), 'q' to quit\n");
    fflush(stdout);

    char c;
    while (read(STDIN_FILENO, &c, 1) == 1) {
        printf("got: 0x%02x '%c'\n", (unsigned char)c, c >= 32 ? c : '.');
        fflush(stdout);
        if (c == 'q') break;
    }

    /* 恢复规范模式 */
    tcsetattr(STDIN_FILENO, TCSANOW, &oldtio);
    printf("\nRestored to canonical mode\n");
    return 0;
}

62.4

–62.6 标志 · 控制字符 · 行控制

本章：TLPI 第 62 章 — Terminals - 上一节：62.3 输入模式（高频） - 下一节：62.7 检测

本节讲什么

c_lflag：ICANON · ECHO · ISIG… ICRNL / ONLCR：\r↔\n 转换。 c_cc：VINTR Ctrl+C→SIGINT；VSUSP Ctrl+Z→SIGTSTP；VEOF Ctrl+D…

要点

c_lflag：ICANON · ECHO · ISIG…
ICRNL / ONLCR：\r↔\n 转换。
c_cc：VINTR Ctrl+C→SIGINT；VSUSP Ctrl+Z→SIGTSTP；VEOF Ctrl+D…
tcflush / tcflow / tcsendbreak。

HFT / 嵌入式关联

路线 要点
嵌入式 关 ISIG 阻止 Ctrl+C 误杀嵌入式服务进程；关 ICRNL 做二进制串口传输，避免 \r→\n 被篡改
HFT ONLCR 控制输出 \n→\r\n，确保发往交易所的数据帧行尾正确；关 OPOST 省去输出加工，减少一次 CPU 拷贝与延迟

代码示例


/* termios 标志位详解与修改 */
#include <stdio.h>
#include <stdlib.h>
#include <termios.h>
#include <unistd.h>

int main(void) {
    struct termios tio;
    tcgetattr(STDIN_FILENO, &tio);

    /* c_lflag 标志位 */
    printf("Local flags (c_lflag):\n");
    printf("  ICANON: %d (line buffering, signal editing)\n", !!(tio.c_lflag & ICANON));
    printf("  ECHO:   %d (echo input characters)\n", !!(tio.c_lflag & ECHO));
    printf("  ECHOE:  %d (echo erase as BS-SP-BS)\n", !!(tio.c_lflag & ECHOE));
    printf("  ISIG:   %d (generate signals: INTR, QUIT, SUSP)\n", !!(tio.c_lflag & ISIG));

    /* c_iflag 标志位 */
    printf("\nInput flags (c_iflag):\n");
    printf("  ICRNL:  %d (translate CR to NL)\n", !!(tio.c_iflag & ICRNL));
    printf("  IXON:   %d (XON/XOFF flow control)\n", !!(tio.c_iflag & IXON));
    printf("  IGNBRK: %d (ignore break condition)\n", !!(tio.c_iflag & IGNBRK));

    /* 修改：关闭 ICRNL（CR 不转 NL，二进制模式常用） */
    tio.c_iflag &= ~ICRNL;
    tcsetattr(STDIN_FILENO, TCSANOW, &tio);
    printf("\nICRNL cleared (CR no longer translated to NL)\n");

    /* tcflush: 刷新缓冲区 */
    tcflush(STDIN_FILENO, TCIFLUSH);
    printf("tcflush: input buffer flushed\n");

    return 0;
}

62.5

Terminal Flags

本章：TLPI 第 62 章

本节讲什么

termios 结构中的 c_iflag（输入）、c_oflag（输出）、c_cflag（控制）和 c_lflag（本地）四组标志位控制终端的行行为。

要点

termios 结构回顾

struct termios {
    tcflag_t  c_iflag;   /* 输入标志 */
    tcflag_t  c_oflag;   /* 输出标志 */
    tcflag_t  c_cflag;   /* 控制标志 */
    tcflag_t  c_lflag;   /* 本地标志 */
    cc_t      c_cc[NCCS]; /* 特殊字符 */
};

输入标志 (c_iflag)

标志 说明
ICRNL 将 CR 映射为 NL
INLCR 将 NL 映射为 CR
IGNCR 忽略 CR
IXON 启用 XON/XOFF 流控
IXOFF 输入时启用 XON/XOFF
IXANY 任意字符恢复输出
ISTRIP 去除第8位
IGNBRK 忽略 BREAK
BRKINT BREAK 产生 SIGINT

输出标志 (c_oflag)

标志 说明
OPOST 启用输出处理
ONLCR 将 NL 映射为 CR-NL
OCRNL 将 CR 映射为 NL
ONOCR 第一列不输出 CR
TABDLY 制表符延迟
NL0/NL1 NL 延迟

控制标志 (c_cflag)

标志 说明
CLOCAL 忽略调制解调器控制
CREAD 启用接收
CS5-CS8 字符大小
CSTOPB 2 个停止位
PARENB 启用奇偶校验
PARODD 奇校验

本地标志 (c_lflag)

标志 说明 关闭效果
ICANON 规范模式 原始模式
ECHO 回显输入 不回显
ECHOE ERASE 删除前一个字符
ECHOK KILL 删除行
ECHONL 回显 NL
ISIG 信号字符 (INTR/QUIT) 不产生信号
IEXTEN 扩展处理

典用组合

模式 设置
规范模式（默认） ICANON | ECHO | ISIG
原始模式 清除 ICANON | ECHO | ISIG | IEXTEN
密码输入 ICANON on, ECHO off
8位原始 CS8 + 清除 ISTRIP + 原始

HFT / 嵌入式关联

  • 嵌入式串口通信：配置波特率、8N1、原始模式
  • 交易终端交互：规范模式读取命令
  • 串口协议解析：原始模式 + VMIN/VTIME

代码示例

#include <termios.h>
#include <unistd.h>
#include <stdio.h>

/* 打印当前终端标志 */
void print_termios_flags(int fd) {
    struct termios t;
    tcgetattr(fd, &t);

    printf("iflag: 0x%lx\n", (long)t.c_iflag);
    printf("  ICRNL:  %s\n", (t.c_iflag & ICRNL)  ? "on" : "off");
    printf("  IXON:   %s\n", (t.c_iflag & IXON)   ? "on" : "off");
    printf("oflag: 0x%lx\n", (long)t.c_oflag);
    printf("  OPOST:  %s\n", (t.c_oflag & OPOST)  ? "on" : "off");
    printf("  ONLCR:  %s\n", (t.c_oflag & ONLCR)  ? "on" : "off");
    printf("cflag: 0x%lx\n", (long)t.c_cflag);
    printf("  CSIZE:  CS%d\n", (t.c_cflag & CSIZE) >> 4);
    printf("  CREAD:  %s\n", (t.c_cflag & CREAD)  ? "on" : "off");
    printf("lflag: 0x%lx\n", (long)t.c_lflag);
    printf("  ICANON: %s\n", (t.c_lflag & ICANON) ? "on" : "off");
    printf("  ECHO:   %s\n", (t.c_lflag & ECHO)   ? "on" : "off");
    printf("  ISIG:   %s\n", (t.c_lflag & ISIG)   ? "on" : "off");
}

代码自测

ICRNL 和 INLCR 有什么区别？
  • ICRNL：将输入的 CR (回车, \r) 转换为 NL (换行, \n)。这是默认行为，让终端的回车键作为行结束符。
  • INLCR：将输入的 NL (换行) 转换为 CR (回车)。较少使用，主要用于特殊协议解析。

两者方向相反：ICRNL 是 CR→NL，INLCR 是 NL→CR。

62.6

Terminal I/O Modes

本章：TLPI 第 62 章

本节讲什么

终端的两种主要 I/O 模式：规范模式（canonical, 行缓冲）和原始模式（raw, 非规范），以及 VMIN/VTIME 超时控制。

要点

规范模式 vs 原始模式

特性 规范模式 (ICANON) 原始模式 (~ICANON)
缓冲 行缓冲（按 NL/CR 提交） 无缓冲（按字符/超时）
行编辑 支持（ERASE/KILL） 不支持
信号 INTR/QUIT/SUSP 产生信号 不产生
回显 可配置 可配置
触发读取 整行或 EOF VMIN/VTIME 控制
适用 shell, 交互命令 串口, TUI, 协议解析

VMIN 和 VTIME

组合 行为 典型用途
VMIN=1, VTIME=0 阻塞等1个字符 交互式，按键响应
VMIN=0, VTIME=0 非阻塞轮询 检测输入但不阻塞
VMIN=0, VTIME=N 超时 N*100ms 无输入返回 超时读取
VMIN=1, VTIME=N 等到1字符或超时 协议帧解析

设置原始模式

struct termios raw, orig;
tcgetattr(fd, &orig);
raw = orig;

/* 关闭 ICANON, ECHO, ISIG, IEXTEN */
raw.c_lflag &= ~(ICANON | ECHO | ISIG | IEXTEN);
/* 关闭输入处理 */
raw.c_iflag &= ~(ICRNL | INLCR | IGNCR | IXON | IXOFF);
/* 关闭输出处理 */
raw.c_oflag &= ~OPOST;
/* 8 位字符 */
raw.c_cflag |= CS8;

/* 非规范参数 */
raw.c_cc[VMIN] = 1;   /* 至少1字符 */
raw.c_cc[VTIME] = 0;  /* 无超时 */

tcsetattr(fd, TCSAFLUSH, &raw);
/* ... 操作终端 ... */
/* 恢复 */
tcsetattr(fd, TCSAFLUSH, &orig);

cfmakeraw() 便捷函数

#include <termios.h>
struct termios t;
tcgetattr(fd, &t);
cfmakeraw(&t);  /* 设置为原始模式 */
tcsetattr(fd, TCSANOW, &t);

tcsetattr 的 when 参数

值 说明
TCSANOW 立即生效
TCSADRAIN 等待输出完成后生效
TCSAFLUSH 刷新输入输出队列后生效

HFT / 嵌入式关联

  • 嵌入式串口通信：原始模式 + VMIN/VTIME 解析协议帧
  • 交易终端 TUI：原始模式捕获按键，关闭回显
  • 串口调试：原始模式透传二进制数据

代码示例

#include <termios.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>

/* 原始模式读取串口数据（带超时） */
int read_serial_raw(int fd, void *buf, size_t len, int timeout_ms) {
    struct termios orig, raw;

    tcgetattr(fd, &orig);
    raw = orig;
    cfmakeraw(&raw);

    /* 设置超时 */
    raw.c_cc[VMIN] = 0;
    raw.c_cc[VTIME] = (timeout_ms + 50) / 100;  /* 100ms 单位 */

    tcsetattr(fd, TCSANOW, &raw);

    int n = read(fd, buf, len);

    /* 恢复 */
    tcsetattr(fd, TCSANOW, &orig);
    return n;
}

/* 交互式 TUI：捕获单键 */
int kbhit(void) {
    struct termios orig, raw;
    tcgetattr(STDIN_FILENO, &orig);
    raw = orig;
    raw.c_lflag &= ~(ICANON | ECHO);
    raw.c_cc[VMIN] = 0;
    raw.c_cc[VTIME] = 0;
    tcsetattr(STDIN_FILENO, TCSANOW, &raw);

    char c;
    int n = read(STDIN_FILENO, &c, 1);

    tcsetattr(STDIN_FILENO, TCSANOW, &orig);
    return (n == 1) ? (int)c : 0;
}

要点

会话最多一控制终端；前台进程组收 Ctrl+C/Z。
后台读 → SIGTTIN；后台写 → SIGTTOU（作业控制基础）。

SIGINT / SIGQUIT / SIGTSTP / SIGTTIN / SIGTTOU。

HFT / 嵌入式关联

路线 要点
嵌入式 后台守护进程 setsid 脱离控制终端，就不会因终端关闭或 Ctrl+C 被杀，是嵌入式常驻服务的标准姿势

代码示例


/* 终端产生的信号：SIGINT / SIGQUIT / SIGTSTP */
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <termios.h>
#include <unistd.h>

static volatile sig_atomic_t got_sigint = 0;
static volatile sig_atomic_t got_sigquit = 0;
static volatile sig_atomic_t got_sigtstp = 0;

static void handler(int sig) {
    switch (sig) {
        case SIGINT:  got_sigint = 1; break;
        case SIGQUIT: got_sigquit = 1; break;
        case SIGTSTP: got_sigtstp = 1; break;
    }
}

int main(void) {
    /* 终端按键产生的信号 */
    signal(SIGINT, handler);   /* Ctrl+C */
    signal(SIGQUIT, handler);  /* Ctrl+\ */
    signal(SIGTSTP, handler);  /* Ctrl+Z */

    printf("Terminal signals demo:\n");
    printf("  Ctrl+C  -> SIGINT  (interrupt)\n");
    printf("  Ctrl+\\  -> SIGQUIT (quit + core dump)\n");
    printf("  Ctrl+Z  -> SIGTSTP (terminal stop)\n");
    printf("  Ctrl+D  -> EOF (not a signal, closes stdin)\n\n");
    printf("Waiting... (run in a terminal to test)\n");

    /* 等待信号 */
    for (int i = 0; i < 30; i++) {
        if (got_sigint) { printf("Got SIGINT\n"); break; }
        if (got_sigquit) { printf("Got SIGQUIT\n"); break; }
        if (got_sigtstp) { printf("Got SIGTSTP\n"); break; }
        sleep(1);
    }

    return 0;
}

事件 信号 默认行为

void winch_handler(int sig) {

/* 注册 */
struct sigaction sa = {0};

问题 说明

  • 嵌入式串口终端重绘
  • 监控面板动态调整行列布局

代码示例

#include <sys/ioctl.h>
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

static volatile sig_atomic_t resized = 1;

static void on_winch(int sig) { resized = 1; }

int get_terminal_size(int *rows, int *cols) {
    struct winsize ws;
    if (ioctl(STDOUT_FILENO, TIOCGWINSZ, &ws) == -1)
        return -1;
    *rows = ws.ws_row;
    *cols = ws.ws_col;
    return 0;
}

int main() {
    /* 注册 SIGWINCH */
    struct sigaction sa = {0};
    sa.sa_handler = on_winch;
    sigaction(SIGWINCH, &sa, NULL);

    int rows, cols;
    while (1) {
        if (resized) {
            resized = 0;
            if (get_terminal_size(&rows, &cols) == 0)
                printf("\rTerminal: %dx%d\n", cols, rows);
        }
        pause(); /* 等待信号 */
    }
    return 0;
}

代码自测

为什么 SIGWINCH 的默认行为是忽略？程序如何知道窗口大小变化？

SIGWINCH 默认被忽略是因为不是所有程序都关心窗口大小变化（如非交互式程序）。关心窗口大小的交互式程序（如编辑器、TUI）需要注册信号处理函数，在信号到达后重新调用 ioctl(TIOCGWINSZ) 获取新大小并重绘界面。信号只通知"变化了"，不包含新尺寸，必须主动查询。

Terminal Identification

本章：TLPI 第 62 章

本节讲什么

终端设备的识别方法：isatty() 检查是否为终端，ttyname() 获取终端设备名，ctermid() 获取控制终端名。

要点

核心 API

#include <unistd.h>
int isatty(int fd);            /* 检查 fd 是否为终端 */

char *ctermid(char *s);        /* 返回控制终端名 */

使用示例

if (isatty(STDIN_FILENO)) {
    printf("STDIN is terminal: %s\n", ttyname(STDIN_FILENO));
} else {
    printf("STDIN is not a terminal (redirected)\n");
}

isatty 行为

输入 fd 返回值 说明
终端设备 1 (true) 如 /dev/tty1, /dev/pts/0
普通文件 0 (false) 重定向到文件
管道 0 (false) echo "data" | app
socket 0 (false) 网络连接
无效 fd 0 (false) errno = EBADF

ttyname 实现

ttyname(fd) 内部： 1. isatty(fd) 确认是终端 2. fstat(fd) 获取设备号 3. 扫描 /dev/ 查找匹配设备号的文件 4. 返回匹配的路径

ctermid

char *name = ctermid(NULL);
/* 返回进程的控制终端路径，如 "/dev/tty" */

/dev/tty 是特殊设备文件，总是指向当前进程的控制终端。

检测重定向

/* 检查 stdin/stdout/stderr 是否被重定向 */
if (!isatty(STDIN_FILENO))  { /* stdin 被重定向 */ }
if (!isatty(STDOUT_FILENO)) { /* stdout 被重定向 */ }
if (!isatty(STDERR_FILENO)) { /* stderr 被重定向 */ }

/proc/self/fd 代替方案

# 查看进程 fd 对应的设备
ls -la /proc/self/fd/
# 0 -> /dev/pts/0
# 1 -> /dev/pts/0
# 2 -> /dev/pts/0

  • 交易系统日志：检测 stdout 是否为终端，决定是否着色
  • 嵌入式串口设备识别：ttyname 确认设备路径
  • 守护进程：isatty 检查是否有关联终端

代码示例

#include <unistd.h>
#include <stdio.h>

int main() {
    /* 检查各标准流 */
    const char *streams[] = { "STDIN", "STDOUT", "STDERR" };
    int fds[] = { STDIN_FILENO, STDOUT_FILENO, STDERR_FILENO };

    for (int i = 0; i < 3; i++) {
        if (isatty(fds[i])) {
            char *name = ttyname(fds[i]);
            printf("%s is terminal: %s\n", streams[i],
                   name ? name : "(unknown)");
        } else {
            printf("%s is NOT a terminal\n", streams[i]);
        }
    }

    /* 控制终端 */
    char term[L_ctermid];
    ctermid(term);
    printf("Control terminal: %s\n", term);

    return 0;
}

规范模式和原始模式的主要区别是什么？

规范模式（ICANON）：内核对输入进行行缓冲和行编辑。用户输入的字符先在内核缓冲区中积累，直到收到 NL/EOF 才提交给 read()。支持 ERASE（删除前一个字符）和 KILL（删除整行）。适合交互式命令输入（如 shell）。原始模式（~ICANON）：无行缓冲，read() 直接返回已输入的字符，无行编辑功能。用 VMIN/VTIME 控制读取行为。适合串口通信和 TUI 程序。

练习 1：密码输入

题目：编写函数 getpass_safe()，提示用户输入密码，输入时回显星号。

参考答案 
#include <termios.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>

char *getpass_safe(const char *prompt, char *buf, size_t len) {
    struct termios orig, noecho;

    /* 输出提示 */
    fputs(prompt, stdout);
    fflush(stdout);

    /* 关闭回显 */
    tcgetattr(STDIN_FILENO, &orig);
    noecho = orig;
    noecho.c_lflag &= ~(ECHO | ECHOE | ECHOK);
    tcsetattr(STDIN_FILENO, TCSANOW, &noecho);

    /* 读取输入 */
    char *result = fgets(buf, len, stdin);

    /* 恢复回显 */
    tcsetattr(STDIN_FILENO, TCSANOW, &orig);
    fputc('\n', stdout);

    /* 去除换行 */
    if (result) {
        char *nl = strchr(buf, '\n');
        if (nl) *nl = '\0';
    }

    return result;
}

练习 2：串口配置

题目：配置串口 /dev/ttyS0 为 115200 波特率、8N1、原始模式、VMIN=1/VTIME=10。

参考答案 
#include <termios.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int open_serial(const char *dev) {
    int fd = open(dev, O_RDWR | O_NOCTTY | O_CLOEXEC);
    if (fd < 0) return -1;

    struct termios t;
    tcgetattr(fd, &t);

    /* 原始模式 */
    cfmakeraw(&t);

    /* 波特率 115200 */
    cfsetispeed(&t, B115200);
    cfsetospeed(&t, B115200);

    /* 8N1 */
    t.c_cflag &= ~PARENB;   /* 无奇偶 */
    t.c_cflag &= ~CSTOPB;   /* 1 停止位 */
    t.c_cflag &= ~CSIZE;    /* 清除字符大小 */
    t.c_cflag |= CS8;       /* 8 位 */

    /* 启用接收 */
    t.c_cflag |= CREAD | CLOCAL;

    /* VMIN/VTIME */
    t.c_cc[VMIN] = 1;        /* 最少1字符 */
    t.c_cc[VTIME] = 10;      /* 1秒超时(100ms*10) */

    tcsetattr(fd, TCSANOW, &t);
    return fd;
}

练习 3：检测重定向

题目：编写程序检测 stdout 是否被重定向到文件或管道，如果被重定向则禁用颜色输出。

参考答案 
#include <unistd.h>
#include <stdio.h>
#include <string.h>

int use_color(void) {
    /* 检查 stdout 是否为终端 */
    if (!isatty(STDOUT_FILENO)) return 0;

    /* 检查 TERM 环境变量 */
    char *term = getenv("TERM");
    if (term == NULL) return 0;
    if (strcmp(term, "dumb") == 0) return 0;

    return 1;
}

int main() {
    const char *red   = use_color() ? "\033[31m" : "";
    const char *reset = use_color() ? "\033[0m"  : "";

    printf("%sERROR:%s Something went wrong\n", red, reset);
    return 0;
}
