章节目标

IPv4/端口；sockaddr_in；字节序；pton/ntop；TCP vs UDP；sockaddr_storage；INADDR_ANY。

初始化模板

struct sockaddr_in addr;
memset(&addr, 0, sizeof(addr));
addr.sin_family = AF_INET;
addr.sin_port = htons(8080);
inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);
/* or: addr.sin_addr.s_addr = htonl(INADDR_ANY); */

陷阱

  1. 忘 htons/htonl
  2. 用 inet_ntoa
  3. 当 UDP 可靠
  4. TCP 一次 send≠一次 recv
  5. sin_zero 未清零
  6. 客户端 connect INADDR_ANY

背诵卡

# 要点
1 端点 = IP + 端口
2 网络大端；htons/htonl
3 pton/ntop；勿 ntoa
4 INADDR_ANY 仅服务端 bind
5 TCP 流 / UDP 报；UDP 可丢
6 UNIX DGRAM ≠ UDP

#include <stdio.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <netdb.h>
#include <string.h>

/* Ch58 TCP/IP 基础 — 地址转换 + DNS 解析。
 * 演示 inet_pton/inet_ntop + getaddrinfo。
 * 编译: gcc -o ch58_demo ch58_demo.c */

int main(void) {
    /* inet_pton: 文本 IP -> 二进制 */
    struct in_addr addr;
    if (inet_pton(AF_INET, "127.0.0.1", &addr) == 1)
        printf("inet_pton: 127.0.0.1 -> 0x%x\n", ntohl(addr.s_addr));

    /* inet_ntop: 二进制 -> 文本 IP */
    char ip_str[INET_ADDRSTRLEN];
    addr.s_addr = htonl(0x08080808);  /* 8.8.8.8 */
    if (inet_ntop(AF_INET, &addr, ip_str, sizeof(ip_str)))
        printf("inet_ntop: 0x08080808 -> %s\n", ip_str);

    /* getaddrinfo: 现代 DNS 解析 (替代过时的 gethostbyname) */
    struct addrinfo hints, *res;
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;       /* IPv4 或 IPv6 */
    hints.ai_socktype = SOCK_STREAM;   /* TCP */

    int err = getaddrinfo("localhost", "80", &hints, &res);
    if (err != 0) {
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(err));
        return 1;
    }

    printf("\ngetaddrinfo(\"localhost\", \"80\"):\n");
    for (struct addrinfo *p = res; p; p = p->ai_next) {
        char host[NI_MAXHOST], service[NI_MAXSERV];
        getnameinfo(p->ai_addr, p->ai_addrlen,
                    host, sizeof(host),
                    service, sizeof(service),
                    NI_NUMERICHOST | NI_NUMERICSERV);
        int family = p->ai_family == AF_INET ? 4 : 6;
        printf("  IPv%d: [%s]:%s\n", family, host, service);
    }
    freeaddrinfo(res);

    /* 字节序转换 */
    printf("\nByte order:\n");
    printf("  htons(80) = %u\n", htons(80));
    printf("  htonl(0x12345678) = 0x%x\n", htonl(0x12345678));
    return 0;
}

58.1

INET vs UNIX

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：本章首页 - 下一节：58.2 –58.3 IPv4 · 端口

本节讲什么

SOCK_STREAM→TCP；SOCK_DGRAM→UDP。 ---

要点

AF_UNIX AF_INET(6)
定位 路径/抽象名 IP + 端口
范围 仅本机 跨机 / 回环

SOCK_STREAM→TCP；SOCK_DGRAM→UDP。

HFT / 嵌入式关联

路线 要点
嵌入式 AF_UNIX 仅本机、定位靠路径名，嵌入式本机服务用 AF_UNIX 省去 IP 栈开销；跨机/回环才用 AF_INET
HFT AF_UNIX 无 IP/TCP 头、零网络栈开销，同机进程 IPC 延迟最低；行情分发/策略进程间通信用 AF_UNIX，跨主机才走 AF_INET

代码示例


/* 网络层与 IPv4 地址结构 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>
#include <netinet/in.h>

int main(void) {
    /* IPv4 地址：32 位，4 字节 */
    printf("IPv4 address: 32 bits, 4 bytes\n");
    printf("INADDR_ANY  = 0x%08x (0.0.0.0)\n", ntohl(htonl(INADDR_ANY)));
    printf("INADDR_LOOPBACK = 0x%08x (127.0.0.1)\n",
           ntohl(htonl(INADDR_LOOPBACK)));
    printf("INADDR_BROADCAST = 0x%08x (255.255.255.255)\n",
           ntohl(htonl(INADDR_BROADCAST)));

    /* 私有地址范围 */
    const char *priv[] = { "10.0.0.1", "172.16.0.1", "192.168.1.1" };
    for (int i = 0; i < 3; i++) {
        struct in_addr addr;
        inet_pton(AF_INET, priv[i], &addr);
        printf("private %s -> 0x%08x\n", priv[i], ntohl(addr.s_addr));
    }

    /* 回环地址 */
    struct in_addr lo;
    inet_pton(AF_INET, "127.0.0.1", &lo);
    printf("loopback: 0x%08x (host byte order)\n", ntohl(lo.s_addr));
    return 0;
}

58.2

–58.3 IPv4 · 端口

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.1 INET vs UNIX - 下一节：58.4 sockaddr_in

本节讲什么

  • 32 位；点分十进制；127.0.0.1 回环（不经物理网卡） - INADDR_ANY (0.0.0.0)：服务端 bind 全网卡；客户端 connect 不用 - 私有：10/8、172.16/12、192.168/16

要点

  • 32 位；点分十进制；127.0.0.1 回环（不经物理网卡）
  • INADDR_ANY (0.0.0.0)：服务端 bind 全网卡；客户端 connect 不用
  • 私有：10/8、172.16/12、192.168/16

端口 0–65535：知名 0–1023（常需 root）；注册 1024–49151；临时 49152–65535（客户端常自动分配，一般不手 bind）。

HFT / 嵌入式关联

路线 要点
嵌入式 INADDR_ANY 绑全网卡，多网卡设备服务端 bind 全网卡；客户端 connect 不用 bind，端口由内核临时分配
HFT INADDR_ANY 监听所有网卡，HFT 跨机房要 bind 特定 IP 走专线网卡；端口分段（临时 49152+）避免与知名端口冲突

代码示例


/* IPv4 地址分类与子网掩码 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>

int main(void) {
    /* A/B/C 类地址判断 */
    const char *addrs[] = {
        "10.1.2.3",      /* A 类: 1.0.0.0 ~ 126.255.255.255 */
        "172.16.1.2",    /* B 类: 128.0.0.0 ~ 191.255.255.255 */
        "192.168.1.1",   /* C 类: 192.0.0.0 ~ 223.255.255.255 */
        "127.0.0.1",     /* 回环 */
        "224.0.0.1"      /* D 类组播 */
    };

    for (int i = 0; i < 5; i++) {
        struct in_addr addr;
        inet_pton(AF_INET, addrs[i], &addr);
        unsigned int a = ntohl(addr.s_addr);
        unsigned char first = (a >> 24) & 0xFF;

        const char *cls;
        if (first < 128) cls = "A";
        else if (first < 192) cls = "B";
        else if (first < 224) cls = "C";
        else if (first < 240) cls = "D (multicast)";
        else cls = "E (reserved)";

        printf("%-16s first=%3d -> class %s\n", addrs[i], first, cls);
    }
    return 0;
}

58.3

The Data-Link Layer

本章：TLPI 第 58 章 — TCP/IP Fundamentals

本节讲什么

介绍 TCP/IP 协议栈的数据链路层（Data-Link Layer）：以太网帧格式、MAC 地址、ARP 协议、以及链路层在 TCP/IP 协议栈中的角色。

要点

TCP/IP 协议栈层次

应用层     HTTP, DNS, SIP, custom
           ↓
传输层     TCP / UDP
           ↓
网络层     IP (IPv4/IPv6), ICMP
           ↓
数据链路层  Ethernet, Wi-Fi, PPP
           ↓
物理层     光纤, 双绞线, 无线电波

数据链路层职责

职责 说明
帧封装 将 IP 数据报封装为以太网帧
MAC 地址 48 位硬件地址，标识网卡接口
ARP 将 IP 地址解析为 MAC 地址
MTU 最大传输单元（以太网 1500 字节）
错误检测 CRC32 帧校验

以太网帧格式

┌───────────┬───────────┬─────┬───────────┬──────────┐
│ 目标 MAC   │ 源 MAC    │类型 │  数据       │  CRC     │
│ 6 字节     │ 6 字节     │2字节│ 46-1500   │ 4 字节   │
└───────────┴───────────┴─────┴───────────┴──────────┘
                           ↑
                    0x0800 = IPv4
                    0x86DD = IPv6
                    0x0806 = ARP
                    0x8100 = VLAN tag

MAC 地址

概念 说明
长度 48 位（6 字节）
格式 XX:XX:XX:XX:XX:XX
厂商 OUI 前 24 位
设备标识 后 24 位
广播地址 FF:FF:FF:FF:FF:FF
组播 第一字节最低位为 1
本地管理 第二位为 1（本地分配）

ARP 协议

进程 A (192.168.1.10) → 通信 192.168.1.20

1. 查 ARP 缓存: 192.168.1.20 → ? (无缓存)
2. 广播 ARP 请求: "谁有 192.168.1.20?"
3. 192.168.1.20 单播回复: "我是 192.168.1.20, MAC=xx:xx:xx:xx:xx:xx"
4. A 更新 ARP 缓存
5. A 发送以太网帧到目标 MAC

ARP 命令

# 查看 ARP 缓存
arp -a
ip neigh

# 静态 ARP 条目（防止 ARP 欺骗）
arp -s 192.168.1.1 00:11:22:33:44:55

MTU

链路类型 MTU
以太网 1500 字节
以太网 Jumbo Frame 9000 字节
PPP 1500 字节
GRE 隧道 1476 字节
IPv6 1280 字节（最小）

MTU 对 TCP 的影响：MSS = MTU - IP头(20) - TCP头(20) = 1460 字节

HFT / 嵌入式关联

  • 组播：HFT 行情数据使用 IP 组播，链路层映射为以太网组播帧
  • Jumbo Frame：HFT 专用网络启用 9000 字节 Jumbo Frame，减少帧头开销
  • ARP 缓存：HFT 服务器对关键网关和行情源设静态 ARP（防欺骗）
  • 网卡亲和性：多队列网卡 RSS，每个队列绑定到不同 CPU 核心
  • DPDK：绕过内核网络栈，用户态直接从网卡 DMA 读取以太网帧
  • 嵌入式：嵌入式以太网通常 MTU 1500，支持 IEEE 1588 PTP

代码示例

# 查看接口 MTU
ip link show eth0

# 修改 MTU（需 root）
ip link set eth0 mtu 9000

# 查看 ARP 缓存
ip neigh show

# 抓取以太网帧（链路层）
tcpdump -i eth0 -e -XX

/* 获取接口 MAC 地址 */
#include <net/if.h>
#include <sys/ioctl.h>
#include <linux/if.h>

int get_mac(const char *ifname, unsigned char *mac) {
    int fd = socket(AF_INET, SOCK_DGRAM, 0);
    struct ifreq ifr;
    strncpy(ifr.ifr_name, ifname, IFNAMSIZ);
    ioctl(fd, SIOCGIFHWADDR, &ifr);
    memcpy(mac, ifr.ifr_hwaddr.sa_data, 6);
    close(fd);
    return 0;
}

代码自测

1. MTU 和 MSS 的关系？

MSS（Maximum Segment Size）是 TCP 数据段的最大数据部分大小。MSS = MTU - IP头(20字节) - TCP头(20字节)。以太网 MTU=1500 → MSS=1460。Jumbo Frame MTU=9000 → MSS=8960。HFT 网络用 Jumbo Frame 减少大消息的分片。

2. 为什么 HFT 网络要用 Jumbo Frame？
  1. 减少帧头开销：每个以太网帧有 38 字节开销（帧头+CRC+间隙），1500 字节帧开销比例 2.5%，9000 字节帧仅 0.4%
  2. 减少中断次数：大消息分片更少，网卡中断更少
  3. 减少 CPU 开销：处理更少的帧，帧处理开销降低
  4. 提升吞吐：尤其适合大块数据传输（行情快照）
3. ARP 欺骗是什么？如何防御？

ARP 欺骗：攻击者发送伪造的 ARP 回复，将自己的 MAC 与网关 IP 关联，使受害者的流量经过攻击者。防御：(1) 静态 ARP 条目（arp -s）；(2) 动态 ARP 检测（交换机层）；(3) 网络隔离（专用 VLAN）；(4) HFT 专用网络通常与办公网络物理隔离。

58.4

`sockaddr_in`

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.2 –58.3 IPv4 · 端口 - 下一节：58.5 网络字节序（高频）

本节讲什么

struct sockaddr_in { sa_family_t sin_family; / AF_INET / in_port_t sin_port; / 网络序 /

要点

struct sockaddr_in {
    sa_family_t    sin_family;  /* AF_INET */
    in_port_t      sin_port;    /* 网络序 */
    struct in_addr sin_addr;    /* s_addr 网络序 */
    unsigned char  sin_zero[8]; /* 置 0 */
};

传给 bind/connect 时转为 struct sockaddr *。

HFT / 嵌入式关联

路线 要点
嵌入式 sockaddr_in 转 sockaddr* 传给 bind/connect，写双栈代码用 sockaddr_storage 兼容 v4/v6；sin_zero 置 0
HFT sin_port/sin_addr 必须网络序（htons/htonl），端序错直接连不上；结构填充一次，热路径不重复构造

代码示例


/* struct sockaddr_in 结构详解 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>
#include <netinet/in.h>
#include <sys/socket.h>

int main(void) {
    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));

    addr.sin_family = AF_INET;       /* 地址族 */
    addr.sin_port = htons(8080);     /* 端口（网络字节序） */
    inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);  /* IP 地址 */

    printf("sizeof(sockaddr_in) = %zu\n", sizeof(addr));
    printf("sin_family = %d\n", addr.sin_family);
    printf("sin_port   = %u (network=%u, host=%u)\n",
           addr.sin_port, addr.sin_port, ntohs(addr.sin_port));
    printf("sin_addr   = 0x%x\n", ntohl(addr.sin_addr.s_addr));

    /* sockaddr_in 可安全转型为 sockaddr */
    struct sockaddr *gen = (struct sockaddr *)&addr;
    printf("cast to sockaddr*: sa_family=%d\n", gen->sa_family);

    /* sockaddr_storage: 足够大，兼容 IPv4/IPv6 */
    struct sockaddr_storage ss;
    memset(&ss, 0, sizeof(ss));
    printf("sizeof(sockaddr_storage) = %zu (enough for IPv6)\n", sizeof(ss));
    return 0;
}

58.5

网络字节序（高频）

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.4 sockaddr_in - 下一节：58.6 地址转换

本节讲什么

网络固定 大端；主机常见小端。 htons/htonl / ntohs/ntohl — 永远调用，勿猜本机端序。 ---

要点

网络固定 大端；主机常见小端。
htons/htonl / ntohs/ntohl — 永远调用，勿猜本机端序。

HFT / 嵌入式关联

路线 要点
嵌入式 网络固定大端、主机常见小端，ARM 嵌入式多为小端，htons/htonl 永远显式调用、勿猜本机端序
HFT 端序转换廉价但别在热路径反复做；报文头解析时端口/IP 一次性 ntohs/ntohl 并缓存；端序错=连错地址/端口，是高频低级 bug

代码示例


/* 字节序：主机字节序 vs 网络字节序 */
#include <stdio.h>
#include <stdlib.h>
#include <arpa/inet.h>

int main(void) {
    /* 检测主机字节序 */
    unsigned int x = 0x01020304;
    char *p = (char *)&x;
    if (p[0] == 0x01) {
        printf("Host byte order: Big-endian\n");
    } else {
        printf("Host byte order: Little-endian\n");
    }

    /* htons/htonl: host to network (short/long) */
    unsigned short h_port = 8080;
    unsigned short n_port = htons(h_port);
    printf("port %u -> htons -> %u\n", h_port, n_port);

    /* ntohs/ntohl: network to host */
    printf("port %u -> ntohs -> %u (round-trip)\n", n_port, ntohs(n_port));

    /* 直接用 inet_addr（已过时，推荐 inet_pton） */
    in_addr_t addr = inet_addr("127.0.0.1");
    printf("inet_addr('127.0.0.1') = 0x%x (already network order)\n",
           ntohl(addr));

    return 0;
}

58.6

地址转换

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.5 网络字节序（高频） - 下一节：58.7 TCP vs UDP

本节讲什么

pton：返回 1 成功，0 非法串，-1 错误。 Demo：code/ ---

要点

推荐 inet_pton / inet_ntop（IPv4+IPv6）
废弃倾向 inet_addr/aton/ntoa（仅 v4；ntoa 静态缓冲、非线程安全）

pton：返回 1 成功，0 非法串，-1 错误。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 inet_ntoa 用静态缓冲非线程安全，多线程服务用 inet_ntop；inet_pton 支持 v4/v6 是双栈唯一选择
HFT 地址转换只在启动/配置期做一次（字符串→二进制），热路径直接用二进制 in_addr 结构零转换；inet_ntoa 静态缓冲多线程下会串数据，禁用

代码示例


/* inet_pton / inet_ntop 地址转换函数 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>

int main(void) {
    /* inet_pton: presentation -> network (字符串 → 二进制) */
    struct in_addr addr4;
    if (inet_pton(AF_INET, "192.168.1.100", &addr4) == 1) {
        printf("inet_pton IPv4: 192.168.1.100 -> 0x%08x\n",
               ntohl(addr4.s_addr));
    }

    /* inet_ntop: network -> presentation (二进制 → 字符串) */
    char str[INET_ADDRSTRLEN];
    inet_ntop(AF_INET, &addr4, str, sizeof(str));
    printf("inet_ntop IPv4: 0x%08x -> %s\n", ntohl(addr4.s_addr), str);

    /* IPv6 支持 */
    struct in6_addr addr6;
    if (inet_pton(AF_INET6, "::1", &addr6) == 1) {
        char str6[INET6_ADDRSTRLEN];
        inet_ntop(AF_INET6, &addr6, str6, sizeof(str6));
        printf("inet_pton/ntop IPv6: ::1 -> %s\n", str6);
    }

    /* 无效地址测试 */
    if (inet_pton(AF_INET, "256.1.1.1", &addr4) != 1) {
        printf("inet_pton: '256.1.1.1' is invalid\n");
    }
    return 0;
}

TCP vs UDP

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.6 地址转换 - 下一节：58.8 –58.9 sockaddr_storage · INADDR_ANY

本节讲什么

再记：UNIX DGRAM 可靠；INET UDP 不可靠。 ---

要点

TCP UDP
连接 有（握手/挥手） 无
可靠 ACK/重传/去重 可丢/乱序/重复
模型 字节流 报文边界
控流 有 无
场景 HTTP/SSH/文件 DNS/直播/低延迟容忍丢包

再记：UNIX DGRAM 可靠；INET UDP 不可靠。

HFT / 嵌入式关联

路线 要点
嵌入式 UNIX DGRAM 可靠、INET UDP 不可靠，局域网内 UDP 可容忍丢包做低延迟采集；TCP 控流适合可靠传输
HFT 行情/订单用 UDP（低开销容忍丢包、报文边界天然）、下单回报要可靠用 TCP；UDP 无控流是双刃剑，洪泛会丢包需应用层限速

代码示例


/* TCP vs UDP socket 创建对比 */
#include <stdio.h>
#include <stdlib.h>
#include <sys/socket.h>
#include <unistd.h>

int main(void) {
    /* TCP: SOCK_STREAM — 面向连接，可靠，有序 */
    int tcp_fd = socket(AF_INET, SOCK_STREAM, 0);
    printf("TCP socket (SOCK_STREAM):\n");
    printf("  - connection-oriented (need connect/accept)\n");
    printf("  - reliable delivery, ordered\n");
    printf("  - flow control, congestion control\n");
    printf("  - use send/recv\n\n");

    /* UDP: SOCK_DGRAM — 无连接，不可靠 */
    int udp_fd = socket(AF_INET, SOCK_DGRAM, 0);
    printf("UDP socket (SOCK_DGRAM):\n");
    printf("  - connectionless (sendto/recvfrom)\n");
    printf("  - unreliable, may lose/reorder\n");
    printf("  - no flow control\n");
    printf("  - lower overhead, suitable for real-time\n");

    close(tcp_fd);
    close(udp_fd);
    return 0;
}

–58.9 `sockaddr_storage` · `INADDR_ANY`

本章：TLPI 第 58 章 — Sockets: Fundamentals of TCP/IP Networks - 上一节：58.7 TCP vs UDP - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

sockaddr_storage：够大装 v4/v6，写双栈通用代码。 服务端：sin_addr.s_addr = htonl(INADDR_ANY)。 ---

要点

sockaddr_storage：够大装 v4/v6，写双栈通用代码。
服务端：sin_addr.s_addr = htonl(INADDR_ANY)。

HFT / 嵌入式关联

路线 要点
嵌入式 sockaddr_storage 足够大装 v4/v6，写双栈通用代码用它避免地址族截断；服务端 INADDR_ANY 绑全网卡
HFT sockaddr_storage 是通用地址容器，同时监听 v4/v6 时用它避免 sockaddr_in 截断；INADDR_ANY 绑所有网卡可能收非预期流量，生产指定网卡 IP

代码示例


/* sockaddr_storage 与 INADDR_ANY 的使用 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

int main(void) {
    /* sockaddr_storage: 通用存储，足够容纳任何地址族 */
    struct sockaddr_storage ss;
    memset(&ss, 0, sizeof(ss));
    printf("sizeof(sockaddr_storage) = %zu\n", sizeof(ss));
    printf("sizeof(sockaddr_in)      = %zu\n", sizeof(struct sockaddr_in));
    printf("sizeof(sockaddr_in6)     = %zu\n", sizeof(struct sockaddr_in6));

    /* INADDR_ANY: 绑定所有网络接口 */
    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);  /* 0.0.0.0 */

    char ipstr[INET_ADDRSTRLEN];
    inet_ntop(AF_INET, &addr.sin_addr, ipstr, sizeof(ipstr));
    printf("\nINADDR_ANY bind: %s:%u\n", ipstr, ntohs(addr.sin_port));
    printf("(server listens on all interfaces: lo, eth0, wlan0, ...)\n");

    /* 对比指定地址 */
    inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);
    inet_ntop(AF_INET, &addr.sin_addr, ipstr, sizeof(ipstr));
    printf("specific bind:  %s:%u (only loopback)\n", ipstr, ntohs(addr.sin_port));
    return 0;
}
