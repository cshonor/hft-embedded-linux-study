17.1

Overview 访问控制列表概览

本章：TLPI 第 17 章 — Access Control Lists ⚠️ 平台标注：POSIX draft ACL（libacl + system.posix_acl_* xattr）是 Linux 专有完整实现；macOS 无对应常量/API。本章判定算法有全平台可跑的用户态复现（本目录 c17_1_acl_algorithm.c），API 类代码标注 Linux 复测平台（Pi5/ext4）。 源码核验：man-pages 6.19 acl(5) / acl_get_file(3) + TLPI dist acl/ - 上一节：本章首页 - 下一节：17.2 ACL Permission-Checking Algorithm

完整源码（code/c17_1_acl_algorithm.c）：

/* c17_1_acl_algorithm.c — Ch17 §17.2：ACL 权限判定算法的纯用户态实现
 *
 * ⚠️ POSIX draft ACL 的内核实现（libacl / system.* xattr）是 Linux 专有；
 *    但 §17.2 的**判定算法本身**是纯逻辑，可以在用户态完整复现并实测。
 *
 * 算法（TLPI §17.2，与 POSIX 1003.1e draft 17 一致）：
 *   ① euid == 文件属主        → ACL_USER_OBJ 三元组，结束（mask 不参与）
 *   ② euid 命中某条 ACL_USER  → 该条 AND mask（若有 mask）；结束
 *   ③ egid == 文件属组 或 补充组 ∩ {ACL_GROUP_OBJ, 命中的 ACL_GROUP}
 *      → 命中可能多条，取"交完 mask 后任一允许"（OR）；结束
 *   ④ 都没命中 → ACL_OTHER（mask 不参与）
 *
 * 关键点（书上的强调 + 本 demo 实测）：
 *   - 属主**只看** ACL_USER_OBJ：即使 named user 给了更大权限，属主也用不上；
 *   - mask 是 named user / named group / group obj 的**上限**，
 *     对 owner 与 other **无效**；
 *   - 组类多条命中是 **OR**（一条交完 mask 后允许即允许）。
 *
 * 编译： cc -Wall -Wextra -o c17_1_acl_algorithm c17_1_acl_algorithm.c
 * 取材： TLPI §17.2（判定算法）；POSIX 1003.1e draft 17 §23.4.24
 */
#include <stdio.h>
#include <stdlib.h>

enum { ACL_USER_OBJ, ACL_USER, ACL_GROUP_OBJ, ACL_GROUP, ACL_MASK, ACL_OTHER };
enum { P_READ = 4, P_WRITE = 2, P_EXEC = 1 };

typedef struct { int type, id, perms; } Ace;
typedef struct { Ace *entries; int n; int has_mask; } Acl;

static const Ace *find(const Acl *a, int type, int id)
{
    for (int i = 0; i < a->n; i++)
        if (a->entries[i].type == type && a->entries[i].id == id)
            return &a->entries[i];
    return NULL;
}

/* uid 的补充组由 supp[] 给出；owner_id / owning_gid 是文件 inode 的属主/属组 */
static int acl_check(const Acl *acl, int uid, const int *supp, int nsupp,
                     int owner_id, int owning_gid, int want, const char **why)
{
    const Ace *mask = acl->has_mask ? find(acl, ACL_MASK, 0) : NULL;
    int cap = mask ? mask->perms : ~0;

    /* ① 属主 */
    if (uid == owner_id) {
        const Ace *e = find(acl, ACL_USER_OBJ, 0);
        *why = "① euid==属主 → ACL_USER_OBJ（mask 不参与）";
        return (e->perms & want) == want ? 0 : -1;
    }
    /* ② named user */
    {
        const Ace *e = find(acl, ACL_USER, uid);
        if (e) {
            *why = acl->has_mask ? "② 命中 named user → 该条 AND mask"
                                 : "② 命中 named user（无 mask）";
            return ((e->perms & cap) & want) == want ? 0 : -1;
        }
    }
    /* ③ 组类：group obj（egid 或补充组 == owning_gid）+ named group（OR） */
    {
        int in_owning_group = 0;
        for (int i = 0; i < nsupp; i++)
            if (supp[i] == owning_gid) in_owning_group = 1;
        int allow = 0, matched = 0;
        for (int i = 0; i < acl->n; i++) {
            const Ace *e = &acl->entries[i];
            int hit = 0;
            if (e->type == ACL_GROUP_OBJ)
                hit = in_owning_group;
            else if (e->type == ACL_GROUP) {
                for (int g = 0; g < nsupp; g++)
                    if (supp[g] == e->id) hit = 1;
            }
            if (hit) {
                matched = 1;
                if (((e->perms & cap) & want) == want) allow = 1;
            }
        }
        if (matched) {
            *why = acl->has_mask ? "③④ 组类命中 → 该条 AND mask（多条 OR）"
                                 : "③④ 组类命中（无 mask，多条 OR）";
            return allow ? 0 : -1;
        }
    }
    /* ④ other */
    {
        const Ace *e = find(acl, ACL_OTHER, 0);
        *why = "④ 未命中 → ACL_OTHER（mask 不参与）";
        return (e->perms & want) == want ? 0 : -1;
    }
}

int main(void)
{
    /* 典型 ACL（对应 setfacl 文本形式）：
     *   user::rw-  user:1001:r--  group::r--  group:1002:-wx  mask::rw-  other::r--
     * 文件属主=1000，属组=1003
     */
    Ace entries[] = {
        { ACL_USER_OBJ,  0,    P_READ | P_WRITE },
        { ACL_USER,      1001, P_READ },
        { ACL_GROUP_OBJ, 0,    P_READ },
        { ACL_GROUP,     1002, P_WRITE | P_EXEC },
        { ACL_MASK,      0,    P_READ | P_WRITE },
        { ACL_OTHER,     0,    P_READ },
    };
    Acl acl = { entries, 6, 1 };
    const int OWNER = 1000, OWNING_GID = 1003;

    int g_1002[] = { 1002 };
    int g_1003[] = { 1003 };
    int g_none[] = { 0 };
    struct { const char *who; int uid, *g, ng, want; } cases[] = {
        { "uid=1000 属主         求 r", 1000, g_none, 0, P_READ },
        { "uid=1000 属主         求 w", 1000, g_none, 0, P_WRITE },
        { "uid=1001 named user   求 r", 1001, g_none, 0, P_READ },
        { "uid=1001 named user   求 w", 1001, g_none, 0, P_WRITE },
        { "uid=1002 ∈组1002      求 w", 1002, g_1002, 1, P_WRITE },
        { "uid=1002 ∈组1002      求 x", 1002, g_1002, 1, P_EXEC },
        { "uid=2000 ∈组1003      求 r", 2000, g_1003, 1, P_READ },
        { "uid=2000 路人         求 r", 2000, g_none, 0, P_READ },
        { "uid=2000 路人         求 w", 2000, g_none, 0, P_WRITE },
    };

    printf("== ACL: user::rw- user:1001:r-- group::r-- group:1002:-wx mask::rw- other::r-- ==\n");
    printf("== 文件属主=1000，属组=1003 ==\n\n");
    for (size_t i = 0; i < sizeof cases / sizeof cases[0]; i++) {
        const char *why = "";
        int r = acl_check(&acl, cases[i].uid, cases[i].g, cases[i].ng,
                          OWNER, OWNING_GID, cases[i].want, &why);
        printf("  %-24s → %-4s  [%s]\n", cases[i].who, r == 0 ? "允许" : "拒绝", why);
    }

    printf("\n== mask 清零：named/组全灭，属主与 other 不受影响 ==\n");
    for (int i = 0; i < acl.n; i++)
        if (entries[i].type == ACL_MASK) entries[i].perms = 0;
    const char *why = "";
    printf("  属主 1000 求 r  → %s（mask 对 owner 无效，仍走 rw-）\n",
           acl_check(&acl, 1000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  named 1001 求 r → %s（mask 生效，r-- 交 0 = 0）\n",
           acl_check(&acl, 1001, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  路人 2000 求 r  → %s（mask 对 other 无效，仍走 r--）\n",
           acl_check(&acl, 2000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    return EXIT_SUCCESS;
}

本节讲什么

传统 9 位权限只能表达「属主 / 属组 / 其他」三个身份的 rwx。ACL（访问控制列表）在这之上加任意多条的"指定用户/指定组"条目——精确到人的授权。本节钉住：

  1. ACL 条目的 6 种 tag 与"最小 ACL / 扩展 ACL"的分界；
  2. ACL 存在哪：system.posix_acl_access xattr（Ch16 的直接应用）；
  3. 为什么 ls -l 会多出一个 +。

要点：6 种条目

tag 文本形式 数量 语义
ACL_USER_OBJ user::rwx 恰 1 文件属主的权限
ACL_USER user:1001:r-- 0..n 指定 uid 的权限
ACL_GROUP_OBJ group::r-x 恰 1 文件属组的权限
ACL_GROUP group:devs:rwx 0..n 指定 gid 的权限
ACL_MASK mask::rwx 0/1 named/group 类的上限（17.4）
ACL_OTHER other::r-- 恰 1 其他人的权限

最小 ACL = USER_OBJ + GROUP_OBJ + OTHER 三条（正好映射传统 9 位权限）。 扩展 ACL = 上述 + 任意 named 条目；一旦有 named 条目，mask 变成必选（内核自动生成/维护）。

ACL 落在哪

system.posix_acl_access   ← 文件的 ACL（Ch16 xattr 的直接应用）
system.posix_acl_default  ← 目录的 default ACL（17.6）

权限语义：与文件本身绑定（读写 ACL = 读写文件的属主/特权）。

ls -l 的 +：扩展 ACL 存在时 ls -l 尾部显示 +，且 group 位显示的实际是 mask（不是 GROUP_OBJ）——这是最经典的混淆点（17.4 展开）。

平台现实（诚实标注）

平台 POSIX draft ACL 备注
Linux + ext4/xfs ✅ 完整 本章的"真机"；getfacl/setfacl 可用
Linux + tmpfs ✅ 支持（实测平台候选）
macOS (APFS) ❌ draft 无 只有 NFSv4 风格 ACL，API/语义完全不同
本机复现方式 — 判定算法纯逻辑复现（c17_1，实测通过）

HFT / 嵌入式关联

场景 本节机制怎么兑现

        return (e->perms & want) == want ? 0 : -1;
    }
}

int main(void)
{
    /* 典型 ACL（对应 setfacl 文本形式）：
     *   user::rw-  user:1001:r--  group::r--  group:1002:-wx  mask::rw-  other::r--
     * 文件属主=1000，属组=1003
     */
    Ace entries[] = {
        { ACL_USER_OBJ,  0,    P_READ | P_WRITE },
        { ACL_USER,      1001, P_READ },
        { ACL_GROUP_OBJ, 0,    P_READ },
        { ACL_GROUP,     1002, P_WRITE | P_EXEC },
        { ACL_MASK,      0,    P_READ | P_WRITE },
        { ACL_OTHER,     0,    P_READ },
    };
    Acl acl = { entries, 6, 1 };
    const int OWNER = 1000, OWNING_GID = 1003;

    int g_1002[] = { 1002 };
    int g_1003[] = { 1003 };
    int g_none[] = { 0 };
    struct { const char *who; int uid, *g, ng, want; } cases[] = {
        { "uid=1000 属主         求 r", 1000, g_none, 0, P_READ },
        { "uid=1000 属主         求 w", 1000, g_none, 0, P_WRITE },
        { "uid=1001 named user   求 r", 1001, g_none, 0, P_READ },
        { "uid=1001 named user   求 w", 1001, g_none, 0, P_WRITE },
        { "uid=1002 ∈组1002      求 w", 1002, g_1002, 1, P_WRITE },
        { "uid=1002 ∈组1002      求 x", 1002, g_1002, 1, P_EXEC },
        { "uid=2000 ∈组1003      求 r", 2000, g_1003, 1, P_READ },
        { "uid=2000 路人         求 r", 2000, g_none, 0, P_READ },
        { "uid=2000 路人         求 w", 2000, g_none, 0, P_WRITE },
    };

    printf("== ACL: user::rw- user:1001:r-- group::r-- group:1002:-wx mask::rw- other::r-- ==\n");
    printf("== 文件属主=1000，属组=1003 ==\n\n");
    for (size_t i = 0; i < sizeof cases / sizeof cases[0]; i++) {
        const char *why = "";
        int r = acl_check(&acl, cases[i].uid, cases[i].g, cases[i].ng,
                          OWNER, OWNING_GID, cases[i].want, &why);
        printf("  %-24s → %-4s  [%s]\n", cases[i].who, r == 0 ? "允许" : "拒绝", why);
    }

    printf("\n== mask 清零：named/组全灭，属主与 other 不受影响 ==\n");
    for (int i = 0; i < acl.n; i++)
        if (entries[i].type == ACL_MASK) entries[i].perms = 0;
    const char *why = "";
    printf("  属主 1000 求 r  → %s（mask 对 owner 无效，仍走 rw-）\n",
           acl_check(&acl, 1000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  named 1001 求 r → %s（mask 生效，r-- 交 0 = 0）\n",
           acl_check(&acl, 1001, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  路人 2000 求 r  → %s（mask 对 other 无效，仍走 r--）\n",
           acl_check(&acl, 2000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    return EXIT_SUCCESS;
}

算法本体（switch 语义，与 15.4.3 一脉相承）

① euid == 文件属主            → ACL_USER_OBJ，结束（mask 不参与）
② euid 命中某条 ACL_USER      → 该条 AND mask（若有），结束
③ egid==属组 或 补充组命中
   ACL_GROUP_OBJ / ACL_GROUP  → 命中可能多条（OR：交完 mask 后任一允许即允许），结束
④ 都没命中                    → ACL_OTHER，结束（mask 不参与）

三条铁律（每条都有实测输出对应）： 1. 属主只看 USER_OBJ——named user 给属主更大权限也没用； 2. mask 只约束②③两类条目，对 owner / other 无效； 3. 组类命中是 OR：多条命中时一条允许就允许（15.4.3 的"属组不叠加"在 ACL 里反而变成 OR）。

实测（c17_1，全部与书上的算法一致）

== ACL: user::rw- user:1001:r-- group::r-- group:1002:-wx mask::rw- other::r-- ==
== 文件属主=1000，属组=1003 ==

  uid=1000 属主         求 r → 允许  [① euid==属主 → ACL_USER_OBJ（mask 不参与）]
  uid=1001 named user   求 r → 允许  [② 命中 named user → 该条 AND mask]
  uid=1001 named user   求 w → 拒绝  [② 命中 named user → 该条 AND mask]
  uid=1002 ∈组1002      求 w → 允许  [③④ 组类命中 → 该条 AND mask（多条 OR）]
  uid=1002 ∈组1002      求 x → 拒绝  [③④ 组类命中 → 该条 AND mask（多条 OR）]
  uid=2000 ∈组1003      求 r → 允许  [③④ 组类命中 → 该条 AND mask（多条 OR）]
  uid=2000 路人         求 r → 允许  [④ 未命中 → ACL_OTHER（mask 不参与）]

== mask 清零：named/组全灭，属主与 other 不受影响 ==
  属主 1000 求 r  → 允许（mask 对 owner 无效，仍走 rw-）
  named 1001 求 r → 拒绝（mask 生效，r-- 交 0 = 0）
  路人 2000 求 r  → 允许（mask 对 other 无效，仍走 r--）

注意 uid=1002 求 x：named group 明明给了 -wx，但 mask 是 rw-——交完只剩 -w-，x 拒绝。这一单条就是 mask 的全部意义。

17.3

Long and Short Text Forms for ACLs ACL 的两种文本形式

本章：TLPI 第 17 章 — Access Control Lists 平台标注：getfacl/setfacl 文本协议属 Linux（libacl）生态；文本形式本身是规范内容，值得掌握 源码核验：man-pages 6.19 acl(5)（TEXT FORM 段）+ TLPI §17.3 - 上一节：17.2 判定算法 - 下一节：17.4 The ACL Mask Entry

两种形式

长形式（long，每条目一行带全名，getfacl 默认输出）：

user::rw-
user:1001:r--           #effective:r--
group::r--
group:1002:-wx          #effective:-w-
mask::rw-
other::r--

短形式（short，setfacl -m 的参数格式）：

u::rw-,u:1001:r--,g::r--,g:1002:-wx,m::rw-,o::r--

要点：

  • tag 可缩写（u/g/o/m），ACL_USER_OBJ 与 ACL_GROUP_OBJ 用 u:: / g::（双冒号表示"无 id"），named 条目 u:1001:r--（单冒号接 id/名字）；
  • id 与名字可互换（u:liming:r--），存储时统一为数字 id；
  • 长形式里的 #effective: 注释是 getfacl 替你算好的"该条 AND mask"——短形式没有它。

注释位（# 开头行）会被 setfacl 的解析器忽略——所以 getfacl 的输出可以直接回灌给 setfacl：

getfacl file > acl.txt        # 备份
setfacl --restore=acl.txt     # 恢复（或 setfacl -M acl.txt file）

这是 ACL 权限模型做版本化/迁移的标准手法（chmod 做不到——9 位权限的语义上下文丢了）。

常见坑

坑 正解
u:1001 与 u:: 混写 双冒号=USER_OBJ（无 id），单冒号=named（带 id）；setfacl 对此敏感
忽略 #effective 注释 它就是 mask 生效后的真实权限，评审 ACL 文件先看它
拿 chmod 恢复权限备份 chmod 只覆盖 USER_OBJ/GROUP_OBJ/OTHER 三条；named 条目与 mask 会原样残留

场景 本节机制怎么兑现
权限即代码 getfacl -R 快照进 git，部署时 --restore——与 IaC 同一哲学
事故复盘 #effective 列是"为什么他进不去"的一手证据（named 条目常被 mask 削）

17.4

The ACL_MASK Entry and the ACL Group Class mask 与组类

本章：TLPI 第 17 章 — Access Control Lists 实测：c17_1 的"mask 清零"实验（见 17.2） 源码核验：man-pages 6.19 acl(5) + TLPI §17.4 - 上一节：17.3 文本形式 - 下一节：17.5 getfacl/setfacl

本节讲什么

mask 是 ACL 里唯一的"集合级上限"：它单独一条条目，却约束所有 named user / named group / GROUP_OBJ 条目（组类），不约束 USER_OBJ 与 OTHER。本节钉住三个事实：

  1. 为什么需要 mask：chmod 与 ACL 的兼容协议；
  2. ls -l 的 group 位显示的是 mask；
  3. mask 何时自动生成、何时被 chmod 覆盖。

1. mask 的设计动机：给 chmod 留一扇门

设想没有 mask：chmod g-w file 应该干什么？named group 条目满天飞，"组权限"根本不是一条数据。POSIX 的解法：

chmod 的 group 位 = mask。chmod 改"组权限"，实际是改 mask——一刀切住所有组类条目的上限。

  • USER_OBJ / OTHER：chmod 直接改，语义清晰；
  • 组类（GROUP_OBJ + named）：chmod 只动 mask。GROUP_OBJ 本身保持不动，被 mask 压着。

2. mask 的生效规则（实测见 17.2）

条目 受 mask 约束
ACL_USER_OBJ ❌
ACL_USER ✅
ACL_GROUP_OBJ ✅
ACL_GROUP ✅
ACL_OTHER ❌

#effective: 注释 = 条目 AND mask 后的真实权限。

3. 什么时候有 mask

  • 无 named 条目：mask 不存在（最小 ACL），GROUP_OBJ 即组权限；
  • 添加第一条 named 条目时：内核自动生成 mask = 当时所有组类条目的并集（所以默认不裁权限）；
  • 此后 setfacl -m m::rwx 手动设；setfacl -n 禁止自动重算；
  • chmod 组位直接覆盖 mask——这就是"加了 named 条目后 chmod 会神秘地砍权限"的原因。

4. ls -l 的 + 与 group 位

-rw-rw----+ 1 user devs  1024 ...   ← 有扩展 ACL：尾缀 +，group 位 = mask

ls -l 的 group 三位在有 mask 时显示 mask 而非 GROUP_OBJ——因为对多数"非属主非 named"的进程，mask 才是生效路径的参与者。看到 + 就该 getfacl 而不是猜。

常见坑

坑 正解
chmod 后 named 用户权限"消失" chmod 组位覆盖了 mask；用 setfacl -m m::rw 恢复，或 chmod 后重审 ACL
以为 mask 是"每条目自己的" 全 ACL 一条 mask，组类共享
用 cp 复制文件后 ACL 不见了 cp 默认不带 ACL（--preserve=xattr/cp -p 行为按平台）；迁移用 getfacl/restore

HFT / 嵌入式关联

场景 本节机制怎么兑现
权限收敛审计 审 mask 而不是逐条 named——mask 是组类的最大暴露面
变更窗口 "chmod 收紧"会同时压住所有 named 授权——灰度操作前先 getfacl 快照

17.6

Default ACLs and File Creation 默认 ACL 与文件创建

本章：TLPI 第 17 章 — Access Control Lists 平台标注：Linux 专有实现（system.posix_acl_default xattr）；语义按 POSIX draft 讲解 源码核验：man-pages 6.19 acl(5)（DEFAULT ACL 段）+ TLPI §17.6 - 上一节：17.5 getfacl/setfacl - 下一节：17.7 Implementation Limits

本节讲什么

default ACL 只属于目录：它不影响目录自身的访问控制（那是 access ACL），而是新文件的权限模板。本节钉住继承规则——它是"SGID 目录"方案的 ACL 加强版。

继承规则

目录带 default ACL 时，在其中新建对象：

新对象 access ACL default ACL
文件 继承 default ACL（权限 = 模板 AND open 的 mode 参数） 无
目录 继承（同上） 同时继承目录的 default ACL（递归传播）

关键细节：

  1. AND 语义：default 模板是上限，open(path, O_CREAT, 0644) 的 0644 与模板逐位相与——模板给 rwx、mode 给 644 → 得 644。mode 参数仍是"第一道裁剪"；
  2. umask 在有 default ACL 时被忽略——这是书上的重点：默认 umask 022 会悄悄砍权限，default ACL 让这个坑消失（对需要"创建即组可写"的共享目录是天大的好事）；
  3. mask 自动生成（继承后取组类并集）。

与 15.3.1 新文件属组规则的关系

机制 作用
SYSV 新文件属组 = 进程 egid 谁创建，谁负责
目录 SGID 新文件属组强制 = 目录属组
default ACL 不止属组：连 named 条目、mask、other 一起做成模板

三层是叠加关系；default ACL 是表达能力最强的一层。

常见坑

坑 正解
以为 default ACL 会改目录自身的权限 不会；它只管"未来新建的对象"。目录自己的控制靠 access ACL
open mode 给 0000 还想靠 default ACL 救 AND 语义下 0000 AND anything = 0000；应用层 mode 必须给足
期望 default ACL 被 cp 继承 cp 创建的新文件走 open mode；只有内核分配 inode 的"新建"才继承
忘了给已有目录补 default 只对"设置之后"的新建生效；存量文件要 setfacl -m 批量补

HFT / 嵌入式关联

场景 本节机制怎么兑现
共享落盘目录 default ACL g:traders:rw- + 忽略 umask → 多账户写入即组可读，监控进程免 root 读
数据采集目录 子目录自动继承 default（递归传播），树型权限零维护
归档分层 归档目录 default 收紧到 other::---，新文件即私有

ACL Implementation Limits 实现限制

本章：TLPI 第 17 章 — Access Control Lists 平台标注：Linux 专有实现；数字以 kernel/ext4 为准 源码核验：man-pages 6.19 acl(5)（实现限制段）+ Linux v6.6 fs/ext4/xattr.c - 上一节：17.6 Default ACLs - 下一节：17.8 The ACL API

要点

限制 值 出处
named 条目数 ≥ 32（POSIX 最低保证）；ext4 实际上限远高于此（受 xattr 空间约束） acl(5)

三条工程推论：

  1. 条目数不是问题（32 条的下限对"人组授权"够用，超量说明建模错了——应该用组而不是逐人 named 条目）；
  2. FS 支持面才是问题：部署脚本要探测（setfacl 试一下 + getfacl 回读验证），squashfs 只读镜像上的 ACL 是幻觉；
  3. 备份必须带 ACL：tar 默认不带（需 --acls），rsync 需 --acls；漏掉的恢复 = 权限模型蒸发。

/* Supplementary program for Chapter 17 */

/* acl_update.c

   Perform updates on the access control lists (ACLs) of files named in the
   command line.  This program provides a subset of the functionality of the
   setfacl(1) command. For usage, see usageError() below.

   This program is Linux-specific. ACLs are supported since Linux 2.6.
   To build this program, you must have the ACL library (libacl) installed
   on your system.
*/
#include <sys/acl.h>
#include <stdbool.h>
#include "ugid_functions.h"
#include "tlpi_hdr.h"

#define MAX_ENTRIES 10000   /* Maximum entries that we can handle in an ACL */

struct AccessControlEntry {     /* Represent a single ACL entry */
    acl_tag_t   tag;            /* Tag type */
    id_t        qual;           /* Optional tag qualifier (UID or GID) */
    int         perms;          /* Permissions bit mask */
};

enum Operation {
    unspecifiedOp, modifyEntries, removeEntries, removeDefaultACL,
    checkValidACL
};

static void
usageError(char *progName, char *msg, bool shortUsage)
{
    if (msg != NULL)
        fprintf(stderr, "%s\n", msg);

    if (shortUsage) {
        fprintf(stderr, "Type '%s --help' for usage information\n", progName);
        exit(EXIT_FAILURE);
    }

    fprintf(stderr, "Usage: %s -m acl [-d] [-n] file...\n", progName);
    fprintf(stderr, "   or: %s -x acl [-d] [-n] file...\n", progName);
    fprintf(stderr, "   or: %s -k file...\n\n", progName);
    fprintf(stderr, "   or: %s -V acl\n\n", progName);
#define fpe(msg) fprintf(stderr, "      " msg);
    fpe("-m  Modify/create ACL entries\n");
    fpe("-x  Remove ACL entries\n");
    fpe("-k  Remove default ACL\n\n");
    fpe("-V  Check validity of 'acl'\n\n");
    fpe("'acl' consists of one or more comma-separated entries of the form:\n");
    fpe("\n            tag:[qualifier][:[perms]]\n\n");
    fpe("      'perms' are specified only for -m\n\n");
    fpe("-d  Apply operation to default ACL\n");
    fpe("-n  Don't recalculate mask entry\n");
    fpe("      NOTE: if you specify this option and have specified\n");
    fpe("      the -m option, then you may encounter errors if the\n");
    fpe("      file does not already have a mask entry.\n");
    exit(EXIT_FAILURE);
}

/* Parse an ACL entry specification (the null-terminated string
   provided in 'entryStr') of the form:

        tag:[qualifier][:[permissions]]

   returning the parsed information in the structure 'ace'.

   If 'permsReqd' is set, then the ACL entries must contain
   permission specifications (i.e., a colon followed by at least
   one of [-rwx], otherwise they must not.

   Return true if the specification parsed okay, or false otherwise. */

static bool
parseEntrySpec(char *entryStr, struct AccessControlEntry *ace,
               bool permsReqd)
{
    char *colon1 = strchr(entryStr, ':');
    if (colon1 == NULL) {
        fprintf(stderr, "Missing initial colon in ACL entry: %s\n", entryStr);
        return false;
    }

    bool hasQual = *(colon1 + 1) != '\0' && *(colon1 + 1) != ':';

    *colon1 = '\0';     /* Add terminator to tag type */

    char *colon2 = strchr(colon1 + 1, ':');
    bool hasPerms = colon2 != NULL && *(colon2 + 1) != '\0';

    if (hasPerms && !permsReqd) {
        fprintf(stderr, "Cannot specify permissions here\n");
        return false;
    }

    if (!hasPerms && permsReqd) {
        fprintf(stderr, "Must specify permissions\n");
        return false;
    }

    /* Determine tag type, depending on tag string and presence
       of qualifier after the first ':' */

    if (strcmp(entryStr, "u") == 0 || strcmp(entryStr, "user") == 0)
        ace->tag = hasQual ? ACL_USER : ACL_USER_OBJ;
    else if (strcmp(entryStr, "g") == 0 || strcmp(entryStr, "group") == 0)
        ace->tag = hasQual ? ACL_GROUP : ACL_GROUP_OBJ;
    else if (strcmp(entryStr, "o") == 0 || strcmp(entryStr, "other") == 0)
        ace->tag = ACL_OTHER;
    else if (strcmp(entryStr, "m") == 0 || strcmp(entryStr, "mask") == 0)
        ace->tag = ACL_MASK;
    else {
        fprintf(stderr, "Bad tag: %s\n", entryStr);
        return false;
    }

    /* For ACL_USER and ACL_GROUP tags, extract a UID / GID from qualifier */

    if (colon2 != NULL)
        *colon2 = '\0';         /* Add terminator to qualifier */

    ace->qual = 0;

    if (ace->tag == ACL_USER) {
        ace->qual = userIdFromName(colon1 + 1);
        if ((int) ace->qual == -1) {
            fprintf(stderr, "Bad user ID: %s\n", colon1 + 1);
            return false;
        }
    } else if (ace->tag == ACL_GROUP) {
        ace->qual = groupIdFromName(colon1 + 1);
        if ((int) ace->qual == -1) {
            fprintf(stderr, "Bad group ID: %s\n", colon1 + 1);
            return false;
        }
    }

    /* If a permissions string was present, return it as a bit mask */

    if (hasPerms) {
        ace->perms = 0;

        /* We're not too thorough here -- we don't check for multiple
           instances of [-rwx], or check if the permissions string is
           longer than three characters... */

        for (char *p = colon2 + 1; *p != '\0'; p++) {
            if (*p == 'r')
                ace->perms |= ACL_READ;
            else if (*p == 'w')
                ace->perms |= ACL_WRITE;
            else if (*p == 'x')
                ace->perms |= ACL_EXECUTE;
            else if (*p != '-') {
                fprintf(stderr, "Bad character in permissions "
                        "string: %c\n", *p);
                return false;
            }
        }
    }

    return true;
}

/* Parse a text form ACL, returning information about the entries in
   'aclist'. On success, return the number of ACL entries found; on
   error return -1. */

static int
parseACL(char *aclStr, struct AccessControlEntry aclist[],
        bool permsReqd)
{
    char *comma;
    int n = 0;

    for (char *nextEntry = aclStr; ; nextEntry = comma + 1) {

        if (n >= MAX_ENTRIES) {
            fprintf(stderr, "Too many entries in ACL\n");
            return -1;
        }

        comma = strchr(nextEntry, ',');
        if (comma != NULL)
            *comma = '\0';

        if (!parseEntrySpec(nextEntry, &aclist[n], permsReqd))
            return -1;

        if (comma == NULL)              /* This was the last entry */
            break;

        n++;
    }

    return n + 1;
}

/* Find the ACL entry in 'acl' corresponding to the tag type and
   qualifier in 'tag' and 'qual'. Return the matching entry, or NULL
   if no entry was found. */

static acl_entry_t
findEntry(acl_t acl, acl_tag_t tag, id_t qual)
{
    acl_entry_t entry;
    acl_tag_t entryTag;

    for (int ent = ACL_FIRST_ENTRY; ; ent = ACL_NEXT_ENTRY) {
        int s = acl_get_entry(acl, ent, &entry);
        if (s == -1)
            errExit("acl_get_entry");

        if (s == 0)
            return NULL;

        if (acl_get_tag_type(entry, &entryTag) == -1)
            errExit("acl_get_tag_type");

        if (tag == entryTag) {
            if (tag == ACL_USER) {
                uid_t *uidp = acl_get_qualifier(entry);
                if (uidp == NULL)
                    errExit("acl_get_qualifier");

                if (qual == *uidp) {
                    if (acl_free(uidp) == -1)
                        errExit("acl_free");
                    return entry;
                } else {
                    if (acl_free(uidp) == -1)
                        errExit("acl_free");
                }

            } else if (tag == ACL_GROUP) {
                gid_t *gidp = acl_get_qualifier(entry);
                if (gidp == NULL)
                    errExit("acl_get_qualifier");

                if (qual == *gidp) {
                    if (acl_free(gidp) == -1)
                        errExit("acl_free");
                    return entry;
                } else {
                    if (acl_free(gidp) == -1)
                        errExit("acl_free");
                }

            } else {
                return entry;
            }
        }
    }
}

/* Set the permissions in 'perms' in the ACL entry 'entry' */

static void
setPerms(acl_entry_t entry, int perms)
{
    acl_permset_t permset;

    if (acl_get_permset(entry, &permset) == -1)
        errExit("acl_get_permset");

    if (acl_clear_perms(permset) == -1)
        errExit("acl_clear_perms");

    if (perms & ACL_READ)
        if (acl_add_perm(permset, ACL_READ) == -1)
            errExit("acl_add_perm");
    if (perms & ACL_WRITE)
        if (acl_add_perm(permset, ACL_WRITE) == -1)
            errExit("acl_add_perm");
    if (perms & ACL_EXECUTE)
        if (acl_add_perm(permset, ACL_EXECUTE) == -1)
            errExit("acl_add_perm");

    if (acl_set_permset(entry, permset) == -1)
        errExit("acl_set_permset");
}

/* Modify or remove entries in the default or in the access ACL of 'file. */

static void
updateACL(char *file, bool useDefaultACL,
          int numEntries, struct AccessControlEntry *aclist,
          enum Operation operation, bool recalcMask)
{
    acl_type_t type = useDefaultACL ? ACL_TYPE_DEFAULT : ACL_TYPE_ACCESS;

    acl_t acl = acl_get_file(file, type);
    if (acl == NULL)
        errExit("acl_get_file");

    /* Apply each of the entries in 'aclist' to the current file */

    for (int en = 0; en < numEntries; en++) {
        acl_entry_t entry = findEntry(acl, aclist[en].tag, aclist[en].qual);

        if (operation == removeEntries) {
            if (entry != NULL)
                if (acl_delete_entry(acl, entry) == -1)
                    errExit("acl_delete_entry");

        } else {        /* modifyEntries */

            if (entry == NULL) {

                /* Entry didn't exist in ACL -- create a new
                   entry with required tag and qualifier */

                if (acl_create_entry(&acl, &entry) == -1)
                    errExit("acl_create_entry");
                if (acl_set_tag_type(entry, aclist[en].tag) == -1)
                    errExit("acl_set_tag_type");
                if (aclist[en].tag == ACL_USER || aclist[en].tag == ACL_GROUP)
                    if (acl_set_qualifier(entry, &aclist[en].qual) == -1)
                        errExit("acl_set_qualifier");
            }

            setPerms(entry, aclist[en].perms);
        }

        /* Recalculate the mask entry if requested */

        if (recalcMask)
            if (acl_calc_mask(&acl) == -1)
                errExit("acl_calc_mask");

        /* Update the file ACL */

        if (acl_valid(acl) == -1)
            errExit("acl_valid");

        if (acl_set_file(file, type, acl) == -1)
            errExit("acl_set_file");
    }

    if (acl_free(acl) == -1)
        errExit("acl_free");
}

int
main(int argc, char *argv[])
{
    if (argc < 2 || strcmp(argv[1], "--help") == 0)
        usageError(argv[0], NULL, false);

    /* Parse command-line options */

    enum Operation operation = unspecifiedOp;
    bool recalcMask = true;
    bool useDefaultACL = false;
    int optCnt = 0;
    char *aclSpec = NULL;
    int opt;

    while ((opt = getopt(argc, argv, "m:x:kdnV:")) != -1) {
        switch (opt) {
        case 'm':
            operation = modifyEntries;
            aclSpec = optarg;
            optCnt++;
            break;

        case 'x':
            operation = removeEntries;
            aclSpec = optarg;
            optCnt++;
            break;

        case 'k':
            operation = removeDefaultACL;
            optCnt++;
            break;

        case 'V':
            operation = checkValidACL;
            aclSpec = optarg;
            optCnt++;
            break;

        case 'd':
            useDefaultACL = true;
            break;

        case 'n':
            recalcMask = false;
            break;

        default:
            usageError(argv[0], "Bad option\n", true);
            break;
        }
    }

    if (optCnt != 1)
        usageError(argv[0], "Specify exactly one of -m, -x, -k, or -V\n", true);

    if ((operation == checkValidACL) && useDefaultACL)
        usageError(argv[0], "Can't specify -d with -V\n", true);

    struct AccessControlEntry aclist[MAX_ENTRIES];

    if (operation == checkValidACL) {
        if (parseACL(aclSpec, aclist, true) == -1) {
            fatal("Bad ACL entry specification");
        } else {
            printf("ACL is valid\n");
            exit(EXIT_SUCCESS);
        }
    }

    int numEntries = 0;
    if (operation == modifyEntries || operation == removeEntries) {
        numEntries = parseACL(aclSpec, aclist, operation == modifyEntries);
        if (numEntries == -1)
            usageError(argv[0], "Bad ACL specification\n", true);
    }

    /* Perform the operation on each file argument */

    for (int j = optind; j < argc; j++) {
        if (operation == removeDefaultACL) {
            if (acl_delete_def_file(argv[j]) == -1)
                errExit("acl_delete_def_file: %s", argv[j]);

        } else if (operation == modifyEntries || operation == removeEntries) {
            updateACL(argv[j], useDefaultACL, numEntries, aclist,
                      operation, recalcMask);

        } else {
            fatal("Bad logic!");
        }
    }

    exit(EXIT_SUCCESS);
}

17-1 显示指定 user/group 的 ACL 条目权限 ⛔→✅（Linux 复测）

任务：程序接 u|g 与用户/组（名字或数字）两个参数，显示该身份对应 ACL 条目的权限；若条目受 mask 约束，显示与 mask 求交后的有效权限；若无对应条目，说明其落入 other/组类判定。

实现（完整源码见下，libacl，Linux 编译 -lacl）。要点：

  • 名字→id 复用 Listing 8-1 的 userIdFromName 思路（passwd/group 查询，数字直接 atoi）；
  • 两遍遍历：先取 mask 条目，再找 ACL_USER/ACL_GROUP 目标条目；
  • 有效权限 = 条目 permset 逐位 AND mask，输出 raw → effective 两列；
  • 无专属条目时明确告知将落入哪条判定路径——这正是 §17.2 算法的应用。

完整源码（code/ex17_1_listacls.c）：

/* ex17_1_listacls.c — 习题 17-1：显示指定 user/group 对应 ACL 条目的权限
 *
 * 题面：程序接两个命令行参数：第一个是 u 或 g，第二个是用户/组（数字或
 * 名字均可）；显示该 user/group 对应 ACL 条目的权限；若该条目落入 mask
 * 作用范围，显示**与 mask 求交后的有效权限**；若无对应条目，说明其将
 * 落入 other/组类判定。
 *
 * ⚠️ 实测状态（诚实标注）：POSIX draft ACL 的完整实现（libacl，
 *    system.posix_acl_access xattr）是 **Linux 专有**；macOS 无对应
 *    常量/API，本文件**只在 Linux 编译运行**，Pi5(ext4) 是复测平台。
 *    判定算法的可运行版本见 c17_1_acl_algorithm.c（纯逻辑，全平台）。
 *
 * 编译（Linux/Pi5）： gcc -Wall -o ex17_1_listacls ex17_1_listacls.c -lacl
 * 取材： TLPI §17.1/§17.8 习题 17-1；结构沿用原书 acl_view.c 的成熟写法
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <pwd.h>
#include <grp.h>
#include <sys/acl.h>
#include <acl/libacl.h>         /* acl_get_perm_np() */
#include "tlpi_hdr.h"

static void perms_str(acl_permset_t ps, char *out)
{
    /* 逐位探测：r w x（draft 只定义这三个；ACL_MASK entry 另说） */
    out[0] = acl_get_perm_np(ps, ACL_READ)  == 1 ? 'r' : '-';
    out[1] = acl_get_perm_np(ps, ACL_WRITE) == 1 ? 'w' : '-';
    out[2] = acl_get_perm_np(ps, ACL_EXECUTE) == 1 ? 'x' : '-';
    out[3] = '\0';
}

static const char *tag_name(acl_tag_t t)
{
    switch (t) {
    case ACL_USER_OBJ:  return "user::";
    case ACL_USER:      return "user:";
    case ACL_GROUP_OBJ: return "group::";
    case ACL_GROUP:     return "group:";
    case ACL_MASK:      return "mask::";
    case ACL_OTHER:     return "other::";
    default:            return "?";
    }
}

int main(int argc, char *argv[])
{
    if (argc != 4 || argv[1][1] != '\0' ||
        (argv[1][0] != 'u' && argv[1][0] != 'g'))
        usageErr("%s u|g user-or-group file\n", argv[0]);

    const char *file = argv[3];
    int is_user = argv[1][0] == 'u';
    id_t want_id;

    if (is_user) {
        struct passwd *pw = getpwnam(argv[2]);
        want_id = pw ? pw->pw_uid : (uid_t) atoi(argv[2]);
    } else {
        struct group *gr = getgrnam(argv[2]);
        want_id = gr ? gr->gr_gid : (gid_t) atoi(argv[2]);
    }

    acl_t acl = acl_get_file(file, ACL_TYPE_ACCESS);
    if (acl == NULL)
        errExit("acl_get_file:%s（FS 需支持 POSIX draft ACL）", file);

    /* 第一遍：找 mask */
    acl_permset_t mask_ps = NULL;
    int have_mask = 0;
    acl_entry_t e;
    for (int st = acl_get_entry(acl, ACL_FIRST_ENTRY, &e);
         st == 1; st = acl_get_entry(acl, ACL_NEXT_ENTRY, &e)) {
        acl_tag_t t;
        acl_get_tag_type(e, &t);
        if (t == ACL_MASK) {
            acl_get_permset(e, &mask_ps);
            have_mask = 1;
        }
    }

    /* 第二遍：找目标条目 */
    int found = 0;
    for (int st = acl_get_entry(acl, ACL_FIRST_ENTRY, &e);
         st == 1; st = acl_get_entry(acl, ACL_NEXT_ENTRY, &e)) {
        acl_tag_t t;
        acl_get_tag_type(e, &t);
        id_t qual = 0;
        int hit = 0;
        if ((t == ACL_USER || t == ACL_GROUP) && is_user == (t == ACL_USER)) {
            qual = *(id_t *) acl_get_qualifier(e);
            hit = (qual == want_id);
        }
        if (hit) {
            acl_permset_t ps;
            acl_get_permset(e, &ps);
            char raw[8], m[8];
            perms_str(ps, raw);
            printf("%s%d 原始权限=%s", tag_name(t), (int) qual, raw);
            if (have_mask && mask_ps) {
                perms_str(mask_ps, m);
                char eff[4];
                for (int i = 0; i < 3; i++)
                    eff[i] = (raw[i] != '-' && m[i] != '-') ? raw[i] : '-';
                eff[3] = '\0';
                printf("  mask=%s  有效权限=%s", m, eff);
            }
            printf("\n");
            found = 1;
        }
    }

    if (!found)
        printf("没有 %s=%d 的专属条目：该身份经 §17.2 算法落入%s判定\n",
               is_user ? "uid" : "gid", (int) want_id,
               is_user ? "组类扫描或 other" : "组类扫描或 other");

    acl_free(acl);
    return EXIT_SUCCESS;
}

完整源码（code/c17_1_acl_algorithm.c）：

/* c17_1_acl_algorithm.c — Ch17 §17.2：ACL 权限判定算法的纯用户态实现
 *
 * ⚠️ POSIX draft ACL 的内核实现（libacl / system.* xattr）是 Linux 专有；
 *    但 §17.2 的**判定算法本身**是纯逻辑，可以在用户态完整复现并实测。
 *
 * 算法（TLPI §17.2，与 POSIX 1003.1e draft 17 一致）：
 *   ① euid == 文件属主        → ACL_USER_OBJ 三元组，结束（mask 不参与）
 *   ② euid 命中某条 ACL_USER  → 该条 AND mask（若有 mask）；结束
 *   ③ egid == 文件属组 或 补充组 ∩ {ACL_GROUP_OBJ, 命中的 ACL_GROUP}
 *      → 命中可能多条，取"交完 mask 后任一允许"（OR）；结束
 *   ④ 都没命中 → ACL_OTHER（mask 不参与）
 *
 * 关键点（书上的强调 + 本 demo 实测）：
 *   - 属主**只看** ACL_USER_OBJ：即使 named user 给了更大权限，属主也用不上；
 *   - mask 是 named user / named group / group obj 的**上限**，
 *     对 owner 与 other **无效**；
 *   - 组类多条命中是 **OR**（一条交完 mask 后允许即允许）。
 *
 * 编译： cc -Wall -Wextra -o c17_1_acl_algorithm c17_1_acl_algorithm.c
 * 取材： TLPI §17.2（判定算法）；POSIX 1003.1e draft 17 §23.4.24
 */
#include <stdio.h>
#include <stdlib.h>

enum { ACL_USER_OBJ, ACL_USER, ACL_GROUP_OBJ, ACL_GROUP, ACL_MASK, ACL_OTHER };
enum { P_READ = 4, P_WRITE = 2, P_EXEC = 1 };

typedef struct { int type, id, perms; } Ace;
typedef struct { Ace *entries; int n; int has_mask; } Acl;

static const Ace *find(const Acl *a, int type, int id)
{
    for (int i = 0; i < a->n; i++)
        if (a->entries[i].type == type && a->entries[i].id == id)
            return &a->entries[i];
    return NULL;
}

/* uid 的补充组由 supp[] 给出；owner_id / owning_gid 是文件 inode 的属主/属组 */
static int acl_check(const Acl *acl, int uid, const int *supp, int nsupp,
                     int owner_id, int owning_gid, int want, const char **why)
{
    const Ace *mask = acl->has_mask ? find(acl, ACL_MASK, 0) : NULL;
    int cap = mask ? mask->perms : ~0;

    /* ① 属主 */
    if (uid == owner_id) {
        const Ace *e = find(acl, ACL_USER_OBJ, 0);
        *why = "① euid==属主 → ACL_USER_OBJ（mask 不参与）";
        return (e->perms & want) == want ? 0 : -1;
    }
    /* ② named user */
    {
        const Ace *e = find(acl, ACL_USER, uid);
        if (e) {
            *why = acl->has_mask ? "② 命中 named user → 该条 AND mask"
                                 : "② 命中 named user（无 mask）";
            return ((e->perms & cap) & want) == want ? 0 : -1;
        }
    }
    /* ③ 组类：group obj（egid 或补充组 == owning_gid）+ named group（OR） */
    {
        int in_owning_group = 0;
        for (int i = 0; i < nsupp; i++)
            if (supp[i] == owning_gid) in_owning_group = 1;
        int allow = 0, matched = 0;
        for (int i = 0; i < acl->n; i++) {
            const Ace *e = &acl->entries[i];
            int hit = 0;
            if (e->type == ACL_GROUP_OBJ)
                hit = in_owning_group;
            else if (e->type == ACL_GROUP) {
                for (int g = 0; g < nsupp; g++)
                    if (supp[g] == e->id) hit = 1;
            }
            if (hit) {
                matched = 1;
                if (((e->perms & cap) & want) == want) allow = 1;
            }
        }
        if (matched) {
            *why = acl->has_mask ? "③④ 组类命中 → 该条 AND mask（多条 OR）"
                                 : "③④ 组类命中（无 mask，多条 OR）";
            return allow ? 0 : -1;
        }
    }
    /* ④ other */
    {
        const Ace *e = find(acl, ACL_OTHER, 0);
        *why = "④ 未命中 → ACL_OTHER（mask 不参与）";
        return (e->perms & want) == want ? 0 : -1;
    }
}

int main(void)
{
    /* 典型 ACL（对应 setfacl 文本形式）：
     *   user::rw-  user:1001:r--  group::r--  group:1002:-wx  mask::rw-  other::r--
     * 文件属主=1000，属组=1003
     */
    Ace entries[] = {
        { ACL_USER_OBJ,  0,    P_READ | P_WRITE },
        { ACL_USER,      1001, P_READ },
        { ACL_GROUP_OBJ, 0,    P_READ },
        { ACL_GROUP,     1002, P_WRITE | P_EXEC },
        { ACL_MASK,      0,    P_READ | P_WRITE },
        { ACL_OTHER,     0,    P_READ },
    };
    Acl acl = { entries, 6, 1 };
    const int OWNER = 1000, OWNING_GID = 1003;

    int g_1002[] = { 1002 };
    int g_1003[] = { 1003 };
    int g_none[] = { 0 };
    struct { const char *who; int uid, *g, ng, want; } cases[] = {
        { "uid=1000 属主         求 r", 1000, g_none, 0, P_READ },
        { "uid=1000 属主         求 w", 1000, g_none, 0, P_WRITE },
        { "uid=1001 named user   求 r", 1001, g_none, 0, P_READ },
        { "uid=1001 named user   求 w", 1001, g_none, 0, P_WRITE },
        { "uid=1002 ∈组1002      求 w", 1002, g_1002, 1, P_WRITE },
        { "uid=1002 ∈组1002      求 x", 1002, g_1002, 1, P_EXEC },
        { "uid=2000 ∈组1003      求 r", 2000, g_1003, 1, P_READ },
        { "uid=2000 路人         求 r", 2000, g_none, 0, P_READ },
        { "uid=2000 路人         求 w", 2000, g_none, 0, P_WRITE },
    };

    printf("== ACL: user::rw- user:1001:r-- group::r-- group:1002:-wx mask::rw- other::r-- ==\n");
    printf("== 文件属主=1000，属组=1003 ==\n\n");
    for (size_t i = 0; i < sizeof cases / sizeof cases[0]; i++) {
        const char *why = "";
        int r = acl_check(&acl, cases[i].uid, cases[i].g, cases[i].ng,
                          OWNER, OWNING_GID, cases[i].want, &why);
        printf("  %-24s → %-4s  [%s]\n", cases[i].who, r == 0 ? "允许" : "拒绝", why);
    }

    printf("\n== mask 清零：named/组全灭，属主与 other 不受影响 ==\n");
    for (int i = 0; i < acl.n; i++)
        if (entries[i].type == ACL_MASK) entries[i].perms = 0;
    const char *why = "";
    printf("  属主 1000 求 r  → %s（mask 对 owner 无效，仍走 rw-）\n",
           acl_check(&acl, 1000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  named 1001 求 r → %s（mask 生效，r-- 交 0 = 0）\n",
           acl_check(&acl, 1001, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    printf("  路人 2000 求 r  → %s（mask 对 other 无效，仍走 r--）\n",
           acl_check(&acl, 2000, g_none, 0, OWNER, OWNING_GID, P_READ, &why) == 0 ? "允许" : "拒绝");
    return EXIT_SUCCESS;
}
