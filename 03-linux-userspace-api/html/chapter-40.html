章节目标

utmp/wtmp/btmp；struct utmp 与 ut_type；遍历 API；更新由谁做；systemd/容器注意点；简易 who。

  2. wtmp 含大量 DEAD/BOOT 历史
  3. 用 API，别当文本读
  4. 并发写须锁
  5. ut_user 空 ≠ 一定无终端槽

  1. 简易 who
  2. utmpname(wtmp) 扫历史
  3. 找 BOOT_TIME
  4. （选）ssh 登录前后 utmp 变化

代码示例

#include <stdio.h>
#include <utmp.h>
#include <utmpx.h>
#include <string.h>
#include <unistd.h>
#include <time.h>

/* Ch40 登录记账 — utmp/wtmp + last 命令原理。
 * 演示读取 /var/run/utmp 获取当前登录用户。
 * 编译: gcc -o ch40_demo ch40_demo.c */

int main(void) {
    /* utmpx: 读取当前登录记录 */
    setutxent();  /* 打开/重置 utmp 文件 */

    printf("Currently logged in users:\n");
    printf("%-12s %-8s %-16s %-24s\n",
           "USER", "TTY", "HOST", "LOGIN TIME");

    struct utmpx *entry;
    while ((entry = getutxent()) != NULL) {
        /* 只显示用户进程登录记录 */
        if (entry->ut_type == USER_PROCESS) {
            char timebuf[32];
            struct tm *tm = localtime(&entry->ut_tv.tv_sec);
            strftime(timebuf, sizeof(timebuf), "%Y-%m-%d %H:%M:%S", tm);

            printf("%-12s %-8s %-16s %-24s\n",
                   entry->ut_user,
                   entry->ut_line,
                   entry->ut_host,
                   timebuf);
        }
    }

    endutxent();

    /* 写入 utmp 记录 (通常由 login 程序做) */
    printf("\nNote: utmp is normally written by login(1) and login(1)\n");
    printf("wtmp (/var/log/wtmp) keeps historical login records\n");
    printf("Use 'last' command to read wtmp\n");
    return 0;
}

40.2

The utmpx API

本章：TLPI 第 40 章

本节讲什么

utmpx API 用于读写登录记账文件 /var/run/utmp（当前登录）和 /var/log/wtmp（历史登录）。记录用户登录、注销和系统启动/关机事件。

要点

核心 API

#include <utmpx.h>
void setutxent(void);            /* 重置到文件开头 */
struct utmpx *getutxent(void);     /* 读取下一条记录 */
struct utmpx *getutxid(const struct utmpx *ut); /* 按 ID 查找 */
struct utmpx *getutxline(const struct utmpx *ut); /* 按行号查找 */
int pututxline(const struct utmpx *ut); /* 写入记录 */
void endutxent(void);             /* 关闭文件 */
int utmpxname(const char *file);  /* 指定替代文件 */

struct utmpx

字段 说明
ut_type 记录类型 (BOOT_TIME/USER_PROCESS/DEAD_PROCESS 等)
ut_pid 登录进程 PID
ut_line 设备名 (tty/pts)
ut_id 行 ID
ut_user 用户名
ut_tv 时间戳
ut_host 远程主机
ut_addr 远程 IP

记录类型

类型 说明
BOOT_TIME 系统启动
USER_PROCESS 用户登录
DEAD_PROCESS 进程退出（注销）
INIT_PROCESS init 生成的进程
LOGIN_PROCESS getty 等待登录

查看命令

who     # 读 /var/run/utmp（当前登录）
last    # 读 /var/log/wtmp（历史登录）
lastlog # 读 /var/log/lastlog（最后登录时间）

HFT / 嵌入式关联

  • 嵌入式系统登录审计
  • HFT 运维监控登录用户
  • 安全合规：记录所有登录/注销事件

代码示例

#include <utmpx.h>
#include <stdio.h>

    setutxent();

代码自测

utmp 和 wtmp 有什么区别？

utmp（/var/run/utmp）记录当前登录的会话，每次登录覆盖旧记录，注销时清除。wtmp（/var/log/wtmp）记录历史登录和注销事件，只追加不覆盖，用于审计。who 读 utmp，last 读 wtmp。

要点

路线 要点
嵌入式 setutent/getutent/endutent 遍历，utmpname 切 wtmp 读历史，getutid/getutline 条件查找
HFT 登录审计脚本遍历 getutent 统计活跃会话，非热路径无性能压力

代码示例

#include <stdio.h>
#include <utmpx.h>
#include <time.h>
/* utmpx 读取操作: getutxent/getutxid/getutxline */
int main(void) {
    /* setutxent: 打开/重置 utmp 文件 */
    setutxent();

    printf("=== All USER_PROCESS entries ===\n");
    struct utmpx *entry;
    while ((entry = getutxent()) != NULL) {
        if (entry->ut_type != USER_PROCESS)
            continue;

        char timebuf[32];
        struct tm *tm = localtime((time_t *)&entry->ut_tv.tv_sec);
        strftime(timebuf, sizeof(timebuf), "%H:%M:%S", tm);

        printf("user=%-8s tty=%-8s host=%-16s login=%s\n",
               entry->ut_user, entry->ut_line,
               entry->ut_host[0] ? entry->ut_host : "(local)",
               timebuf);
    }
    endutxent();

    /* getutxline: 按终端查找 */
    printf("\n=== getutxline(\"tty1\") ===\n");
    setutxent();
    struct utmpx query;
    query.ut_type = LOGIN_PROCESS;
    strncpy(query.ut_line, "tty1", sizeof(query.ut_line));
    entry = getutxline(&query);
    if (entry)
        printf("Found: user=%s\n", entry->ut_user);
    else
        printf("Not found\n");
    endutxent();

    printf("\nwtmp (/var/log/wtmp) keeps historical records\n");
    printf("Use 'last' command to read wtmp\n");
    return 0;
}

40.7

The lastlog File

本章：TLPI 第 40 章

本节讲什么

/var/log/lastlog 记录每个用户的最后登录时间和来源。以 UID 为索引的固定大小记录结构。

要点

lastlog 文件结构

偏移 = UID × sizeof(struct lastlog)

每个 UID 对应一条记录，未使用的 UID 为全零。

lastlog 结构

字段 类型 说明
ll_time time_t 最后登录时间
ll_line char[UT_LINESIZE] 终端设备名
ll_host char[UT_HOSTSIZE] 远程主机

查看

lastlog          # 所有用户
lastlog -u root  # 特定用户
lastlog -t 7     # 最近7天

写入

登录程序（如 login/sshd）在用户成功登录后写入 lastlog：

struct lastlog ll;
ll.ll_time = time(NULL);
strncpy(ll.ll_line, ttyname, sizeof(ll.ll_line));
strncpy(ll.ll_host, remote_host, sizeof(ll.ll_host));

int fd = open("/var/log/lastlog", O_WRONLY);
lseek(fd, uid * sizeof(ll), SEEK_SET);
write(fd, &ll, sizeof(ll));
close(fd);

练习 2：模拟登录记录

题目：编写程序向 wtmp 追加一条登录记录和一条注销记录。

参考答案 
#include <utmpx.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

void write_wtmp(short type, const char *user, const char *line) {
    struct utmpx ut = {0};
    ut.ut_type = type;
    ut.ut_pid = getpid();
    strncpy(ut.ut_user, user, sizeof(ut.ut_user));
    strncpy(ut.ut_line, line, sizeof(ut.ut_line));
    ut.ut_tv.tv_sec = time(NULL);

    utmpxname("/var/log/wtmp");
    setutxent();
    pututxline(&ut);
    endutxent();
}

/* 写入登录 + 注销 */
write_wtmp(USER_PROCESS, "hft", "pts/0");
/* ... 一段时间后 ... */
