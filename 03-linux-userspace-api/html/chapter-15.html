15.10 易错清单

  1. stat 跟随；lstat 才见 S_ISLNK

代码示例

#include <stdio.h>
#include <sys/stat.h>
#include <time.h>

/* Ch15 文件属性 — stat/fstat/lstat 获取 inode 信息。
 * 演示文件类型判断 + 权限/大小/时间戳。
 * 编译: gcc -o ch15_demo ch15_demo.c */

int main(void) {
    struct stat sb;
    if (lstat("/etc/hostname", &sb) < 0) {
        perror("lstat");
        return 1;
    }

    /* 文件类型 */
    if (S_ISREG(sb.st_mode))  printf("type: regular file\n");
    else if (S_ISDIR(sb.st_mode)) printf("type: directory\n");
    else if (S_ISLNK(sb.st_mode)) printf("type: symlink\n");
    else if (S_ISCHR(sb.st_mode)) printf("type: char device\n");
    else if (S_ISBLK(sb.st_mode)) printf("type: block device\n");

    /* 权限 */
    printf("mode: %04o\n", sb.st_mode & 07777);
    printf("size: %ld bytes\n", (long)sb.st_size);
    printf("links: %ld\n", (long)sb.st_nlink);
    printf("uid: %u, gid: %u\n", sb.st_uid, sb.st_gid);

    /* 时间戳 */
    printf("mtime: %s", ctime(&sb.st_mtime));
    printf("atime: %s", ctime(&sb.st_atime));
    printf("ctime: %s", ctime(&sb.st_ctime));

    /* inode 号 */
    printf("inode: %lu\n", (unsigned long)sb.st_ino);
    return 0;
}

15.1

`stat` / `lstat` / `fstat`

本章：TLPI 第 15 章 — File Attributes - 上一节：本章首页 - 下一节：15.2 st_mode：类型与权限

本节讲什么

include int stat(const char pathname, struct stat buf); / 跟随符号链接 / int lstat(const char pathname, struct stat buf); / 链接自身 /

要点

#include <sys/stat.h>
int  stat(const char *pathname, struct stat *buf);   /* 跟随符号链接 */
int lstat(const char *pathname, struct stat *buf);   /* 链接自身 */
int fstat(int fd, struct stat *buf);                 /* 已打开 fd；减 TOCTOU */
调用 符号链接
stat 跟到目标
lstat 得到链接 inode（S_ISLNK）
fstat 看 fd 所指对象（打开时已解析）

struct stat 核心字段

字段 含义
st_dev / st_ino 设备 + inode（同 FS 内唯一）
st_mode 类型 + 权限 + 特殊位
st_nlink 硬链接计数
st_uid / st_gid 属主 / 属组
st_rdev 设备文件：主/次设备号
st_size 字节大小
st_blksize / st_blocks 优选 IO 块；占用块数（常 512B 单位）
st_atim / st_mtim / st_ctim 访问 / 内容修改 / inode 变更（现代为 timespec）

Demo：code/print_stat.c

HFT / 嵌入式关联

路线 要点
嵌入式 fstat 对已打开 fd 避免路径重解析的 TOCTOU；lstat 不跟随符号链接，取链接自身信息
HFT 热路径已持有 fd 用 fstat 而非 stat(路径)，省一次 dentry 查找与权限检查，降低延迟

代码示例

#include <stdio.h>
#include <sys/stat.h>
#include <unistd.h>
/* stat: 跟随符号链接; lstat: 不跟随 */
int main(void) {
    struct stat sb;

    /* stat: 如果是符号链接, 获取目标文件信息 */
    symlink("/etc/passwd", "/tmp/stat_test_link");
    stat("/tmp/stat_test_link", &sb);
    printf("stat: inode=%lu (target's inode)\n", (unsigned long)sb.st_ino);

    /* lstat: 获取链接本身的信息 */
    lstat("/tmp/stat_test_link", &sb);
    printf("lstat: inode=%lu (link's own inode)\n", (unsigned long)sb.st_ino);
    printf("is symlink: %s\n", S_ISLNK(sb.st_mode) ? "yes" : "no");

    /* fstat: 通过 fd 获取 */
    int fd = open("/etc/passwd", O_RDONLY);
    fstat(fd, &sb);
    printf("fstat: size=%ld\n", (long)sb.st_size);
    close(fd);
    remove("/tmp/stat_test_link");
    return 0;
}

15.2

`st_mode`：类型与权限

本章：TLPI 第 15 章 — File Attributes - 上一节：15.1 stat / lstat / fstat - 下一节：15.3 chmod / fchmod

本节讲什么

S_ISREG · S_ISDIR · S_ISLNK · S_ISCHR · S_ISBLK · S_ISFIFO · S_ISSOCK S_IRUSR/WUSR/XUSR · S_IRGRP/... · S_IROTH/... ---

要点

类型宏

S_ISREG · S_ISDIR · S_ISLNK · S_ISCHR · S_ISBLK · S_ISFIFO · S_ISSOCK

九位权限

S_IRUSR/WUSR/XUSR · S_IRGRP/... · S_IROTH/...

特殊位

位 含义
S_ISUID Set-User-ID
S_ISGID Set-Group-ID
S_ISVTX 粘滞位：在目录上 → 仅属主/root 可删目录内条目（如 /tmp）

普通文件上的 sticky：现代 Linux 基本无实际意义。

HFT / 嵌入式关联

路线 要点
嵌入式 st_mode 编码文件类型+权限+特殊位，用 S_ISREG 等宏判断类型而非位掩码手算
HFT 权限位检查（S_IXUSR 等）决定可执行文件与目录访问；误判权限会阻断交易进程启动

代码示例

#include <sys/stat.h>
#include <stdio.h>

int main(void) {
    struct stat sb;
    if (stat("/etc/passwd", &sb) != 0) { perror("stat"); return 1; }

    /* st_mode 包含: 文件类型 + 权限 */
    printf("st_mode = 0%o\n", sb.st_mode);

    /* 文件类型判断 */
    if (S_ISREG(sb.st_mode))  printf("类型: 普通文件\n");
    if (S_ISDIR(sb.st_mode))  printf("类型: 目录\n");
    if (S_ISCHR(sb.st_mode))  printf("类型: 字符设备\n");
    if (S_ISBLK(sb.st_mode))  printf("类型: 块设备\n");
    if (S_ISLNK(sb.st_mode))  printf("类型: 符号链接\n");
    if (S_ISFIFO(sb.st_mode)) printf("类型: 管道\n");
    if (S_ISSOCK(sb.st_mode)) printf("类型: 套接字\n");

    /* 权限位 */
    printf("权限: %o\n", sb.st_mode & 0777);
    printf("setuid: %s\n", (sb.st_mode & S_ISUID) ? "YES" : "NO");
    printf("setgid: %s\n", (sb.st_mode & S_ISGID) ? "YES" : "NO");
    printf("sticky: %s\n", (sb.st_mode & S_ISVTX) ? "YES" : "NO");
    return 0;
}

15.3

`chmod` / `fchmod`

本章：TLPI 第 15 章 — File Attributes - 上一节：15.2 st_mode：类型与权限 - 下一节：15.4 chown / fchown / lchown

本节讲什么

int chmod(const char *pathname, mode_t mode); int fchmod(int fd, mode_t mode); - 属主或 root 可改

要点

int chmod(const char *pathname, mode_t mode);
int fchmod(int fd, mode_t mode);
  • 属主或 root 可改
  • mode 是最终权限，非增量
  • 普通用户设 SUID/SGID 有安全限制
  • 创建时 open/mkdir 的 mode 还受 umask 裁剪

HFT / 嵌入式关联

路线 要点
嵌入式 chmod 修改权限位，setuid/setgid 特殊位有安全风险需谨慎，服务降权后权限易错
HFT 服务进程降权后需正确设置配置/日志文件权限，避免普通用户可写导致被篡改或注入

代码示例

#include <sys/stat.h>
#include <fcntl.h>
#include <stdio.h>

int main(void) {
    /* chmod: 按路径改权限 */
    if (chmod("/tmp/chmod_test.txt", 0644) != 0) {
        perror("chmod");
        /* 文件可能不存在, 先创建 */
        int fd = open("/tmp/chmod_test.txt", O_CREAT, 0644);
        if (fd >= 0) { close(fd); chmod("/tmp/chmod_test.txt", 0644); }
    }
    printf("chmod 0644 (rw-r--r--)\n");

    /* fchmod: 按 fd 改权限 */
    int fd = open("/tmp/chmod_test.txt", O_RDWR);
    if (fd >= 0) {
        fchmod(fd, 0600);  /* rw------- */
        printf("fchmod 0600 (rw-------)\n");
        close(fd);
    }

    /* 权限只能由文件属主或 root 修改 */
    /* 注意: chmod 不改变 umask (umask 只影响 open 时的默认权限) */
    return 0;
}

15.4

`chown` / `fchown` / `lchown`

本章：TLPI 第 15 章 — File Attributes - 上一节：15.3 chmod / fchmod - 下一节：15.5 atime / mtime / ctime

本节讲什么

int chown(const char path, uid_t owner, gid_t group); int lchown(const char path, uid_t owner, gid_t group); / 不跟随链接 / int fchown(int fd, uid_t owner, gid_t group);

要点

int  chown(const char *path, uid_t owner, gid_t group);
int lchown(const char *path, uid_t owner, gid_t group);  /* 不跟随链接 */
int fchown(int fd, uid_t owner, gid_t group);

Linux 常见约束：

谁 能做什么
root 任意改 uid/gid
普通用户 不能把属主改成别人；属组通常只能改成自己所属组

owner/group 传 (uid_t)-1 / (gid_t)-1：不改该项。

HFT / 嵌入式关联

路线 要点
嵌入式 chown 改属主属组，普通用户无法改属主；lchown 对符号链接本身操作
HFT 多用户部署用 chown 明确文件归属，配合权限位实现最小权限原则，防越权读写

代码示例

#include <unistd.h>
#include <sys/stat.h>
#include <stdio.h>

int main(void) {
    const char *path = "/tmp/chown_test.txt";

    /* 创建测试文件 */
    struct stat sb;
    if (stat(path, &sb) != 0) {
        /* 文件不存在, 仅演示 */
        printf("文件不存在, 演示 API 用法\n");
    }

    printf("当前 UID=%d GID=%d\n", getuid(), getgid());

    /* chown: 按路径改属主 (需 root 或属主) */
    /* chown(path, uid, gid);  -- 需 root */

    /* fchown: 按 fd 改 */
    /* int fd = open(path, O_RDWR); fchown(fd, uid, gid); */

    /* lchown: 改符号链接本身 (不改目标文件) */
    /* lchown("/tmp/symlink", uid, gid); */

    /* 只有 root 能改 UID; 属主可改 GID 为自己所属的组 */
    printf("chown(path, uid, gid) — 需 root 权限改 UID\n");
    printf("fchown(fd, uid, gid)  — 通过 fd 改\n");
    printf("lchown(link, uid, gid) — 改符号链接本身\n");
    return 0;
}

15.5

atime / mtime / ctime

本章：TLPI 第 15 章 — File Attributes - 上一节：15.4 chown / fchown / lchown - 下一节：15.6 umask

本节讲什么

⚠️ ctime 不是创建时间；程序不能直接设定 ctime。 经典 ext 不强调 birth time；较新内核可用 statx 取 btime。 int utime(const char path, const struct utimbuf times); / 旧 /

要点

时间 何时更新
atime 读数据（read 等）；挂载 noatime 可关，提性能
mtime 内容变（write/truncate）
ctime inode 元数据变（chmod/chown/link/unlink/rename 等）

⚠️ ctime 不是创建时间；程序不能直接设定 ctime。
经典 ext 不强调 birth time；较新内核可用 statx 取 btime。

改 atime/mtime

int utime(const char *path, const struct utimbuf *times);      /* 旧 */
int utimes(const char *path, const struct timeval times[2]);
int futimens(int fd, const struct timespec times[2]);          /* 推荐，纳秒 */
int utimensat(int dirfd, const char *pathname,
              const struct timespec times[2], int flags);
  • times[0]=atime，times[1]=mtime
  • times == NULL → 都设为现在
  • UTIME_NOW / UTIME_OMIT 可选改一项
  • AT_SYMLINK_NOFOLLOW：不跟随链接

Demo：code/futimens_demo.c

HFT / 嵌入式关联

路线 要点
嵌入式 读文件会更新 atime 触发元数据写盘，noatime 挂载避免读操作写盘，延长闪存寿命
HFT 关闭 atime 减少读文件触发的元数据写，降低延迟抖动；日志轮转依据 mtime 判断

代码示例

#include <stdio.h>
#include <sys/stat.h>
#include <time.h>
#include <fcntl.h>
#include <unistd.h>
/* atime(访问) / mtime(修改) / ctime(状态改变) */
int main(void) {
    const char *path = "/tmp/atime_test.txt";
    int fd = open(path, O_CREAT | O_WRONLY | O_TRUNC, 0644);
    write(fd, "hello", 5);
    close(fd);

    struct stat sb;
    stat(path, &sb);
    printf("After create+write:\n");
    printf("  atime: %s", ctime(&sb.st_atime));
    printf("  mtime: %s", ctime(&sb.st_mtime));
    printf("  ctime: %s", ctime(&sb.st_ctime));

    /* 读文件: 更新 atime */
    fd = open(path, O_RDONLY);
    char buf[5];
    read(fd, buf, 5);
    close(fd);

    /* chmod: 更新 ctime (不改 mtime) */
    chmod(path, 0600);

    stat(path, &sb);
    printf("\nAfter read + chmod:\n");
    printf("  atime: %s", ctime(&sb.st_atime));
    printf("  mtime: %s (unchanged)\n", ctime(&sb.st_mtime));
    printf("  ctime: %s (updated by chmod)\n", ctime(&sb.st_ctime));

    remove(path);
    return 0;
}

15.6

`umask`

本章：TLPI 第 15 章 — File Attributes - 上一节：15.5 atime / mtime / ctime - 下一节：15.7 access() — 尽量别用

本节讲什么

mode_t umask(mode_t mask); / 设新值，返回旧值 / 最终权限 ≈ 创建 mode & ~umask 例：0666 + umask 022 → 0644

要点

mode_t umask(mode_t mask);   /* 设新值，返回旧值 */

最终权限 ≈ 创建 mode & ~umask
例：0666 + umask 022 → 0644
只影响新建，不改已有文件。

Demo：code/umask_demo.c

HFT / 嵌入式关联

路线 要点
嵌入式 umask 裁剪新文件默认权限（0666 & ~umask），进程启动时设定防止权限过宽
HFT 部署脚本显式设置 umask，确保日志/临时文件不被其他用户读写，防止敏感数据泄露

代码示例

#include <sys/stat.h>
#include <fcntl.h>
#include <stdio.h>

int main(void) {
    /* umask: 设置创建文件时的权限掩码 */
    /* 最终权限 = open 请求的权限 & ~umask */

    mode_t old = umask(0022);  /* 022 是常见默认值 */
    printf("旧 umask = 0%o\n", old);
    printf("新 umask = 0%o\n", 0022);

    /* umask=022 时: open(..., 0666) → 实际 0644 */
    int fd = open("/tmp/umask_test1.txt", O_CREAT, 0666);
    if (fd >= 0) {
        struct stat sb;
        fstat(fd, &sb);
        printf("open 0666 + umask 022 → 实际 0%o\n", sb.st_mode & 0777);
        close(fd);
    }

    /* 改 umask 为 0: 不屏蔽任何权限 */
    umask(0);
    fd = open("/tmp/umask_test2.txt", O_CREAT, 0666);
    if (fd >= 0) {
        struct stat sb;
        fstat(fd, &sb);
        printf("open 0666 + umask 000 → 实际 0%o\n", sb.st_mode & 0777);
        close(fd);
    }

    /* 恢复 */
    umask(old);
    return 0;
}

15.7

`access()` — 尽量别用

本章：TLPI 第 15 章 — File Attributes - 上一节：15.6 umask - 下一节：15.9 statx（Linux）

本节讲什么

int access(const char pathname, int mode); / R_OK W_OK X_OK F_OK */ 安全做法：直接 open，处理 EACCES。 ---

要点

int access(const char *pathname, int mode);  /* R_OK W_OK X_OK F_OK */
问题
用 真实 UID/GID 测 实际打开用 有效 UID/GID → setuid 下易误判
TOCTOU 测完再 open 之间状态可变

安全做法：直接 open，处理 EACCES。

HFT / 嵌入式关联

路线 要点
嵌入式 access() 用真实 UID 判断，与实际 open 存在 TOCTOU 竞态窗口，应避免预检查
HFT 用 access() 预检查权限再 open 会引入竞态，直接 open 并处理错误更安全可靠

代码示例

#include <unistd.h>
#include <stdio.h>

int main(void) {
    /* access: 检查真实 UID/GID 的权限 (不是 effective) */
    const char *path = "/etc/passwd";

    if (access(path, R_OK) == 0)
        printf("可读 %s\n", path);
    else
        printf("不可读: %s\n", path);

    if (access(path, W_OK) == 0)
        printf("可写 %s\n", path);
    else
        printf("不可写\n");

    if (access(path, X_OK) == 0)
        printf("可执行 %s\n", path);
    else
        printf("不可执行\n");

    if (access(path, F_OK) == 0)
        printf("文件存在\n");

    /* 注意: access 有 TOCTOU 竞态!
     * access 检查和后续 open 之间文件可能被替换
     * 安全做法: 直接 open 并检查错误 */
    return 0;
}
