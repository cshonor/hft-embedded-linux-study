导读

章节目标

理清目录 / 硬链 / 软链；掌握 mkdir/rmdir、目录遍历、link/symlink/unlink、rename 原子性；用 *at + flags 缓解路径竞态。

18.7 易错清单

  1. 目录要访问内容：常需 x，不只 r
  2. 硬链禁目录；软链可以
  3. unlink 减计数，不是立刻抹数据
  4. rename 原子性限于同一文件系统
  5. readlink ≠ 解析最终路径
  6. 硬链不新建数据 inode；软链新建 inode
  7. 无「递归 rmdir」syscall；自行遍历 unlink + rmdir
  8. 根 .. 指自己，遍历防环

练习

  1. opendir/readdir + lstat 区分类型
  2. link vs symlink，比 st_ino
  3. 临时文件 + rename 安全写
  4. open 后 unlink 仍可读
  5. 目录仅 r 无 x 的行为

背诵卡

# 要点
1 目录项存名字→inode；权限 r/x/w 语义不同
2 硬链共享 inode，同 FS，不链目录
3 软链独立 inode，可跨 FS / 悬空
4 nlink==0 且无 fd → 才释放
5 rename 同 FS 原子；可靠写 = write tmp + rename
6 用 *at + AT_SYMLINK_NOFOLLOW 减 TOCTOU

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/stat.h>
#include <dirent.h>
#include <string.h>
#include <fcntl.h>

/* Ch18 目录与链接 — mkdir/symlink/readlink/rename/link。
 * 演示硬链接/符号链接的区别。
 * 编译: gcc -o ch18_demo ch18_demo.c */

int main(void) {
    /* 创建目录 */
    mkdir("/tmp/ch18_dir", 0755);

    /* 创建文件并硬链接 */
    int fd = open("/tmp/ch18_dir/file.txt", O_CREAT | O_WRONLY, 0644);
    write(fd, "hello\n", 6);
    close(fd);

    link("/tmp/ch18_dir/file.txt", "/tmp/ch18_dir/hardlink.txt");
    printf("hard link created (same inode)\n");

    /* 符号链接 */
    symlink("/tmp/ch18_dir/file.txt", "/tmp/ch18_dir/softlink.txt");

    /* readlink 读取符号链接目标 */
    char buf[256];
    ssize_t n = readlink("/tmp/ch18_dir/softlink.txt", buf, sizeof(buf) - 1);
    if (n > 0) {
        buf[n] = '\0';
        printf("symlink target: %s\n", buf);
    }

    /* stat vs lstat: lstat 不跟随符号链接 */
    struct stat sb1, sb2;
    stat("/tmp/ch18_dir/softlink.txt", &sb1);
    lstat("/tmp/ch18_dir/softlink.txt", &sb2);
    printf("stat inode:  %lu (target file)\n", (unsigned long)sb1.st_ino);
    printf("lstat inode: %lu (link itself)\n", (unsigned long)sb2.st_ino);

    /* rename */
    rename("/tmp/ch18_dir/hardlink.txt", "/tmp/ch18_dir/renamed.txt");
    printf("renamed hardlink\n");

    /* 清理 */
    remove("/tmp/ch18_dir/softlink.txt");
    remove("/tmp/ch18_dir/renamed.txt");
    remove("/tmp/ch18_dir/file.txt");
    rmdir("/tmp/ch18_dir");
    return 0;
}

18.1

目录基础

本章：TLPI 第 18 章 — Directories and Links - 上一节：本章首页 - 下一节：18.2 读目录

本节讲什么

目录是特殊文件：数据块存目录项 (文件名, inode#)。 . = 自己；.. = 父；根的 .. 指向自己。 int mkdir(const char pathname, mode_t mode); / 受 umask */

要点

目录是特殊文件：数据块存目录项 (文件名, inode#)。
. = 自己；.. = 父；根的 .. 指向自己。

目录权限语义

位 含义
r 可列目录项（见名字）；不能单靠它访问内容
x 可进入 / 解析路径内名字（访问内部文件必备）
w 可在目录内创建、删除、重命名条目

仅 r 无 x：能 ls 见名，常无法 stat/open 内部文件。

int mkdir(const char *pathname, mode_t mode);      /* 受 umask */
int mkdirat(int dirfd, const char *pathname, mode_t mode);
int rmdir(const char *pathname);                   /* 仅空目录 */
int chdir(const char *path);
int fchdir(int fd);
char *getcwd(char *buf, size_t size);

rmdir 非空失败；递归删除须自写（rm -r 逻辑）。CWD 是进程属性。

#include <stdio.h>
#include <dirent.h>
#include <sys/stat.h>
/* 目录操作: mkdir/opendir/readdir/closedir */
int main(void) {
    mkdir("/tmp/ch18_dir_test", 0755);

    DIR *dir = opendir("/tmp");
    if (!dir) { perror("opendir"); return 1; }

    printf("Entries in /tmp:\n");
    struct dirent *ent;
    while ((ent = readdir(dir)) != NULL) {
        if (ent->d_name[0] == '.') continue;
        printf("  %s (type=%d, ino=%lu)\n",
               ent->d_name, ent->d_type, (unsigned long)ent->d_ino);
    }
    closedir(dir);
    rmdir("/tmp/ch18_dir_test");
    return 0;
}

代码示例

#include <stdio.h>
#include <dirent.h>
#include <string.h>
/* readdir vs scandir: 遍历 vs 排序过滤 */
static int filter(const struct dirent *d) {
    return d->d_name[0] != '.';  /* 跳过隐藏文件 */
}

int main(void) {
    /* scandir: 自动排序 + 过滤 */
    struct dirent **namelist;
    int n = scandir("/tmp", &namelist, filter, alphasort);
    if (n < 0) { perror("scandir"); return 1; }

    printf("scandir results (sorted):\n");
    for (int i = 0; i < n; i++) {
        printf("  %s\n", namelist[i]->d_name);
        free(namelist[i]);
    }
    free(namelist);

    /* rewinddir: 重置目录流位置 */
    DIR *dir = opendir("/tmp");
    struct dirent *e;
    int count1 = 0;
    while ((e = readdir(dir))) count1++;

    rewinddir(dir);  /* 重置到开头 */
    int count2 = 0;
    while ((e = readdir(dir))) count2++;

    printf("\nrewinddir: first pass=%d, second pass=%d\n", count1, count2);
    closedir(dir);
    return 0;
}

18.3

硬链接

本章：TLPI 第 18 章 — Directories and Links - 上一节：18.2 读目录 - 下一节：18.4 符号链接

本节讲什么

int link(const char oldpath, const char newpath); int linkat(...); int unlink(const char *pathname);

要点

int link(const char *oldpath, const char *newpath);
int linkat(...);
int unlink(const char *pathname);
int unlinkat(int dirfd, const char *pathname, int flags); /* AT_REMOVEDIR≈rmdir */

多个目录项 → 同一 inode；st_nlink++。

规则
无主次 共享权限/大小/时间戳
同 FS 不能跨分区
不能链目录 用户空间禁目录硬链（防环；./.. 内核特例）
目标须已存在

删除： unlink 减链接计数；nlink==0 且无打开 fd → 释放数据。
技巧：open 后立刻 unlink → 进程仍可读写，关 fd 后消失（临时文件）。

Demo：code/links_demo.c · code/unlink_open.c

HFT / 嵌入式关联

路线 要点
嵌入式 硬链接 在嵌入式系统的应用场景
HFT 硬链接 对低延迟和性能的影响

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/stat.h>
#include <string.h>
/* 硬链接 vs 符号链接 */
int main(void) {
    /* 创建测试文件 */
    int fd = open("/tmp/link_test.txt", 2 | 64 | 512, 0644); /* O_RDWR|O_CREAT|O_TRUNC */
    write(fd, "data", 4);
    close(fd);

    /* 硬链接: 同一 inode, 不能跨文件系统 */
    link("/tmp/link_test.txt", "/tmp/link_hard.txt");

    /* 符号链接: 独立 inode, 存储目标路径 */
    symlink("/tmp/link_test.txt", "/tmp/link_soft.txt");

    /* 对比 inode */
    struct stat s1, s2, s3;
    stat("/tmp/link_test.txt", &s1);
    stat("/tmp/link_hard.txt", &s2);
    lstat("/tmp/link_soft.txt", &s3);

    printf("original inode: %lu\n", (unsigned long)s1.st_ino);
    printf("hardlink inode: %lu (same!)\n", (unsigned long)s2.st_ino);
    printf("symlink inode:  %lu (different)\n", (unsigned long)s3.st_ino);
    printf("link count: %lu (2 because of hardlink)\n", (unsigned long)s1.st_nlink);

    /* readlink: 读取符号链接目标 */
    char buf[256];
    ssize_t n = readlink("/tmp/link_soft.txt", buf, sizeof(buf));
    buf[n] = '\0';
    printf("symlink points to: %s\n", buf);

    /* unlink: 删除硬链接, 引用计数减1 */
    unlink("/tmp/link_hard.txt");
    stat("/tmp/link_test.txt", &s1);
    printf("after unlink hardlink: nlink=%lu\n", (unsigned long)s1.st_nlink);

    remove("/tmp/link_test.txt");
    remove("/tmp/link_soft.txt");
    return 0;
}

18.4

符号链接

本章：TLPI 第 18 章 — Directories and Links - 上一节：18.3 硬链接 - 下一节：18.5 rename（同 FS 原子）

本节讲什么

int symlink(const char target, const char linkpath); ssize_t readlink(const char pathname, char buf, size_t bufsiz); 独立 inode，内容 = 目标路径字符串。可跨 FS、可指目录、可悬空。

要点

int symlink(const char *target, const char *linkpath);
ssize_t readlink(const char *pathname, char *buf, size_t bufsiz);

独立 inode，内容 = 目标路径字符串。可跨 FS、可指目录、可悬空。
stat 跟随；lstat 看链接本身；readlink 只读字符串（不解析）。

硬链 vs 软链

硬链接 符号链接
inode 共享 独立
跨 FS ❌ ✅
指目录 ❌（用户） ✅
目标删后 数据仍在 悬空
stat 跟 / lstat 看链

路线 要点
嵌入式 符号链接 在嵌入式系统的应用场景
HFT 符号链接 对低延迟和性能的影响

代码示例

#include <unistd.h>
#include <stdio.h>

int main(void) {
    /* 硬链接: link() — 多个文件名指向同一 inode */
    int ret = link("/tmp/link_demo.txt", "/tmp/hardlink.txt");
    if (ret == 0) {
        printf("硬链接创建成功 (同一 inode, nlink+1)\n");
    } else {
        /* 先创建源文件 */
        FILE *fp = fopen("/tmp/link_demo.txt", "w");
        if (fp) { fprintf(fp, "hello"); fclose(fp); }
        link("/tmp/link_demo.txt", "/tmp/hardlink.txt");
        printf("硬链接创建成功\n");
    }

    /* 符号链接: symlink() — 特殊文件, 内容是目标路径 */
    ret = symlink("/tmp/link_demo.txt", "/tmp/softlink.txt");
    if (ret == 0) {
        printf("符号链接创建成功\n");
    }

    /* 读符号链接目标 */
    char buf[256];
    ssize_t n = readlink("/tmp/softlink.txt", buf, sizeof(buf) - 1);
    if (n > 0) {
        buf[n] = '\0';
        printf("符号链接指向: %s\n", buf);
    }

    /* 区别:
     * 硬链接: 不能跨文件系统, 不能链接目录, 删除源文件仍可用
     * 符号链接: 可跨文件系统, 可链接目录, 删除源文件则失效
     */
    return 0;
}

18.5

`rename`（同 FS 原子）

本章：TLPI 第 18 章 — Directories and Links - 上一节：18.4 符号链接 - 下一节：18.6 路径与 TOCTOU

本节讲什么

int rename(const char oldpath, const char newpath); int renameat(...); 可靠写入：写临时文件 → rename(tmp, target)。

要点

int rename(const char *oldpath, const char *newpath);
int renameat(...);
情况 行为（同 FS）
new 不存在 改名
new 是文件 原子替换
new 空目录 可替换
new 非空目录 失败

可靠写入：写临时文件 → rename(tmp, target)。

#include <stdio.h>
#include <unistd.h>

int main(void) {
    /* rename: 原子性改名/移动 */
    FILE *fp = fopen("/tmp/rename_old.txt", "w");
    if (fp) { fprintf(fp, "hello rename"); fclose(fp); }

    /* 同目录改名 */
    if (rename("/tmp/rename_old.txt", "/tmp/rename_new.txt") == 0) {
        printf("改名成功: old → new\n");
    } else {
        perror("rename");
    }

    /* 跨目录移动 (同文件系统内) */
    /* rename("/tmp/file.txt", "/tmp/subdir/file.txt"); */

    /* rename 是原子操作:
     * - 不会出现"旧名已删, 新名还没建"的中间状态
     * - 目标已存在则原子替换
     * - 常用于原子更新文件: 写临时文件 → rename 覆盖
     */
    return 0;
}

18.6

路径与 TOCTOU

本章：TLPI 第 18 章 — Directories and Links - 上一节：18.5 rename（同 FS 原子） - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

默认多跟随符号链接 → 检查与使用之间可被换链。 优先 openat/unlinkat/linkat 等，基于目录 fd；常用 AT_SYMLINK_NOFOLLOW、AT_REMOVEDIR。

要点

默认多跟随符号链接 → 检查与使用之间可被换链。
优先 openat/unlinkat/linkat 等，基于目录 fd；常用
AT_SYMLINK_NOFOLLOW、AT_REMOVEDIR。

HFT / 嵌入式关联

路线 要点
嵌入式 路径与 TOCTOU 在嵌入式系统的应用场景
HFT 路径与 TOCTOU 对低延迟和性能的影响

代码示例

#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

/* TOCTOU: Time-of-Check to Time-of-Use 竞态
 * access() 检查和 open() 使用之间文件可能被替换
 */
int main(void) {
    const char *path = "/tmp/toctou_demo.txt";

    /* 不安全: access + open 之间有间隙 */
    if (access(path, W_OK) == 0) {
        /* ← 攻击者可在此刻替换文件为符号链接指向 /etc/passwd */
        int fd = open(path, O_WRONLY);
        if (fd >= 0) {
            write(fd, "oops", 4);  /* 可能写到了错误文件! */
            close(fd);
        }
    }

    /* 安全: 直接 open, 检查错误 */
    int fd = open(path, O_WRONLY | O_NOFOLLOW);  /* O_NOFOLLOW 防符号链接 */
    if (fd < 0) {
        printf("安全方式: 直接 open 并检查错误 (无 TOCTOU)\n");
    } else {
        write(fd, "safe", 4);
        close(fd);
    }

    printf("TOCTOU: 检查和使用之间有间隙 → 用原子操作避免\n");
    return 0;
}

18.7

Removing a File or Directory: remove()

本章：TLPI 第 18 章

本节讲什么

remove() 是 C 标准库函数，用于删除文件或空目录。它根据路径类型自动选择 unlink() 或 rmdir()，是文件删除的统一入口。

要点

remove() 函数

#include <stdio.h>
int remove(const char *pathname);
// 返回：0 成功，-1 失败

行为规则

路径类型 底层调用 效果
普通文件 unlink() 减少硬链接计数，计数为0时释放
符号链接 unlink() 删除链接本身，不影响目标
空目录 rmdir() 删除空目录
非空目录 失败 ENOTEMPTY 无法删除

unlink() vs rmdir() vs remove()

函数 来源 适用文件 适用目录
unlink() POSIX 是 否 (EISDIR)
rmdir() POSIX 否 (ENOTDIR) 仅空目录
remove() C 标准 是 仅空目录

删除非空目录

remove() 无法直接删除非空目录。需递归删除内容：

#include <ftw.h>
static int rm_fn(const char *path, const struct stat *sb, int type, struct FTW *ftwbuf) {
    return remove(path);  /* 后序遍历：先删文件再删目录 */
}
int rm_recursive(const char *dir) {
    return nftw(dir, rm_fn, 64, FTW_DEPTH | FTW_PHYS);
}

HFT / 嵌入式关联

  • 交易系统日志轮转时需删除过期日志文件
  • 嵌入式 OTA 升级后清理旧固件包
  • 临时 socket 文件清理（UNIX domain socket）

代码示例

#include <stdio.h>
#include <errno.h>
#include <string.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <path>\n", argv[0]);
        return 1;
    }

    if (remove(argv[1]) == -1) {
        fprintf(stderr, "remove %s: %s\n", argv[1], strerror(errno));
        return 1;
    }

    printf("Removed %s\n", argv[1]);
    return 0;
}

remove() 删除目录时返回 ENOTEMPTY，如何处理？

remove() 内部调用 rmdir()，只能删除空目录。非空目录需递归删除：使用 nftw() 配合 FTW_DEPTH 标志（后序遍历），先删除子项再删父目录。也可用 opendir/readdir 手动递归。

18.8

Reading Directories: opendir() and readdir()

本章：TLPI 第 18 章

本节讲什么

目录的读取操作：opendir() 打开目录流，readdir() 逐条读取目录项，readdir_r() 可重入版本（已废弃），telldir()/seekdir() 定位。closedir() 关闭目录流。

要点

核心 API

#include <dirent.h>
DIR *opendir(const char *name);           // 打开目录流
struct dirent *readdir(DIR *dirp);         // 读取下一项（非线程安全）
int closedir(DIR *dirp);                  // 关闭目录流
long telldir(DIR *dirp);                  // 获取当前位置
void seekdir(DIR *dirp, long loc);        // 设置位置
void rewinddir(DIR *dirp);                // 回到开头

struct dirent 结构

字段 类型 说明
d_ino ino_t inode 号
d_name char[] 文件名（不含路径）
d_type unsigned char 文件类型（见下表）
d_reclen unsigned short 记录长度

d_type 文件类型

宏 值 含义
DT_REG 8 普通文件
DT_DIR 4 目录
DT_LNK 10 符号链接
DT_BLK 6 块设备
DT_CHR 2 字符设备
DT_FIFO 1 管道
DT_SOCK 12 套接字
DT_UNKNOWN 0 未知（需 stat 补充）

readdir() vs readdir_r()

特性 readdir() readdir_r()
线程安全 否（共享静态缓冲） 是
返回方式 指针，下次调用覆盖 写入调用者缓冲
POSIX.1-2008 保留 已废弃
替代方案 配合 readdir_r 或目录流互斥 用 readdir() + 互斥锁

HFT / 嵌入式关联

  • 日志目录扫描：遍历查找最新的日志文件
  • 配置热加载：监控目录变化（配合 inotify）
  • 嵌入式文件系统：存储空间有限时遍历计算目录大小

代码示例

#include <dirent.h>
#include <stdio.h>
#include <string.h>

void list_dir(const char *path) {
    DIR *d = opendir(path);
    if (!d) { perror("opendir"); return; }

    struct dirent *entry;
    while ((entry = readdir(d)) != NULL) {
        /* 跳过 . 和 .. */
        if (strcmp(entry->d_name, ".") == 0 ||
            strcmp(entry->d_name, "..") == 0)
            continue;

        const char *type_str = "unknown";
        switch (entry->d_type) {
            case DT_REG: type_str = "file";   break;
            case DT_DIR: type_str = "dir";    break;
            case DT_LNK: type_str = "link";   break;
        }
        printf("%-20s %s\n", entry->d_name, type_str);
    }
    closedir(d);
}

代码自测

readdir() 返回 NULL 时如何区分"读完"和"出错"？

readdir() 返回 NULL 可能是遍历完毕，也可能是读取出错。必须在返回 NULL 后检查 errno：如果 errno 在调用前清零、调用后仍为 0，则是正常结束；如果 errno 被设置，则表示出错。但 POSIX 规定 readdir 不保证设置 errno，因此需在调用前 errno = 0，调用后检查。

18.9

File Tree Walking: nftw()

本章：TLPI 第 18 章

本节讲什么

nftw() (new file tree walk) 递归遍历目录树，对每个文件调用用户提供的回调函数。相比旧的 ftw()，支持更多标志和文件信息，是目录树遍历的标准接口。

要点

nftw() 函数

#include <ftw.h>
int nftw(const char *dirpath,
         int (*fn)(const char *fpath, const struct stat *sb,
                   int typeflag, struct FTW *ftwbuf),
         int nopenfd, int flags);
// 返回：0 成功完成，非0 回调返回值，-1 错误

struct FTW

字段 说明
base fpath 中文件名起始偏移
level 目录深度（根为0）

typeflag 类型

宏 含义
FTW_F 普通文件
FTW_D 目录（先序，进入前）
FTW_DNR 目录不可读
FTW_NS stat 失败
FTW_DP 目录（后序，离开时）
FTW_SL 符号链接
FTW_SLN 断链符号链接

flags 标志

标志 效果
FTW_CHDIR 回调前 chdir 到包含目录
FTW_DEPTH 后序遍历（先子后父，适合删除）
FTW_MOUNT 不跨文件系统
FTW_PHYS 不跟随符号链接

nftw() vs ftw()

特性 ftw() nftw()
回调参数 path + stat + type path + stat + type + FTW
后序遍历 不支持 FTW_DEPTH
符号链接 总是跟随 FTW_PHYS 不跟随
跨文件系统 无控制 FTW_MOUNT 控制
POSIX 已废弃 推荐

HFT / 嵌入式关联

  • 递归计算目录大小（磁盘配额监控）
  • 批量删除日志目录（FTW_DEPTH 后序删除）
  • 构建文件索引（配置文件树序列化）
  • 嵌入式固件包打包遍历

代码示例

#include <ftw.h>
#include <stdio.h>
#include <sys/stat.h>

static long long total_size = 0;

static int sum_cb(const char *fpath, const struct stat *sb,
                  int typeflag, struct FTW *ftwbuf) {
    if (S_ISREG(sb->st_mode))
        total_size += sb->st_size;
    return 0;  /* 继续遍历 */
}

int main(int argc, char *argv[]) {
    const char *dir = argc > 1 ? argv[1] : ".";
    /* nopenfd=64 限制同时打开的目录数 */
    if (nftw(dir, sum_cb, 64, FTW_PHYS) == -1) {
        perror("nftw");
        return 1;
    }
    printf("Total: %lld bytes\n", total_size);
    return 0;
}

递归删除目录树

static int rm_cb(const char *path, const struct stat *sb,
                 int typeflag, struct FTW *ftwbuf) {
    return remove(path);  /* FTW_DEPTH 保证先删子项 */
}
nftw(dirpath, rm_cb, 64, FTW_DEPTH | FTW_PHYS);

代码自测

nftw 回调返回非0值时会发生什么？

回调函数返回非0值会立即终止遍历，nftw() 将该值返回给调用者。这使得可以在找到目标文件后提前退出，或遇到错误时中止。注意：返回值为0表示成功继续，非0表示终止。如果回调想表示错误，应返回非0值，nftw 会原样返回（不是-1）。

18.10

The Current Working Directory of a Process

本章：TLPI 第 18 章

本节讲什么

每个进程有一个当前工作目录（CWD），作为相对路径解析的基准。getcwd() 获取当前路径，chdir() 修改当前目录。这是 shell cd 命令的底层机制。

要点

核心 API

#include <unistd.h>
char *getcwd(char *buf, size_t size);   // 获取当前工作目录
char *getwd(char *buf);                  // 已废弃，不安全
char *get_current_dir_name(void);        // GNU 扩展，malloc 分配
int chdir(const char *path);             // 修改当前工作目录
int fchdir(int fd);                       // 通过已打开的目录 fd 修改

getcwd 行为

情况 返回 errno
成功 buf 指针 -
buf 太小 NULL ERANGE
当前目录已被删除 NULL ENOENT
无权限访问父目录 NULL EACCES

CWD 内核机制

task_struct {
    struct fs_struct *fs;  // 文件系统信息
    // fs->pwd 是 CWD 的 path 结构
    // 内核解析相对路径时以此为基准
}

fchdir() 的优势

特性 chdir() fchdir()
参数 路径字符串 已打开的 fd
竞争安全 有 TOCTOU 无
适用场景 普通切换 保存/恢复 CWD

保存和恢复 CWD 的模式

/* 模式1：open + fchdir（推荐） */
int saved = open(".", O_RDONLY | O_DIRECTORY);
chdir("/some/path");
/* ... 在新目录工作 ... */
fchdir(saved);  /* 恢复 */
close(saved);

/* 模式2：getcwd + chdir（有竞争窗口） */
char cwd[PATH_MAX];
getcwd(cwd, sizeof(cwd));
chdir("/some/path");
/* ... */
chdir(cwd);  /* 如果 cwd 被删除/改名会失败 */

HFT / 嵌入式关联

  • 交易策略切换工作目录加载不同配置文件集
  • 日志系统切换到日志目录写入
  • 嵌入式系统 fchdir 避免 chdir 的符号链接竞争

代码示例

#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>
#include <limits.h>

int main() {
    char buf[PATH_MAX];
    if (getcwd(buf, sizeof(buf)) == NULL) {
        perror("getcwd");
        return 1;
    }
    printf("CWD: %s\n", buf);

    /* 保存 CWD，切换到 /tmp，再恢复 */
    int saved = open(".", O_RDONLY | O_DIRECTORY);
    if (saved < 0) { perror("open"); return 1; }

    chdir("/tmp");
    getcwd(buf, sizeof(buf));
    printf("Now in: %s\n", buf);

    fchdir(saved);
    close(saved);
    getcwd(buf, sizeof(buf));
    printf("Restored: %s\n", buf);
    return 0;
}

当前目录被删除后 getcwd() 返回什么？

如果当前目录被其他进程删除（unlink 或 rmdir），getcwd() 会失败并返回 NULL，errno 设置为 ENOENT。因为内核中的 CWD 路径已无效。在 Linux 上，getcwd 会返回带 (deleted) 后缀的路径，表示目录已被删除。程序应检测此情况并 chdir 到有效目录。

18.11

Operating Relative to a Directory File Descriptor

本章：TLPI 第 18 章

本节讲什么

Linux 2.6.39+ 和 POSIX.1-2008 引入了以目录 fd 为基准的文件操作 API：openat()、fstatat()、linkat()、unlinkat() 等，避免 CWD 全局状态带来的竞争和安全问题。

要点

*at() 函数族

函数 对应旧 API 说明
openat() open() 相对目录 fd 打开文件
fstatat() stat() 相对目录 fd 获取属性
linkat() link() 相对目录 fd 创建硬链接
unlinkat() unlink() 相对目录 fd 删除
renameat() rename() 相对目录 fd 重命名
readlinkat() readlink() 相对目录 fd 读取符号链接
fchmodat() chmod() 相对目录 fd 修改权限
fchownat() chown() 相对目录 fd 修改属主
futimensat() utimensat() 相对目录 fd 修改时间戳

openat() 原型

#include <fcntl.h>
int openat(int dirfd, const char *pathname, int flags, ... /* mode_t */);

dirfd 参数规则

dirfd 值 pathname 行为
AT_FDCWD 相对路径 相对 CWD（等同 open）
AT_FDCWD 绝对路径 忽略 dirfd，使用绝对路径
有效目录 fd 相对路径 相对该目录
有效目录 fd 绝对路径 忽略 dirfd，使用绝对路径

unlinkat() flags

标志 效果
0 删除文件（等同 unlink）
AT_REMOVEDIR 删除空目录（等同 rmdir）

优势：消除 TOCTOU 竞争

/* 旧方式：有竞争窗口 */
chdir("/srv/data");        // step 1
fd = open("config", ...);  // step 2: 可能在 1-2 之间被替换

/* 新方式：原子操作 */
dirfd = open("/srv/data", O_RDONLY | O_DIRECTORY);
fd = openat(dirfd, "config", ...);  // 基于已验证的目录

HFT / 嵌入式关联

  • 多线程交易系统：每个线程用自己的目录 fd，避免 CWD 竞争
  • 安全审计：验证目录未被符号链接劫持后操作
  • 容器/沙箱：openat 配合 O_PATH 实现受限文件系统访问

代码示例

#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <sys/stat.h>

int main() {
    /* 打开基准目录 */
    int dirfd = open("/srv/data", O_RDONLY | O_DIRECTORY);
    if (dirfd < 0) { perror("open dir"); return 1; }

    /* 相对 dirfd 打开文件 */
    int fd = openat(dirfd, "config.json", O_RDONLY);
    if (fd < 0) { perror("openat"); close(dirfd); return 1; }

    /* 相对 dirfd 获取文件属性 */
    struct stat st;
    if (fstatat(dirfd, "config.json", &st, 0) == 0)
        printf("size: %ld\n", st.st_size);

    /* 相对 dirfd 删除临时文件 */
    unlinkat(dirfd, "config.tmp", 0);

    close(fd);
    close(dirfd);
    return 0;
}

AT_FDCWD 表示使用进程的当前工作目录作为基准。此时 openat(AT_FDCWD, "file", flags) 等同于 open("file", flags)。如果 pathname 是绝对路径，则忽略 dirfd（无论是 AT_FDCWD 还是有效 fd），直接使用绝对路径。

18.12

Changing the Root Directory of a Process: chroot()

本章：TLPI 第 18 章

本节讲什么

chroot() 改变进程的根目录（/），使其只能访问该目录子树内的文件。这是早期的沙箱机制，但本身不提供完整的安全隔离。

要点

chroot() 函数

#include <unistd.h>
int chroot(const char *path);  // 改变调用者的根目录
// 返回：0 成功，-1 失败
// 需要 CAP_SYS_CHROOT

效果与限制

方面 说明
根目录 改为 path
CWD 不变（可能在新根之外）
权限 需要 CAP_SYS_CHROOT（root）
继承 fork 子进程共享新根；exec 不变
恢复 无对应"unchroot"调用

chroot 安全性问题

  1. CWD 不变：如果 CWD 在新根之外，getcwd 可能逃逸
  2. root 可逃逸：chroot(".") 可以恢复原根
  3. 不隔离进程/网络/IPC：只限制文件系统路径
  4. 信号和进程可见：kill 仍可达外部进程

正确的 chroot 流程

/* 安全 chroot 四步 */
chroot("/srv/sandbox");     // 1. 改根
chdir("/");                  // 2. 切换 CWD 到新根（关键！）
drop_privileges();           // 3. 丢弃 root 权限（关键！）
execve("/bin/app", argv, envp);  // 4. 执行目标程序

chroot vs 容器

特性 chroot 容器 (namespaces)
文件系统隔离 部分 完全 (mount namespace)
进程隔离 无 有 (pid namespace)
网络隔离 无 有 (net namespace)
用户映射 无 有 (user namespace)
安全等级 低 中-高

HFT / 嵌入式关联

  • 交易策略沙箱化执行：限制文件系统访问范围
  • 嵌入式系统恢复模式：chroot 到备用根文件系统
  • CI/CD 测试环境：在隔离根中运行测试
  • 注意：现代更推荐使用 containers/namespaces 替代

代码示例

#include <unistd.h>
#include <stdio.h>
#include <sys/prctl.h>

int main() {
    /* 1. chroot */
    if (chroot("/srv/sandbox") == -1) {
        perror("chroot");
        return 1;
    }

    /* 2. chdir 到新根（否则 CWD 仍在旧根） */
    if (chdir("/") == -1) {
        perror("chdir");
        return 1;
    }

    /* 3. 丢弃 root 权限（防止逃逸） */
    if (setgid(1000) == -1 || setuid(1000) == -1) {
        perror("setuid");
        return 1;
    }

    /* 4. 执行程序 */
    execl("/bin/sh", "sh", NULL);
    perror("execl");
    return 1;
}

代码自测

chroot 后不调用 chdir("/") 有什么风险？

chroot() 只改变根目录，不改变 CWD。如果 CWD 仍在新根之外的旧路径上，进程可以通过相对路径（如 ../../）访问新根之外的文件。必须在 chroot() 后立即 chdir("/") 使 CWD 也在新根内。此外，还必须丢弃 root 权限，因为 root 用户可以通过 chroot(".") 逃逸。

18.13

Resolving a Pathname: realpath()

本章：TLPI 第 18 章

本节讲什么

realpath() 将包含符号链接、.、.. 的路径解析为规范的绝对路径。它会展开所有符号链接并消除冗余路径分隔符，返回最终的真实路径。

要点

realpath() 函数

#include <stdlib.h>
char *realpath(const char *path, char *resolved_path);
// 返回：成功 resolved_path，失败 NULL

使用规则

resolved_path 参数 行为 注意
NULL malloc 分配缓冲区 需手动 free
非 NULL 写入调用者缓冲 缓冲需 ≥ PATH_MAX
POSIX.1-2008 支持 NULL 参数 GNU 扩展先于标准

realpath 解析过程

  1. 沿路径逐段解析
  2. 遇到符号链接时展开（递归，最多 MAXSYMLINKS 次）
  3. 处理 . 和 ..
  4. 消除多余 /
  5. 返回绝对路径

错误条件

errno 含义
EACCES 路径中某目录无搜索权限
EINVAL path 为 NULL
ENAMETOOLONG 路径超长或符号链接递归超限
ENOENT 路径不存在的部分
ENOMEM 内存不足（NULL 参数时）
ELOOP 符号链接循环

安全用法

/* 旧用法：缓冲区可能溢出（已废弃） */
char buf[PATH_MAX];
realpath(path, buf);  /* PATH_MAX 足够大时安全 */

/* 新用法：NULL 参数让系统分配 */
char *resolved = realpath(path, NULL);
if (resolved) { /* 使用 */ free(resolved); }

realpath vs readlink

特性 realpath() readlink()
输入 任意路径 符号链接路径
输出 绝对规范路径 链接目标原始内容
展开链接 全部递归展开 不展开

  • 安全路径验证：防止目录穿越攻击（../ 注入）
  • 配置文件路径规范化：确保唯一性避免重复加载
  • 日志文件路径解析：展开符号链接获取真实存储位置

代码示例

#include <stdlib.h>
#include <stdio.h>
#include <limits.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <path>\n", argv[0]);
        return 1;
    }

    /* POSIX.1-2008 方式：NULL 参数 */
    char *resolved = realpath(argv[1], NULL);
    if (resolved == NULL) {
        perror("realpath");
        return 1;
    }
    printf("Input:    %s\n", argv[1]);
    printf("Resolved: %s\n", resolved);
    free(resolved);

    /* 经典方式：调用者提供缓冲 */
    char buf[PATH_MAX];
    if (realpath(argv[1], buf) == NULL) {
        perror("realpath(buf)");
        return 1;
    }
    printf("Buffer:   %s\n", buf);

    return 0;
}

代码自测

realpath(path, NULL) 为什么比传缓冲区更安全？

传入 NULL 让 realpath 自行 malloc 正确大小的缓冲区，避免了调用者猜测大小的问题。旧版本需要 PATH_MAX 大小的缓冲，但某些系统上 PATH_MAX 可能不够大或未定义。缺点是需要调用者 free 返回值，且分配可能失败。

18.14

Parsing Pathname Strings: dirname() and basename()

本章：TLPI 第 18 章

本节讲什么

dirname() 和 basename() 将路径字符串拆分为目录部分和文件部分。这些是纯字符串操作，不检查文件系统。

要点

函数原型

#include <libgen.h>
char *dirname(char *path);   // 返回目录部分
char *basename(char *path);   // 返回文件部分

拆分规则

输入 path dirname basename
/usr/lib/libc.so /usr/lib libc.so
/usr/ / usr
usr . usr
/ / /
. . .
.. . ..
NULL . .
"" (空串) . .

重要陷阱

  1. 修改输入：dirname/basename 可能修改 path 字符串内容
  2. 返回静态/内部指针：返回值指向 path 内部或静态缓冲，下次调用会覆盖
  3. 线程不安全：POSIX 版本使用静态缓冲；GNU 版本修改输入

安全使用模式

/* 必须先复制再使用 */
char *dircpy = strdup(path);
char *basecopy = strdup(path);
char *dir = dirname(dircpy);
char *base = basename(basecopy);
/* 使用 dir 和 base */
free(dircpy);
free(basecopy);

POSIX vs GNU 版本

特性 POSIX (<libgen.h>) GNU (<libgen.h>)
basename 行为 可能修改输入 修改输入
线程安全 否 否（修改输入）
线程安全替代 basename_r（GNU） 无标准替代
链接 需要 -lgen（某些系统） 内置

HFT / 嵌入式关联

  • 配置文件路径处理：从完整路径提取文件名做 key
  • 日志轮转：分离日志目录和文件名
  • 路径拼接：dirname + 新文件名

代码示例

#include <libgen.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    const char *path = argc > 1 ? argv[1] : "/usr/lib/libc.so";

    /* 必须复制：dirname 和 basename 都可能修改输入 */
    char *d1 = strdup(path);
    char *d2 = strdup(path);

    char *dir = dirname(d1);
    char *base = basename(d2);

    printf("Path:  %s\n", path);
    printf("Dir:   %s\n", dir);
    printf("Base:  %s\n", base);

    free(d1);
    free(d2);
    return 0;
}

手动实现（线程安全版）

/* 线程安全的 basename */
const char *safe_basename(const char *path) {
    if (!path || !*path) return ".";
    const char *end = path + strlen(path);
    /* 去掉尾部斜杠 */
    while (end > path && end[-1] == '/') end--;
    if (end == path) return "/";
    const char *start = end;
    while (start > path && start[-1] != '/') start--;
    return start;
}

代码自测

为什么 dirname 和 basename 不能同时处理同一个 path 变量？

dirname() 和 basename() 都可能修改输入字符串的内容（插入 '\0' 来截断）。如果先调用 dirname(path) 修改了 path，再调用 basename(path) 得到的是修改后的结果而非原始路径的文件名。正确做法是分别传入 strdup 复制的副本。

练习 1：硬链接计数

题目：为什么删除一个文件后磁盘空间不一定释放？在什么条件下空间才会真正释放？

参考答案

Linux 文件系统中，每个 inode 有一个 i_link_count 字段记录硬链接数。unlink() 只是将链接计数减 1，当计数归零时才真正释放磁盘空间。此外，即使链接计数为 0，如果有进程仍打开该文件（持有 fd），inode 也不会释放，直到所有 fd 关闭。因此文件删除后空间释放需满足两个条件：(1) 硬链接计数为 0；(2) 没有进程持有打开的 fd。

练习 2：符号链接 vs 硬链接

题目：创建一个指向目录的符号链接，然后对它执行 stat() 和 lstat()，结果有何不同？为什么？

参考答案

stat() 跟随符号链接，返回目标目录的属性：st_mode 包含 S_IFDIR，st_ino 是目录的 inode。 lstat() 不跟随链接，返回符号链接文件本身的属性：st_mode 包含 S_IFLNK，st_ino 是链接文件的 inode，st_size 是目标路径字符串长度。

symlink /home/user/data → ./link
stat("./link")  → st_mode: S_IFDIR, st_ino: 目录inode
lstat("./link") → st_mode: S_IFLNK, st_ino: 链接inode

练习 3：nftw 遍历

题目：使用 nftw() 编写一个程序，统计指定目录下各类文件的数量（普通文件、目录、符号链接、其他），不跟随符号链接。

参考答案 
#include <ftw.h>
#include <stdio.h>
#include <sys/stat.h>

static int counts[4] = {0};  /* regular, dir, link, other */

static int cb(const char *path, const struct stat *sb,
              int typeflag, struct FTW *ftwbuf) {
    if (S_ISREG(sb->st_mode))        counts[0]++;
    else if (S_ISDIR(sb->st_mode))  counts[1]++;
    else if (S_ISLNK(sb->st_mode))  counts[2]++;
    else                             counts[3]++;
    return 0;
}

int main(int argc, char *argv[]) {
    const char *dir = argc > 1 ? argv[1] : ".";
    /* FTW_PHYS: 不跟随符号链接 */
    if (nftw(dir, cb, 64, FTW_PHYS) == -1) {
        perror("nftw");
        return 1;
    }
    printf("Regular: %d\nDir:     %d\nLink:    %d\nOther:   %d\n",
           counts[0], counts[1], counts[2], counts[3]);
    return 0;
}

练习 4：安全 chroot

题目：以下 chroot 代码有何安全问题？给出修正版本。

chroot("/srv/sandbox");
execl("/bin/sh", "sh", NULL);
参考答案

问题： 1. 未 chdir("/")：CWD 可能在沙箱外，通过相对路径访问外部文件 2. 未丢弃权限：root 可以通过 chroot(".") 逃逸 3. 未验证路径：/bin/sh 可能不存在

修正：

chroot("/srv/sandbox");
chdir("/");                    /* 将 CWD 设到新根内 */
setgid(1000);                  /* 先丢 GID */
setgroups(0, NULL);            /* 清除补充组 */
setuid(1000);                  /* 再丢 UID（不可逆） */
execl("/bin/sh", "sh", NULL); /* 执行 */
