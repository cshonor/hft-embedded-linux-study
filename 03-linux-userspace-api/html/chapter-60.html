#include <stdio.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <string.h>
#include <signal.h>
#include <sys/wait.h>
#include <stdlib.h>

/* Ch60 服务器设计 — 迭代 vs 并发 vs prefork。
 * 演示并发服务器 (fork per client) 模型。
 * 编译: gcc -o ch60_demo ch60_demo.c */

#define PORT 9998
#define MAX_CLIENTS 3

/* SIGCHLD 处理器: 回收僵尸子进程 */
void sigchld_handler(int sig) {
    while (waitpid(-1, NULL, WNOHANG) > 0)
        ;
}

int main(void) {
    signal(SIGCHLD, sigchld_handler);

    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {
        .sin_family = AF_INET,
        .sin_addr.s_addr = htonl(INADDR_LOOPBACK),
        .sin_port = htons(PORT)
    };
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("Concurrent server on port %d (fork per client)\n", PORT);

    /* 模拟: 创建 MAX_CLIENTS 个客户端子进程 */
    for (int i = 0; i < MAX_CLIENTS; i++) {
        pid_t cpid = fork();
        if (cpid == 0) {
            /* 客户端 */
            int cfd = socket(AF_INET, SOCK_STREAM, 0);
            connect(cfd, (struct sockaddr *)&addr, sizeof(addr));
            char msg[32];
            snprintf(msg, sizeof(msg), "client %d", i);
            write(cfd, msg, strlen(msg) + 1);
            char buf[64];
            int n = read(cfd, buf, sizeof(buf));
            if (n > 0) printf("Client %d: reply='%s'\n", i, buf);
            close(cfd);
            _exit(0);
        }
    }

    /* 服务器: accept 循环, 每个 client fork 一个子进程处理 */
    int handled = 0;
    while (handled < MAX_CLIENTS) {
        struct sockaddr_in cli;
        socklen_t len = sizeof(cli);
        int cfd = accept(sfd, (struct sockaddr *)&cli, &len);
        if (cfd < 0) continue;

        pid_t pid = fork();
        if (pid == 0) {
            /* 子进程: 处理单个客户端 */
            close(sfd);  /* 子进程不需要监听套接字 */
            char buf[64];
            int n = read(cfd, buf, sizeof(buf));
            if (n > 0) printf("Server child: handling '%s'\n", buf);
            write(cfd, "served", 7);
            close(cfd);
            _exit(0);
        }
        close(cfd);  /* 父进程不需要连接套接字 */
        handled++;
    }

    /* 等待所有客户端完成 */
    while (wait(NULL) > 0)
        ;
    close(sfd);
    return 0;
}

60.1

迭代服务器

本章：TLPI 第 60 章 — Sockets: Server Design - 上一节：本章首页 - 下一节：60.2 –60.3 fork 并发

本节讲什么

accept → 同步处理完 → close(conn)。无并发；慢客户端堵死全体。 适合极低流量短请求。示例见 Ch59 tcp_iter_sv。 ---

要点

accept → 同步处理完 → close(conn)。无并发；慢客户端堵死全体。
适合极低流量短请求。示例见 Ch59 tcp_iter_sv。

/* 迭代服务器：accept → 处理 → close → 循环 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9995) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("iterative server: port 9995\n");
    printf("accept -> handle -> close -> loop\n");
    printf("(one client at a time, others wait in backlog)\n");

    /* 实际循环（简化演示） */
    for (int i = 0; i < 1; i++) {
        int cfd = accept(sfd, NULL, NULL);
        if (cfd < 0) continue;
        char buf[64];
        ssize_t n = recv(cfd, buf, sizeof(buf)-1, 0);
        if (n > 0) { buf[n]='\0'; printf("handled: %s\n", buf); }
        send(cfd, "done", 5, 0);
        close(cfd);
    }

    close(sfd);
    return 0;
}

–60.3 fork 并发

本节讲什么

accept → fork 子: close(listen); handle; close(conn); _exit 父: close(conn); 继续 accept

要点

accept → fork
  子: close(listen); handle; close(conn); _exit
  父: close(conn); 继续 accept
要点
fd 引用 fork 后 +1；双方关无用端
僵尸 SIGCHLD 里循环 waitpid(-1,NULL,WNOHANG)
隔离 强；崩溃不易拖垮主服务
代价 fork/页表贵；海量连接受限

/* fork 并发服务器：每个连接一个子进程 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/wait.h>

static void handle_client(int cfd) {
    char buf[128];
    ssize_t n = recv(cfd, buf, sizeof(buf)-1, 0);
    if (n > 0) {
        buf[n] = '\0';
        printf("child %d: received '%s'\n", getpid(), buf);
    }
    send(cfd, "fork reply", 11, 0);
    close(cfd);
    _exit(0);
}

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9994) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("fork server: port 9994\n");

    /* SIGCHLD 回收子进程 */
    signal(SIGCHLD, SIG_IGN);  /* 简化：自动回收 */

    for (int i = 0; i < 3; i++) {
        int cfd = accept(sfd, NULL, NULL);
        if (cfd < 0) continue;

        pid_t pid = fork();
        if (pid == 0) {
            close(sfd);  /* 子关闭监听 fd */
            handle_client(cfd);
        }
        close(cfd);  /* 父关闭连接 fd */
    }

    close(sfd);
    return 0;
}

60.3

A Concurrent TCP Echo Server

本章：TLPI 第 60 章 — Server Design

本节讲什么

介绍并发 TCP echo 服务器的实现：每接受一个连接 fork 一个子进程处理，展示经典并发服务器模式。

要点

并发服务器模型

1. socket()     → 创建监听套接字
2. bind()       → 绑定地址
3. listen()     → 开始监听
4. 循环:
   a. accept()   → 接受新连接
   b. fork()     → 子进程处理连接
   c. 父进程 close(accept 返回的 fd) → 继续 accept
   d. 子进程 close(listen_fd) → 处理 echo
   e. 子进程完毕 → exit

fork 并发服务器代码结构

int listen_fd = socket(...);
bind(listen_fd, ...);
listen(listen_fd, SOMAXCONN);

while (1) {
    int conn_fd = accept(listen_fd, NULL, NULL);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程 */
        close(listen_fd);          /* 不需要监听 fd */
        echo_service(conn_fd);     /* 处理连接 */
        close(conn_fd);
        _exit(0);
    }
    /* 父进程 */
    close(conn_fd);               /* 父不需要连接 fd */
    /* 信号处理回收子进程（SIGCHLD） */
}

子进程回收

/* SIGCHLD 处理器：回收所有已退出子进程 */
void sigchld_handler(int sig) {
    int saved_errno = errno;  /* 防止 errno 被覆盖 */
    while (waitpid(-1, NULL, WNOHANG) > 0)
        ;
    errno = saved_errno;
}

/* 安装时使用 SA_RESTART 自动重启被中断的系统调用 */
struct sigaction sa;
sa.sa_handler = sigchld_handler;
sigemptyset(&sa.sa_mask);
sa.sa_flags = SA_RESTART;
sigaction(SIGCHLD, &sa, NULL);

accept 与 SIGCHLD 竞态

问题 说明
accept 被 SIGCHLD 中断 SA_RESTART 自动重启，无问题
慢速 accept + SIGCHLD 如果无 SA_RESTART，accept 返回 EINTR

fork 并发 vs 其他模型

模型 优点 缺点 适用
fork 每连接 简单、隔离好 fork 开销、内存开销 简单服务
线程每连接 比 fork 轻 线程安全复杂度 中等并发
线程池 固定线程数 连接数受限于线程数 中高并发
select/poll 单进程多连接 fd 限制、性能 低并发
epoll 高效事件驱动 异步编程复杂 高并发
SO_REUSEPORT 多进程 epoll 需内核支持 HFT/高并发

HFT / 嵌入式关联

  • HFT 不用 fork 每连接：fork 开销约 50-100μs，HFT 需要更低延迟
  • HFT 用 epoll：单线程事件驱动 + 非阻塞 I/O，避免上下文切换
  • SO_REUSEPORT：多进程/多线程各自 epoll 同一端口，内核负载均衡
  • 嵌入式：资源受限用 select 或单线程状态机
  • UDP 并发：HFT 行情数据用 UDP，无连接，不需 fork

代码示例

#include "tlpi_hdr.h"
#include <sys/socket.h>
#include <signal.h>
#include <netinet/in.h>

void echo_service(int fd) {
    char buf[4096];
    ssize_t n;
    while ((n = read(fd, buf, sizeof(buf))) > 0) {
        if (write(fd, buf, n) != n) break;
    }
}

void sigchld_handler(int sig) {
    int saved_errno = errno;
    while (waitpid(-1, NULL, WNOHANG) > 0) ;
    errno = saved_errno;
}

int main(void) {
    /* 安装 SIGCHLD 处理器 */
    struct sigaction sa;
    sa.sa_handler = sigchld_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_RESTART;
    sigaction(SIGCHLD, &sa, NULL);

    /* 创建监听套接字 */
    int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(listen_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {0};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(8080);
    bind(listen_fd, (struct sockaddr *)&addr, sizeof(addr));
    listen(listen_fd, 128);

    printf("listening on :8080\n");

    while (1) {
        int conn_fd = accept(listen_fd, NULL, NULL);
        if (conn_fd == -1) {
            if (errno == EINTR) continue;
            errExit("accept");
        }

        pid_t pid = fork();
        if (pid == 0) {
            close(listen_fd);
            echo_service(conn_fd);
            close(conn_fd);
            _exit(0);
        }
        close(conn_fd);
    }
    return 0;
}

代码自测

1. 为什么子进程要 close(listen_fd)？

子进程从 fork 继承了父进程所有 fd，包括 listen_fd。子进程不需要监听套接字，不关闭会导致 fd 泄漏——每个子进程占用一个 listen_fd，大量子进程会耗尽 fd 表。close 后子进程只持有连接 fd，处理完即可退出。

2. SIGCHLD 处理器中为什么要用 while + WNOHANG？

多个子进程可能同时退出，产生一个 SIGCHLD 信号（可能被合并）。如果只 waitpid 一次，可能只回收一个子进程，其余成为僵尸进程。while (waitpid(-1, NULL, WNOHANG) > 0) 循环回收所有已退出的子进程。WNOHANG 确保没有子进程退出时立即返回 0，不阻塞。

3. 为什么 SIGCHLD 处理器要保存 errno？

waitpid 可能修改 errno。如果信号中断了主进程的某个系统调用（如 accept），该调用返回后需要检查 errno。如果 SIGCHLD 处理器的 waitpid 修改了 errno，主进程会看到错误的 errno 值。在处理器入口保存 errno、出口恢复是异步信号安全编程的标准做法。

60.4

多线程

本章：TLPI 第 60 章 — Sockets: Server Design - 上一节：60.2 –60.3 fork 并发 - 下一节：60.5 事件驱动（select/poll 入门）

本节讲什么

accept → pthread_create(handler)。 坑：传 &connfd → 主循环覆盖 → 多线程同一 fd。 修：堆上分配 fd 副本，或加锁传递。

要点

accept → pthread_create(handler)。
坑：传 &connfd → 主循环覆盖 → 多线程同一 fd。
修：堆上分配 fd 副本，或加锁传递。
优：轻、易共享状态；劣：一线程崩整进程、须加锁、线程过多调度差。进阶：线程池。

/* 线程并发服务器：每个连接一个线程 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <pthread.h>

static void *handle_client(void *arg) {
    int cfd = *(int *)arg;
    free(arg);

    char buf[128];
    ssize_t n = recv(cfd, buf, sizeof(buf)-1, 0);
    if (n > 0) {
        buf[n] = '\0';
        printf("thread: received '%s'\n", buf);
    }
    send(cfd, "thread reply", 13, 0);
    close(cfd);
    return NULL;
}

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9993) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("thread server: port 9993\n");

    for (int i = 0; i < 3; i++) {
        int cfd = accept(sfd, NULL, NULL);
        if (cfd < 0) continue;

        int *arg = malloc(sizeof(int));
        *arg = cfd;

        pthread_t tid;
        pthread_create(&tid, NULL, handle_client, arg);
        pthread_detach(tid);  /* 自动回收 */
    }

    close(sfd);
    sleep(1);  /* 等线程完成 */
    return 0;
}

#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/select.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9992) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("select server: port 9992\n");

    fd_set readfds;
    int clients[FD_SETSIZE];
    for (int i = 0; i < FD_SETSIZE; i++) clients[i] = -1;
    int maxfd = sfd;

    for (int loop = 0; loop < 3; loop++) {
        FD_ZERO(&readfds);
        FD_SET(sfd, &readfds);

        for (int i = 0; i < FD_SETSIZE; i++) {
            if (clients[i] > 0) {
                FD_SET(clients[i], &readfds);
                if (clients[i] > maxfd) maxfd = clients[i];
            }
        }

        struct timeval tv = { .tv_sec = 5, .tv_usec = 0 };
        int n = select(maxfd + 1, &readfds, NULL, NULL, &tv);
        if (n <= 0) { printf("select timeout\n"); break; }

        if (FD_ISSET(sfd, &readfds)) {
            int cfd = accept(sfd, NULL, NULL);
            printf("new client: fd=%d\n", cfd);
            for (int i = 0; i < FD_SETSIZE; i++) {
                if (clients[i] < 0) { clients[i] = cfd; break; }
            }
        }

        for (int i = 0; i < FD_SETSIZE; i++) {
            if (clients[i] > 0 && FD_ISSET(clients[i], &readfds)) {
                char buf[64];
                ssize_t len = recv(clients[i], buf, sizeof(buf)-1, 0);
                if (len <= 0) { close(clients[i]); clients[i] = -1; }
                else { buf[len]='\0'; printf("recv: %s\n", buf); }
            }
        }
    }

    close(sfd);
    return 0;
}

    printf("3. Thread per connection:\n");

    printf("   cons: shared state, race conditions\n\n");

    setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    printf("SO_REUSEADDR: allows quick restart after crash\n");
    close(fd);

}
