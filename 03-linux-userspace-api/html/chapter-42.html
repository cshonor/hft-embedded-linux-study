代码示例

#include <stdio.h>
#include <dlfcn.h>
#include <stdlib.h>
#include <link.h>

/* Ch42 共享库高级 — dlopen 标志/符号版本/初始化/析构。
 * 演示 RTLD_GLOBAL vs RTLD_LOCAL + 构造/析构函数。
 * 编译: gcc -o ch42_demo ch42_demo.c -ldl */

/* 共享库的构造和析构函数 */
__attribute__((constructor))
static void init_func(void) {
    printf("[constructor] Library loaded\n");
}

__attribute__((destructor))
static void fini_func(void) {
    printf("[destructor] Library unloaded\n");
}

/* 遍历已加载的共享库 */
static int callback(struct dl_phdr_info *info, size_t size, void *data) {
    static int count = 0;
    if (info->dlpi_name && info->dlpi_name[0])
        printf("  [%d] %s\n", count++, info->dlpi_name);
    return 0;
}

int main(void) {
    /* RTLD_NOW: 立即解析所有符号 (vs RTLD_LAZY 延迟) */
    void *h1 = dlopen("libm.so.6", RTLD_NOW | RTLD_LOCAL);
    if (h1) {
        printf("Loaded libm.so.6 with RTLD_NOW|RTLD_LOCAL\n");

        /* 深绑定: RTLD_DEEPBIND 优先搜索自己 */
        void *h2 = dlopen("libc.so.6", RTLD_NOW | RTLD_GLOBAL);
        if (h2) {
            printf("Loaded libc.so.6 with RTLD_GLOBAL (symbols available to later libs)\n");
            dlclose(h2);
        }
        dlclose(h1);
    }

    /* 遍历已加载库 */
    printf("\nLoaded shared objects:\n");
    dl_iterate_phdr(callback, NULL);

    /* RPATH/RUNPATH: 编译时指定搜索路径
     * gcc -Wl,-rpath,/custom/lib -o demo demo.c
     * ldd demo 查看依赖
     * LD_LIBRARY_PATH 也可以设置运行时搜索路径
     */
    printf("\nRuntime search paths:\n");
    printf("  LD_LIBRARY_PATH: %s\n", getenv("LD_LIBRARY_PATH") ?: "(not set)");
    printf("  Use 'ldd <binary>' to see dependencies\n");
    return 0;
}

42.1

–42.4 dlopen 家族（`<dlfcn.h>`，`-ldl`）

本章：TLPI 第 42 章 — Advanced Features of Shared Libraries - 上一节：本章首页 - 下一节：42.5 dladdr

本节讲什么

void dlopen(const char filename, int flags); void dlsym(void handle, const char symbol); int dlclose(void handle);

要点

void *dlopen(const char *filename, int flags);
void *dlsym(void *handle, const char *symbol);
int   dlclose(void *handle);
char *dlerror(void);   /* 读一次即清零 */

Flags

组 标志 含义
绑定 RTLD_LAZY 首次调用再解析（默认）
RTLD_NOW 打开时解析完；缺符号立即失败
作用域 RTLD_LOCAL 符号不进全局空间（默认）
RTLD_GLOBAL 后续库可解析本库符号
附加 RTLD_NODELETE dlclose 后仍不卸载映射 
dlopen("./plugin.so", RTLD_NOW | RTLD_LOCAL);

插件实践：RTLD_NOW | RTLD_LOCAL + 通过 dlsym 取一个 plugin_api 结构体指针，避免库间互相 dlsym 依赖。

dlsym

返回 void*；转函数指针宜用 typedef + 中转赋值。C++ 需 extern "C" 防 name mangling。

dlerror

每次调用会清除状态。模板：先 dlerror() 清空 → dlsym → 再 dlerror() 判错（区分「符号不存在」与「合法 NULL」）。

dlclose

引用计数到 0 → 跑 destructor → 默认卸载。
陷阱：插件线程/回调仍存活 → SIGSEGV。先注销回调/停线程，或 RTLD_NODELETE。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 dlopen/dlsym/dlclose/dlerror；RTLD_NOW|RTLD_LOCAL 插件实践；dlclose 引用计数到 0 才卸载
HFT 策略/网关插件化 dlopen 热加载，RTLD_NOW 立即解析避免运行时缺符号；插件线程存活时 dlclose 会 SIGSEGV

代码示例

#include <stdio.h>
#include <dlfcn.h>
#include <stdlib.h>
#include <string.h>
/* dlopen/dlsym/dlclose/dlerror: 运行时动态加载 */
int main(void) {
    /* dlerror: 清除之前的错误 */
    dlerror();

    /* dlopen: 打开共享库
     * RTLD_LAZY: 延迟解析符号
     * RTLD_NOW: 立即解析所有符号
     * RTLD_GLOBAL: 符号可供后续加载的库使用
     * RTLD_LOCAL: 符号仅本库可见 (默认)
     */
    void *handle = dlopen("libm.so.6", RTLD_LAZY);
    if (!handle) {
        fprintf(stderr, "dlopen: %s\n", dlerror());
        return 1;
    }

    /* dlsym: 查找符号 */
    dlerror();  /* 清除错误 */
    double (*sqrt_fn)(double) = (double (*)(double))dlsym(handle, "sqrt");
    const char *err = dlerror();
    if (err) {
        fprintf(stderr, "dlsym: %s\n", err);
        dlclose(handle);
        return 1;
    }

    printf("sqrt(144.0) = %.2f\n", sqrt_fn(144.0));

    /* dlsym 查找变量 */
    dlerror();
    int *errno_loc = dlsym(handle, "errno");  /* 可能找不到 */
    printf("dlsym('errno'): %s\n", dlerror() ? "not found" : "found");

    /* dlclose: 关闭 (引用计数减1) */
    dlclose(handle);
    printf("Library closed\n");

    /* 编译时链接: gcc -ldl */
    /* 运行时搜索路径: LD_LIBRARY_PATH / RPATH / RUNPATH */
    return 0;
}

42.2

Controlling Symbol Visibility

本章：TLPI 第 42 章 — Shared Libraries: Advanced Topics

本节讲什么

介绍如何控制共享库中符号的可见性：哪些符号导出给使用者，哪些隐藏为内部实现。

要点

符号可见性控制方式

方式 说明 优点
默认（全导出） 所有非 static 符号导出 简单
__attribute__((visibility("hidden"))) 隐藏单个符号 精细控制
-fvisibility=hidden + __attribute__((visibility("default"))) 默认隐藏，显式导出 最佳实践
版本脚本（version script） 声明式批量控制 最灵活
static 关键字 编译单元内隐藏 C 语言基础

visibility 属性

/* 导出（可被使用者链接） */
__attribute__((visibility("default")))
int public_function(void);

/* 隐藏（仅库内部使用） */
__attribute__((visibility("hidden")))
int internal_helper(void);
visibility 说明
default 导出，外部可见（默认）
hidden 不导出，仅本共享库内可见
protected 导出但不允许被覆盖（BIND_NOW 时不插入 PLT）
internal 不导出，不能通过函数指针调用

默认隐藏 + 显式导出模式

# 编译时默认隐藏所有符号
gcc -fvisibility=hidden -shared -o libfoo.so foo.c

/* foo.c */
/* 头文件中用宏统一标记导出 */
#define API_EXPORT __attribute__((visibility("default")))

API_EXPORT int public_api(void);   /* 导出 */
int internal_helper(void) { ... }  /* 自动隐藏（因 -fvisibility=hidden） */

符号可见性检查

# 查看导出的动态符号
nm -D libfoo.so
# 或
readelf -s libfoo.so | grep FUNC

# 比较隐藏前后
gcc -shared -o libfoo.so foo.c              # 全导出
gcc -fvisibility=hidden -shared -o libfoo.so foo.c  # 默认隐藏

隐藏的好处

好处 说明
减少 PLT/GOT 隐藏符号不经过 PLT，直接调用
优化 编译器可内联隐藏函数
ABI 稳定 只暴露公共 API，内部可自由修改
安全 不暴露内部实现细节
加载速度 动态链接器解析更少符号

HFT / 嵌入式关联

  • 延迟优化：隐藏内部符号减少 PLT 跳转，每个 PLT 调用约 2-5ns
  • ABI 稳定：HFT 策略库只导出公共接口，内部实现可自由修改
  • 嵌入式：减少符号表大小，节省 flash 空间
  • 最佳实践：-fvisibility=hidden + API_EXPORT 宏统一管理

代码示例

/* foo.h */
#pragma once
#define API_EXPORT __attribute__((visibility("default")))

API_EXPORT int foo_init(void);
API_EXPORT void foo_destroy(void);

/* foo.c (编译: gcc -fvisibility=hidden -shared -fPIC -o libfoo.so foo.c) */
#include "foo.h"

static int initialized = 0;
static void *internal_state = NULL;   /* hidden（static + -fvisibility=hidden） */

int internal_helper(int x) {         /* hidden（因 -fvisibility=hidden） */
    return x * 2;
}

API_EXPORT int foo_init(void) {       /* default（显式导出） */
    if (initialized) return 0;
    internal_state = malloc(1024);
    initialized = 1;
    return 0;
}

API_EXPORT void foo_destroy(void) {   /* default */
    free(internal_state);
    initialized = 0;
}

代码自测

1. -fvisibility=hidden 和 static 的区别？

static 限制符号在编译单元（.c 文件）内可见，链接器完全不可见。visibility("hidden") 允许符号在多个 .o 文件间通过链接器可见（可被同一库的其他文件调用），但不导出到动态符号表（外部使用者不可见）。共享库由多个 .o 组成时，hidden 更适合内部跨文件调用。

42.3

Linker Version Scripts

本章：TLPI 第 42 章 — Shared Libraries: Advanced Topics

本节讲什么

介绍 GNU ld 版本脚本（version script，.map 文件）：声明式控制符号导出和符号版本管理。

要点

版本脚本语法

/* libfoo.map */
LIBFOO_1.0 {
    global:
        foo_init;
        foo_destroy;
        foo_process;
    local:
        *;       /* 其他所有符号隐藏 */
};

LIBFOO_1.1 {
    global:
        foo_batch;     /* 1.1 新增 */
} LIBFOO_1.0;          /* 依赖 1.0 */

编译使用

gcc -shared -fPIC -Wl,--version-script,libfoo.map -o libfoo.so foo.o

版本脚本能力

能力 说明
控制符号导出 global: 列出导出符号
隐藏内部符号 local: *; 隐藏其他所有
符号版本管理 允许多版本同名符号共存
ABI 兼容 新版本可依赖旧版本

符号版本

/* foo.c */
__asm__(".symver foo_init_v1,foo_init@LIBFOO_1.0");
int foo_init_v1(void) { return 1; }

__asm__(".symver foo_init_v2,foo_init@@LIBFOO_1.1");  /* @@ 表示默认版本 */
int foo_init_v2(void) { return 2; }

/* libfoo.map */
LIBFOO_1.0 {
    global: foo_init;
    local: *;
};
LIBFOO_1.1 {
    global: foo_init;
} LIBFOO_1.0;
  • @@ 表示默认版本（链接器选择）
  • @ 表示非默认版本（特定版本请求时选择）
  • 运行时 dlsym 默认返回 @@ 版本

版本脚本 vs visibility 属性

特性 版本脚本 visibility 属性
位置 外部 .map 文件 源码内
粒度 按通配符批量 每符号
版本管理 支持 不支持
维护 集中管理 分散在源码
可读性 声明式 代码侵入

实际应用：glibc 符号版本

# 查看 glibc 的符号版本
readelf -s /lib/x86_64-linux-gnu/libc.so.6 | grep printf
# printf@GLIBC_2.2.5
# __printf_chk@GLIBC_2.7

glibc 用版本脚本保持 ABI 兼容：旧程序链接 printf@GLIBC_2.2.5，新程序链接 @@GLIBC_2.2.5（默认版本）。

HFT / 嵌入式关联

  • ABI 管理：HFT 策略库用版本脚本管理接口版本，支持热更新
  • 符号隐藏：local: *; 等价于 -fvisibility=hidden，集中管理
  • 嵌入式：版本脚本减少导出符号，节省空间
  • glibc 兼容：链接特定 glibc 版本避免运行时版本不匹配

代码示例

# 完整示例

# libfoo.map
LIBFOO_1.0 {
    global:
        foo_init;
        foo_process;
    local:
        *;
};

LIBFOO_1.1 {
    global:
        foo_batch;
} LIBFOO_1.0;

# 编译
gcc -c -fPIC foo.c -o foo.o
gcc -shared -Wl,--version-script,libfoo.map -o libfoo.so foo.o

# 验证
nm -D libfoo.so
# 应该只有 foo_init, foo_process, foo_batch

代码自测

1. local: *; 的作用是什么？

local: *; 是通配符，将所有未在 global: 中列出的符号标记为 local（隐藏）。这是一种"白名单"模式——默认隐藏所有符号，只导出明确列出的。等价于 -fvisibility=hidden + 逐个标记导出，但集中在外部脚本文件中管理。

2. @@ 和 @ 的区别？

@@ 标记默认版本：链接器在未指定版本时自动选择该版本。一个符号只能有一个 @@ 版本。@ 标记非默认版本：只有显式请求该版本的程序才会链接到它。运行时 dlsym 返回 @@ 版本的符号。

42.4

Initialization and Finalization Functions

本章：TLPI 第 42 章 — Shared Libraries: Advanced Topics

本节讲什么

介绍共享库的初始化（constructor）和终结化（destructor）函数：在库加载/卸载时自动执行的函数，用于替代 _init()/_fini()。

要点

constructor / destructor 属性

__attribute__((constructor))
void my_init(void) {
    /* 在库加载时（dlopen 或程序启动时）执行 */
}

__attribute__((destructor))
void my_fini(void) {
    /* 在库卸载时（dlclose 或程序退出时）执行 */
}

执行时机

事件 触发
dlopen() 加载库 constructor 执行
dlclose() 卸载库 destructor 执行
程序启动（静态链接的库） constructor 在 main 之前执行
程序退出（exit 或 return） destructor 执行
_exit() destructor 不执行

优先级

__attribute__((constructor(101)))  /* 优先级高（先执行） */
void early_init(void) { ... }

__attribute__((constructor(65535))) /* 优先级低（后执行） */
void late_init(void) { ... }
优先级 执行顺序 说明
1-100 保留（系统库） 不要使用
101 最高 最先执行
65535 最低 最后执行
不指定 默认 65535 —

constructor vs _init() vs LD_PRELOAD

方式 说明
constructor 属性 现代 GCC 推荐
_init() / _fini() 旧方式，不推荐
__attribute__((constructor)) 自动调用，无需手动

dlopen 时的初始化

void *handle = dlopen("./libfoo.so", RTLD_NOW);
/* → libfoo 的 constructor 在此处执行 */
...
dlclose(handle);
/* → libfoo 的 destructor 在此处执行 */

RTLD_NOW vs RTLD_LAZY

flag constructor 时机
RTLD_NOW dlopen 时立即执行
RTLD_LAZY 符号首次使用时（但 constructor 仍在 dlopen 时执行）

HFT / 嵌入式关联

  • 策略库加载：HFT 用 dlopen 加载策略库，constructor 初始化策略状态
  • 资源管理：constructor 分配内存/打开连接，destructor 释放
  • 嵌入式：避免 constructor，显式调用 init/destroy 更可控
  • 顺序依赖：constructor 间有优先级依赖时需小心

代码示例

/* libfoo.c */
#include <stdio.h>
#include <stdlib.h>

static void *foo_state = NULL;

__attribute__((constructor(200)))
void foo_init(void) {
    fprintf(stderr, "[foo] constructor\n");
    foo_state = malloc(4096);
}

__attribute__((destructor(200)))
void foo_destroy(void) {
    fprintf(stderr, "[foo] destructor\n");
    free(foo_state);
}

/* 公共 API */
int foo_process(const char *data) {
    /* 使用 foo_state */
    return 0;
}

/* main.c */
#include <dlfcn.h>
#include <stdio.h>

int main(void) {
    void *handle = dlopen("./libfoo.so", RTLD_NOW);
    if (!handle) { fprintf(stderr, "%s\n", dlerror()); return 1; }

    /* constructor 已在 dlopen 时执行 */

    int (*process)(const char *) = dlsym(handle, "foo_process");
    process("hello");

    dlclose(handle);  /* destructor 在此执行 */
    return 0;
}

代码自测

1. constructor 和 C++ 全局对象构造有什么关系？

C++ 全局对象的构造函数通过 .init_array 段实现，执行时机与 constructor 属性类似（在 main 之前）。但两者不同：constructor 用 GCC 属性，C++ 全局对象用编译器生成的构造器。执行顺序：constructor 优先级 < C++ 全局对象构造 < main。

2. _exit() 会执行 destructor 吗？

不会。_exit() 直接进入内核，不执行 atexit 处理器、stdio flush、destructor。只有 exit() 或 return 从 main 返回才会执行 destructor。如果需要确保 destructor 执行，不要用 _exit()。

要点

    const char *dli_fname;

地址 → 所属库/符号；用于日志、简易栈信息。

HFT / 嵌入式关联

路线 要点
嵌入式 dladdr 地址→所属库/符号（Dl_info），用于日志、简易栈信息
HFT 崩溃/性能日志用 dladdr 反查函数地址归属库，辅助定位热点

代码示例

#include <stdio.h>
#include <dlfcn.h>
#include <string.h>
/* dladdr: 通过函数地址反查所在的库和符号 */
/* 需要编译为 position-independent: gcc -fPIC -pie -o demo demo.c -ldl */

int my_function(void) { return 42; }

int main(void) {
    /* dladdr: 已知函数地址, 查找所在库信息 */
    Dl_info info;

    /* 查询 libc 的 printf */
    if (dladdr((void *)printf, &info)) {
        printf("printf:\n");
        printf("  library:  %s\n", info.dli_fname);
        printf("  base:     %p\n", info.dli_fbase);
        printf("  nearest:  %s\n", info.dli_sname ? info.dli_sname : "(unknown)");
        printf("  addr:     %p\n", info.dli_saddr);
    }

    /* 查询自己的函数 */
    if (dladdr((void *)my_function, &info)) {
        printf("\nmy_function:\n");
        printf("  library:  %s\n", info.dli_fname);
        printf("  nearest:  %s\n", info.dli_sname ? info.dli_sname : "(unknown)");
    }

    /* 查询 main 本身 */
    if (dladdr((void *)main, &info)) {
        printf("\nmain:\n");
        printf("  library:  %s\n", info.dli_fname);
    }
    return 0;
}
