章节目标

分清线程安全 / 可重入；pthread_once；TSD 四 API 与 strerror 改造范式；对比 __thread TLS；关联 C++/errno/嵌入式与低延迟选型。

与 C++ / 嵌入式 / HFT

栈
C++ thread_local ≈ TLS；可直接调 pthread；勿混两套线程生命周期模型
用户态 only 内核驱动无 POSIX TSD
HFT __thread 放连接/缓冲，少锁、少抖动；禁 strtok/localtime 一类

实验清单

  1. TSD 版「线程安全缓冲」
  2. __thread 对比
  3. （选）静态缓冲多线程错乱复现

背诵卡

# 要点
1 可重入 ⊂ 更强；线程安全可用锁/TLS
2 用 _r 或自带缓冲，别用静态内部状态 API
3 pthread_once + key_create 改造老接口
4 key 全局、值每线程；退出跑 destructor
5 key_delete 不析构各线程数据
6 低延迟优先 __thread / thread_local

代码示例

#include <stdio.h>
#include <pthread.h>
#include <string.h>
#include <unistd.h>

/* Ch31 线程安全与线程特定数据 (TSD)。
 * 演示 strerror_r 线程安全版 + pthread_key 线程局部存储。
 * 编译: gcc -o ch31_demo ch31_demo.c -lpthread */

static pthread_key_t tsd_key;

/* 线程特定数据: 每个线程有独立的副本 */
void tsd_destructor(void *value) {
    printf("TSD destructor: freeing %s\n", (char *)value);
    free(value);
}

void *worker(void *arg) {
    int id = *(int *)arg;

    /* 线程安全函数: strerror_r 而非 strerror */
    char errbuf[256];
    strerror_r(ENOENT, errbuf, sizeof(errbuf));
    printf("Thread %d: strerror_r(ENOENT) = %s\n", id, errbuf);

    /* 设置线程特定数据 */
    char *msg = malloc(64);
    snprintf(msg, 64, "TSD data for thread %d", id);
    pthread_setspecific(tsd_key, msg);

    /* 读取线程特定数据 */
    char *my_data = pthread_getspecific(tsd_key);
    printf("Thread %d: TSD = %s\n", id, my_data);

    return NULL;
}

int main(void) {
    pthread_key_create(&tsd_key, tsd_destructor);

    pthread_t t1, t2;
    int id1 = 1, id2 = 2;
    pthread_create(&t1, NULL, worker, &id1);
    pthread_create(&t2, NULL, worker, &id2);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    pthread_key_delete(tsd_key);
    return 0;
}

31.1

线程安全与可重入

本章：TLPI 第 31 章 — Thread Safety and Per-Thread Storage - 上一节：本章首页 - 下一节：31.2 pthread_once

本节讲什么

不安全经典：strtok、localtime、ctime → 用 strtok_r、localtime_r 等，状态由调用者提供。 并发代码：禁止无 _r / 无文档保证的老接口。 ---

要点

含义
线程安全 多线程并发调用无 UB；可用锁/原子/TLS 保护共享
可重入 信号打断自身也安全；不用静态全局可变状态 → 通常也线程安全（更强）

不安全经典：strtok、localtime、ctime → 用 strtok_r、localtime_r 等，状态由调用者提供。
并发代码：禁止无 _r / 无文档保证的老接口。

代码示例

#include <pthread.h>
#include <stdio.h>
/* 非线程安全: 使用静态缓冲 */
char *unsafe(void) { static char buf[32]; snprintf(buf, 32, "tid=%lu", (unsigned long)pthread_self()); return buf; }
/* 线程安全: 调用者提供缓冲 */
void safe(char *buf, size_t n) { snprintf(buf, n, "tid=%lu", (unsigned long)pthread_self()); }
int main(void) {
    char buf[32];
    safe(buf, sizeof(buf));
    printf("safe: %s\n", buf);
    return 0;
}

31.2

`pthread_once`

本章：TLPI 第 31 章 — Thread Safety and Per-Thread Storage - 上一节：31.1 线程安全与可重入 - 下一节：31.3 TSD：pthread_key_*（核心）

本节讲什么

pthread_once_t once = PTHREAD_ONCE_INIT; / 必须静态初始化 / int pthread_once(pthread_once_t once_control, void (init_routine)(void)); 多线程下 init 只跑一次；常与 pthread_key_create 搭配。

要点

pthread_once_t once = PTHREAD_ONCE_INIT;   /* 必须静态初始化 */
int pthread_once(pthread_once_t *once_control, void (*init_routine)(void));

多线程下 init 只跑一次；常与 pthread_key_create 搭配。
once_control 不要 malloc。

代码示例

#include <pthread.h>
#include <stdio.h>
static pthread_once_t once = PTHREAD_ONCE_INIT;
void init_func(void) { printf("初始化 (只执行一次)\n"); }
void *worker(void *arg) { pthread_once(&once, init_func); return NULL; }
int main(void) {
    pthread_t t1, t2, t3;
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_create(&t3, NULL, worker, NULL);
    pthread_join(t1, NULL); pthread_join(t2, NULL); pthread_join(t3, NULL);
    return 0;
}

31.3

TSD：`pthread_key_*`（核心）

本章：TLPI 第 31 章 — Thread Safety and Per-Thread Storage - 上一节：31.2 pthread_once - 下一节：31.4 静态 TLS：__thread / _Thread_local

本节讲什么

痛点：接口不能加参数，但内部曾靠静态变量 → 用「每线程一份指针」。 int pthread_key_create(pthread_key_t key, void (destructor)(void )); int pthread_setspecific(pthread_key_t key, const void value);

要点

痛点：接口不能加参数，但内部曾靠静态变量 → 用「每线程一份指针」。

int pthread_key_create(pthread_key_t *key, void (*destructor)(void *));
int pthread_setspecific(pthread_key_t key, const void *value);
void *pthread_getspecific(pthread_key_t key);
int pthread_key_delete(pthread_key_t key);
点
key 进程内全局一份
值 每线程独立指针
线程退出 value≠NULL → 调 destructor
key_delete 不对现有线程跑析构；易泄漏

范式（类 strerror）：

  1. pthread_once → key_create
  2. 首次：malloc + setspecific
  3. 之后：getspecific
  4. 退出：destructor free

Demo：code/strerror_tsd.c

代码示例

#include <stdio.h>
#include <pthread.h>
#include <string.h>
#include <stdlib.h>
/* 线程特定数据 (TSD): pthread_key_create/setspecific/getspecific */
static pthread_key_t key;

void destructor(void *value) {
    printf("TSD destructor: freeing '%s'\n", (char *)value);
    free(value);
}

void *worker(void *arg) {
    int id = *(int *)arg;

    /* 每个线程设置自己的 TSD */
    char *msg = malloc(64);
    snprintf(msg, 64, "private data of thread %d", id);
    pthread_setspecific(key, msg);

    /* 读取自己的 TSD */
    char *my = pthread_getspecific(key);
    printf("Thread %d: TSD = %s\n", id, my);
    return NULL;
}

int main(void) {
    pthread_key_create(&key, destructor);

    pthread_t t1, t2;
    int id1 = 1, id2 = 2;
    pthread_create(&t1, NULL, worker, &id1);
    pthread_create(&t2, NULL, worker, &id2);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    pthread_key_delete(key);
    return 0;
}

31.4

静态 TLS：`__thread` / `_Thread_local`

本章：TLPI 第 31 章 — Thread Safety and Per-Thread Storage - 上一节：31.3 TSD：pthread_key_*（核心） - 下一节：31.5 易错 / 原理

本节讲什么

static __thread char buf[256]; / GCC；C11: _Thread_local / Demo：code/thread_local_buf.c ---

要点

static __thread char buf[256];   /* GCC；C11: _Thread_local */
pthread_key __thread
时机 运行时 key 编译期变量
开销 函数查表 通常更快（x86_64 FS/GS；ARM 也有 TLS 寄存器）
场景 改老接口、动态数量 已知线程私有全局/静态；HFT 优先

Demo：code/thread_local_buf.c

代码示例

#include <pthread.h>
#include <stdio.h>
static __thread int tls_var = 0;  /* 线程局部存储 */
void *worker(void *arg) {
    tls_var = (int)(long)arg;
    printf("线程 %ld: tls_var=%d\n", (long)arg, tls_var);
    return NULL;
}
int main(void) {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, worker, (void *)1);
    pthread_create(&t2, NULL, worker, (void *)2);
    pthread_join(t1, NULL); pthread_join(t2, NULL);
    printf("主线程 tls_var=%d (独立)\n", tls_var);
    return 0;
}

31.5

易错 / 原理

本章：TLPI 第 31 章 — Thread Safety and Per-Thread Storage - 上一节：31.4 静态 TLS：__thread / _Thread_local - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

  1. key_delete 不跑析构 2. destructor 内勿再 setspecific 同 key（循环析构风险） 3. errno：每线程一份（glibc 用 TLS），不是普通全局 int

要点

  1. key_delete 不跑析构
  2. destructor 内勿再 setspecific 同 key（循环析构风险）
  3. errno：每线程一份（glibc 用 TLS），不是普通全局 int

代码示例

#include <pthread.h>
#include <stdio.h>
/* 线程安全原则:
 * 1. 不用静态/全局可变数据, 或加锁
 * 2. 用 __thread 线程局部存储
 * 3. 调用者提供缓冲
 * 4. 用 pthread_once 做一次性初始化
 */
static pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;
static int counter = 0;
int safe_increment(void) {
    pthread_mutex_lock(&lock);
    int v = ++counter;
    pthread_mutex_unlock(&lock);
    return v;
}
int main(void) { printf("count=%d\n", safe_increment()); return 0;
}

练习 1：线程安全的 strerror

实现一个线程安全的 strerror_r() 包装函数，使用 per-thread 变量存储错误字符串。

参考答案

练习 1 参考代码

#include <stdio.h>
#include <string.h>
#include <errno.h>

/* 方法 1：使用 strerror_r（POSIX 版本） */
void safe_strerror(int err, char *buf, size_t len) {
    strerror_r(err, buf, len);
}

/* 方法 2：使用 __thread（TLS） */
static __thread char err_buf[128];

const char *my_strerror(int err) {
    strerror_r(err, err_buf, sizeof(err_buf));
    return err_buf;
}

练习 2 参考代码

#include <pthread.h>

static pthread_key_t errno_key;

void errno_destructor(void *ptr) {
    free(ptr);
}

void tsd_init(void) {
    pthread_key_create(&errno_key, errno_destructor);
}

int tsd_get_errno(void) {
    int *p = pthread_getspecific(errno_key);
    if (p == NULL) {
        p = malloc(sizeof(int));
        *p = 0;
        pthread_setspecific(errno_key, p);
    }
    return *p;
}

void tsd_set_errno(int e) {
    int *p = pthread_getspecific(errno_key);
    if (p == NULL) {
        p = malloc(sizeof(int));
        pthread_setspecific(errno_key, p);
    }
    *p = e;
}

练习 3 答案

特性 可重入 (reentrant) 线程安全 (thread-safe)
定义 可被信号处理器安全调用 可被多线程安全调用
静态/全局变量 不使用 可用锁保护
标准 严格 较宽松
关系 可重入 → 线程安全 线程安全 ≠ 可重入

示例： - 可重入：strtok_r(str, delim, &saveptr) — 不使用静态变量，可被信号处理器调用 - 线程安全但不可重入：strtok + mutex — 多线程安全，但信号处理器中调用可能死锁（如果被信号打断的线程已持有 mutex） - 既不可重入也不线程安全：strtok() — 使用静态变量

HFT / 嵌入式关联

  • TLS：HFT 使用 __thread 存储每线程上下文（如线程 ID、策略句柄）
  • 无锁设计：HFT 热路径避免锁，用 per-thread 数据 + 原子操作
  • errno：系统调用的 errno 在 glibc 中是 TLS 变量，每线程独立
  • 嵌入式：__thread 需要编译器和运行时支持，裸机/RTOS 可能不支持
