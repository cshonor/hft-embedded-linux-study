章节目标

sockaddr_un；STREAM/DGRAM；权限；socketpair；抽象命名空间；工程选型。

选型（嵌入式 + HFT）

  1. 本机组件通信 → UDS（优先于 127.0.0.1 TCP）
  2. 守护进程 → 抽象名（Linux）
  3. 父子临时双向 → socketpair
  4. 要消息边界且本机 → UNIX DGRAM

思考题要点

  1. 崩溃残留：启动 unlink 或改用抽象名。
  2. 抽象不受 umask/文件权限。
  3. fork 后关无用 fd；需半关闭用 shutdown。

背诵卡

# 要点
1 路径型 bind 前 unlink
2 UNIX DGRAM ≠ UDP：本地可靠
3 connect 需写权限 + 目录 x
4 socketpair：无名互连对
5 抽象：path[0]=0，无文件
6 本机延迟：UDS > loopback TCP

代码示例

#include <stdio.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

/* Ch57 UNIX 域套接字 — socketpair + fd 传递 (SCM_RIGHTS)。
 * UNIX 域套接字仅限本机, 但支持传递文件描述符。
 * 编译: gcc -o ch57_demo ch57_demo.c */

int main(void) {
    /* socketpair: 创建一对已连接的套接字 */
    int sv[2];
    if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) < 0) {
        perror("socketpair"); return 1;
    }

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 通过 sv[1] 接收 fd */
        close(sv[0]);

        /* 接收普通消息 */
        char buf[64];
        read(sv[1], buf, 12);
        printf("Child: received message: %s\n", buf);

        /* 接收文件描述符 (SCM_RIGHTS) */
        struct msghdr msg = {0};
        char mbuf[256];
        struct iovec iov = { .iov_base = buf, .iov_len = sizeof(buf) };
        msg.msg_iov = &iov;
        msg.msg_iovlen = 1;
        msg.msg_control = mbuf;
        msg.msg_controllen = sizeof(mbuf);

        recvmsg(sv[1], &msg, 0);

        struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
        if (cmsg && cmsg->cmsg_level == SOL_SOCKET &&
            cmsg->cmsg_type == SCM_RIGHTS) {
            int recv_fd;
            memcpy(&recv_fd, CMSG_DATA(cmsg), sizeof(int));
            printf("Child: received fd=%d\n", recv_fd);

            /* 使用接收到的 fd 读文件 */
            char fbuf[64];
            int n = read(recv_fd, fbuf, sizeof(fbuf) - 1);
            if (n > 0) { fbuf[n] = '\0'; printf("Child: fd content: %s\n", fbuf); }
            close(recv_fd);
        }
        close(sv[1]);
        _exit(0);
    }

    /* 父进程: 通过 sv[0] 发送消息 + fd */
    close(sv[1]);

    /* 发送普通消息 */
    write(sv[0], "hello fd!", 12);

    /* 打开一个文件, 通过套接字传递 fd */
    int file_fd = open("/etc/hostname", O_RDONLY);

    /* 构造带 ancillary data 的消息 */
    struct msghdr msg = {0};
    char buf[] = "fd coming";
    struct iovec iov = { .iov_base = buf, .iov_len = sizeof(buf) };
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;

    char cbuf[CMSG_SPACE(sizeof(int))];
    msg.msg_control = cbuf;
    msg.msg_controllen = sizeof(cbuf);

    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    cmsg->cmsg_level = SOL_SOCKET;
    cmsg->cmsg_type = SCM_RIGHTS;
    cmsg->cmsg_len = CMSG_LEN(sizeof(int));
    memcpy(CMSG_DATA(cmsg), &file_fd, sizeof(int));

    sendmsg(sv[0], &msg, 0);
    printf("Parent: sent fd=%d\n", file_fd);

    close(file_fd);
    close(sv[0]);
    waitpid(pid, NULL, 0);
    return 0;
}

`struct sockaddr_un`

本章：TLPI 第 57 章 — Sockets: UNIX Domain - 上一节：本章首页 - 下一节：57.2 STREAM

本节讲什么

struct sockaddr_un { sa_family_t sun_family; / AF_UNIX / char sun_path[108]; / Linux 常见长度；勿假定可移植 /

要点

struct sockaddr_un {
    sa_family_t sun_family;  /* AF_UNIX */
    char sun_path[108];      /* Linux 常见长度；勿假定可移植 */
};
  • 路径用 snprintf 防溢出
  • bind 的 addrlen = offsetof(sun_path) + strlen(path) + 1（路径型）；抽象名另算（含前导 \0 后的名字长度）
  • 路径型：文件系统可见 socket 文件

代码示例


/* struct sockaddr_un 结构体与 bind */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>

int main(void) {
    int fd = socket(AF_UNIX, SOCK_STREAM, 0);
    if (fd < 0) { perror("socket"); exit(1); }

    /* sockaddr_un 结构 */
    struct sockaddr_un addr;
    memset(&addr, 0, sizeof(addr));
    addr.sun_family = AF_UNIX;  /* 必须是 AF_UNIX */
    strncpy(addr.sun_path, "/tmp/ch57_addr.sock", sizeof(addr.sun_path) - 1);

    printf("sizeof(struct sockaddr_un) = %zu\n", sizeof(addr));
    printf("sun_family = %d (AF_UNIX)\n", addr.sun_family);
    printf("sun_path = '%s'\n", addr.sun_path);
    printf("sun_path max len = %zu\n", sizeof(addr.sun_path));

    unlink(addr.sun_path);  /* bind 前清理 */
    if (bind(fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind"); exit(1);
    }
    printf("bind OK\n");

    close(fd);
    unlink(addr.sun_path);
    return 0;
}

57.2

STREAM

本章：TLPI 第 57 章 — Sockets: UNIX Domain - 上一节：57.1 struct sockaddr_un - 下一节：57.3 DGRAM（高频）

本节讲什么

同 TCP 流程：bind→listen→accept / connect；全双工字节流。 退出不删 socket 文件 → 再 bind EADDRINUSE → bind 前 unlink。 相对 pipe：双向。路径流式示例见 Ch56 us_stream。

要点

同 TCP 流程：bind→listen→accept / connect；全双工字节流。
退出不删 socket 文件 → 再 bind EADDRINUSE → bind 前 unlink。
相对 pipe：双向。路径流式示例见 Ch56 us_stream。

代码示例


/* AF_UNIX SOCK_STREAM 服务器+客户端完整示例 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/wait.h>

int main(void) {
    const char *path = "/tmp/ch57_stream.sock";
    unlink(path);

    int sfd = socket(AF_UNIX, SOCK_STREAM, 0);
    struct sockaddr_un addr = { .sun_family = AF_UNIX };
    strncpy(addr.sun_path, path, sizeof(addr.sun_path) - 1);
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_UNIX, SOCK_STREAM, 0);
        struct sockaddr_un caddr = { .sun_family = AF_UNIX };
        strncpy(caddr.sun_path, path, sizeof(caddr.sun_path) - 1);
        connect(cfd, (struct sockaddr *)&caddr, sizeof(caddr));

        /* 发送多段数据 */
        send(cfd, "part1-", 6, 0);
        send(cfd, "part2-", 6, 0);
        send(cfd, "part3", 5, 0);

        char buf[64];
        ssize_t n = recv(cfd, buf, sizeof(buf) - 1, 0);
        if (n > 0) { buf[n] = '\0'; printf("client: reply='%s'\n", buf); }
        close(cfd);
        _exit(0);
    }

    int afd = accept(sfd, NULL, NULL);
    /* 流式 socket 保证有序可靠，但可能合并多次 send */
    char buf[128];
    ssize_t n = recv(afd, buf, sizeof(buf) - 1, 0);
    if (n > 0) { buf[n] = '\0'; printf("server: received='%s' (%zd bytes)\n", buf, n); }
    send(afd, "ack", 4, 0);
    close(afd);

    wait(NULL);
    close(sfd);
    unlink(path);
    return 0;
}

57.3

DGRAM（高频）

本章：TLPI 第 57 章 — Sockets: UNIX Domain - 上一节：57.2 STREAM - 下一节：57.4 权限

本节讲什么

两端宜各自 bind。包大小受缓冲限制。 ---

要点

网络 UDP UNIX DGRAM
可靠/有序 否 是（内核转发）
边界 有 有
满队列 可丢 发送方可阻塞

两端宜各自 bind。包大小受缓冲限制。

代码示例


/* AF_UNIX SOCK_DGRAM 数据报通信 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/wait.h>

int main(void) {
    const char *srv_path = "/tmp/ch57_dgram_srv.sock";
    const char *cli_path = "/tmp/ch57_dgram_cli.sock";
    unlink(srv_path); unlink(cli_path);

    int sfd = socket(AF_UNIX, SOCK_DGRAM, 0);
    struct sockaddr_un saddr = { .sun_family = AF_UNIX };
    strncpy(saddr.sun_path, srv_path, sizeof(saddr.sun_path) - 1);
    bind(sfd, (struct sockaddr *)&saddr, sizeof(saddr));

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_UNIX, SOCK_DGRAM, 0);
        struct sockaddr_un caddr = { .sun_family = AF_UNIX };
        strncpy(caddr.sun_path, cli_path, sizeof(caddr.sun_path) - 1);
        bind(cfd, (struct sockaddr *)&caddr, sizeof(caddr));

        struct sockaddr_un dest = { .sun_family = AF_UNIX };
        strncpy(dest.sun_path, srv_path, sizeof(dest.sun_path) - 1);

        /* sendto 指定目标地址 */
        sendto(cfd, "dgram msg", 10, 0,
               (struct sockaddr *)&dest, sizeof(dest));
        printf("client: sent dgram\n");

        char buf[64];
        ssize_t n = recvfrom(cfd, buf, sizeof(buf) - 1, 0, NULL, NULL);
        if (n > 0) { buf[n] = '\0'; printf("client: reply='%s'\n", buf); }
        close(cfd);
        _exit(0);
    }

    /* recvfrom 获取发送方地址 */
    char buf[64];
    struct sockaddr_un src;
    socklen_t srclen = sizeof(src);
    ssize_t n = recvfrom(sfd, buf, sizeof(buf) - 1, 0,
                         (struct sockaddr *)&src, &srclen);
    if (n > 0) { buf[n] = '\0'; printf("server: received='%s' from %s\n", buf, src.sun_path); }

    /* 回复 */
    sendto(sfd, "dgram reply", 12, 0,
           (struct sockaddr *)&src, srclen);

    wait(NULL);
    close(sfd);
    unlink(srv_path); unlink(cli_path);
    return 0;
}

57.4

权限

本章：TLPI 第 57 章 — Sockets: UNIX Domain - 上一节：57.3 DGRAM（高频） - 下一节：57.5 socketpair

本节讲什么

connect/发数据需对 socket 文件 写；父目录 x。 默认创建常很宽 → 用 umask 收紧。简易访问控制，免自建鉴权。 ---

要点

connect/发数据需对 socket 文件 写；父目录 x。
默认创建常很宽 → 用 umask 收紧。简易访问控制，免自建鉴权。

代码示例


/* AF_UNIX socket 文件权限控制 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/stat.h>

int main(void) {
    int fd = socket(AF_UNIX, SOCK_STREAM, 0);
    struct sockaddr_un addr = { .sun_family = AF_UNIX };
    strncpy(addr.sun_path, "/tmp/ch57_perm.sock", sizeof(addr.sun_path) - 1);
    unlink(addr.sun_path);

    /* bind 创建 socket 文件，权限受 umask 影响 */
    mode_t old = umask(0077);  /* 只有 owner 可访问 */
    bind(fd, (struct sockaddr *)&addr, sizeof(addr));
    umask(old);

    /* 查看权限 */
    struct stat sb;
    stat(addr.sun_path, &sb);
    printf("socket file permissions: %04o\n", sb.st_mode & 0777);
    printf("(0077 umask -> only owner can connect)\n");

    /* 也可以 bind 后 chmod */
    chmod(addr.sun_path, 0666);
    stat(addr.sun_path, &sb);
    printf("after chmod 0666: %04o (all can connect)\n", sb.st_mode & 0777);

    close(fd);
    unlink(addr.sun_path);
    return 0;
}

57.5

`socketpair`

本章：TLPI 第 57 章 — Sockets: UNIX Domain - 上一节：57.4 权限 - 下一节：57.6 抽象命名空间（Linux）

本节讲什么

int socketpair(int domain, int type, int protocol, int sv[2]); 仅 AF_UNIX；一对已互连、无名（无路径文件）。父子 IPC、双向替代 pipe。 fork 后各方关闭不用的端（同 pipe 纪律）。

要点

int socketpair(int domain, int type, int protocol, int sv[2]);

仅 AF_UNIX；一对已互连、无名（无路径文件）。父子 IPC、双向替代 pipe。
fork 后各方关闭不用的端（同 pipe 纪律）。

常见陷阱

  1. SIGPIPE
send(fd, buf, n, 0);  /* 对端关闭 → SIGPIPE */
send(fd, buf, n, MSG_NOSIGNAL);  /* 安全 */

向已关闭连接的 socket 写数据触发 SIGPIPE。用 MSG_NOSIGNAL 或忽略 SIGPIPE。

  1. TIME_WAIT
bind(fd, (struct sockaddr*)&addr, len);
/* EADDRINUSE if port in TIME_WAIT */

服务端重启时端口可能还在 TIME_WAIT。用 SO_REUSEADDR 允许绑定。

  1. accept 返回值未检查
int cfd = accept(lfd, NULL, NULL);
/* cfd 可能 -1 (EMFILE/ENFILE) */

accept 可能因 fd 耗尽返回 -1。不检查直接用会导致后续操作失败。

自测题

题目 1（选择）： listen() 的第二个参数 backlog 表示什么？ A. 最大连接数 B. 等待 accept 的队列长度 C. 缓冲区大小 D. 超时秒数

参考答案

B。backlog 是已完成连接但尚未 accept 的队列长度。超过后新连接可能被拒绝。

题目 2（判断）： bind 可以绑定任意端口号。

参考答案

错误。1-1023 是特权端口，需要 root 权限。已占用的端口会返回 EADDRINUSE。

题目 3（踩坑）： 以下代码有什么问题？

int sfd = socket(AF_INET, SOCK_STREAM, 0);
connect(sfd, (struct sockaddr*)&addr, sizeof(addr));
send(sfd, "data", 4, 0);
参考答案

没有检查 connect 返回值。connect 可能失败（连接被拒绝、超时等），不检查直接 send 会导致错误。

题目 4（应用）： 写代码创建 TCP 服务端 socket 并监听。

参考答案 
int sfd = socket(AF_INET, SOCK_STREAM, 0);
int opt = 1;
setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
struct sockaddr_in addr = {0};
addr.sin_family = AF_INET;
addr.sin_port = htons(8080);
addr.sin_addr.s_addr = INADDR_ANY;

代码示例


/* socketpair 创建双向通信管道 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/wait.h>

int main(void) {
    int sv[2];
    /* socketpair 创建一对已连接的 socket */
    if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) < 0) {
        perror("socketpair"); exit(1);
    }
    printf("socketpair: sv[0]=%d, sv[1]=%d (connected)\n", sv[0], sv[1]);

    pid_t pid = fork();
    if (pid == 0) {
        close(sv[1]);  /* 子关闭另一端 */
        char buf[64];
        ssize_t n = read(sv[0], buf, sizeof(buf) - 1);
        if (n > 0) { buf[n] = '\0'; printf("child: received '%s'\n", buf); }
        write(sv[0], "child reply", 12);
        close(sv[0]);
        _exit(0);
    }

    close(sv[0]);  /* 父关闭另一端 */
    write(sv[1], "parent msg", 11);

    char buf[64];
    ssize_t n = read(sv[1], buf, sizeof(buf) - 1);
    if (n > 0) { buf[n] = '\0'; printf("parent: received '%s'\n", buf); }

    wait(NULL);
    close(sv[1]);
    return 0;
}

/* AF_UNIX 抽象命名空间：不创建文件系统条目 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/wait.h>

int main(void) {
    int sfd = socket(AF_UNIX, SOCK_STREAM, 0);

    /* 抽象命名空间：sun_path[0] = '\0'，后面是名字 */
    struct sockaddr_un addr;
    memset(&addr, 0, sizeof(addr));
    addr.sun_family = AF_UNIX;
    addr.sun_path[0] = '\0';  /* 抽象命名空间标志 */
    strncpy(addr.sun_path + 1, "ch57_abstract", sizeof(addr.sun_path) - 1);

    /* bind 不会在文件系统创建文件 */
    if (bind(sfd, (struct sockaddr *)&addr, sizeof(struct sockaddr_un)) < 0) {
        perror("bind"); exit(1);
    }
    listen(sfd, 1);
    printf("abstract socket: \\0ch57_abstract (no file on disk)\n");

    /* 优点：不需要清理文件，不会残留 */
    /* 生命周期：随 socket fd 关闭自动消失 */

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_UNIX, SOCK_STREAM, 0);
        struct sockaddr_un caddr;
        memset(&caddr, 0, sizeof(caddr));
        caddr.sun_family = AF_UNIX;
        caddr.sun_path[0] = '\0';
        strncpy(caddr.sun_path + 1, "ch57_abstract", sizeof(caddr.sun_path) - 1);
        connect(cfd, (struct sockaddr *)&caddr, sizeof(struct sockaddr_un));
        send(cfd, "abstract", 9, 0);
        close(cfd);
        _exit(0);
    }

    int afd = accept(sfd, NULL, NULL);
    char buf[32];
    ssize_t n = recv(afd, buf, sizeof(buf) - 1, 0);
    if (n > 0) { buf[n] = '\0'; printf("received: %s\n", buf); }
    close(afd);

    wait(NULL);
    close(sfd);
    /* 无需 unlink（抽象命名空间没有文件） */
    return 0;
}

代码自测

1. Unix 域套接字比 TCP 快多少？为什么？

Unix 域套接字比 TCP 快约 2-5 倍。原因：(1) 无需 TCP/IP 协议栈处理（无分片/重组/校验和/ACK）；(2) 数据在内核内存中直接拷贝，不经过网络层；(3) 无需路由表查找。典型延迟：Unix 域 1-5μs，TCP 本地 5-20μs。

2. SCM_RIGHTS 的作用是什么？

SCM_RIGHTS 允许通过 Unix 域套接字传递文件描述符。发送进程将一个 fd 序列化到辅助数据中，接收进程获得一个新的 fd（编号不同，指向同一个内核打开文件表项）。用途：传递共享内存 fd、网络连接 fd、pipe fd 等，实现高效的进程间资源共享。

3. 抽象地址和路径名绑定的区别？

路径名绑定在文件系统创建一个 sock 文件，需要清理（unlink），有文件权限控制。抽象地址绑定不创建文件，使用 \0 + 名字，无需清理（fd 关闭自动消失），无权限检查，但仅 Linux 支持。HFT 临时通信推荐抽象地址。

/* 发送 fd */
int send_fd(int sock, int fd) {
    struct msghdr msg = {0};
    char buf[CMSG_SPACE(sizeof(int))];
    struct iovec iov;
    char dummy = '\0';

    iov.iov_base = &dummy;
    iov.iov_len = 1;
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = buf;
    msg.msg_controllen = sizeof(buf);

    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    cmsg->cmsg_level = SOL_SOCKET;
    cmsg->cmsg_type = SCM_RIGHTS;
    cmsg->cmsg_len = CMSG_LEN(sizeof(int));
    memcpy(CMSG_DATA(cmsg), &fd, sizeof(int));

    return sendmsg(sock, &msg, 0);
}

/* 接收 fd */
int recv_fd(int sock) {
    struct msghdr msg = {0};
    char buf[CMSG_SPACE(sizeof(int))];
    char dummy;
    struct iovec iov = { &dummy, 1 };

    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = buf;
    msg.msg_controllen = sizeof(buf);

    recvmsg(sock, &msg, 0);
    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    int fd;
    memcpy(&fd, CMSG_DATA(cmsg), sizeof(int));
    return fd;
}
