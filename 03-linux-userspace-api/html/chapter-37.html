章节目标

守护特征；标准守护化步骤与 becomeDaemon()；syslog；SIGHUP/SIGTERM；PID 文件单实例；对比 daemon()。

易错清单

  1. 只 fork 一次
  2. 不 chdir
  3. 只 close 012 不重定向
  4. SIGHUP handler 里做重 IO
  5. 靠 daemon()
  6. 无 PID 锁多实例
  7. 依赖 stdout

实验清单

  1. 双重 fork 后查 SID/无 tty
  2. becomeDaemon
  3. syslog
  4. SIGHUP 标志位重载
  5. （选）PID 文件锁

背诵卡

# 要点
1 无终端；双重 fork + setsid
2 二次 fork：非会话首，防再抢 tty
3 chdir + 关 fd + 012→null
4 SIGHUP=重载；SIGTERM=退出

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <signal.h>
#include <syslog.h>

/* Ch37 守护进程 — 标准守护进程化步骤。
 * 1. fork + setsid 脱离终端
 * 2. 再次 fork 防止重新获取终端
 * 3. chdir("/") 不占用挂载点
 * 4. umask(0) 清除文件权限掩码
 * 5. 关闭/重定向标准 fd
 * 编译: gcc -o ch37_demo ch37_demo.c */

void daemonize(void) {
    /* Step 1: fork, 父进程退出 */
    pid_t pid = fork();
    if (pid < 0) exit(1);
    if (pid > 0) exit(0);

    /* Step 2: setsid — 成为新会话首领 */
    if (setsid() < 0) exit(1);

    /* Step 3: 再次 fork, 不再是会话首领 */
    pid = fork();
    if (pid < 0) exit(1);
    if (pid > 0) exit(0);

    /* Step 4: chdir + umask */
    chdir("/");
    umask(0);

    /* Step 5: 关闭所有继承的 fd, 重定向 0/1/2 */
    for (int fd = sysconf(_SC_OPEN_MAX); fd >= 0; fd--)
        close(fd);

    int nullfd = open("/dev/null", O_RDWR);
    if (nullfd >= 0) {
        dup2(nullfd, STDIN_FILENO);
        dup2(nullfd, STDOUT_FILENO);
        dup2(nullfd, STDERR_FILENO);
        if (nullfd > 2) close(nullfd);
    }
}

int main(void) {
    printf("Becoming a daemon...\n");
    fflush(stdout);

    daemonize();

    /* 现在是守护进程，用 syslog 输出日志 */
    openlog("ch37_demo", LOG_PID | LOG_CONS, LOG_DAEMON);
    syslog(LOG_NOTICE, "Daemon started, pid=%d", (int)getpid());

    /* 模拟工作: 每 5 秒写一次日志 */
    for (int i = 0; i < 3; i++) {
        sleep(5);
        syslog(LOG_INFO, "Daemon working: iteration %d", i);
    }

    syslog(LOG_NOTICE, "Daemon exiting");
    closelog();
    return 0;
}

/* 守护进程特征:
 * - 无控制终端 (PPID=1)

int main(void) {

    if (getppid() == 1) printf("→ 守护进程 (被 init 收养)\n");
    else printf("→ 普通进程\n");
    return 0;
}

37.2

标准 7 步（及原因）

本章：TLPI 第 37 章 — Daemons - 上一节：37.1 特征 - 下一节：37.3 编写规范

本节讲什么

glibc daemon()：常缺第二次 fork → 仍可能是会话首；生产宜自写 becomeDaemon。 Demo：code/become_daemon.c · code/mini_daemon.c ---

要点

步 动作 为何
1 fork，父 _exit shell 归还提示符；子非组 Leader → 可 setsid
2 setsid 新会话+新组，脱离控制终端
3 再 fork，中间父退出 最终进程非会话首，难再抢终端
4 umask(0) 自控创建 mode
5 chdir("/")（或 /var） 不挡 umount
6 关继承 fd 防泄漏
7 0/1/2 → /dev/null 用 open+dup2，勿只 close 被抢 fd

glibc daemon()：常缺第二次 fork → 仍可能是会话首；生产宜自写 becomeDaemon。

Demo：code/become_daemon.c · code/mini_daemon.c

HFT / 嵌入式关联

路线 要点
嵌入式 7 步：fork→setsid→再 fork→umask(0)→chdir("/")→关 fd→0/1/2 到 /dev/null；glibc daemon() 缺第二次 fork
HFT 生产用自写 becomeDaemon（含第二次 fork），彻底脱离终端防信号干扰

代码示例

#include <unistd.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <stdio.h>
/* 创建守护进程步骤 */
int main(void) {
    /* 1. fork + 父退出 */
    pid_t pid = fork();
    if (pid > 0) return 0;  /* 父退出 */
    if (pid < 0) return 1;

    /* 2. setsid: 脱离控制终端 */
    setsid();

    /* 3. fork 再次 (防止重新获取终端) */
    pid = fork();
    if (pid > 0) return 0;

    /* 4. chdir("/") */
    chdir("/");

    /* 5. 关闭/重定向 fd 0/1/2 */
    int devnull = open("/dev/null", O_RDWR);
    dup2(devnull, 0); dup2(devnull, 1); dup2(devnull, 2);

    /* 6. umask(0) */
    umask(0);

    /* 守护进程已创建 */
    for (;;) sleep(1);
    return 0;
}

要点

kill -HUP <pid>。handler 只设 volatile sig_atomic_t；主循环里重载。

Demo：code/daemon_sighup.c（演示标志位范式；可配合 becomeDaemon）

HFT / 嵌入式关联

路线 要点
嵌入式 kill -HUP 触发，handler 只设 volatile sig_atomic_t 标志，主循环里重载
HFT 配置热重载不重启进程，避免交易中断；handler 里只置标志不做重活（async-safe）

代码示例

#include <signal.h>
#include <unistd.h>
#include <stdio.h>
/* 守护进程无控制终端 → SIGHUP 不会因终端断开触发
 * 惯例: 用 SIGHUP 做配置重载
 */
int reload = 0;
void handler(int sig) { reload = 1; }
int main(void) {
    signal(SIGHUP, handler);
    printf("守护进程: kill -SIGHUP %d → 重载配置\n", getpid());
    for (;;) { if (reload) { printf("重载配置\n"); reload = 0; } sleep(1); }
    return 0;
}

37.6

Summary

本章：TLPI 第 37 章 — Daemons

本章回顾

Daemon 化流程

1. fork() + 父 exit      ← 脱离 shell
2. setsid()               ← 新会话 + 新进程组 + 脱离控制终端
3. fork() + 父 exit       ← 不再是会话首，不能再获得终端
4. chdir("/")              ← 不占用目录
5. umask(0)                ← 文件权限不受继承 umask 影响
6. close all fd / dup→null ← 关闭继承的 fd
7. open syslog             ← 日志重定向

Daemon 相关 API

函数 用途
setsid() 创建新会话
fork() × 2 脱离 shell 和会话首
chdir("/") 切换工作目录
umask(0) 清除 umask
open("/dev/null") + dup2 重定向 stdio
openlog()/syslog() 系统日志
umask() 设置 umask

日志系统

openlog("my_daemon", LOG_PID | LOG_NDELAY, LOG_DAEMON);
syslog(LOG_INFO, "started, pid=%d", getpid());
syslog(LOG_ERR, "error: %s", strerror(errno));
closelog();
facility 说明
LOG_DAEMON 系统服务（默认）
LOG_LOCAL0~LOG_LOCAL7 自定义
priority 级别
LOG_EMERG 紧急
LOG_ALERT 警报
LOG_CRIT 严重
LOG_ERR 错误
LOG_WARNING 警告
LOG_NOTICE 通知
LOG_INFO 信息
LOG_DEBUG 调试

SIGHUP 处理

Daemon 脱离控制终端后不会收到 SIGHUP，因此常用作"重新加载配置"信号：

void sighup_handler(int sig) {
    reload_config();
}

signal(SIGHUP, sighup_handler);

HFT / 嵌入式关联

  • HFT 服务 daemon 化：行情/策略/风控进程都需 daemon 化
  • 日志：用 syslog 或自定义日志框架（HFT 通常用自建日志，syslog 延迟不可控）
  • systemd 替代：现代 Linux 用 systemd 管理服务，Type=forking 或 Type=simple，自动处理 daemon 化
  • 嵌入式：BusyBox 的 start-stop-daemon 管理 daemon 生命周期
  • PID 文件：在 /var/run/myapp.pid 写入 PID，供运维脚本管理

代码自测

1. 为什么 daemon 需要两次 fork？

第一次 fork 脱离 shell。setsid 后成为会话首进程，但会话首进程可能重新获得控制终端（通过打开终端设备）。第二次 fork 后子进程不是会话首进程，不会再获得控制终端。这确保 daemon 永远脱离终端控制。

2. systemd Type=simple 和 Type=forking 的区别？
  • Type=simple：进程不 daemon 化，systemd 认为主进程是服务进程。简单直接，stdout/stderr 自动被 journald 捕获。
  • Type=forkting：传统 daemon 模式，主进程 fork 子进程后退出。systemd 需指定 PIDFile 追踪子进程。复杂但兼容传统 daemon。

现代推荐 Type=simple（或 Type=exec），不需要程序自己做 daemon 化——systemd 接管。

3. 为什么 daemon 要关闭所有 fd？
  1. 释放资源：从父进程继承的 fd（如终端、管道）可能不再需要
  2. 防止干扰：继承的终端 fd 可能影响终端操作
  3. 文件描述符泄漏：未关闭的 fd 占用资源
  4. 安全：防止通过继承的 fd 访问不应访问的文件

重定向 stdin/stdout/stderr 到 /dev/null 确保标准 I/O 操作不失败。

参考答案

练习 1 + 3 参考代码

#include "tlpi_hdr.h"
#include <sys/stat.h>
#include <syslog.h>

#define PID_FILE "/var/run/mydaemon.pid"

int write_pid_file(void) {
    int fd = open(PID_FILE, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) return -1;
    char buf[16];
    snprintf(buf, sizeof(buf), "%d\n", (int)getpid());
    write(fd, buf, strlen(buf));
    close(fd);
    return 0;
}

int become_daemon(void) {
    /* 1. fork + 父 exit */
    pid_t pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 2. setsid */
    if (setsid() == -1) return -1;

    /* 3. 第二次 fork */
    pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 4. chdir + umask */
    chdir("/");
    umask(0);

    /* 5. 关闭所有 fd */
    int maxfd = sysconf(_SC_OPEN_MAX);
    if (maxfd == -1) maxfd = 1024;
    for (int fd = 0; fd < maxfd; fd++) close(fd);

    /* 重定向到 /dev/null */
    open("/dev/null", O_RDWR);  /* stdin = fd 0 */
    dup2(0, 1);                 /* stdout */
    dup2(0, 2);                 /* stderr */

    /* 6. syslog */
    openlog("mydaemon", LOG_PID, LOG_DAEMON);
    return 0;
}

void cleanup(void) {
    unlink(PID_FILE);
    closelog();
}

int main(void) {
    if (become_daemon() != 0) {
        fprintf(stderr, "become_daemon failed\n");
        exit(1);
    }
    atexit(cleanup);
    write_pid_file();

    syslog(LOG_INFO, "daemon started, pid=%d", (int)getpid());

    /* 主循环 */
    for (int i = 0; i < 60; i++) {
        syslog(LOG_INFO, "tick %d", i);
        sleep(1);
    }

    syslog(LOG_INFO, "daemon exiting");
    return 0;
}

#include <signal.h>

static atomic_int reload_flag = 0;

void sighup_handler(int sig) {

    reload_flag = 1;

/* 在 main 中 */
struct sigaction sa;
sa.sa_handler = sighup_handler;
sigemptyset(&sa.sa_mask);
sa.sa_flags = SA_RESTART;
sigaction(SIGHUP, &sa, NULL);

/* 主循环 */
while (running) {

        reload_flag = 0;
