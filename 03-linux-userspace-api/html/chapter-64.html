导读

章节目标

主从模型；vs 管道；POSIX 打开流程；fork/setsid 架构；包模式与 winsize；BSD 旧式了解。

陷阱

  1. 忘 unlockpt
  2. 无 setsid → 无控制终端 / 作业控制失效
  3. 在 master 调 tcgetattr
  4. 子未关 master → PTY 不销毁
  5. 用 pipe 冒充交互终端

背诵卡

# 要点
1 master↔slave；slave=终端
2 openpt→grant→unlock→open
3 子 setsid+dup2(slave)+exec
4 交互 shell 必须 PTY
5 winsize → SIGWINCH
6 禁用 BSD 固定 pty 对

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/wait.h>
#include <string.h>
#include <termios.h>

/* Ch64 伪终端 — openpty/forkpty/posix_openpt。
 * 伪终端 (PTY) 让子进程以为自己在和真实终端交互。
 * 常用于 ssh, xterm, script 等程序。
 * 编译: gcc -o ch64_demo ch64_demo.c -lutil */

int main(void) {
    int master_fd, slave_fd;
    char slave_name[256];

    /* === openpty: 打开一对伪终端 === */
    if (openpty(&master_fd, &slave_fd, slave_name, NULL, NULL) < 0) {
        perror("openpty (need -lutil)");
        return 1;
    }

    printf("PTY pair created:\n");
    printf("  master fd: %d\n", master_fd);
    printf("  slave fd:  %d\n", slave_fd);
    printf("  slave name: %s\n", slave_name);

    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid == 0) {
        /* 子进程: 打开 slave 端, 作为自己的终端 */
        close(master_fd);

        /* 设置 slave 为控制终端 */
        setsid();
        ioctl(slave_fd, TIOCSCTTY, 0);

        /* 重定向 stdin/stdout/stderr 到 slave */
        dup2(slave_fd, STDIN_FILENO);
        dup2(slave_fd, STDOUT_FILENO);
        dup2(slave_fd, STDERR_FILENO);
        if (slave_fd > 2) close(slave_fd);

        /* 子进程以为自己在和真实终端交互 */
        /* 运行 tty 命令查看终端名 */
        execlp("tty", "tty", NULL);
        _exit(1);
    }

    /* 父进程: 从 master 端读取子进程输出 */
    close(slave_fd);

    char buf[256];
    ssize_t n = read(master_fd, buf, sizeof(buf) - 1);
    if (n > 0) {
        buf[n] = '\0';
        printf("Master received from PTY: %s", buf);
    }

    waitpid(pid, NULL, 0);

    /* === posix_openpt: POSIX 标准方式 === */
    printf("\n=== posix_openpt (POSIX standard) ===\n");
    int pty_master = posix_openpt(O_RDWR | O_NOCTTY);
    if (pty_master >= 0) {
        if (grantpt(pty_master) == 0 && unlockpt(pty_master) == 0) {
            char *name = ptsname(pty_master);
            printf("POSIX PTY slave: %s\n", name ? name : "?");
        }
        close(pty_master);
    }

    close(master_fd);
    return 0;
}

64.1

–64.2 概念

本章：TLPI 第 64 章 — Pseudoterminals - 上一节：本章首页 - 下一节：64.3 vs 管道

本节讲什么

一对虚拟字符设备： master（/dev/ptmx）↔ slave（/dev/pts/N）。 写 master → 到 slave；写 slave → 到 master。

要点

一对虚拟字符设备：
master（/dev/ptmx）↔ slave（/dev/pts/N）。
写 master → 到 slave；写 slave → 到 master。
Slave 走终端子系统（规范模式、ECHO、SIGINT…）。

场景：xterm、sshd、tmux、expect、script。

HFT / 嵌入式关联

路线 要点
嵌入式 无头设备用 PTY 给远程登录提供完整终端语义；master/slave 对让调试器、交互 shell 在嵌入式板子上有标准 tty 行为
HFT PTY 是 ssh/终端模拟器的底座，运维经 PTY 远程登录交易主机；master 侧可编程截获 slave 输入输出，用于会话回放与操作审计

代码示例


/* PTY 概念：master/slave 对，终端模拟 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>

int main(void) {
    printf("=== Pseudoterminal (PTY) Concepts ===\n\n");

    printf("PTY = pair of devices:\n");
    printf("  master: one end (like keyboard)\n");
    printf("  slave:  other end (looks like a real terminal)\n\n");

    printf("Typical use cases:\n");
    printf("  - terminal emulators (xterm, ssh)\n");
    printf("  - expect scripts (automate interactive programs)\n");
    printf("  - screen/tmux multiplexers\n\n");

    /* 实际创建一个 PTY */
    int master = posix_openpt(O_RDWR | O_NOCTTY);
    if (master < 0) { perror("posix_openpt"); exit(1); }
    grantpt(master);
    unlockpt(master);

    char *name = ptsname(master);
    printf("Created PTY:\n");
    printf("  master fd:  %d (write here → slave reads)\n", master);
    printf("  slave name: %s (open this to get terminal fd)\n", name);
    printf("  slave appears as a tty to programs\n");

    close(master);
    return 0;
}

64.2

Unix 98 Pseudoterminals

本章：TLPI 第 64 章

本节讲什么

Unix 98 标准伪终端（PTY）接口：posix_openpt() 创建主设备，grantpt()/unlockpt()/ptsname() 准备从设备，fork + setsid 建立伪终端会话。

要点

PTY 架构

┌───────────┐           ┌───────────┐
│ Master PTY │ ←数据流→  │ Slave PTY  │
│ (pty master)│          │ (pts/N)    │
└─────┬─────┘           └─────┬─────┘
      │                        │
  应用程序                  子进程（如 shell）
  写 master → 从 slave 读    读写 slave

Unix 98 PTY API

#include <stdlib.h>
int posix_openpt(int flags);         /* 打开主设备 */
int grantpt(int fd);                  /* 设置从设备权限 */
int unlockpt(int fd);                 /* 解锁从设备 */
char *ptsname(int fd);                /* 获取从设备路径 */

完整流程

/* 1. 打开主设备 */
int masterfd = posix_openpt(O_RDWR | O_NOCTTY);

/* 2. 设置从设备权限 */
grantpt(masterfd);

/* 3. 解锁从设备 */
unlockpt(masterfd);

/* 4. 获取从设备名 */
char *slavename = ptsname(masterfd);

/* 5. fork，子进程打开从设备 */
pid_t pid = fork();
if (pid == 0) {
    setsid();                    /* 新会话 */
    int slavefd = open(slavename, O_RDWR);
    dup2(slavefd, STDIN_FILENO);
    dup2(slavefd, STDOUT_FILENO);
    dup2(slavefd, STDERR_FILENO);
    close(slavefd);
    close(masterfd);
    execlp("bash", "bash", NULL);
}

/* 6. 父进程通过 masterfd 与子进程通信 */

grantpt 替代

方式 说明
grantpt(fd) 标准方式，修改从设备属主和权限
手动 chown/chmod 非标准，不推荐

现代 Linux 上 posix_openpt 已设置适当权限，grantpt 通常是无操作。

openpty() 便捷函数

#include <pty.h>
int masterfd, slavefd;
char name[256];
openpty(&masterfd, &slavefd, name, NULL, NULL);
/* 一步完成 master+slave 创建和打开 */

HFT / 嵌入式关联

  • 交易系统远程管理：通过 PTY 提供交互式 shell
  • 嵌入式设备调试终端：通过串口+PTY 桥接
  • 自动化测试：PTY 模拟终端交互
  • 终端录制/回放：捕获 PTY 数据流

代码示例

#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>

/* 创建 PTY 并启动子进程 */
int pty_fork(int *masterfd) {
    /* 1. 打开主设备 */
    *masterfd = posix_openpt(O_RDWR | O_NOCTTY);
    if (*masterfd < 0) return -1;

    /* 2. 准备从设备 */
    if (grantpt(*masterfd) < 0 ||
        unlockpt(*masterfd) < 0) {
        close(*masterfd);
        return -1;
    }

    char *slavename = ptsname(*masterfd);
    if (!slavename) { close(*masterfd); return -1; }

    /* 3. fork */
    pid_t pid = fork();
    if (pid < 0) { close(*masterfd); return -1; }

    if (pid == 0) {
        /* 子进程 */
        close(*masterfd);
        setsid();

        int slavefd = open(slavename, O_RDWR);
        if (slavefd < 0) _exit(1);

        dup2(slavefd, STDIN_FILENO);
        dup2(slavefd, STDOUT_FILENO);
        dup2(slavefd, STDERR_FILENO);
        close(slavefd);

        return 0;  /* 子进程返回 0 */
    }

    /* 父进程 */
    return pid;
}

int main() {
    int masterfd;
    pid_t pid = pty_fork(&masterfd);

    if (pid == 0) {
        execlp("bash", "bash", "--norc", NULL);
        _exit(1);
    }

    /* 父进程：读写 masterfd 与子进程通信 */
    char buf[4096];
    ssize_t n;
    while ((n = read(masterfd, buf, sizeof(buf))) > 0) {
        write(STDOUT_FILENO, buf, n);  /* 转发输出 */
    }
    close(masterfd);
    return 0;
}

代码自测

为什么子进程需要调用 setsid()？

setsid() 创建新会话，使子进程脱离控制终端。然后打开从设备作为新会话的控制终端。如果不调用 setsid()，子进程仍属于父进程的会话组，从设备不会成为控制终端，终端相关的功能（如 Ctrl+C 产生 SIGINT）不会正常工作。setsid() 还使子进程成为新会话的首进程和新进程组的组长，避免被父进程的作业控制干扰。

64.3

vs 管道

本章：TLPI 第 64 章 — Pseudoterminals - 上一节：64.1 –64.2 概念 - 下一节：64.4 –64.5 POSIX 打开 · 典型架构

本节讲什么

纯字节流 → pipe；跑交互 shell → PTY。 ---

要点

Pipe PTY slave
isatty false true
termios / 信号 / 作业控制 无 有

纯字节流 → pipe；跑交互 shell → PTY。

HFT / 嵌入式关联

路线 要点
嵌入式 管道只是纯字节流、无终端语义，跑交互程序（如 busybox shell）必须用 PTY，让程序自认连着真终端
HFT 自动化脚本驱动交互式 CLI（expect 式）用 PTY 而非管道：交互程序检测到 isatty 才会输出提示符并做行缓冲，避免协议错乱

代码示例


/* PTY vs pipe：PTY 提供终端语义 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <termios.h>

int main(void) {
    /* pipe: 原始字节流，无终端处理 */
    int pipefd[2];
    pipe(pipefd);
    write(pipefd[1], "hello\r\n", 7);
    char buf[16];
    int n = read(pipefd[0], buf, sizeof(buf));
    printf("pipe: read %d bytes: '%.*s' (raw, no terminal processing)\n",
           n, n, buf);
    close(pipefd[0]); close(pipefd[1]);

    /* PTY: 有终端处理（ICRNL 等） */
    int master = posix_openpt(O_RDWR | O_NOCTTY);
    grantpt(master);
    unlockpt(master);
    char *name = ptsname(master);
    int slave = open(name, O_RDWR);

    /* PTY slave 是一个真正的终端 */
    printf("\nPTY slave isatty: %s\n", isatty(slave) ? "yes" : "no");
    printf("pipe read end isatty: (would be no)\n");

    /* PTY 会做终端处理：ICRNL 把 \r 转成 \n */
    write(master, "hello\r", 6);
    n = read(slave, buf, sizeof(buf));
    printf("PTY: read %d bytes (ICRNL converts CR to NL)\n", n);

    close(slave);
    close(master);
    return 0;
}

64.4

–64.5 POSIX 打开 · 典型架构

本章：TLPI 第 64 章 — Pseudoterminals - 上一节：64.3 vs 管道 - 下一节：64.6 –64.7 特性

本节讲什么

mfd = posix_openpt(O_RDWR | O_NOCTTY); grantpt(mfd); unlockpt(mfd); slave = ptsname(mfd);

要点

mfd = posix_openpt(O_RDWR | O_NOCTTY);
grantpt(mfd); unlockpt(mfd);
slave = ptsname(mfd);
sfd = open(slave, O_RDWR | O_NOCTTY);

父持 master；子：close(master) → setsid() → dup2(slave, 0/1/2) → exec shell。
setsid 后打开 slave → 常成为控制终端。
termios 只改 slave，勿在 master 上改。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 标准打开四步 posix_openpt→grantpt→unlockpt→ptsname；子进程 setsid 后把 slave dup2 到 0/1/2 再 exec，是嵌入式 shell/登录的固定套路
HFT 用 forkpty/openpty 一步建 PTY 并 fork，让受管交易子进程自认连着终端、输出走 slave，父进程经 master 统一采集子进程输出做集中日志与风控审计

代码示例


/* PTY 架构：fork 子进程在 slave 端运行 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/wait.h>

int main(void) {
    int master = posix_openpt(O_RDWR | O_NOCTTY);
    if (master < 0) { perror("posix_openpt"); exit(1); }
    grantpt(master);
    unlockpt(master);
    char *slave_name = ptsname(master);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程：打开 slave，设为控制终端 */
        int slave = open(slave_name, O_RDWR);
        if (slave < 0) { perror("open slave"); _exit(1); }

        /* 重定向 stdin/stdout/stderr 到 slave */
        dup2(slave, STDIN_FILENO);
        dup2(slave, STDOUT_FILENO);
        dup2(slave, STDERR_FILENO);
        if (slave > STDERR_FILENO) close(slave);
        close(master);

        /* 子进程认为自己连着终端 */
        printf("child: isatty(stdout)=%d\n", isatty(STDOUT_FILENO));
        printf("child: running with PTY as terminal\n");
        _exit(0);
    }

    /* 父进程：通过 master 读取子进程输出 */
    close(STDIN_FILENO);  /* 简化：不重定向 */
    char buf[256];
    ssize_t n = read(master, buf, sizeof(buf) - 1);
    if (n > 0) {
        buf[n] = '\0';
        printf("parent read from PTY master:\n%s", buf);
    }

    wait(NULL);
    close(master);
    return 0;
}

64.5

Pseudoterminal I/O

本章：TLPI 第 64 章

本节讲什么

通过 PTY 主设备读写数据的语义：行模式/原始模式下的缓冲行为、数据传递规则，以及半关闭和关闭行为。

要点

数据流方向

写 masterfd → 从 slave 读出          （写入→终端输出）
从 slave 写入 → 从 masterfd 读出      （终端输入←键盘）
操作 方向
write(masterfd, ...) 数据出现在从设备端（如终端输出）
read(masterfd, ...) 读取从设备写入的数据（如键盘输入）
子进程 write(STDOUT) 父进程从 masterfd 读到
子进程 read(STDIN) 父进程写到 masterfd 的数据

规范模式下的行为

条件 read(masterfd) 返回
从设备写入了 NL 返回整行
从设备写入 EOF (^D) 返回已缓冲数据
无数据 阻塞

原始模式下的行为

条件 read(masterfd) 返回
有数据 立即返回可用数据
无数据 阻塞（或 VMIN/VTIME）
数据量 受 VMIN/VTIME 控制

关闭行为

操作 效果
关闭 masterfd 从设备 read 返回 0 (EOF)
子进程退出 masterfd read 返回 0 或 EIO
关闭 slavefd 不影响 masterfd

SIGPIPE 风险

/* 如果子进程已退出，write(masterfd) 可能触发 SIGPIPE */
/* 解决：忽略 SIGPIPE 或用 send(fd, ..., MSG_NOSIGNAL) */
signal(SIGPIPE, SIG_IGN);

数据完整性与 termios 处理

父进程 write(masterfd, "hello\n")
  → 内核 termios 处理（如果从设备端 OPOST on，NL→CRNL）
  → 子进程从 STDIN 读到 "hello\r\n"

注意：masterfd 的写入会经过从设备的 termios 处理（输出处理），如果 OPOST 开启，NL 会被转为 CR-NL。

HFT / 嵌入式关联

  • PTY 通信协议解析：设置原始模式避免 termios 干扰
  • 嵌入式设备远程终端：通过 TCP + PTY 桥接
  • 终端自动化：程序化输入命令、读取输出

代码示例

#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>
#include <termios.h>
#include <signal.h>
#include <sys/select.h>

int main() {
    /* 忽略 SIGPIPE */
    signal(SIGPIPE, SIG_IGN);

    int masterfd = posix_openpt(O_RDWR | O_NOCTTY);
    grantpt(masterfd);
    unlockpt(masterfd);
    char *slavename = ptsname(masterfd);

    /* 设置从设备为原始模式（避免输出处理干扰） */
    struct termios t;
    tcgetattr(masterfd, &t);
    cfmakeraw(&t);
    tcsetattr(masterfd, TCSANOW, &t);

    pid_t pid = fork();
    if (pid == 0) {
        close(masterfd);
        setsid();
        int slavefd = open(slavename, O_RDWR);
        dup2(slavefd, 0);
        dup2(slavefd, 1);
        dup2(slavefd, 2);
        close(slavefd);
        execlp("cat", "cat", NULL);  /* 回显程序 */
        _exit(1);
    }

    /* 父进程：写入命令，读取回显 */
    const char *cmd = "hello world\n";
    write(masterfd, cmd, strlen(cmd));

    /* 使用 select 等待输出 */
    fd_set rfds;
    struct timeval tv = {1, 0};
    FD_ZERO(&rfds);
    FD_SET(masterfd, &rfds);

    char buf[4096];
    while (select(masterfd + 1, &rfds, NULL, NULL, &tv) > 0) {
        ssize_t n = read(masterfd, buf, sizeof(buf));
        if (n <= 0) break;
        write(STDOUT_FILENO, buf, n);
        FD_ZERO(&rfds);
        FD_SET(masterfd, &rfds);
        tv.tv_sec = 1; tv.tv_usec = 0;
    }

    close(masterfd);
    return 0;
}

代码自测

为什么在 PTY 通信中建议设置从设备为原始模式？

规范模式下，从设备的 termios 会对数据进行处理：NL→CRNL 映射、行缓冲、回显等。这些处理会干扰二进制协议解析：额外的 \r 字符混入数据流、行缓冲导致数据延迟提交。设置原始模式（cfmakeraw）禁用所有 termios 处理，使数据透明传输，write 的数据原样出现在 read 端。对于二进制协议尤其重要。

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pty.h>
#include <sys/wait.h>

int main(void) {
    int master, slave;
    /* openpty: 一步创建 PTY 对，返回 master+slave fd */
    if (openpty(&master, &slave, NULL, NULL, NULL) < 0) {
        perror("openpty"); exit(1);
    }
    printf("openpty: master=%d, slave=%d\n", master, slave);
    printf("slave name: %s\n", ttyname(slave));

    write(master, "via openpty\n", 12);
    char buf[32];
    int n = read(slave, buf, sizeof(buf) - 1);
    if (n > 0) { buf[n] = '\0'; printf("slave read: %s", buf); }

    close(master);
    close(slave);

    /* forkpty: 创建 PTY + fork，一步到位 */
    pid_t pid = forkpty(&master, NULL, NULL, NULL);
    if (pid == 0) {
        /* 子进程：stdin/stdout 已连到 PTY slave */
        printf("child via forkpty\n");

    char fbuf[64];
    n = read(master, fbuf, sizeof(fbuf) - 1);

    close(master);
    return 0;
}

本节讲什么

PTY 主从两端的 termios 属性和窗口大小的关系：修改从设备属性会影响主设备的数据处理，窗口大小通过 TIOCSWINSZ 设置。

要点

主从 termios 关系

tcgetattr(slavefd) ←→ 内核 termios ←→ tcgetattr(masterfd)
tcsetattr(slavefd) ←→ 修改内核 termios ←→ 影响数据流

tcsetattr(slavefd, ...) 修改 termios，影响数据处理

tcsetattr(masterfd, ...) 等效于设置从设备（某些实现）

关键：termios 属性作用于从设备端，但影响两端的数据流。

/* 子进程的 SIGWINCH 处理函数被调用 */

#include <sys/ioctl.h>
#include <termios.h>
#include <unistd.h>
#include <stdio.h>
#include <signal.h>
#include <fcntl.h>
#include <stdlib.h>

/* 同步当前终端属性到 PTY 从设备 */
void sync_terminal_attrs(int masterfd, int slavefd) {
    struct termios t;
    struct winsize ws;

    /* 获取当前终端属性 */
    tcgetattr(STDIN_FILENO, &t);
    ioctl(STDIN_FILENO, TIOCGWINSZ, &ws);

    /* 设置到 PTY 从设备 */
    tcsetattr(slavefd, TCSANOW, &t);
    ioctl(slavefd, TIOCSWINSZ, &ws);
}

/* SIGWINCH 处理：转发窗口大小变化 */
static int g_masterfd;
static void forward_winsize(int sig) {
    struct winsize ws;
    if (ioctl(STDIN_FILENO, TIOCGWINSZ, &ws) == 0)
        ioctl(g_masterfd, TIOCSWINSZ, &ws);
}

int main() {
    /* 创建 PTY */
    g_masterfd = posix_openpt(O_RDWR | O_NOCTTY);
    grantpt(g_masterfd);
    unlockpt(g_masterfd);
    char *slavename = ptsname(g_masterfd);

    /* fork 子进程 */
    pid_t pid = fork();
    if (pid == 0) {
        close(g_masterfd);
        setsid();
        int slavefd = open(slavename, O_RDWR);
        /* 同步终端属性 */
        struct termios t;
        struct winsize ws;
        tcgetattr(STDIN_FILENO, &t);
        tcsetattr(slavefd, TCSANOW, &t);
        ioctl(STDIN_FILENO, TIOCGWINSZ, &ws);
        ioctl(slavefd, TIOCSWINSZ, &ws);

        dup2(slavefd, 0);
        dup2(slavefd, 1);
        dup2(slavefd, 2);
        close(slavefd);
        execlp("vim", "vim", NULL);
        _exit(1);
    }

    /* 父进程：注册 SIGWINCH 转发 */
    struct sigaction sa = {0};
    sa.sa_handler = forward_winsize;
    sigaction(SIGWINCH, &sa, NULL);

    /* 数据转发循环... */
    /* ... */

    close(g_masterfd);
    return 0;
}

代码自测

在 PTY 中修改从设备的窗口大小后，子进程如何得知？

设置 ioctl(slavefd, TIOCSWINSZ, &ws) 时，如果 ws_row 或 ws_col 与当前值不同，内核会向从设备的会话组发送 SIGWINCH 信号。子进程需注册 SIGWINCH 处理函数，在信号处理中调用 ioctl(STDIN_FILENO, TIOCGWINSZ, &ws) 获取新大小并重绘界面。如果子进程未注册处理函数，信号被忽略，子进程不会自动感知变化。

64.8

BSD Pseudoterminals

本章：TLPI 第 64 章

本节讲什么

BSD 风格的伪终端接口，与 Unix 98 标准的对比。BSD PTY 使用固定的 /dev/ptyXX（主）和 /dev/ttyXX（从）设备文件对，通过遍历查找可用设备。

要点

BSD vs Unix 98 对比

特性 BSD PTY Unix 98 PTY
主设备 /dev/pty[p-za-e][0-9a-f] posix_openpt() 动态分配
从设备 /dev/tty[p-za-e][0-9a-f] ptsname() 返回
查找方式 遍历设备文件直到成功 自动分配
并发限制 固定设备数（通常 256） 动态，几乎无限制
标准 BSD 历史 SUSv3+
推荐 不推荐（遗留） 推荐

BSD PTY 使用流程

/* 1. 遍历主设备名查找可用 */
char name[] = "/dev/ptyXY";
for (char major = 'p'; major <= 'z'; major++) {
    for (char minor = '0'; minor <= '9'; minor++) {
        name[8] = major; name[9] = minor;
        int masterfd = open(name, O_RDWR);
        if (masterfd >= 0) {
            /* 找到可用主设备 */
            name[5] = 't'; name[6] = 't';  /* /dev/ttyXY */
            char *slavename = strdup(name);
            break;
        }
    }
}

/* 2. fork，子进程打开从设备 */
/* 3. grantpt/unlockpt 不需要 */

问题 说明
竞争窗口 遍历查找是 racy 的
有限设备数 固定设备文件数量

#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

/* BSD PTY 遍历查找（回退方案） */
int open_bsd_pty(char **slavename) {
    char master[] = "/dev/ptyXY";
    char slave[]  = "/dev/ttyXY";

    for (char p1 = 'p'; p1 <= 'z'; p1++) {
        for (char p2 = '0'; p2 <= 'f'; p2++) {
            if (p2 > '9' && p2 < 'a') continue;

            master[9] = p1; master[10] = p2;
            int fd = open(master, O_RDWR | O_NOCTTY);
            if (fd >= 0) {
                slave[8] = p1; slave[9] = p2;
                *slavename = strdup(slave);
                return fd;
            }
        }
    }
    return -1;  /* 无可用 PTY */
}

/* 优先 Unix 98，回退 BSD */
int open_pty(char **slavename) {
    /* 尝试 Unix 98 */
    int fd = posix_openpt(O_RDWR | O_NOCTTY);
    if (fd >= 0) {
        grantpt(fd);
        unlockpt(fd);
        *slavename = strdup(ptsname(fd));
        return fd;
    }
    /* 回退 BSD */
    return open_bsd_pty(slavename);
}

代码自测

为什么现代系统推荐 Unix 98 PTY 而非 BSD PTY？
  1. 无竞争：posix_openpt 原子分配可用设备，BSD 需要遍历查找，存在竞争窗口
  2. 无限数量：Unix 98 动态分配从设备号（/dev/pts/N），几乎无数量限制；BSD 固定 256 对
  3. 自动权限：grantpt 自动设置从设备权限，BSD 需手动 chown
  4. 标准化：Unix 98 是 POSIX/SUS 标准，BSD PTY 各系统实现不一致
  5. 安全：BSD 遍历可能遗漏权限设置，Unix 98 自动处理

    for (int i = 0; i < n; i++) {
        int fd = events[i].data.fd;
        int len = read(fd, buf, sizeof(buf));
        if (len <= 0) goto done;
        int dest = (fd == clientfd) ? masterfd : clientfd;
        write(dest, buf, len);
    }
}
done:
close(clientfd);
close(masterfd);
