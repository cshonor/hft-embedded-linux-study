章节目标

掌握 inotify 实例 / watch / 事件掩码；理解目录 vs 文件监控差异、变长事件缓冲、溢出与 watch 自动删除；能与 poll/epoll 搭配；规避软链接、移动 cookie、非递归等陷阱。

19.2 核心 API

#include <sys/inotify.h>

int inotify_init(void);
int inotify_init1(int flags);   /* IN_NONBLOCK | IN_CLOEXEC 推荐 */

int inotify_add_watch(int fd, const char *pathname, uint32_t mask);
int inotify_rm_watch(int fd, int wd);

对 inotify fd 做 read() 取事件；该 fd 可进 select/poll/epoll。

19.7 速查：陷阱

# 陷阱
1 以为天然递归
2 跨目录移动：源 MOVED_FROM、目标 MOVED_TO
3 硬链：仅最后一链 unlink 才 DELETE_SELF；中间只是父目录 IN_DELETE
4 改名监控对象自身 → IN_MOVE_SELF
5 缓冲太小 → 截断/解析错乱
6 忽略 IN_Q_OVERFLOW → 状态不一致
7 NFS 远端变更通常看不见

练习

  1. 监控目录：打印 create/delete/modify
  2. epoll + 非阻塞 inotify
  3. （选）子目录创建时动态 add_watch
  4. （选）压测溢出
  5. 移动文件观察 cookie

背诵卡

# 要点
1 Linux 专属；fd + read 取变长事件
2 不递归；要自管子目录 watch
3 目录事件才有子 name；移动用 cookie
4 必处理 IN_Q_OVERFLOW / IN_IGNORED
5 生产搭配 epoll/poll
6 非 POSIX；NFS 远端无效

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <sys/inotify.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>

/* Ch19 文件事件监控 — inotify_init/inotify_add_watch/read。
 * 监控目录中文件的创建/修改/删除事件。
 * 编译: gcc -o ch19_demo ch19_demo.c */

#define EVENT_SIZE (sizeof(struct inotify_event))
#define BUF_LEN (1024 * (EVENT_SIZE + 16))

int main(void) {
    int fd = inotify_init1(IN_NONBLOCK);
    if (fd < 0) { perror("inotify_init"); return 1; }

    int wd = inotify_add_watch(fd, "/tmp", IN_CREATE | IN_DELETE | IN_MODIFY);
    if (wd < 0) { perror("inotify_add_watch"); return 1; }

    printf("Watching /tmp for 3 seconds...\n");

    /* 简化: 等3秒，读一次事件 */
    sleep(3);

    char buf[BUF_LEN];
    ssize_t len = read(fd, buf, sizeof(buf));
    if (len > 0) {
        for (ssize_t i = 0; i < len; ) {
            struct inotify_event *ev = (struct inotify_event *)&buf[i];
            if (ev->len > 0) {
                printf("event: mask=0x%x name=%s\n", ev->mask, ev->name);
            }
            i += EVENT_SIZE + ev->len;
        }
    } else {
        printf("No events in 3 seconds\n");
    }

    inotify_rm_watch(fd, wd);
    close(fd);
    return 0;
}

19.2

The inotify API

本章：TLPI 第 19 章 — Monitoring File Events

本节讲什么

介绍 Linux inotify 机制：监控文件/目录的创建、修改、删除等事件。涵盖 inotify_init1()、inotify_add_watch()、read() 事件流的完整编程模型。

要点

inotify 核心 API

函数 原型 说明
inotify_init1(flags) int inotify_init1(int flags) 创建 inotify 实例 fd
inotify_add_watch(fd, path, mask) int inotify_add_watch(int fd, const char *path, uint32_t mask) 添加监控，返回 watch fd
inotify_rm_watch(fd, wd) int inotify_rm_watch(int fd, int wd) 移除监控
read(fd, buf, n) 标准 read 读取事件数据

inotify_init1 flags

flag 说明
IN_NONBLOCK 设置 fd 为非阻塞
IN_CLOEXEC exec 时自动关闭

事件类型（mask）

事件 含义
IN_ACCESS 文件被读取
IN_MODIFY 文件被修改
IN_ATTRIB 属性改变（mode/uid/size）
IN_CLOSE_WRITE 可写文件被关闭
IN_CLOSE_NOWRITE 只读文件被关闭
IN_OPEN 文件被打开
IN_MOVED_FROM 文件被移走
IN_MOVED_TO 文件被移入
IN_CREATE 创建文件/目录
IN_DELETE 删除文件/目录
IN_DELETE_SELF 监控对象本身被删除
IN_MOVE_SELF 监控对象本身被移走
IN_ALL_EVENTS 所有事件
IN_MOVE MOVED_FROM | MOVED_TO
IN_CLOSE CLOSE_WRITE | CLOSE_NOWRITE
IN_DONT_FOLLOW 不跟随符号链接
IN_MASK_ADD 追加到已有 mask
IN_ONESHOT 只报告一次事件
IN_ONLYDIR 只监控目录（非目录返回错误）

inotify_event 结构

struct inotify_event {
    int      wd;       /* watch descriptor */
    uint32_t mask;     /* 触发的事件 mask */
    uint32_t cookie;   | 重命名关联 cookie（同一操作的 FROM/TO cookie 相同） */
    uint32_t len;      /* name 字段长度 */
    char     name[];   /* 可选：子文件名（监控目录时） */
};

事件读取模式

fd = inotify_init1(IN_NONBLOCK | IN_CLOEXEC);
wd = inotify_add_watch(fd, "/data/orders", IN_CREATE | IN_MODIFY | IN_DELETE);

char buf[4096] __attribute__((aligned(8)));
ssize_t n = read(fd, buf, sizeof(buf));

/* buf 中包含多个 struct inotify_event，需遍历 */
char *p = buf;
while (p < buf + n) {
    struct inotify_event *ev = (struct inotify_event *)p;
    /* 处理 ev... */
    p += sizeof(struct inotify_event) + ev->len;
}

事件合并与溢出

情况 行为
多个相同事件 内核可能合并（丢弃中间事件）
队列满 IN_Q_OVERFLOW 事件，wd=-1
读取太慢 事件被丢弃

HFT / 嵌入式关联

  • 配置热加载：监控配置文件 IN_MODIFY，自动重新加载策略参数
  • 订单文件监控：监控订单目录 IN_CREATE，新订单文件到达时自动处理
  • 日志轮转：监控日志文件 IN_MOVE_SELF，触发重打开日志 fd
  • 嵌入式：inotify 需要内核 CONFIG_INOTIFY_USER=y，精简系统可能未启用
  • 与 epoll 集成：inotify fd 可以加入 epoll 事件循环，与网络 I/O 统一管理

代码示例

#include <sys/inotify.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>

static void handle_event(struct inotify_event *ev) {
    printf("wd=%2d mask=0x%08x ", ev->wd, ev->mask);
    if (ev->mask & IN_CREATE)  printf("[CREATE] ");
    if (ev->mask & IN_MODIFY)  printf("[MODIFY] ");
    if (ev->mask & IN_DELETE)  printf("[DELETE] ");
    if (ev->mask & IN_MOVED_FROM) printf("[MOVED_FROM] ");
    if (ev->mask & IN_MOVED_TO)   printf("[MOVED_TO] ");
    if (ev->len > 0) printf("name=%s", ev->name);
    printf("\n");
}

int watch_dir(const char *path) {
    int ifd = inotify_init1(IN_CLOEXEC);
    if (ifd == -1) { perror("inotify_init1"); return -1; }

    uint32_t mask = IN_CREATE | IN_MODIFY | IN_DELETE
                  | IN_MOVED_FROM | IN_MOVED_TO;
    int wd = inotify_add_watch(ifd, path, mask);
    if (wd == -1) { perror("inotify_add_watch"); return -1; }

    char buf[4096] __attribute__((aligned(8)));
    while (1) {
        ssize_t n = read(ifd, buf, sizeof(buf));
        if (n <= 0) break;
        char *p = buf;
        while (p < buf + n) {
            struct inotify_event *ev = (struct inotify_event *)p;
            handle_event(ev);
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    close(ifd);
    return 0;
}

代码自测

1. 监控目录时，IN_MODIFY 事件中 name 字段的含义？

当监控的是目录，目录中某个文件被修改时，name 字段包含被修改文件的文件名。当监控的是文件本身，事件中 name 为空（len=0），因为被修改的就是被监控的文件本身。

2. IN_MOVED_FROM 和 IN_MOVED_TO 的 cookie 有什么作用？

重命名操作产生两个事件：IN_MOVED_FROM（旧路径）和 IN_MOVED_TO（新路径）。两个事件可能有不同的 watch wd（如果源和目标在不同监控目录）。cookie 在同一重命名操作的两个事件中相同，用于关联它们，重建完整的重命名操作。

3. inotify 事件队列满了怎么办？

队列满时，内核丢弃溢出的事件并发送一个 IN_Q_OVERFLOW 事件（wd=-1, mask=0x2000）。处理溢出：(1) 记录警告日志；(2) 重新扫描所有监控目录，比对当前状态与上次记录的差异；(3) 如果无法恢复完整状态，可能需要重启监控。

19.3

`struct inotify_event`

本章：TLPI 第 19 章 — Monitoring File Events - 上一节：19.1 技术演进 - 下一节：19.4 常用 mask

本节讲什么

struct inotify_event { int wd; uint32_t mask;

要点

struct inotify_event {
    int      wd;
    uint32_t mask;
    uint32_t cookie;   /* 关联 IN_MOVED_FROM / IN_MOVED_TO */
    uint32_t len;
    char     name[];   /* 变长；监控目录时多为子名 */
};
监控对象 name
普通文件 通常空
目录 子项事件时常有子文件名

一次 read 可能含多个变长事件；缓冲要够大，按 sizeof(struct inotify_event) + NAME_MAX + 1 倍数预留。

HFT / 嵌入式关联

路线 要点
嵌入式 struct inotify_event 变长 name 字段，读取需按"结构体大小 + name len"解析对齐
HFT 解析事件按 name 字段长度对齐，避免越界读崩溃；结合 cookie 配对 MOVED_FROM/TO

代码示例

#include <stdio.h>
#include <stdlib.h>
#include <sys/inotify.h>
#include <unistd.h>
#include <fcntl.h>
/* struct inotify_event 结构解析 */
int main(void) {
    int fd = inotify_init1(IN_NONBLOCK);
    int wd = inotify_add_watch(fd, "/tmp", IN_CREATE | IN_DELETE | IN_MODIFY);

    /* 触发事件 */
    int tfd = open("/tmp/inotify_struct_test.txt", O_CREAT, 0644);
    close(tfd);
    remove("/tmp/inotify_struct_test.txt");
    usleep(100000);  /* 等内核推送事件 */

    char buf[4096] __attribute__((aligned(8)));
    ssize_t len = read(fd, buf, sizeof(buf));
    if (len > 0) {
        for (char *p = buf; p < buf + len; ) {
            struct inotify_event *ev = (struct inotify_event *)p;
            printf("wd=%d mask=0x%x cookie=%u len=%u",
                   ev->wd, ev->mask, ev->cookie, ev->len);
            if (ev->len > 0)
                printf(" name=%s", ev->name);
            if (ev->mask & IN_CREATE) printf(" [CREATE]");
            if (ev->mask & IN_DELETE) printf(" [DELETE]");
            if (ev->mask & IN_MODIFY) printf(" [MODIFY]");
            printf("\n");
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    inotify_rm_watch(fd, wd);
    close(fd);
    return 0;
}

IN_ACCESS · IN_MODIFY · IN_ATTRIB · IN_CLOSE_WRITE · IN_CLOSE_NOWRITE · IN_OPEN IN_DONT_FOLLOW · IN_ONESHOT · IN_ONLYDIR ---

要点

变更

IN_ACCESS · IN_MODIFY · IN_ATTRIB · IN_CLOSE_WRITE · IN_CLOSE_NOWRITE · IN_OPEN

创建 / 删除 / 移动

掩码 含义
IN_CREATE / IN_DELETE 目录内建/删
IN_DELETE_SELF / IN_MOVE_SELF 被监控对象自身删/改名
IN_MOVED_FROM / IN_MOVED_TO 移出/移入；同 cookie 配对

掩码 含义

IN_Q_OVERFLOW 队列溢出，丢事件

HFT / 嵌入式关联

路线 要点
嵌入式 IN_CREATE/IN_MODIFY/IN_DELETE 等 mask 过滤关注事件，减少无关唤醒

代码示例

#include <sys/inotify.h>
#include <stdio.h>
int main(void) {
    /* inotify 事件掩码 */
    printf("IN_ACCESS=%u  IN_MODIFY=%u  IN_ATTRIB=%u\n", IN_ACCESS, IN_MODIFY, IN_ATTRIB);
    printf("IN_CREATE=%u  IN_DELETE=%u  IN_MOVE=%u\n", IN_CREATE, IN_DELETE, IN_MOVE);
    printf("IN_DELETE_SELF=%u  IN_MOVE_SELF=%u\n", IN_DELETE_SELF, IN_MOVE_SELF);
    printf("IN_ALL_EVENTS=0x%x\n", IN_ALL_EVENTS);
    int fd = inotify_init();
    int wd = inotify_add_watch(fd, "/tmp", IN_CREATE | IN_DELETE);
    printf("\n监听 /tmp 的创建+删除事件 (wd=%d)\n", wd);
    close(fd);
    return 0;
}

19.5

关键行为

本章：TLPI 第 19 章 — Monitoring File Events - 上一节：19.4 常用 mask - 下一节：19.6 典型流程

本节讲什么

  1. 不递归：只看直接子项；子树要自建 watch，并在 IN_CREATE 目录时动态 add_watch 2. watch 文件 vs 目录：见上表；目录自身 IN_ATTRIB 常无 name 3. 默认跟随软链接；IN_DONT_FOLLOW 盯链接本身

要点

  1. 不递归：只看直接子项；子树要自建 watch，并在 IN_CREATE 目录时动态 add_watch
  2. watch 文件 vs 目录：见上表；目录自身 IN_ATTRIB 常无 name
  3. 默认跟随软链接；IN_DONT_FOLLOW 盯链接本身
  4. watch 自动没：inode 删除、卸载、IN_ONESHOT；主动 rm_watch
  5. 溢出：读太慢 → IN_Q_OVERFLOW；调 /proc/sys/fs/inotify/max_user_instances|max_user_watches|max_queued_events

HFT / 嵌入式关联

路线 要点
嵌入式 inotify 队列有上限，溢出丢事件并设 IN_Q_OVERFLOW，必须处理否则丢监控
HFT 高事件频率下处理 IN_Q_OVERFLOW，溢出后全量重扫配置而非信任增量，防止状态漂移

代码示例

#include <sys/inotify.h>
#include <unistd.h>
#include <stdio.h>
int main(void) {
    int fd = inotify_init1(IN_NONBLOCK);
    int wd = inotify_add_watch(fd, "/tmp", IN_ALL_EVENTS);
    printf("非阻塞 inotify 监听 /tmp\n");
    char buf[4096];
    while (1) {
        ssize_t n = read(fd, buf, sizeof(buf));
        if (n <= 0) break;
        for (char *p = buf; p < buf + n; ) {
            struct inotify_event *ev = (struct inotify_event *)p;
            printf("mask=0x%x cookie=%u name=%s\n", ev->mask, ev->cookie, ev->name);
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    inotify_rm_watch(fd, wd);
    close(fd);
    return 0;
}

代码自测

1. inotify 和 dnotify 的核心区别？
  • dnotify（旧机制）：通过信号（SIGIO）通知目录变化，目录级粒度，信号处理复杂，无法知道具体哪个文件变化
  • inotify（现代）：通过文件描述符读取事件流，文件级粒度，每个事件包含 watch fd、mask、文件名，可与 epoll 集成
  • inotify 完全替代了 dnotify，dnotify 已废弃
2. 监控大量文件时，inotify 可能遇到什么问题？
  1. watch 数量限制：max_user_watches 默认可能较小（8192），监控大量目录/文件时需调高
  2. 事件风暴：大量文件同时修改（如 tar 解压），队列溢出丢事件
  3. 内存开销：每个 watch 占用内核内存（约 1KB）
  4. 递归监控：inotify 不支持递归，监控子树需手动添加每个子目录的 watch

编写程序，监控配置文件变化，文件修改后自动重新加载配置并应用。

要求： - 使用 IN_MODIFY 事件 - 防抖处理（编辑器可能多次保存，合并短时间内的多次事件） - 打印重新加载的时间戳

参考答案

练习 1 参考代码

#include <sys/inotify.h>
#include <dirent.h>
#include <unistd.h>
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define EVENT_BUF_LEN (16 * (sizeof(struct inotify_event) + NAME_MAX + 1))
#define MAX_WATCHES 1024

int ifd;
int watches[MAX_WATCHES];
char paths[MAX_WATCHES][PATH_MAX];
int watch_count = 0;

void add_watch(const char *path) {
    uint32_t mask = IN_CREATE | IN_DELETE | IN_MODIFY
                  | IN_MOVED_FROM | IN_MOVED_TO | IN_DELETE_SELF;
    int wd = inotify_add_watch(ifd, path, mask);
    if (wd >= 0 && watch_count < MAX_WATCHES) {
        watches[watch_count] = wd;
        strncpy(paths[watch_count], path, PATH_MAX);
        watch_count++;
    }
}

void watch_recursive(const char *dir) {
    add_watch(dir);
    DIR *d = opendir(dir);
    if (!d) return;
    struct dirent *de;
    while ((de = readdir(d)) != NULL) {
        if (de->d_type == DT_DIR && strcmp(de->d_name, ".") != 0
            && strcmp(de->d_name, "..") != 0) {
            char sub[PATH_MAX];
            snprintf(sub, sizeof(sub), "%s/%s", dir, de->d_name);
            watch_recursive(sub);
        }
    }
    closedir(d);
}

int main(int argc, char *argv[]) {
    if (argc != 2) { fprintf(stderr, "usage: %s <dir>\n", argv[0]); return 1; }

    ifd = inotify_init1(IN_CLOEXEC);
    watch_recursive(argv[1]);
    printf("watching %d directories\n", watch_count);

    char buf[EVENT_BUF_LEN] __attribute__((aligned(8)));
    while (1) {
        ssize_t n = read(ifd, buf, sizeof(buf));
        if (n <= 0) break;
        char *p = buf;
        while (p < buf + n) {
            struct inotify_event *ev = (struct inotify_event *)p;
            if (ev->mask & IN_CREATE && ev->len > 0) {
                /* 新目录自动添加 watch */
                char full[PATH_MAX];
                for (int i = 0; i < watch_count; i++)
                    if (watches[i] == ev->wd) { snprintf(full, sizeof(full), "%s/%s", paths[i], ev->name); break; }
                add_watch(full);
            }
            printf("wd=%d mask=0x%x %s\n", ev->wd, ev->mask,
                   ev->len > 0 ? ev->name : "");
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    close(ifd);
    return 0;
}

练习 2 参考代码

#include <sys/inotify.h>
#include <unistd.h>
#include <time.h>
#include <stdio.h>
#include <string.h>

volatile int config_changed = 0;
time_t last_event_time = 0;

void load_config(const char *path) {
    printf("[%ld] config reloaded: %s\n", (long)time(NULL), path);
    /* 实际加载配置逻辑 */
}

int main(int argc, char *argv[]) {
    const char *config = argc > 1 ? argv[1] : "app.conf";

    int ifd = inotify_init1(IN_CLOEXEC);
    inotify_add_watch(ifd, config, IN_MODIFY | IN_CLOSE_WRITE);

    char buf[4096] __attribute__((aligned(8)));
    while (1) {
        ssize_t n = read(ifd, buf, sizeof(buf));
        if (n <= 0) break;
        char *p = buf;
        while (p < buf + n) {
            struct inotify_event *ev = (struct inotify_event *)p;
            if (ev->mask & (IN_MODIFY | IN_CLOSE_WRITE)) {
                time_t now = time(NULL);
                /* 防抖：3 秒内的事件合并 */
                if (now - last_event_time >= 3) {
                    load_config(config);
                }
                last_event_time = now;
            }
            p += sizeof(struct inotify_event) + ev->len;
        }
    }
    close(ifd);
    return 0;
}
