章节目标

命名 / 匿名；wait/post/timed；fork/exec；vs SysV / mutex。

陷阱

  1. 跨进程匿名：pshared+共享内存
  2. API 混用 destroy/close
  3. 无 SEM_UNDO
  4. timed 绝对时间
  5. getvalue TOCTOU
  6. pshared=0 fork 后不可跨进程
  7. 名 /a/b 非法

背诵卡

1 命名 vs 匿名；单个计数器
2 open 原子初值；无 init 竞态

4 无 SEM_UNDO；无所有权

代码示例

#include <stdio.h>
#include <semaphore.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <unistd.h>
#include <sys/wait.h>

/* Ch53 POSIX 信号量 — sem_open/sem_wait/sem_post/sem_unlink。
 * 命名信号量可在无关进程间共享; 无名信号量在共享内存中使用。
 * 编译: gcc -o ch53_demo ch53_demo.c -lpthread */

#define SEM_NAME "/ch53_demo"

int main(void) {
    /* 创建命名信号量, 初始值=1 (可用) */
    sem_t *sem = sem_open(SEM_NAME, O_CREAT, 0644, 1);
    if (sem == SEM_FAILED) { perror("sem_open"); return 1; }

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 获取信号量 */
        printf("Child: waiting for semaphore...\n");
        sem_wait(sem);   /* P: value 1->0, 如果为0则阻塞 */
        printf("Child: acquired, working 2s...\n");
        sleep(2);
        printf("Child: releasing\n");
        sem_post(sem);   /* V: value 0->1 */
        _exit(0);
    }

    sleep(1);
    /* 父进程: 也会被阻塞直到子进程释放 */
    printf("Parent: waiting for semaphore...\n");
    sem_wait(sem);   /* 会阻塞 */
    printf("Parent: acquired after child released\n");
    sem_post(sem);

    waitpid(pid, NULL, 0);
    sem_close(sem);
    sem_unlink(SEM_NAME);

    /* 无名信号量 (进程内线程同步) */
    printf("\nUnnamed semaphore (for threads, in shared memory):\n");
    sem_t unnamed;
    sem_init(&unnamed, 0, 1);
    sem_wait(&unnamed);
    printf("  acquired unnamed semaphore\n");
    sem_post(&unnamed);
    sem_destroy(&unnamed);
    return 0;
}

53.1

概念

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：本章首页 - 下一节：53.2 命名

本节讲什么

sem_wait P / sem_post V；值非负；无所有权。 ---

要点

命名 匿名
定位 /name，无关进程 内存中的 sem_t
场景 任意本机进程 线程；或 pshared+共享内存的亲缘/共享区进程
生命周期 unlink + 全 close 随内存；sem_destroy

sem_wait P / sem_post V；值非负；无所有权。

HFT / 嵌入式关联

路线 要点
嵌入式 命名信号量 /name 供无关进程（守护 vs 采集）同步，无需亲缘关系
HFT 无所有权计数信号量做跨进程资源计数/限流，sem_wait/post 轻量 futex 快于 SysV semop

代码示例


/* POSIX 信号量概念：计数信号量 vs 二值信号量 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <semaphore.h>

int main(void) {
    /* 二值信号量：初值 1，用作互斥锁 */
    sem_t *mutex = sem_open("/bin_sem", O_CREAT, 0644, 1);
    if (mutex == SEM_FAILED) { perror("sem_open"); exit(1); }

    int val;
    sem_getvalue(mutex, &val);
    printf("binary semaphore (mutex): initial value=%d\n", val);

    sem_wait(mutex);  /* P: value 1 -> 0 */
    sem_getvalue(mutex, &val);
    printf("after sem_wait: value=%d (locked)\n", val);

    sem_post(mutex);  /* V: value 0 -> 1 */
    sem_getvalue(mutex, &val);
    printf("after sem_post: value=%d (unlocked)\n", val);

    /* 计数信号量：初值 3，允许 3 个并发 */
    sem_t *count = sem_open("/cnt_sem", O_CREAT, 0644, 3);
    sem_getvalue(count, &val);
    printf("\ncounting semaphore: initial value=%d\n", val);

    sem_wait(count); sem_wait(count);
    sem_getvalue(count, &val);
    printf("after 2x sem_wait: value=%d (1 slot left)\n", val);

    sem_close(mutex); sem_unlink("/bin_sem");
    sem_close(count); sem_unlink("/cnt_sem");
    return 0;
}

53.2

命名

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：53.1 概念 - 下一节：53.3 操作（两形态共用）

本节讲什么

sem_t sem_open(const char name, int oflag, mode_t mode, unsigned value); int sem_close(sem_t sem); int sem_unlink(const char name);

要点

sem_t *sem_open(const char *name, int oflag, mode_t mode, unsigned value);
int sem_close(sem_t *sem);
int sem_unlink(const char *name);

O_CREAT|O_EXCL：创建+初值原子，无 SysV 式初始化竞态。
unlink 删名；全 close 销毁。

HFT / 嵌入式关联

路线 要点
嵌入式 O_CREAT|O_EXCL 原子创建+初值，多进程竞态下首个创建者唯一，无 SysV get+ctl 竞态
HFT 多进程网关按 /name 打开同一信号量，初值原子初始化避免锁状态错乱

代码示例


/* 命名信号量 vs 未命名（内存）信号量 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <semaphore.h>
#include <sys/mman.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    /* 1) 命名信号量：sem_open，跨进程共享 */
    sem_t *named = sem_open("/named_sem", O_CREAT, 0644, 1);
    if (named == SEM_FAILED) { perror("sem_open"); exit(1); }
    printf("named semaphore: /named_sem\n");

    /* 2) 未命名信号量：sem_init + 共享内存 */
    sem_t *unnamed = mmap(NULL, sizeof(sem_t), PROT_READ | PROT_WRITE,
                          MAP_SHARED | MAP_ANONYMOUS, -1, 0);
    if (unnamed == MAP_FAILED) { perror("mmap"); exit(1); }
    if (sem_init(unnamed, 1, 1) < 0) { perror("sem_init"); exit(1); }
    printf("unnamed semaphore: in shared memory (pshared=1)\n");

    /* fork 子进程测试未命名信号量 */
    pid_t pid = fork();
    if (pid == 0) {
        sem_wait(unnamed);
        printf("child: got unnamed sem\n");
        sem_post(unnamed);
        _exit(0);
    }
    wait(NULL);

    sem_close(named); sem_unlink("/named_sem");
    sem_destroy(unnamed);
    munmap(unnamed, sizeof(sem_t));
    return 0;
}

53.3

操作（两形态共用）

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：53.2 命名 - 下一节：53.4 匿名

本节讲什么

sem_wait · sem_trywait · sem_timedwait（绝对 CLOCK_REALTIME）· sem_post · sem_getvalue（瞬时快照，勿当同步条件）。 ---

要点

sem_wait · sem_trywait · sem_timedwait（绝对 CLOCK_REALTIME）· sem_post · sem_getvalue（瞬时快照，勿当同步条件）。

HFT / 嵌入式关联

路线 要点
嵌入式 sem_trywait 非阻塞探测、sem_timedwait 绝对超时做看门狗/限时抢锁
HFT 热路径用 sem_trywait 免阻塞，超时走降级路径；绝对时钟便于全局 deadline 对齐

代码示例


/* sem_wait / sem_trywait / sem_timedwait / sem_post */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <semaphore.h>
#include <time.h>

int main(void) {
    sem_t *sem = sem_open("/ops_sem", O_CREAT, 0644, 1);
    if (sem == SEM_FAILED) { perror("sem_open"); exit(1); }

    /* sem_wait：阻塞 P 操作 */
    sem_wait(sem);
    printf("sem_wait: acquired (value now 0)\n");

    /* sem_trywait：非阻塞 P，返回 -1 + EAGAIN */
    if (sem_trywait(sem) < 0) {
        printf("sem_trywait: would block (EAGAIN)\n");
    }

    /* sem_timedwait：限时等待 */
    struct timespec ts = { .tv_sec = 2, .tv_nsec = 0 };
    if (sem_timedwait(sem, &ts) < 0) {
        printf("sem_timedwait: timed out after 2s\n");
    }

    /* sem_post：V 操作 */
    sem_post(sem);
    printf("sem_post: released (value now 1)\n");

    /* 现在可以再次 sem_wait */
    sem_wait(sem);
    printf("sem_wait: acquired again\n");
    sem_post(sem);

    sem_close(sem); sem_unlink("/ops_sem");
    return 0;
}

53.4

匿名

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：53.3 操作（两形态共用） - 下一节：53.5 fork / exec

本节讲什么

int sem_init(sem_t sem, int pshared, unsigned value); int sem_destroy(sem_t sem); 勿对命名调 sem_destroy；勿对匿名调 sem_open/close。

要点

int sem_init(sem_t *sem, int pshared, unsigned value);
int sem_destroy(sem_t *sem);
pshared
0 线程间；普通进程内存
≠0 进程间；sem_t 必须在共享内存

勿对命名调 sem_destroy；勿对匿名调 sem_open/close。

Demo：code/

HFT / 嵌入式关联

路线 要点
嵌入式 pshared≠0 + sem_t 放共享内存可跨进程；线程间 pshared=0 无内核对象开销
HFT 线程内互斥优先匿名 pshared=0，绕开 sem_open 命名解析路径

代码示例


/* 未命名信号量 sem_init / sem_destroy */
#include <stdio.h>
#include <stdlib.h>
#include <semaphore.h>
#include <sys/mman.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    /* 在匿名共享内存中创建未命名信号量 */
    sem_t *sem = mmap(NULL, sizeof(sem_t), PROT_READ | PROT_WRITE,
                      MAP_SHARED | MAP_ANONYMOUS, -1, 0);
    if (sem == MAP_FAILED) { perror("mmap"); exit(1); }

    /* pshared=1 表示可跨进程共享 */
    if (sem_init(sem, 1, 2) < 0) { perror("sem_init"); exit(1); }
    printf("unnamed sem initialized: pshared=1, value=2\n");

    /* fork 两个子进程，各消耗一个信号量 */
    for (int i = 0; i < 2; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            sem_wait(sem);
            printf("child %d: acquired sem\n", i);
            sleep(1);
            sem_post(sem);
            _exit(0);
        }
    }
    for (int i = 0; i < 2; i++) wait(NULL);

    int val;
    sem_getvalue(sem, &val);
    printf("final value=%d\n", val);

    sem_destroy(sem);
    munmap(sem, sizeof(sem_t));
    return 0;
}

53.5

fork / exec

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：53.4 匿名 - 下一节：53.6 vs SysV · vs mutex

本节讲什么

要点

命名 匿名 pshared=0 匿名 pshared≠0
fork 同内核对象 各自副本，不同步 共享内存内同一 sem
exec 自动 close 映射毁 → 失效 同左

HFT / 嵌入式关联

路线 要点
嵌入式 fork 后匿名 pshared=0 各自副本不同步（陷阱）；跨进程同步必须 pshared≠0 放共享内存
HFT 命名信号量 exec 后同内核对象可重开；匿名随 exec 失效，避免 exec 边界传递锁

代码示例


/* POSIX 信号量在 fork/exec 后的继承 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <semaphore.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    /* 命名信号量：fork 后子进程可用 sem_open 重新打开（同名） */
    sem_t *sem = sem_open("/inherit_sem", O_CREAT, 0644, 1);
    if (sem == SEM_FAILED) { perror("sem_open"); exit(1); }

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程继承父进程的 sem_t* 指针（同一引用） */
        sem_wait(sem);
        printf("child: acquired inherited sem\n");
        sem_post(sem);
        _exit(0);
    }
    wait(NULL);

    /* exec 后：命名信号量可用 sem_open 重新获取 */
    /* 未命名信号量：如果在共享内存中，fork 后仍可用，exec 后丢失 */
    printf("fork: sem_t* inherited (same handle)\n");
    printf("exec: named sem re-openable, unnamed sem lost\n");

    sem_close(sem); sem_unlink("/inherit_sem");
    return 0;
}

53.6

vs SysV · vs mutex

本章：TLPI 第 53 章 — POSIX Semaphores - 上一节：53.5 fork / exec - 下一节：下一章+1:02d}-*/notes.md)

本节讲什么

工程：多锁原子 / SEM_UNDO → SysV；简洁单计数器 → POSIX。 线程互斥优先 pthread_mutex（有所有权、常更快）；跨进程用信号量。 ---

要点

SysV POSIX
形态 集、多计数器 单个
初始化 get+ctl 竞态 open/init 原子
多 op 原子 semop ✅ ❌
崩溃撤销 SEM_UNDO 无
超时 弱 sem_timedwait

工程：多锁原子 / SEM_UNDO → SysV；简洁单计数器 → POSIX。
线程互斥优先 pthread_mutex（有所有权、常更快）；跨进程用信号量。

HFT / 嵌入式关联

路线 要点
嵌入式 线程互斥优先 pthread_mutex（有所有权、更快）；跨进程计数用 POSIX sem；多锁原子/SEM_UNDO 才 SysV
HFT 单计数器同步选 POSIX sem，免 SysV 内核持久清理；mutex 所有者内联 futex 最低延迟

代码示例


/* POSIX 信号量实现互斥锁模式（对比 SysV semop P/V） */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <semaphore.h>
#include <sys/wait.h>
#include <unistd.h>

static int counter = 0;  /* 简化：仅父子共享通过 fork 的 COW */

int main(void) {
    /* 用共享内存做计数器，信号量做互斥 */
    int *shared = mmap(NULL, sizeof(int), PROT_READ | PROT_WRITE,
                       MAP_SHARED | MAP_ANONYMOUS, -1, 0);
    *shared = 0;

    sem_t *sem = sem_open("/mutex_sem", O_CREAT, 0644, 1);
    if (sem == SEM_FAILED) { perror("sem_open"); exit(1); }

    /* 5 个子进程并发递增 */
    for (int i = 0; i < 5; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            for (int j = 0; j < 1000; j++) {
                sem_wait(sem);         /* P 操作 = lock */
                (*shared)++;
                sem_post(sem);         /* V 操作 = unlock */
            }
            _exit(0);
        }
    }
    for (int i = 0; i < 5; i++) wait(NULL);

    printf("counter = %d (expected 5000)\n", *shared);
    /* POSIX: sem_wait/sem_post == SysV: semop(SEM_UNDO, P/V) */

    sem_close(sem); sem_unlink("/mutex_sem");
    munmap(shared, sizeof(int));
    return 0;
}

Summary

本章回顾

POSIX 信号量 API

函数 说明
sem_open(name, oflag, mode, value) 创建/打开命名信号量
sem_close(sem) 关闭（减少引用计数）
sem_unlink(name) 删除命名信号量
sem_init(sem, pshared, value) 初始化未命名信号量
sem_destroy(sem) 销毁未命名信号量
sem_wait(sem) 等待（P 操作，-1）
sem_trywait(sem) 非阻塞等待
sem_timedwait(sem, abs_timeout) 超时等待
sem_post(sem) 释放（V 操作，+1）
sem_getvalue(sem, &val) 获取当前值

命名 vs 未命名

特性 命名信号量 未命名信号量
创建 sem_open sem_init
共享方式 通过名字 共享内存
适用 无亲缘进程 有亲缘进程/线程
清理 sem_unlink sem_destroy
持久性 内核级 跟随共享内存

POSIX vs SysV 信号量

P/V 操作语义

P (wait):  while (sem <= 0) block; sem--;
V (post):  sem++; wake one waiter;

  • 共享内存同步：POSIX 未命名信号量 + 共享内存保护环形缓冲
  • 轻量级：比 SysV 信号量更轻量，适合 HFT

代码自测

1. sem_open 和 sem_init 的区别？什么时候用哪个？
  • sem_open：创建/打开命名信号量，用于无亲缘关系的进程间同步。信号量存放在 /dev/shm 下，通过名字访问。
  • sem_init：初始化未命名信号量，存储在调用者提供的内存中。配合共享内存用于进程间，或用于线程间。

选择：无亲缘进程用 sem_open；有共享内存的进程/线程用 sem_init。

2. POSIX 信号量为什么比 SysV 更易用？
  1. 单个信号量操作而非信号量集——概念简单
  2. 创建+初始化原子（SysV 需要两步，有竞态）
  3. 标准 API 接口简洁（sem_wait/sem_post）
  4. 支持超时等待（sem_timedwait）
  5. 与共享内存自然配合（未命名信号量）

练习 1 参考代码

#include <semaphore.h>
#include <sys/mman.h>
#include <stdio.h>

typedef struct {
    int buf[1024];
    int head, tail;
    sem_t sem_data;   /* 数据可用 */
    sem_t sem_space;  /* 空间可用 */
} shm_queue_t;

shm_queue_t *q;

void producer(void) {
    for (int i = 0; i < 10000; i++) {
        sem_wait(&q->sem_space);    /* 等空间 */
        q->buf[q->tail] = i;
        q->tail = (q->tail + 1) % 1024;
        sem_post(&q->sem_data);     /* 通知有数据 */
    }
}

void consumer(void) {
    for (int i = 0; i < 10000; i++) {
        sem_wait(&q->sem_data);     /* 等数据 */
        int item = q->buf[q->head];
        q->head = (q->head + 1) % 1024;
        sem_post(&q->sem_space);    /* 通知有空间 */
    }
}

int main(void) {
    q = mmap(NULL, sizeof(*q), PROT_READ|PROT_WRITE,
             MAP_SHARED|MAP_ANONYMOUS, -1, 0);
    sem_init(&q->sem_data, 1, 0);     /* pshared=1 */
    sem_init(&q->sem_space, 1, 1024);

    if (fork() == 0) { consumer(); _exit(0); }
    producer();
    wait(NULL);
    sem_destroy(&q->sem_data);
    sem_destroy(&q->sem_space);
    munmap(q, sizeof(*q));
    return 0;
}

练习 3 参考代码

#include <semaphore.h>
#include <time.h>
#include <stdio.h>

int main(void) {
    sem_t *sem = sem_open("/hft_timeout", O_CREAT, 0644, 0);

    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);
    ts.tv_sec += 5;  /* 5 秒超时 */

    printf("waiting with 5s timeout...\n");
    if (sem_timedwait(sem, &ts) == -1) {
        if (errno == ETIMEDOUT)
            printf("timeout! no signal received\n");
        else perror("sem_timedwait");
    } else {
        printf("signal received\n");
    }
    sem_close(sem);
    sem_unlink("/hft_timeout");
    return 0;
}
