代码示例

#include <stdio.h>
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/sem.h>
#include <unistd.h>
#include <sys/wait.h>

/* Ch47 SysV 信号量 — semget/semop/semctl。
 * 信号量用于同步: P 操作(等待) / V 操作(释放)。
 * 编译: gcc -o ch47_demo ch47_demo.c */

/* SysV 信号量是集合, 操作比较复杂 */
union semun {
    int val;
    struct semid_ds *buf;
    unsigned short *array;
};

void sem_op(int semid, int op) {
    struct sembuf sb = {
        .sem_num = 0,
        .sem_op = op,    /* -1 = P (wait), +1 = V (signal) */
        .sem_flg = 0
    };
    semop(semid, &sb, 1);
}

int main(void) {
    /* 创建包含 1 个信号量的集合 */
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    if (semid < 0) { perror("semget"); return 1; }

    /* 初始化信号量值为 1 (可用) */
    union semun arg;
    arg.val = 1;
    semctl(semid, 0, SETVAL, arg);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: P 操作 (获取信号量) */
        printf("Child: waiting for semaphore...\n");
        sem_op(semid, -1);  /* P: value 1->0 */
        printf("Child: got semaphore, working...\n");
        sleep(2);
        printf("Child: releasing semaphore\n");
        sem_op(semid, +1);  /* V: value 0->1 */
        _exit(0);
    }

    sleep(1);
    /* 父进程: 也尝试 P 操作, 会被阻塞直到子进程 V */
    printf("Parent: waiting for semaphore...\n");
    sem_op(semid, -1);  /* 会阻塞 */
    printf("Parent: got semaphore after child released\n");
    sem_op(semid, +1);

    waitpid(pid, NULL, 0);
    semctl(semid, 0, IPC_RMID);
    return 0;
}

47.1

概念

本章：TLPI 第 47 章 — System V Semaphores - 上一节：本章首页 - 下一节：47.2 –47.4 API

本节讲什么

  • 对象是信号量集（semid + 下标 0..nsems-1），非单个变量。 - 操作：P（减，不足则阻塞）· V（加，可唤醒）· 等值变为 0（sem_op==0）。 - 不传数据，只同步；典型保护 SysV shm。

要点

  • 对象是信号量集（semid + 下标 0..nsems-1），非单个变量。
  • 操作：P（减，不足则阻塞）· V（加，可唤醒）· 等值变为 0（sem_op==0）。
  • 不传数据，只同步；典型保护 SysV shm。
  • 内核持久 → IPC_RMID / 重启。

代码示例

#include <stdio.h>
#include <sys/sem.h>
/* 信号量概念: 计数器, P(减) / V(加), 用于同步 */
int main(void) {
    /* 信号量是一个内核维护的计数器 */
    /* P (wait): value--, 如果 < 0 则阻塞 */
    /* V (signal): value++, 如果 <= 0 则唤醒一个等待者 */

    /* SysV 信号量是"集合", 可以同时操作多个 */
    int semid = semget(IPC_PRIVATE, 3, IPC_CREAT | 0666);
    printf("Created semaphore set: id=%d, 3 semaphores\n", semid);

    /* 初始化每个信号量 */
    semctl(semid, 0, SETVAL, 1);  /* sem 0 = 1 (mutex) */
    semctl(semid, 1, SETVAL, 0);  /* sem 1 = 0 (event) */
    semctl(semid, 2, SETVAL, 5);  /* sem 2 = 5 (counting) */

    printf("sem[0]=%d (mutex, binary)\n", semctl(semid, 0, GETVAL));
    printf("sem[1]=%d (event, initially blocked)\n", semctl(semid, 1, GETVAL));
    printf("sem[2]=%d (counting, 5 resources)\n", semctl(semid, 2, GETVAL));

    semctl(semid, 0, IPC_RMID);
    return 0;
}

–47.4 API

本章：TLPI 第 47 章 — System V Semaphores - 上一节：47.1 概念 - 下一节：47.5 初始化竞态（经典坑）

本节讲什么

创建时指定 nsems；打开已有集时 nsems 不能大于集大小。 ⚠️ 只分配，不初始化数值。 struct sembuf { unsigned short sem_num; short sem_op; short sem_flg; };

要点

semget(key, nsems, semflg)

创建时指定 nsems；打开已有集时 nsems 不能大于集大小。
⚠️ 只分配，不初始化数值。

semctl · 须自声明 union semun

cmd
SETVAL / GETVAL 单个
SETALL / GETALL 整集
IPC_STAT / IPC_SET 属性
IPC_RMID 立即删整集；唤醒阻塞者

semop(semid, sops, nsops)

struct sembuf { unsigned short sem_num; short sem_op; short sem_flg; };
sem_op 行为
>0 V，不阻塞
<0 P；值不够则阻塞（或 IPC_NOWAIT→EAGAIN）
==0 等变为 0

sem_flg：IPC_NOWAIT · SEM_UNDO。
多条 sembuf：原子 all-or-nothing（可一次拿多锁，减死锁）。

路线 要点

代码示例

#include <stdio.h>
#include <sys/sem.h>
#include <unistd.h>
/* 信号量 API: semget/semctl/semop */
union semun { int val; struct semid_ds *buf; unsigned short *array; };

int main(void) {
    /* semget: 创建信号量集 */
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);

    /* semctl: 控制操作 */
    union semun arg;
    arg.val = 1;
    semctl(semid, 0, SETVAL, arg);          /* 设置值为 1 */
    printf("After SETVAL(1): %d\n", semctl(semid, 0, GETVAL));

    /* semop: P/V 操作 */
    struct sembuf sop;
    /* P 操作 (等待) */
    sop.sem_num = 0; sop.sem_op = -1; sop.sem_flg = 0;
    semop(semid, &sop, 1);
    printf("After P: %d\n", semctl(semid, 0, GETVAL));

    /* V 操作 (释放) */
    sop.sem_op = 1;
    semop(semid, &sop, 1);
    printf("After V: %d\n", semctl(semid, 0, GETVAL));

    /* IPC_NOWAIT: 非阻塞 P */
    sop.sem_op = -1; sop.sem_flg = IPC_NOWAIT;
    if (semop(semid, &sop, 1) < 0)
        perror("non-blocking P (EAGAIN expected)");

    semctl(semid, 0, IPC_RMID);
    return 0;
}

47.3

Semaphore Control Operations: semctl()

本章：TLPI 第 47 章

本节讲什么

semctl() 是 System V 信号量的控制接口，用于获取/设置信号量权限、删除信号量集、获取统计信息等。

要点

semctl() 函数

#include <sys/sem.h>
int semctl(int semid, int semnum, int cmd, ... /* union semun arg */);

union semun

union semun {
    int val;              /* SETVAL 用 */
    struct semid_ds *buf; /* IPC_STAT / IPC_SET 用 */
    unsigned short *array; /* GETALL / SETALL 用 */
};
/* 注意：需程序自定义，glibc 不提供 */

命令一览

cmd 说明 需要 arg?
IPC_RMID 立即删除信号量集 否
IPC_STAT 获取 semid_ds 是 (buf)
IPC_SET 设置权限 是 (buf)
GETVAL 获取单个信号量值 否
SETVAL 设置单个信号量值 是 (val)
GETALL 获取所有信号量值 是 (array)
SETALL 设置所有信号量值 是 (array)
GETPID 获取最后操作者 PID 否
GETNCNT 等待值增加的进程数 否
GETZCNT 等待值为0的进程数 否

struct semid_ds

struct semid_ds {
    struct ipc_perm sem_perm;  /* 权限 */
    time_t sem_otime;          /* 最后操作时间 */
    time_t sem_ctime;         /* 最后修改时间 */
    unsigned long sem_nsems;   /* 信号量数量 */
};

关键操作示例

/* 删除信号量集 */
semctl(semid, 0, IPC_RMID);

/* 设置信号量值为 1 */
union semun arg;
arg.val = 1;
semctl(semid, 0, SETVAL, arg);

/* 获取所有信号量值 */
unsigned short vals[10];
union semun arg;
arg.array = vals;
semctl(semid, 0, GETALL, arg);

/* 获取权限和统计信息 */
struct semid_ds ds;
union semun arg;
arg.buf = &ds;
semctl(semid, 0, IPC_STAT, arg);

HFT / 嵌入式关联

  • SysV 信号量在 HFT 中使用减少，主要被 POSIX 信号量/互斥锁替代
  • 嵌入式遗留系统可能使用 SysV 信号量做进程间同步
  • IPC_RMID 清理泄漏的信号量集

代码示例

#include <sys/sem.h>
#include <stdio.h>
#include <string.h>
#include <time.h>

/* union semun 需自定义 */
union semun {
    int val;
    struct semid_ds *buf;
    unsigned short *array;
};

void dump_sem_info(int semid, int num_sems) {
    struct semid_ds ds;
    union semun arg;
    arg.buf = &ds;

    if (semctl(semid, 0, IPC_STAT, arg) == -1) {
        perror("semctl IPC_STAT");
        return;
    }

    printf("=== Semaphore Set %d ===\n", semid);
    printf("Owner UID: %d  GID: %d\n",
           ds.sem_perm.cuid, ds.sem_perm.cgid);
    printf("Permissions: %04o\n", ds.sem_perm.mode);
    printf("Num sems: %lu\n", ds.sem_nsems);
    printf("Last op:  %s", ctime(&ds.sem_otime));
    printf("Last chg: %s", ctime(&ds.sem_ctime));

    /* 获取所有信号量值 */
    unsigned short vals[num_sems];
    arg.array = vals;
    if (semctl(semid, 0, GETALL, arg) != -1) {
        for (int i = 0; i < num_sems; i++)
            printf("  sem[%d] = %d\n", i, vals[i]);
    }
}

代码自测

为什么 union semun 需要程序自己定义？

POSIX 标准规定 union semun 由应用程序定义，而非系统头文件提供。不同系统的定义方式可能不同（某些 BSD 系统在头文件中提供），为保证可移植性，应用程序需要自己定义。这是 POSIX 规范的设计选择，避免将实现细节固化在标准中。

47.4

Semaphore Associated Data Structure

本章：TLPI 第 47 章

本节讲什么

System V 信号量集关联的 semid_ds 结构，包含权限信息、时间戳和统计字段。了解其布局有助于调试和监控。

要点

struct semid_ds 完整字段

struct semid_ds {
    struct ipc_perm sem_perm;   /* 所有权和权限 */
    time_t          sem_otime;  /* 最后 semop 时间 */
    time_t          sem_ctime;  /* 最后 semctl 修改时间 */
    unsigned long   sem_nsems;  /* 信号量集中的数量 */
    /* 内核内部字段... */
};

struct ipc_perm

struct ipc_perm {
    key_t          __key;   /* IPC key (ftok) */
    uid_t          uid;     /* 所有者 UID */
    gid_t          gid;     /* 所有者 GID */
    uid_t          cuid;    /* 创建者 UID */
    gid_t          cgid;    /* 创建者 GID */
    unsigned short mode;    /* 权限位 */
    unsigned short __seq;   /* 序列号 */
};

权限位语义

权限 读 修改(写)
所有者 GETVAL/GETALL SETVAL/SETALL/semop
组 同上 同上
其他 同上 同上

sem_otime vs sem_ctime

字段 何时更新 更新者
sem_otime 成功 semop() 内核
sem_ctime semctl(IPC_SET/SETVAL/SETALL) 内核

字段 监控用途

sem_ctime 配置变更时间
sem_perm.mode 权限审计

  • SysV 信号量监控：定期 IPC_STAT 检查活跃度
  • 权限审计：确保只有授权进程可操作
  • 嵌入式调试：用 ipcs -s 检查信号量泄漏

代码示例

#include <sys/sem.h>
#include <stdio.h>
#include <time.h>

union semun { int val; struct semid_ds *buf; unsigned short *array; };

void monitor_semaphore(int semid) {
    struct semid_ds ds;
    union semun arg;
    arg.buf = &ds;

    if (semctl(semid, 0, IPC_STAT, arg) == -1) {
        perror("semctl");
        return;
    }

    time_t now = time(NULL);
    printf("Semaphore set ID:  %d\n", semid);
    printf("Key:               0x%x\n", ds.sem_perm.__key);
    printf("Owner:             %d:%d\n", ds.sem_perm.uid, ds.sem_perm.gid);
    printf("Creator:           %d:%d\n", ds.sem_perm.cuid, ds.sem_perm.cgid);
    printf("Permissions:       %04o\n", ds.sem_perm.mode);
    printf("Num semaphores:    %lu\n", ds.sem_nsems);

    if (ds.sem_otime == 0)
        printf("Last semop:        never\n");
    else
        printf("Last semop:        %ld sec ago\n", now - ds.sem_otime);

    printf("Last change:       %ld sec ago\n", now - ds.sem_ctime);
}

代码自测

sem_otime 为 0 表示什么？

sem_otime 记录最后一次成功 semop() 的时间。值为 0 表示该信号量集创建后从未被 semop() 操作过。这是判断信号量是否被正确初始化的关键线索：semget() 创建的信号量初始值为 0，必须通过 semctl(SETVAL/SETALL) 设置初始值后才能使用。sem_otime 为 0 可能意味着初始化尚未完成。

47.5

初始化竞态（经典坑）

本章：TLPI 第 47 章 — System V Semaphores - 上一节：47.2 –47.4 API - 下一节：47.8 SEM_UNDO

本节讲什么

A semget(CREAT) 尚未 SETVAL，B 已 semop → 值未定义。 解法：仅 IPC_CREAT|IPC_EXCL 成功者初始化；EEXIST 则再 semget 打开。 semid = semget(key, 1, IPC_CREAT | IPC_EXCL | 0600);

要点

A semget(CREAT) 尚未 SETVAL，B 已 semop → 值未定义。

解法：仅 IPC_CREAT|IPC_EXCL 成功者初始化；EEXIST 则再 semget 打开。

semid = semget(key, 1, IPC_CREAT | IPC_EXCL | 0600);
if (semid >= 0) {
    union semun su = { .val = 1 };
    semctl(semid, 0, SETVAL, su);
} else if (errno == EEXIST) {
    semid = semget(key, 1, 0600);
}

代码示例

#include <stdio.h>
#include <sys/sem.h>
#include <string.h>
/* 信号量初始化: SETVAL vs SETALL (注意竞态!) */
union semun { int val; struct semid_ds *buf; unsigned short *array; };

int main(void) {
    int semid = semget(IPC_PRIVATE, 3, IPC_CREAT | 0666);

    /* 方式1: SETVAL 逐个设置 */
    for (int i = 0; i < 3; i++) {
        union semun arg;
        arg.val = i + 1;
        semctl(semid, i, SETVAL, arg);
    }
    printf("After SETVAL: %d %d %d\n",
           semctl(semid, 0, GETVAL),
           semctl(semid, 1, GETVAL),
           semctl(semid, 2, GETVAL));

    /* 方式2: SETALL 一次性设置所有 */
    union semun arg;
    unsigned short vals[3] = {10, 20, 30};
    arg.array = vals;
    semctl(semid, 0, SETALL, arg);
    printf("After SETALL: %d %d %d\n",
           semctl(semid, 0, GETVAL),
           semctl(semid, 1, GETVAL),
           semctl(semid, 2, GETVAL));

    /* GETALL: 一次性读取所有 */
    unsigned short get_vals[3];
    arg.array = get_vals;
    semctl(semid, 0, GETALL, arg);
    printf("GETALL: %d %d %d\n", get_vals[0], get_vals[1], get_vals[2]);

    /* 竞态问题: semget 创建后到 SETVAL 之前, 信号量值为 0 */
    /* 如果另一个进程在这期间做 P 操作, 可能意外阻塞 */
    /* 解决: 用 IPC_CREAT|IPC_EXCL, 只有创建者做初始化 */
    semctl(semid, 0, IPC_RMID);
    return 0;
}

47.6

Semaphore Operations: semop()

本章：TLPI 第 47 章

本节讲什么

semop() 是 System V 信号量的核心操作函数，执行原子性的信号量增减和等待操作。支持操作数组、非阻塞和撤销操作。

要点

semop() 函数

#include <sys/sem.h>
int semop(int semid, struct sembuf *sops, size_t nsops);
// 返回：0 成功，-1 失败

struct sembuf

struct sembuf {
    unsigned short sem_num;  /* 信号量编号（0-based） */
    short          sem_op;   /* 操作值 */
    short          sem_flg;  /* 标志 */
};

sem_op 含义

sem_op 行为
> 0 V 操作：值增加 sem_op，唤醒等待者
= 0 等待：阻塞直到值变为 0
< 0 P 操作：若值 ≥

sem_flg 标志

标志 效果
0 阻塞等待
IPC_NOWAIT 不阻塞，条件不满足时返回 EAGAIN
SEM_UNDO 进程退出时自动撤销操作

原子操作数组

struct sembuf ops[2] = {
    {0, -1, SEM_UNDO},  /* P 操作 sem[0] */
    {1, +1, SEM_UNDO},  /* V 操作 sem[1] */
};
semop(semid, ops, 2);  /* 原子执行两个操作 */

所有操作要么全部成功，要么全部不执行。

SEM_UNDO 机制

内核维护每个进程的 semadj（调整值）： - SEM_UNDO 操作记录在 semadj 中 - 进程正常/异常退出时，内核自动回滚 semadj 记录的操作 - 防止进程崩溃时持有锁导致死锁

常见操作模式

模式 sem_op sem_flg 用途
获取锁 -1 SEM_UNDO P 操作
释放锁 +1 SEM_UNDO V 操作
等待零 0 0 阻塞到归零
非阻塞尝试 -1 IPC_NOWAIT 条件获取

超时控制

semtimedop() (Linux 扩展) 支持超时：

#include <time.h>
struct timespec timeout = {.tv_sec = 5};
int ret = semtimedop(semid, ops, nops, &timeout);
// 超时返回 -1，errno = EAGAIN

HFT / 嵌入式关联

  • SysV 信号量做进程间互斥（多策略进程共享资源）
  • SEM_UNDO 防止进程崩溃导致死锁
  • semtedop 超时避免无限等待
  • 现代更推荐 POSIX 信号量/pthread 互斥

代码示例

#include <sys/sem.h>
#include <stdio.h>
#include <time.h>

union semun { int val; struct semid_ds *buf; unsigned short *array; };

/* 获取信号量锁 */
int sem_lock(int semid, int semnum) {
    struct sembuf op = {
        .sem_num = semnum,
        .sem_op = -1,
        .sem_flg = SEM_UNDO,
    };
    return semop(semid, &op, 1);
}

/* 释放信号量锁 */
int sem_unlock(int semid, int semnum) {
    struct sembuf op = {
        .sem_num = semnum,
        .sem_op = +1,
        .sem_flg = SEM_UNDO,
    };
    return semop(semid, &op, 1);
}

/* 带超时的获取 */
int sem_lock_timeout(int semid, int semnum, int sec) {
    struct sembuf op = {
        .sem_num = semnum,
        .sem_op = -1,
        .sem_flg = SEM_UNDO,
    };
    struct timespec ts = {.tv_sec = sec};
    return semtimedop(semid, &op, 1, &ts);
}

/* 原子操作：获取一个释放另一个 */
int sem_transfer(int semid, int acquire, int release) {
    struct sembuf ops[2] = {
        {acquire,  -1, SEM_UNDO},
        {release,  +1, SEM_UNDO},
    };
    return semop(semid, ops, 2);
}

代码自测

SEM_UNDO 的原理和限制是什么？

原理：内核为每个进程-信号量对维护一个 semadj（调整计数器）。SEM_UNDO 标志的 semop 会累加 semadj（P 操作记负，V 操作记正）。进程退出时内核自动用 semadj 的反值恢复信号量值，释放可能持有的锁。

限制：(1) semadj 是 int，可能溢出；(2) 进程被 SIGKILL 杀死时 SEM_UNDO 仍然生效（内核处理）；(3) 多线程中 semadj 是进程级共享的，不是线程级；(4) 只对单个信号量值回滚，无法处理复杂的操作序列。

47.7

Handling of Multiple Blocked Semaphore Operations

本章：TLPI 第 47 章

本节讲什么

当 semop() 的操作数组中部分操作可执行、部分需阻塞时的内核处理策略，以及死锁检测和取消行为。

要点

原子性规则

semop() 操作数组的所有操作是原子的： - 要么全部立即执行 - 要么全部阻塞等待 - 不会部分执行

内核处理流程

1. 检查所有 sops 的 sem_op 能否满足
   - sem_op > 0: 总能满足
   - sem_op = 0: 需要值 == 0
   - sem_op < 0: 需要值 >= |sem_op|

2. 如果全部可满足 → 全部执行 → 返回 0

3. 如果任一不满足:
   a. 无 IPC_NOWAIT → 阻塞等待
   b. 有 IPC_NOWAIT → 立即返回 EAGAIN

阻塞时的唤醒策略

当信号量值变化，内核按 FIFO 顺序检查阻塞队列中的等待者：

等待者队列: [A: sem[0] -= 1] [B: sem[1] = 0] [C: sem[0] -= 2]
sem[0] 变为 3:
  → 尝试 A: 3 >= 1 → 执行 → sem[0] = 2
  → 尝试 B: sem[1] 未变 → 不满足
  → 尝试 C: 2 >= 2 → 执行 → sem[0] = 0

唤醒的不确定性

  • 不保证公平性：唤醒顺序由内核策略决定
  • 饥饿可能：低优先级等待者可能被反复跳过
  • 批量唤醒：一个 V 操作可能唤醒多个等待者

死锁风险

进程 A: semop(semid, [{0, -1}, {1, +1}])  // 等 sem[0]
进程 B: semop(semid, [{1, -1}, {0, +1}])  // 等 sem[1]
→ A 持有 sem[0] 等 sem[1]，B 持有 sem[1] 等 sem[0]
→ 经典死锁！

信号中断处理

/* semop 被信号中断时返回 -1, errno = EINTR */
struct sigaction sa = {.sa_handler = handler};
sigaction(SIGINT, &sa, NULL);

int ret;
do {
    ret = semop(semid, &op, 1);
} while (ret == -1 && errno == EINTR);
/* 自动重启（Linux 默认不自动重启 semop） */

semop 与信号交互

场景 行为
信号中断 返回 -1, errno = EINTR
SA_RESTART semop 不自动重启（与 read 不同）
信号处理 + semop 需手动重试

HFT / 嵌入式关联

  • HFT 进程间同步避免使用 SysV 信号量（死锁风险高）
  • 必须使用时：确保操作顺序一致，避免交叉等待
  • semtimedop 限制等待时间，降低死锁影响
  • 嵌入式系统使用信号量保护共享硬件资源

代码示例

#include <sys/sem.h>
#include <stdio.h>
#include <errno.h>
#include <string.h>
#include <time.h>

/* 安全的 semop：处理 EINTR */
int safe_semop(int semid, struct sembuf *ops, size_t nops) {
    int ret;
    do {
        ret = semop(semid, ops, nops);
    } while (ret == -1 && errno == EINTR);
    return ret;
}

/* 带超时和 EINTR 处理 */
int safe_semop_timeout(int semid, struct sembuf *ops, size_t nops,
                        int timeout_sec) {
    int ret;
    struct timespec ts = {.tv_sec = timeout_sec};
    do {
        ret = semtimedop(semid, ops, nops, &ts);
        /* 超时后重置 timeout（已消耗的时间需要重新计算） */
    } while (ret == -1 && errno == EINTR);
    return ret;
}

/* 避免死锁的资源获取：按固定顺序 */
int acquire_two_semaphores(int semid) {
    /* 总是先 sem[0] 再 sem[1]，避免交叉死锁 */
    struct sembuf ops[2] = {
        {0, -1, SEM_UNDO},  /* 先获取 sem[0] */
        {1, -1, SEM_UNDO},  /* 再获取 sem[1] */
    };
    /* 原子执行：如果 sem[0] 够但 sem[1] 不够，全部阻塞 */
    return safe_semop_timeout(semid, ops, 2, 5);
}

代码自测

semop 操作数组中一个操作需要阻塞，其他操作可以立即执行，内核如何处理？

内核不会部分执行操作数组。如果任何一个操作不满足（需阻塞），整个操作数组都不会执行，进程阻塞直到所有操作都能满足。这保证了原子性：不会出现"执行了前半部分但后半部分阻塞"的中间状态。如果设置了 IPC_NOWAIT，任一操作不满足就立即返回 EAGAIN，不执行任何操作。

47.8

`SEM_UNDO`

本章：TLPI 第 47 章 — System V Semaphores - 上一节：47.5 初始化竞态（经典坑） - 下一节：47.9 二元信号量 ≈ 互斥

要点

带此标志的 semop 记入进程 undo 表；进程退出（含 kill）内核回放反向操作。
局限：只跟本进程；IPC_RMID 后 undo 失效；多次 P 叠加 undo；不能替代正常 V。

HFT / 嵌入式关联

路线 要点
嵌入式 SEM_UNDO 在嵌入式系统的应用场景

代码示例

#include <stdio.h>
#include <sys/sem.h>
#include <unistd.h>
#include <sys/wait.h>
/* SEM_UNDO: 进程退出时自动撤销信号量操作 */
int main(void) {
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    semctl(semid, 0, SETVAL, 1);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: P 操作带 SEM_UNDO 标志 */
        struct sembuf sop = {0, -1, SEM_UNDO};
        semop(semid, &sop, 1);
        printf("Child: P done, value=%d\n", semctl(semid, 0, GETVAL));

        /* 模拟异常退出 (不 V) */
        printf("Child: exiting without V (SEM_UNDO will auto-undo)\n");
        _exit(0);  /* 内核自动恢复信号量值 */
    }

    waitpid(pid, NULL, 0);
    /* 子进程退出后, SEM_UNDO 自动将值恢复 */
    printf("Parent: after child exit, value=%d (auto-restored to 1)\n",
           semctl(semid, 0, GETVAL));

    /* 不用 SEM_UNDO 的情况: 子进程 P 后崩溃, 信号量永远卡在 0 */
    semctl(semid, 0, IPC_RMID);
    return 0;
}

二元信号量 ≈ 互斥

本章：TLPI 第 47 章 — System V Semaphores - 上一节：47.8 SEM_UNDO - 下一节：47.10 –47.11 限额 · 缺陷

本节讲什么

HFT / 嵌入式关联

路线 要点
嵌入式 二元信号量 ≈ 互斥 在嵌入式系统的应用场景

代码示例

#include <stdio.h>
#include <sys/sem.h>
#include <unistd.h>
#include <sys/wait.h>
/* 用信号量实现互斥锁 (mutex) */
union semun { int val; };

int main(void) {
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    /* 初始化为 1 (未锁定) */
    union semun arg = {.val = 1};
    semctl(semid, 0, SETVAL, arg.val);

    /* 两个子进程争抢锁 */
    for (int i = 0; i < 2; i++) {
        if (fork() == 0) {
            struct sembuf lock = {0, -1, SEM_UNDO};   /* P = lock */
            struct sembuf unlock = {0, 1, SEM_UNDO};   /* V = unlock */

            semop(semid, &lock, 1);    /* 获取锁 */
            printf("Process %d: acquired lock\n", i);
            sleep(1);                   /* 临界区 */
            printf("Process %d: releasing lock\n", i);
            semop(semid, &unlock, 1);  /* 释放锁 */
            _exit(0);
        }
    }

    for (int i = 0; i < 2; i++) wait(NULL);
    printf("All done, final value=%d\n", semctl(semid, 0, GETVAL));
    semctl(semid, 0, IPC_RMID);
    return 0;
}

47.11

Disadvantages of System V Semaphores

本章：TLPI 第 47 章

本节讲什么

System V 信号量的固有缺陷和设计问题，以及为什么现代应用推荐使用 POSIX 信号量或 pthread 互斥锁替代。

要点

主要缺点

缺点 说明 影响
信号量集而非单个 API 操作的是信号量集合 使用复杂
创建和初始化非原子 semget + semctl 两步 竞争窗口
semop 不自动重启 信号中断返回 EINTR 需手动重试
内核限制多 SEMMNI/SEMMNS/SEMMSL 可扩展性差
无超时（标准） 需 semtimedop 扩展 不便控制
引用计数语义 创建者崩溃导致残留 资源泄漏
不支持进程内线程同步 设计目标为进程间 性能差

创建-初始化竞争

/* 问题：两步操作 */
int semid = semget(key, 1, IPC_CREAT | 0666);  /* step 1: 创建 */
semctl(semid, 0, SETVAL, 1);                    /* step 2: 初始化 */
/* 如果 step 1 后进程崩溃，信号量值为 0（未初始化） */
/* 另一个进程 semget 获得未初始化的信号量 */

解决方案（但仍不便）： 1. 使用 sem_otime 检查是否已初始化 2. 只允许创建者初始化（IPC_CREAT | IPC_EXCL）

IPC 资源持久性

SysV IPC 对象在进程退出后仍存在，直到： - 显式 IPC_RMID 删除 - 系统重启

这导致： - 崩溃的程序留下残留信号量 - 累积的 IPC 对象耗尽系统限制 - 需要手动 ipcrm 清理

SysV vs POSIX 信号量对比

特性 SysV POSIX (named) POSIX (unnamed)
类型 内核持久 文件系统持久 进程/共享内存内
API 复杂度 高 低 低
初始化 两步 原子 原子
操作 信号量集 单个 单个
超时 扩展 API sem_timedwait sem_timedwait
移植性 Linux/Unix POSIX.1-2001 POSIX.1-2001
性能 中 高 高

替代方案推荐

场景 推荐
进程间互斥 POSIX 信号量 / fcntl 文件锁
线程间同步 pthread 互斥锁 / 条件变量
进程间通知 eventfd / 管道
带超时等待 POSIX sem_timedwait

HFT / 嵌入式关联

  • HFT 进程间同步优先使用 POSIX 信号量或共享内存 + pthread 互斥锁
  • 避免 SysV 信号量的 EINTR 重试开销
  • 嵌入式系统内存有限，避免 SysV IPC 资源泄漏
  • 遗留系统维护：用 ipcs/ipcrm 清理

代码示例

/* SysV 信号量竞争安全初始化 */
#include <sys/sem.h>
#include <errno.h>

union semun { int val; struct semid_ds *buf; unsigned short *array; };

int init_semaphore(key_t key, int initial_val) {
    int semid, created = 0;

    /* 尝试创建 */
    semid = semget(key, 1, IPC_CREAT | IPC_EXCL | 0666);
    if (semid != -1) {
        created = 1;  /* 我们是创建者 */
    } else if (errno == EEXIST) {
        /* 已存在，打开 */
        semid = semget(key, 1, 0666);
        if (semid == -1) return -1;
    } else {
        return -1;
    }

    if (created) {
        /* 创建者负责初始化 */
        union semun arg;
        arg.val = initial_val;
        semctl(semid, 0, SETVAL, arg);
    }

    /* 等待初始化完成（非创建者） */
    struct semid_ds ds;
    union semun arg;
    arg.buf = &ds;
    int retry = 0;
    while (!created && retry++ < 100) {
        semctl(semid, 0, IPC_STAT, arg);
        if (ds.sem_otime != 0) break;  /* 已初始化 */
        usleep(10000);
    }

    return semid;
}

代码自测

为什么说 SysV 信号量的创建和初始化存在竞争窗口？

semget() 创建信号量集时，所有信号量初始值为 0。必须用 semctl(SETVAL) 单独设置初始值。在 semget 和 semctl 之间，如果另一个进程也调用 semget（不带 IPC_EXCL），它会获得一个值仍为 0 的信号量，而非期望的初始值。解决方案是检查 sem_otime（非 0 表示创建者已完成初始化并执行了 semop），但这需要忙等或轮询。

本节讲什么

第 47 章总结：System V 信号量的完整 API、操作模式和设计缺陷。

要点

API 总览

API 用途
ftok() 生成 IPC key
semget() 创建/打开信号量集
semop() 原子操作（P/V/等待零）
semtimedop() 带超时的 semop
semctl(IPC_RMID) 删除信号量集
semctl(SETVAL/SETALL) 设置初始值
semctl(IPC_STAT) 获取信息
ipcs -s 查看系统信号量
ipcrm -s 删除信号量

操作模式

操作 sem_op 效果
P (获取锁) -1 值减1，不足则阻塞
V (释放锁) +1 值加1，唤醒等待者
等待零 0 阻塞到值=0

关键标志

标志 用途
IPC_CREAT 创建信号量
IPC_EXCL 独占创建
SEM_UNDO 进程退出自动撤销
IPC_NOWAIT 非阻塞模式

信号量集 vs 单信号量

SysV 信号量操作的是集合（多个信号量值），而非单个。semop 可以原子操作集合中的多个信号量。

生命周期

semget(IPC_CREAT) → 创建（值为0）
  ↓
semctl(SETVAL) → 初始化
  ↓
semop() → P/V 操作
  ↓
semctl(IPC_RMID) → 显式删除

注意：进程退出后信号量仍存在，直到 IPC_RMID 或系统重启。

主要缺点

  1. 创建+初始化非原子
  2. semop 不自动重启
  3. 内核限制多
  4. IPC 资源泄漏
  5. API 复杂

现代 API 推荐

场景 推荐
进程间互斥 POSIX 信号量
线程同步 pthread 互斥
超时等待 sem_timedwait()
文件锁 fcntl(F_SETLK)

HFT / 嵌入式关联

  • HFT 新开发优先使用 POSIX 信号量或 pthread 互斥锁
  • SysV 信号量维护遗留系统：用 ipcs -s 监控，ipcrm -s 清理
  • SEM_UNDO 防止崩溃死锁（但 POSIX 互斥不提供等价机制）

代码自测

SEM_UNDO 在什么场景下特别重要？为什么 POSIX 信号量没有等价机制？

SEM_UNDO 在进程可能崩溃的锁场景下重要：进程获取锁后崩溃，内核自动回滚（释放锁），其他进程不会被永久阻塞。POSIX 命名信号量和互斥锁在进程崩溃时不会自动释放（sem_post 不会被调用），其他等待者会永久阻塞。POSIX 的 robust mutex（pthread_mutexattr_setrobust）提供了类似功能：持有者崩溃时，下一个获取者会得到 EOWNERDEAD，需调用 pthread_mutex_consistent 恢复。

练习 1：二元互斥锁

题目：使用 SysV 信号量实现二元互斥锁（初始值为1），包含 SEM_UNDO 保护。

参考答案 
#include <sys/sem.h>

union semun { int val; struct semid_ds *buf; unsigned short *array; };

typedef struct {
    int semid;
    int semnum;
} sem_mutex_t;

int sem_mutex_init(sem_mutex_t *m, key_t key) {
    m->semid = semget(key, 1, IPC_CREAT | 0666);
    if (m->semid == -1) return -1;
    m->semnum = 0;

    /* 初始化为 1（解锁状态） */
    union semun arg;
    arg.val = 1;
    return semctl(m->semid, 0, SETVAL, arg);
}

int sem_mutex_lock(sem_mutex_t *m) {
    struct sembuf op = {
        .sem_num = m->semnum,
        .sem_op = -1,        /* P: 值减1 */
        .sem_flg = SEM_UNDO, /* 退出时自动恢复 */
    };
    int ret;
    do {
        ret = semop(m->semid, &op, 1);
    } while (ret == -1 && errno == EINTR);
    return ret;
}

int sem_mutex_unlock(sem_mutex_t *m) {
    struct sembuf op = {
        .sem_num = m->semnum,
        .sem_op = +1,        /* V: 值加1 */
        .sem_flg = SEM_UNDO,
    };
    return semop(m->semid, &op, 1);
}

void sem_mutex_destroy(sem_mutex_t *m) {
    semctl(m->semid, 0, IPC_RMID);
}

练习 2：生产者-消费者

题目：用 SysV 信号量集（3个信号量）实现生产者-消费者同步。

参考答案 
/* 信号量集：
 * sem[0] = mutex (1, 互斥访问缓冲区)
 * sem[1] = empty (N, 空位数量)
 * sem[2] = full  (0, 满位数量)
 */
#define SEM_MUTEX 0
#define SEM_EMPTY 1
#define SEM_FULL  2
#define BUF_SIZE 10

void producer(int semid) {
    struct sembuf ops[2];
    /* P(empty), P(mutex) */
    ops[0] = (struct sembuf){SEM_EMPTY, -1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_MUTEX, -1, SEM_UNDO};
    semop(semid, ops, 2);

    /* 写入数据到缓冲区 */
    /* ... */

    /* V(mutex), V(full) */
    ops[0] = (struct sembuf){SEM_MUTEX, +1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_FULL,  +1, SEM_UNDO};
    semop(semid, ops, 2);
}

void consumer(int semid) {
    struct sembuf ops[2];
    /* P(full), P(mutex) */
    ops[0] = (struct sembuf){SEM_FULL,  -1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_MUTEX, -1, SEM_UNDO};
    semop(semid, ops, 2);

    /* 从缓冲区读取数据 */
    /* ... */

    /* V(mutex), V(empty) */
    ops[0] = (struct sembuf){SEM_MUTEX, +1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_EMPTY, +1, SEM_UNDO};
    semop(semid, ops, 2);
}

练习 3：初始化竞争

题目：以下代码有什么竞争？如何修复？

int semid = semget(key, 1, IPC_CREAT | 0666);
semctl(semid, 0, SETVAL, 1);  /* 初始化 */
参考答案

竞争：两个进程同时执行此代码，semget 返回同一个 semid（已存在），但两者都调用 SETVAL 设置为 1。如果在这期间有第三个进程执行 semop(P)，可能看到值 0（未初始化）。

修复：使用 IPC_EXCL 确保只有创建者初始化：

int semid = semget(key, 1, IPC_CREAT | IPC_EXCL | 0666);
if (semid != -1) {
    /* 我们是创建者，负责初始化 */
    union semun arg = {.val = 1};
    semctl(semid, 0, SETVAL, arg);
} else if (errno == EEXIST) {
    /* 已存在，打开即可 */
    semid = semget(key, 1, 0666);
    /* 等待创建者初始化完成 */
    struct semid_ds ds;
    union semun arg = {.buf = &ds};
    while (ds.sem_otime == 0) {
        semctl(semid, 0, IPC_STAT, arg);
        usleep(1000);
    }
}
