#include <stdio.h>
#include <time.h>
#include <sys/time.h>
#include <unistd.h>

/* Ch10 时间 — time/gettimeofday/clock_gettime/nanosleep。
 * 演示多种获取时间的方式 + 精度差异。
 * 编译: gcc -o ch10_demo ch10_demo.c */

int main(void) {
    /* time(): 秒级精度 */
    time_t t = time(NULL);
    printf("time():       %ld (%s)", (long)t, ctime(&t));

    /* gettimeofday(): 微秒精度 */
    struct timeval tv;
    gettimeofday(&tv, NULL);
    printf("gettimeofday: %ld.%06ld\n", (long)tv.tv_sec, (long)tv.tv_usec);

    /* clock_gettime(CLOCK_MONOTONIC): 纳秒精度，不受系统时间调整影响 */
    struct timespec ts1, ts2;
    clock_gettime(CLOCK_MONOTONIC, &ts1);

    nanosleep(&(struct timespec){0, 100 * 1000 * 1000}, NULL);  /* 100ms */

    clock_gettime(CLOCK_MONOTONIC, &ts2);
    long delta_ns = (ts2.tv_sec - ts1.tv_sec) * 1000000000L
                  + (ts2.tv_nsec - ts1.tv_nsec);
    printf("nanosleep(100ms) actual: %ld ns (%.2f ms)\n",
           delta_ns, delta_ns / 1000000.0);
    return 0;
}

与 C 语言学习的关系

时间 API 是 C 系统编程的常见需求。K&R 没有深入讲解时间 API，但 time_t 和 struct tm 是 C 标准的一部分。在 HFT 场景中，高精度时钟（clock_gettime + CLOCK_MONOTONIC）是延迟测量的基础。注意 time_t 的类型不要假设为 long——应使用 inttypes.h 的格式化宏。

HFT / 嵌入式关联

路线 要点
嵌入式 要点梳理 在嵌入式系统的应用场景
HFT 要点梳理 对低延迟和性能的影响

常见陷阱

  1. 混淆 CLOCK_REALTIME 和 CLOCK_MONOTONIC
/* CLOCK_REALTIME: 可被 NTP 调整，会跳变 */
/* CLOCK_MONOTONIC: 单调递增，不可调 */
clock_gettime(CLOCK_MONOTONIC, &ts);

测量时间间隔必须用 CLOCK_MONOTONIC。CLOCK_REALTIME 可能被 NTP 往回调整，导致负的时间差。

  1. time_t 类型假设
long t = time(NULL);  /* 错误假设 */
time_t t = time(NULL); /* 正确 */

time_t 的底层类型是实现定义的（可能是 int32_t 或 int64_t）。不要假设为 long，用 time_t 类型。

  1. sleep 被信号中断
sleep(10);  /* 被信号中断后返回，不保证睡满 */
/* 返回值 = 剩余秒数 */

sleep 被信号中断后返回剩余秒数。需要精确延时应检查返回值并继续 sleep 剩余时间，或用 nanosleep。

自测题

题目 1（选择）： 测量两个事件之间的时间间隔，应该用哪个时钟？ A. CLOCK_REALTIME B. CLOCK_MONOTONIC C. CLOCK_PROCESS_CPUTIME_ID D. time()

参考答案

B。CLOCK_MONOTONIC 单调递增不受 NTP 影响。CLOCK_REALTIME 可能被调整导致负间隔。

题目 2（判断）： time_t 一定是 long 类型。

参考答案

错误。time_t 的底层类型是实现定义的。在 32 位系统上可能是 32 位整数（2038 年问题）。应直接用 time_t 类型。

题目 3（踩坑）： 以下代码有什么问题？

struct timespec start, end;
clock_gettime(CLOCK_REALTIME, &start);
/* ... do work ... */
clock_gettime(CLOCK_REALTIME, &end);
double elapsed = (end.tv_sec - start.tv_sec) +
                 (end.tv_nsec - start.tv_nsec) / 1e9;
参考答案

用 CLOCK_REALTIME 测量时间间隔。如果 NTP 在期间调整了时钟，elapsed 可能为负。应改用 CLOCK_MONOTONIC。

题目 4（应用）： 写代码用 clock_gettime 测量函数执行时间（纳秒精度）。

参考答案 
struct timespec start, end;
clock_gettime(CLOCK_MONOTONIC, &start);
foo();
clock_gettime(CLOCK_MONOTONIC, &end);
long ns = (end.tv_sec - start.tv_sec) * 1000000000L
         + (end.tv_nsec - start.tv_nsec);
printf("elapsed: %ld ns
", ns);

代码示例

#include <stdio.h>
#include <time.h>
#include <sys/time.h>
/* 时间测量: CLOCK_REALTIME vs CLOCK_MONOTONIC vs CLOCK_PROCESS_CPUTIME_ID */
int main(void) {
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);
    printf("CLOCK_REALTIME:     %ld.%09ld\n", (long)ts.tv_sec, (long)ts.tv_nsec);
    clock_gettime(CLOCK_MONOTONIC, &ts);
    printf("CLOCK_MONOTONIC:    %ld.%09ld\n", (long)ts.tv_sec, (long)ts.tv_nsec);
    clock_gettime(CLOCK_PROCESS_CPUTIME_ID, &ts);
    printf("CLOCK_PROC_CPUTIME: %ld.%09ld\n", (long)ts.tv_sec, (long)ts.tv_nsec);
    return 0;
}

10.5

Updating the System Clock

本章：TLPI 第 10 章 — Time

本节讲什么

介绍如何修改系统时钟，包括 settimeofday()、clock_settime() 和 adjtime() 三种方式，以及 NTP 守护进程如何微调时钟。

要点

时钟设置 API

API 原型 说明 特权要求
settimeofday(tv, tz) int settimeofday(const struct timeval *tv, const struct timezone *tz) 设置墙上时钟（已废弃） CAP_SYS_TIME
clock_settime(clockid, ts) int clock_settime(clockid_t clk, const struct timespec *ts) 设置指定时钟（推荐） CAP_SYS_TIME
adjtime(delta, olddelta) int adjtime(const struct timeval *delta, struct timeval *olddelta) 微调时钟（渐变） CAP_SYS_TIME

三种方式对比

方式 效果 跳变 适用
settimeofday 直接设置 跳变（时间突变） 时钟偏差很大时
clock_settime(REALTIME, &ts) 直接设置 跳变 推荐替代 settimeofday
adjtime(&delta, NULL) 渐进微调 平滑（每秒最多微调约 500μs） NTP 日常校正

adjtime 工作原理

当前时间: 100.000000
delta:    +0.010000 (10ms 快了)

adjtime 后：
  → 不是立即减 10ms
  → 而是在约 20 秒内，每秒减 500μs
  → 20 秒后时钟准确，无跳变
  • 每秒最多调整约 500 微秒（500 / 1000000 比例）
  • 如果 delta 为正：时钟加速（每秒多跑一点）
  • 如果 delta 为负：时钟减速
  • 正在进行的 adjtime 可以被新调用取消（olddelta 返回剩余调整量）

NTP 与 adjtime

NTP 守护进程 (ntpd/chronyd)
  │
  ├── 定期查询 NTP 服务器
  ├── 计算本地时钟偏差 (offset)
  ├── 如果偏差 < 128ms → 调用 adjtime() 平滑校正
  └── 如果偏差 > 128ms → 直接 clock_settime() 跳变（需人工确认）

struct timeval vs struct timespec

结构 精度 定义
timeval 微秒（μs） {time_t tv_sec; suseconds_t tv_usec;}
timespec 纳秒（ns） {time_t tv_sec; long tv_nsec;}

HFT / 嵌入式关联

  • PTP（精确时间协议）：HFT 场景下纳秒级时钟同步，用 clock_adjtime() 更细粒度调整
  • chrony：现代 Linux 默认时间同步服务，比传统 ntpd 更适合低延迟场景
  • 嵌入式：无 RTC 的板子需从 NTP/PTP 获取时间后 clock_settime 设置
  • 硬件时间戳：网卡支持 SO_TIMESTAMPING，与软件时钟配合校准
  • 禁止跳变：HFT 系统绝不允许 settimeofday 跳变，所有校时用 adjtime 平滑

代码示例

#include <sys/time.h>
#include <time.h>
#include <stdio.h>

/* 直接设置系统时间（需 CAP_SYS_TIME） */
int set_clock(time_t new_sec) {
    struct timespec ts = { .tv_sec = new_sec, .tv_nsec = 0 };
    if (clock_settime(CLOCK_REALTIME, &ts) != 0) {
        perror("clock_settime");
        return -1;
    }
    return 0;
}

/* NTP 风格的微调（需 CAP_SYS_TIME） */
int nudge_clock(long delta_usec) {
    struct timeval delta = { .tv_sec = 0, .tv_usec = delta_usec };
    if (adjtime(&delta, NULL) != 0) {
        perror("adjtime");
        return -1;
    }
    return 0;
}

/* 查看 adjtime 剩余调整量 */
void show_pending_adj(void) {
    struct timeval remaining;
    if (adjtime(NULL, &remaining) == 0) {
        printf("pending adjustment: %ld.%06ld s\n",
               (long)remaining.tv_sec, (long)remaining.tv_usec);
    }
}

代码自测

1. 为什么 NTP 用 adjtime 而不是 clock_settime？

clock_settime 会造成时间跳变，影响： 1. 定时器（基于时间差的定时可能立刻触发或永远不会触发） 2. 日志时间戳不连续，难以追踪 3. 测量代码性能时数据不连续 4. 分布式系统的时间排序逻辑出错

adjtime 每秒只微调约 500μs，时钟平滑移动，避免跳变，对上述场景无影响。

2. settimeofday 为什么被废弃？
  1. POSIX 已标记为废弃
  2. timezone 参数无意义（Linux 不使用时区信息）
  3. 功能被 clock_settime(CLOCK_REALTIME, &ts) 完全覆盖，后者精度更高（纳秒 vs 微秒）
  4. 非标准扩展的 timezone 参数容易误用
3. 如果进程没有 CAP_SYS_TIME，调用 clock_settime 会怎样？

返回 -1，errno 设为 EPERM（操作不允许）。只有具有 CAP_SYS_TIME 能力的进程（通常是 root 或通过 capabilities 授权的进程）才能修改系统时钟。

10.6

The Software Clock (Jiffies)

本章：TLPI 第 10 章 — Time

本节讲什么

介绍 Linux 内核软件时钟的概念：jiffies（时钟中断滴答）、HZ 配置、分辨率限制，以及 clock_getres() 如何查询时钟精度。

要点

jiffies 概念

概念 说明
jiffies 内核全局计数器，每次时钟中断 +1
HZ 时钟中断频率（每秒中断次数），编译时配置
jiffy 一个时钟滴答的时长 = 1/HZ 秒
分辨率 软件时钟的最小可分辨时间间隔

HZ 配置值

HZ jiffy 内核版本 典型场景
100 10ms 2.4 及更早 桌面/服务器
250 4ms 2.6 默认 通用服务器
300 3.33ms 某些发行版 桌面/服务器
1000 1ms 2.6 可选 低延迟/桌面

HZ 对系统的影响

高 HZ（1000） 低 HZ（100）
计时更精确 计时粗糙
定时器精度更高 定时器精度低
中断开销更大 中断开销小
上下文切换更频繁 CPU 利用率更高
低延迟应用受益 吞吐量应用受益
功耗更高 笔记本省电

查询分辨率

struct timespec res;
clock_getres(CLOCK_REALTIME, &res);
/* res.tv_sec = 0, res.tv_nsec = 4000000 (4ms) 当 HZ=250 */
clockid 典型分辨率 说明
CLOCK_REALTIME 1/HZ 墙上时钟
CLOCK_MONOTONIC 1/HZ 单调时钟
CLOCK_MONOTONIC_RAW 硬件精度 不受 NTP 调整
CLOCK_MONOTONIC_COARSE 1/HZ 快速但粗糙
CLOCK_REALTIME_COARSE 1/HZ 快速但粗糙

clock_getres API

int clock_getres(clockid_t clk_id, struct timespec *res);
/* 返回 0 成功，res 填入分辨率 */

高分辨率定时器（hrtimer）

Linux 2.6.21+ 引入高分辨率定时器： - 不依赖 jiffies，直接使用硬件时钟源（TSC/HPET） - 支持纳秒级定时器 - clock_nanosleep() / timer_settime() 可达纳秒精度 - 前提：内核 CONFIG_HIGH_RES_TIMERS=y

定时器类型 分辨率 依赖
传统定时器 jiffies（4ms） HZ
hrtimer 纳秒 硬件时钟源
CLOCK_MONOTONIC_RAW 硬件精度 TSC

HFT / 嵌入式关联

  • HFT 时钟需求：微秒级时间戳，必须用 hrtimer + CLOCK_MONOTONIC
  • TSC（Time Stamp Counter）：x86 CPU 硬件计数器，纳秒精度，HFT 首选时钟源
  • rdtsc 指令直接读 TSC，无需系统调用
  • 嵌入式：ARM 的 Generic Timer（如 aarch64 的 CNTVCT）替代 jiffies 做高精度计时
  • 内核编译：HFT 专用内核通常 CONFIG_HZ_1000=y + CONFIG_HIGH_RES_TIMERS=y + CONFIG_PREEMPT_RT=y
  • 时钟源选择：/sys/devices/system/clocksource/clocksource0/current_clocksource 查看当前时钟源

代码示例

#include <time.h>
#include <stdio.h>

void show_clock_resolutions(void) {
    clockid_t clocks[] = {
        CLOCK_REALTIME, CLOCK_MONOTONIC,
        CLOCK_MONOTONIC_RAW, CLOCK_REALTIME_COARSE,
        CLOCK_MONOTONIC_COARSE
    };
    const char *names[] = {
        "REALTIME", "MONOTONIC",
        "MONOTONIC_RAW", "REALTIME_COARSE",
        "MONOTONIC_COARSE"
    };
    struct timespec res;
    for (int i = 0; i < 5; i++) {
        clock_getres(clocks[i], &res);
        printf("%-20s: %ld.%09ld s (%ld ns)\n", names[i],
               (long)res.tv_sec, res.tv_nsec,
               res.tv_sec * 1000000000L + res.tv_nsec);
    }
}

/* 使用 rdtsc 直接读 TSC（x86） */
#ifdef __x86_64__
#include <stdint.h>
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ __volatile__ ("rdtsc" : "=a"(lo), "=d"(hi));
    return ((uint64_t)hi << 32) | lo;
}
#endif

代码自测

1. HZ=250 时，clock_gettime(CLOCK_MONOTONIC) 返回的时间精度是多少？

理论上分辨率为 4ms（1/250 秒），但实际精度取决于实现： - 如果使用 hrtimer（高分辨率定时器），CLOCK_MONOTONIC 可达纳秒精度，不受 HZ 限制 - 如果未启用 hrtimer，精度为 4ms - 现代内核默认启用 hrtimer，所以 clock_gettime 通常为纳秒精度 - clock_getres 可以查询实际分辨率

2. CLOCK_MONOTONIC_COARSE 为什么比 CLOCK_MONOTONIC 快？

CLOCK_MONOTONIC_COARSE 是 Linux 特有时钟，它直接读取内核的 jiffies 计数器（jiffies 全局变量），不陷入硬件时钟源查询，开销极低。但精度只有 1/HZ（通常 4ms）。适用于不需要高精度但需要频繁获取时间戳的场景（如日志记录、统计计数）。CLOCK_MONOTONIC 则通过硬件时钟源（TSC/HPET）获取纳秒级时间，开销较高。

3. 为什么 HFT 场景通常编译 HZ=1000？
  1. 更高的 HZ 意味着更频繁的调度检查，减少调度延迟
  2. 定时器精度更高（1ms vs 4ms）
  3. CLOCK_MONOTONIC 在无 hrtimer 时精度更高
  4. HFT 对延迟敏感，中断开销可接受
  5. 现代 HFT 系统通常还启用 CONFIG_PREEMPT_RT（实时抢占），配合高 HZ 实现可预测的延迟

代码自测

1. CLOCK_REALTIME 和 CLOCK_MONOTONIC 的区别？HFT 应该用哪个？
  • CLOCK_REALTIME：墙上时钟，表示实际日期时间，受 NTP/管理员修改，可能跳变
  • CLOCK_MONOTONIC：单调时钟，从某个未定义起点递增，不受 settimeofday 影响，但受 NTP 平滑调整

HFT 应用： - 测量间隔、延迟 → CLOCK_MONOTONIC（必须） - 时间戳日志（需人可读日期） → CLOCK_REALTIME - 最好同时记录两者：REALTIME 用于人可读时间，MONOTONIC 用于间隔计算

2. 为什么 gettimeofday 被废弃？
  1. clock_gettime 提供纳秒精度（timespec），而 gettimeofday 只有微秒（timeval）
  2. gettimeofday 的 timezone 参数无意义
  3. POSIX.1-2008 标记为废弃
  4. clock_gettime 支持多种时钟（MONOTONIC 等），更灵活
  5. 现代 glibc 中 gettimeofday 内部调用 clock_gettime，但接口不够好
3. 如何测量一段代码的纳秒级执行时间？ 
struct timespec start, end;
clock_gettime(CLOCK_MONOTONIC, &start);
/* 待测代码 */
clock_gettime(CLOCK_MONOTONIC, &end);
long long ns = (end.tv_sec - start.tv_sec) * 1000000000LL
              + (end.tv_nsec - start.tv_nsec);
printf("耗时: %lld ns\n", ns);

如果需要更低开销，用 x86 的 rdtsc 指令直接读 TSC，配合 CPU 频率换算。

练习 3：时间格式化

编写程序，获取当前时间并格式化输出为 "2026-08-12 19:30:45.123456" 格式（带微秒）。

参考答案

练习 1 参考代码

#include <time.h>
#include <stdio.h>

#define TIMING_START(ts) clock_gettime(CLOCK_MONOTONIC, &(ts))
#define TIMING_NS(start, end) \
    (((end).tv_sec - (start).tv_sec) * 1000000000LL \
   + ((end).tv_nsec - (start).tv_nsec))

int main(void) {
    struct timespec start, end, raw_start, raw_end;

    clock_gettime(CLOCK_MONOTONIC, &start);
    clock_gettime(CLOCK_MONOTONIC_RAW, &raw_start);

    /* 待测代码：计算 sum */
    volatile long long sum = 0;
    for (long long i = 0; i < 100000000LL; i++) sum += i;

    clock_gettime(CLOCK_MONOTONIC, &end);
    clock_gettime(CLOCK_MONOTONIC_RAW, &raw_end);

    long long ns_monotonic = TIMING_NS(start, end);
    long long ns_raw       = TIMING_NS(raw_start, raw_end);

    printf("CLOCK_MONOTONIC     : %lld.%09lld s (%lld ns)\n",
           ns_monotonic / 1000000000LL, ns_monotonic % 1000000000LL, ns_monotonic);
    printf("CLOCK_MONOTONIC_RAW : %lld.%09lld s (%lld ns)\n",
           ns_raw / 1000000000LL, ns_raw % 1000000000LL, ns_raw);
    printf("diff                 : %lld ns\n", ns_monotonic - ns_raw);
    return 0;
}

练习 2 答案

特性 CLOCK_REALTIME CLOCK_MONOTONIC
含义 实际日期时间 从启动起单调递增
单调性 否（可能跳变） 是（单调递增）
受 settimeofday 影响 是 否
受 adjtime 影响 是 是（平滑调整速率）
起点 1970-01-01 UTC 未定义（通常系统启动）

使用场景： - CLOCK_REALTIME：需要日期时间的场景（日志时间戳、文件 mtime、定时唤醒到特定时刻） - CLOCK_MONOTONIC：测量时间间隔（延迟、性能、超时），绝不能用于计算日期

练习 3 参考代码

#include <time.h>
#include <stdio.h>

int main(void) {
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);

    struct tm tm_buf;
    localtime_r(&ts.tv_sec, &tm_buf);

    /* 格式: 2026-08-12 19:30:45.123456 */
    char buf[64];
    strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &tm_buf);
    printf("%s.%06ld\n", buf, ts.tv_nsec / 1000);  /* 纳秒→微秒 */
    return 0;
}
