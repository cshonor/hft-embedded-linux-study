章节目标

掌握 exec 六兄弟与 execve；理清保留/销毁资源；FD_CLOEXEC / O_CLOEXEC；PATH 与 shebang；熟练 fork + exec + waitpid。

27.6 工业范式：fork + exec

pid = fork();
if (pid == 0) {
    /* close / redirect fds */
    execvp(prog, argv);
    _exit(127);          /* 禁止 exit() */
}
waitpid(pid, &st, 0);    /* 父 */

立刻 exec → COW 几乎不触发；多线程场景也相对安全。

27.8 易错清单

  1. exec 成功后无后续业务代码
  2. handler 重置；掩码保留
  3. CLOEXEC 只对 exec
  4. root + execvp PATH 风险
  5. 失败用 _exit
  6. exec 不改 PID；新进程靠 fork
  7. argv[0] 约定为名，内核不强制

练习 / 实验清单

  1. execvp / execl
  2. FD_CLOEXEC 跨 exec
  3. （选）shebang
  4. fork+exec+重定向模板
  5. （选）exec 前后 handler 对比

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <string.h>
#include <stdlib.h>

/* Ch27 程序执行 — execl/execlp/execv/execvp。
 * exec 替换当前进程映像，pid 不变。
 * 编译: gcc -o ch27_demo ch27_demo.c */

int main(void) {
    /* 方式1: execlp — 搜索 PATH，变长参数 */
    pid_t pid = fork();
    if (pid == 0) {
        printf("Child: calling execlp(\"ls\")\n");
        execlp("ls", "ls", "-l", "/tmp", NULL);
        perror("execlp");  /* 只有失败才返回 */
        _exit(1);
    }
    waitpid(pid, NULL, 0);

    /* 方式2: execvp — 搜索 PATH，数组参数 */
    pid = fork();
    if (pid == 0) {
        char *args[] = {"echo", "hello from execvp", NULL};
        printf("Child: calling execvp(\"echo\")\n");
        execvp("echo", args);
        perror("execvp");
        _exit(1);
    }
    waitpid(pid, NULL, 0);

    /* 方式3: execv — 完整路径，数组参数 */
    pid = fork();
    if (pid == 0) {
        char *args[] = {"/bin/date", NULL};
        printf("Child: calling execv(\"/bin/date\")\n");
        execv("/bin/date", args);
        perror("execv");
        _exit(1);
    }
    waitpid(pid, NULL, 0);

    /* 传递环境变量给 exec */
    pid = fork();
    if (pid == 0) {
        char *args[] = {"env", NULL};
        char *envp[] = {"MY_VAR=hello_from_parent", NULL};
        printf("Child: calling execve with custom env\n");
        execve("/usr/bin/env", args, envp);
        perror("execve");
        _exit(1);
    }
    waitpid(pid, NULL, 0);
    return 0;
}

六兄弟

本章：TLPI 第 27 章 — Program Execution - 上一节：本章首页 - 下一节：27.2 保留 vs 销毁

本节讲什么

真正的系统调用是 execve；其余为库封装。 int execve(const char pathname, char const argv[], char *const envp[]); int execl / execlp / execle(...);

要点

真正的系统调用是 execve；其余为库封装。

后缀 含义
l 参数列表，末尾 (char *)NULL
v argv[] 向量
p 用 PATH 搜文件名
e 自带 envp[]；否则继承 environ 
int execve(const char *pathname, char *const argv[], char *const envp[]);
int execl / execlp / execle(...);
int execv / execvp(...);

成功永不返回；失败返回 -1。

execvp("ls", argv);
_exit(127);

Demo：code/fork_exec.c

HFT / 嵌入式关联

路线 要点
嵌入式 execve 是唯一系统调用，其余为库封装；l/v/p/e 后缀语义
HFT exec 变体按传参方式选；execvp 按 PATH 搜，execve 显式 envp

#include <unistd.h>

        char *args[] = {"ls", "-la", "/tmp", NULL};
        execvp("ls", args);
        perror("execvp");  /* 只有失败才返回 */
        _exit(1);

}

✅ 大致保留

PID/PPID/PGID/SID · cwd · umask · 未设 CLOEXEC 的 fd · 凭证（SUID 见 Ch9）· 信号掩码 · rlimit 等

❌ 销毁 / 重置

#include <unistd.h>
#include <stdio.h>
int main(void) {
    char *args[] = {"echo", "hello", "exec", NULL};

     * execl/execlp/execle: 参数列表 (l)
     * execv/execvp/execve: 参数数组 (v)
     * p 后缀: 搜索 PATH
     * e 后缀: 自定义环境变量
     */

    if (pid == 0) { execvp("echo", args); perror("execvp"); _exit(1); }
    wait(NULL);
    return 0;
}

27.3

`FD_CLOEXEC` / `O_CLOEXEC`

本章：TLPI 第 27 章 — Program Execution - 上一节：27.2 保留 vs 销毁 - 下一节：27.4 PATH（*p）

本节讲什么

打开时带 O_CLOEXEC（首选，减竞态）；或 fcntl(F_SETFD, FD_CLOEXEC)。 Demo：code/cloexec_demo.c ---

要点

fork exec
CLOEXEC fd 不关 内核关掉

打开时带 O_CLOEXEC（首选，减竞态）；或 fcntl(F_SETFD, FD_CLOEXEC)。

Demo：code/cloexec_demo.c

HFT / 嵌入式关联

路线 要点
嵌入式 O_CLOEXEC 首选（减竞态），exec 时内核关 fd 防泄漏
HFT O_CLOEXEC 防 fd 泄漏到 worker，安全 + 资源不失控（HFT 最佳实践）

代码示例

#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>
int main(void) {
    /* O_CLOEXEC: exec 时自动关闭 fd, 防泄漏 */
    int fd = open("/tmp/cloexec.txt", O_CREAT | O_RDWR | O_CLOEXEC, 0644);

    pid_t pid = fork();
    if (pid == 0) {
        char *args[] = {"ls", "-la", "/proc/self/fd", NULL};
        execvp("ls", args);
        _exit(1);
    }
    wait(NULL);
    /* 没有 O_CLOEXEC 的 fd 在 exec 后仍然打开 (泄漏) */
    printf("O_CLOEXEC: exec 后 fd 自动关闭\n");
    close(fd);
    return 0;
}

27.4

PATH（`*p`）

本章：TLPI 第 27 章 — Program Execution - 上一节：27.3 FD_CLOEXEC / O_CLOEXEC - 下一节：27.5 shebang #!

本节讲什么

  • 名中无 / → 按 PATH 搜 - 含 / → 当路径，不搜 PATH - 特权程序慎用 *p（PATH 含 . 等劫持风险）

要点

  • 名中无 / → 按 PATH 搜
  • 含 / → 当路径，不搜 PATH
  • 特权程序慎用 *p（PATH 含 . 等劫持风险）

HFT / 嵌入式关联

路线 要点
嵌入式 名无 / 按 PATH 搜，含 / 当路径；特权程序慎用 *p
HFT execvp 按 PATH 搜有劫持风险，生产用完整路径 execv

代码示例

#include <unistd.h>
#include <stdio.h>
int main(void) {
    /* execvp/execlp: 搜索 PATH */
    char *args[] = {"date", NULL};

    pid_t pid = fork();
    if (pid == 0) {
        execvp("date", args);  /* 自动找 /bin/date */
        perror("execvp");
        _exit(1);
    }
    wait(NULL);

    /* execv/execl: 不搜索 PATH, 需要完整路径 */
    pid = fork();
    if (pid == 0) {
        char *args2[] = {"date", NULL};
        execv("/bin/date", args2);  /* 必须完整路径 */
        _exit(1);
    }
    wait(NULL);
    return 0;
}

27.6

Executing a Shell Command: system()

本章：TLPI 第 27 章 — Program Execution

本节讲什么

介绍 system() 函数：执行 shell 命令的高层接口，内部通过 fork() + exec("/bin/sh") + waitpid() 实现。涵盖其返回值语义、安全问题和替代方案。

要点

system() 函数

#include <stdlib.h>
int system(const char *command);
  • command 为 NULL：检查 shell 是否可用（非 NULL 返回非 0）
  • 内部实现：fork() → 子进程 execl("/bin/sh", "sh", "-c", command, NULL)
  • 父进程 waitpid() 等待子进程结束

返回值

返回值 含义
-1 fork() 失败（errno 有值）
127 exec() 失败（shell 不可用）
其他 子进程退出状态（需 WEXITSTATUS 解析）

返回值解析

int status = system("ls -l");
if (status == -1) {
    /* fork 失败 */
} else if (WIFEXITED(status)) {
    int exit_code = WEXITSTATUS(status);  /* 0-255 */
    /* shell 命令的退出码 */
} else if (WIFSIGNALED(status)) {
    int sig = WTERMSIG(status);
    /* 被信号终止 */
}

system() 的安全问题

/* 危险！命令注入 */
char user_input[256];
scanf("%255s", user_input);
char cmd[512];
snprintf(cmd, sizeof(cmd), "echo %s > /tmp/out", user_input);
system(cmd);  /* user_input = "; rm -rf /" → 灾难！ */
风险 说明
命令注入 用户输入被 shell 解释为元字符（; \``$()
环境变量 PATH/SHELL/LD_PRELOAD 等可被篡改
信号继承 system() 执行期间 SIGCHLD 被阻塞
竞争条件 fork/exec 之间的状态变化

安全替代方案

方案 说明
fork() + execvp() 不经过 shell，无命令注入
posix_spawn() POSIX 标准替代
exec* 系列 直接执行，无 shell 解析

system() vs fork+exec 对比

特性 system() fork() + exec()
经过 shell 是 否
命令注入风险 有 无
支持管道/重定向 是（shell 语法） 需手动处理
信号处理 阻塞 SIGCHLD 完全控制
返回值 退出状态 完全控制
灵活性 低 高
安全性 低（除非命令完全可信） 高

HFT / 嵌入式关联

  • 绝不使用 system()：HFT 服务中任何用户可控的输入都不能传给 system()
  • 安全执行：用 fork() + execvp() 替代，完全控制参数和环境变量
  • 嵌入式：BusyBox 系统中 /bin/sh 可能是 ash/dash，system() 行为可能有差异
  • 脚本调用：HFT 运维脚本可用 system() 执行简单命令，但核心服务不行
  • 信号风险：system() 执行时阻塞 SIGCHLD，可能干扰信号处理逻辑

代码示例

#include "tlpi_hdr.h"
#include <sys/wait.h>

/* system() 的简化实现（教学用） */
int my_system(const char *command) {
    if (command == NULL)
        return 1;  /* shell 可用 */

    pid_t pid = fork();
    if (pid == -1) return -1;  /* fork 失败 */

    if (pid == 0) {  /* 子进程 */
        execl("/bin/sh", "sh", "-c", command, (char *)NULL);
        _exit(127);  /* exec 失败 */
    }

    int status;
    if (waitpid(pid, &status, 0) == -1) return -1;
    return status;
}

/* 安全替代方案：不经过 shell */
int safe_exec(const char *file, char *const argv[]) {
    pid_t pid = fork();
    if (pid == -1) return -1;
    if (pid == 0) {
        execvp(file, argv);
        _exit(127);
    }
    int status;
    waitpid(pid, &status, 0);
    return status;
}

int main(void) {
    /* 危险：可能命令注入 */
    int s1 = system("ls -l /tmp");

    /* 安全：不经过 shell */
    char *argv[] = {"ls", "-l", "/tmp", NULL};
    int s2 = safe_exec("ls", argv);

    return 0;
}

代码自测

1. system(NULL) 的作用是什么？

system(NULL) 用于检查当前系统是否有可用的 shell。如果 /bin/sh 存在且可执行，返回非 0（通常为 1）。如果 shell 不可用，返回 0。这用于在调用实际命令前检查环境是否支持 system()。

2. 为什么 setuid 程序中绝不能用 system()？
  1. 环境变量不安全：PATH、IFS、LD_PRELOAD 等可被用户篡改，shell 可能执行恶意程序
  2. 命令注入：即使命令看起来安全，shell 元字符可能导致执行额外命令
  3. 权限继承：setuid 程序以 root 运行，system() 产生的 shell 也以 root 运行，风险极大
  4. POSIX 明确禁止：setuid 程序不应使用 system()，应使用 exec() 直接执行
3. system() 执行期间 SIGCHLD 信号如何处理？

system() 实现中，执行期间会阻塞 SIGCHLD 信号（sigprocmask），防止父进程的 SIGCHLD 处理器干扰 waitpid。同时也会阻塞 SIGINT 和 SIGQUIT，防止 Ctrl-C 影响子进程但不影响父进程。这是 POSIX 规定的行为，确保 system() 的原子性。

无 e：用 environ。有 e：自定义 envp，末项 NULL。
特权启动宜清理危险变量（如 LD_*）。

HFT / 嵌入式关联

路线 要点
嵌入式 无 e 用 environ，有 e 自定义 envp；特权启动清理 LD_*
HFT 环境变量传配置；特权 worker 清理 LD_* 防劫持

代码示例

#include <unistd.h>
#include <stdio.h>
extern char **environ;
int main(void) {
    /* execle/execve: 传自定义环境变量 */
    char *args[] = {"env", NULL};
    char *envp[] = {"MY_VAR=hello", "PATH=/usr/bin", NULL};

    pid_t pid = fork();
    if (pid == 0) {
        execve("/usr/bin/env", args, envp);
        perror("execve");
        _exit(1);
    }
    wait(NULL);
    /* execvp/execlp: 继承当前环境变量 */
    return 0;
}

代码自测

1. exec 后哪些属性会变？哪些不变？

变：程序代码/数据/栈/堆全部替换、信号处理程序恢复为默认、CPU 时间统计清零 不变：PID/PPID、real UID/GID、打开的 fd（无 CLOEXEC 的）、CWD、umask、CPU 亲和性、nice 值、已 attach 的共享内存

2. execvp 如何搜索 PATH？

execvp("ls", argv) 依次搜索 PATH 环境变量中的每个目录（用 : 分隔），查找名为 ls 的可执行文件。如果找到且有执行权限就执行。如果文件名包含 /（如 ./myprog），不搜索 PATH 直接使用。PATH="" 表示只搜索当前目录。

3. 如何防止 fd 泄漏到 exec 的子进程？

方法 1：对每个 fd 设置 FD_CLOEXEC（fcntl(fd, F_SETFD, FD_CLOEXEC)） 方法 2：打开时使用 O_CLOEXEC flag（open(path, O_RDONLY | O_CLOEXEC)） 方法 3：exec 前手动 close() 所有不需要的 fd 方法 4：closefrom(fd) 关闭 fd 及以上（非标准，但 Linux/BSD 有）

推荐方法 2，因为它是原子的——在 open 和 fcntl 之间不会被信号打断。

练习 1 参考代码

#include "tlpi_hdr.h"
#include <sys/wait.h>
#include <wordexp.h>

#define MAX_ARGS 64

int main(void) {
    char line[1024];
    char *argv[MAX_ARGS];

    while (1) {
        printf("$ ");
        fflush(stdout);
        if (fgets(line, sizeof(line), stdin) == NULL) break;  /* EOF */

        /* 解析参数 */
        int argc = 0;
        char *tok = strtok(line, " \t\n");
        while (tok && argc < MAX_ARGS - 1) {
            argv[argc++] = tok;
            tok = strtok(NULL, " \t\n");
        }
        argv[argc] = NULL;

        if (argc == 0) continue;  /* 空行 */
        if (strcmp(argv[0], "exit") == 0) break;

        pid_t pid = fork();
        if (pid == -1) { perror("fork"); continue; }
        if (pid == 0) {
            execvp(argv[0], argv);
            /* exec 失败 */
            fprintf(stderr, "%s: %s\n", argv[0], strerror(errno));
            _exit(127);
        }
        int status;
        waitpid(pid, &status, 0);
        if (WIFEXITED(status) && WEXITSTATUS(status) != 0)
            fprintf(stderr, "exit code: %d\n", WEXITSTATUS(status));
    }
    return 0;
}

练习 3 参考代码

#include <unistd.h>
#include <sys/wait.h>
#include <stdarg.h>
#include <string.h>

int safe_system(const char *file, char *const argv[]) {
    pid_t pid = fork();
    if (pid == -1) return -1;

    if (pid == 0) {
        execvp(file, argv);
        _exit(127);  /* exec 失败 */
    }

    int status;
    if (waitpid(pid, &status, 0) == -1) return -1;
    return status;  /* 调用者用 WEXITSTATUS 解析 */
}

/* 用法示例 */
int main(void) {
    char *argv[] = {"ls", "-l", "/tmp", NULL};
    int status = safe_system("ls", argv);
    if (status != -1 && WIFEXITED(status))
        printf("exit code: %d\n", WEXITSTATUS(status));
    return 0;
}
