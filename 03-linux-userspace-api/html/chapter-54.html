代码示例

#include <stdio.h>
#include <sys/mman.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <unistd.h>
#include <string.h>
#include <sys/wait.h>

/* Ch54 POSIX 共享内存 — shm_open/ftruncate/mmap/shm_unlink。
 * 比 SysV shm 更简洁的 API。
 * 编译: gcc -o ch54_demo ch54_demo.c -lrt */

#define SHM_NAME "/ch54_demo"
#define SHM_SIZE 4096

int main(void) {
    /* 创建/打开 POSIX 共享内存对象 */
    int fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0644);
    if (fd < 0) { perror("shm_open"); return 1; }

    /* 设置大小 */
    ftruncate(fd, SHM_SIZE);

    /* 映射 */
    char *shm = mmap(NULL, SHM_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (shm == MAP_FAILED) { perror("mmap"); return 1; }
    close(fd);  /* 映射后可以关闭 fd */

    /* 写入数据 */
    strcpy(shm, "Hello from parent via POSIX shm!");

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 映射同一共享内存对象 */
        int cfd = shm_open(SHM_NAME, O_RDWR, 0);
        char *cshm = mmap(NULL, SHM_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, cfd, 0);
        close(cfd);

        printf("Child reads: %s\n", cshm);
        strcpy(cshm, "Child modified the shared memory!");

        munmap(cshm, SHM_SIZE);
        _exit(0);
    }

    waitpid(pid, NULL, 0);
    printf("Parent reads: %s\n", shm);

    munmap(shm, SHM_SIZE);
    shm_unlink(SHM_NAME);
    return 0;
}

/* POSIX 共享内存原理：shm_open 创建文件 → mmap 映射 → 多进程共享 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/mman.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    /* 步骤 1: shm_open 创建共享内存对象（tmpfs 中的文件） */
    int fd = shm_open("/principle_shm", O_CREAT | O_RDWR, 0644);
    if (fd < 0) { perror("shm_open"); exit(1); }
    ftruncate(fd, 256);

    /* 步骤 2: mmap 映射到虚拟地址空间 */
    char *shm = mmap(NULL, 256, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (shm == MAP_FAILED) { perror("mmap"); exit(1); }
    close(fd);

    strcpy(shm, "shared data");

    /* fork 后子进程继承映射 */
    pid_t pid = fork();
    if (pid == 0) {
        printf("child reads: %s\n", shm);
        strcpy(shm, "child modified");
        _exit(0);
    }
    wait(NULL);
    printf("parent reads: %s\n", shm);

    munmap(shm, 256);
    shm_unlink("/principle_shm");
    return 0;
}

/* shm_open / shm_unlink API 完整用法 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    /* 创建共享内存 */
    int fd = shm_open("/api_shm", O_CREAT | O_RDWR, 0644);
    if (fd < 0) { perror("shm_open"); exit(1); }
    printf("shm_open: fd=%d\n", fd);

    /* 设置大小（必须，否则 mmap 报 BUS error） */
    ftruncate(fd, 4096);

    /* 查看属性 */
    struct stat sb;
    fstat(fd, &sb);
    printf("size=%lld, mode=%04o\n", (long long)sb.st_size, sb.st_mode & 0777);

    /* 映射并使用 */
    char *p = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);  /* 映射后 fd 不再需要 */

    memset(p, 'A', 4096);
    printf("filled 4096 bytes with 'A'\n");

    munmap(p, 4096);

    /* shm_unlink 删除对象名（引用计数为 0 时释放） */
    shm_unlink("/api_shm");
    printf("shm_unlink: object marked for deletion\n");
    return 0;
}

54.3

Using Shared Memory Objects

本章：TLPI 第 54 章 — POSIX Shared Memory

本节讲什么

介绍 POSIX 共享内存对象的使用流程：shm_open() 创建/打开 → ftruncate() 设置大小 → mmap() 映射 → 读写 → munmap() 解除。

要点

POSIX 共享内存 API

函数 说明
shm_open(name, oflag, mode) 创建/打开共享内存对象（类似 open）
shm_unlink(name) 删除共享内存对象
ftruncate(fd, size) 设置对象大小
mmap(addr, len, prot, flags, fd, 0) 映射到进程地址空间
fstat(fd, &sb) 查看对象信息（大小等）
fchmod(fd, mode) 修改权限
close(fd) 关闭 fd（对象仍存在）

使用流程

1. shm_open("/hft_data", O_CREAT|O_RDWR, 0644)  → fd
2. ftruncate(fd, 1024*1024)                      → 设置 1MB 大小
3. mmap(NULL, 1024*1024, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0)  → ptr
4. 读写 ptr 指向的内存
5. munmap(ptr, 1024*1024)                        → 解除映射
6. close(fd)                                      → 关闭 fd
7. shm_unlink("/hft_data")                        → 删除对象

shm_open flags

flag 说明
O_CREAT 不存在则创建
O_EXCL 配合 O_CREAT，存在则失败
O_RDWR 读写
O_RDONLY 只读
O_TRUNC 截断为 0

与 SysV 共享内存对比

特性 POSIX shm_open SysV shmget
接口 文件式（open/ftruncate/mmap） 系统调用式
位置 /dev/shm/ 内核匿名
管理工具 ls /dev/shm/ ipcs -m
大小控制 ftruncate 创建时指定
权限 文件权限 ipc_perm
可移植性 POSIX SysV

mmap 选项

/* POSIX 共享内存的标准 mmap 用法 */
void *p = mmap(NULL, size,
               PROT_READ | PROT_WRITE,
               MAP_SHARED | MAP_HASSEMICOLON,
               fd, 0);

/* 常用 flags 组合 */
MAP_SHARED          /* 共享映射（多进程可见） */
MAP_SHARED | MAP_LOCKED       /* 锁定在内存中 */
MAP_SHARED | MAP_HUGETLB      /* 使用大页 */
MAP_SHARED | MAP_POPULATE     /* 预填充页表 */

/dev/shm 文件系统

$ ls -l /dev/shm/
total 1024
-rw-r--r-- 1 hft hft 1048576 Aug 12 19:00 hft_data
  • POSIX 共享内存对象以文件形式存在于 /dev/shm（tmpfs）
  • df -h /dev/shm 查看可用空间
  • 可用 ls/rm 管理（但推荐用 shm_unlink）

HFT / 嵌入式关联

  • 行情共享：HFT 行情进程创建共享内存对象，策略进程 mmap 读取
  • MAP_LOCKED：锁定页面防止换页延迟
  • MAP_HUGETLB：使用 2MB 大页减少 TLB miss
  • MAP_POPULATE：启动时预填充页表，避免运行时缺页
  • NUMA：mbind 绑定共享内存到特定 NUMA 节点
  • 嵌入式：/dev/shm 可能需要手动挂载或增大

代码示例

#include <sys/mman.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>

#define SHM_NAME "/hft_market_data"
#define SHM_SIZE (64 * 1024 * 1024)  /* 64MB */

int main(void) {
    /* 创建共享内存对象 */
    int fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0644);
    if (fd == -1) { perror("shm_open"); return 1; }

    /* 设置大小 */
    if (ftruncate(fd, SHM_SIZE) == -1) { perror("ftruncate"); return 1; }

    /* 映射 */
    void *ptr = mmap(NULL, SHM_SIZE,
                     PROT_READ | PROT_WRITE,
                     MAP_SHARED | MAP_LOCKED,
                     fd, 0);
    if (ptr == MAP_FAILED) { perror("mmap"); return 1; }

    /* 写入数据 */
    memset(ptr, 0, SHM_SIZE);
    printf("shared memory ready: %p (%d MB)\n", ptr, SHM_SIZE/(1024*1024));

    /* 另一个进程可以 shm_open + mmap 访问同一内存 */

    /* 清理 */
    munmap(ptr, SHM_SIZE);
    close(fd);
    shm_unlink(SHM_NAME);
    return 0;
}

代码自测

1. shm_open 后为什么需要 ftruncate？

shm_open 创建的对象初始大小为 0 字节。如果直接 mmap 一个大小为 0 的对象，mmap 会失败（ EINVAL）。ftruncate 将对象扩展到所需大小，之后 mmap 才能映射。这是 POSIX 共享内存的标准流程。

2. close(fd) 后共享内存还有效吗？

是的。close(fd) 只关闭文件描述符，共享内存对象仍然存在于 /dev/shm 中，其他已 mmap 的进程仍然可以访问。对象在 shm_unlink 被调用且所有映射解除后才真正释放。

54.4

Removing Shared Memory Objects

本章：TLPI 第 54 章 — POSIX Shared Memory

本节讲什么

介绍如何正确删除 POSIX 共享内存对象：shm_unlink() 的语义、延迟删除行为和清理策略。

要点

shm_unlink 语义

int shm_unlink(const char *name);
行为 说明
从 /dev/shm 删除名字 新进程无法通过名字打开
已有 fd/mapping 不受影响 正在使用的进程不受影响
最后一个映射解除后释放内存 引用计数为 0 时回收
返回值 0 成功，-1 失败

延迟删除模型

进程 A: shm_open("/hft_data", O_CREAT, 0644)
进程 A: mmap(...) → refcount = 1
进程 B: shm_open("/hft_data", 0) → fd
进程 B: mmap(...) → refcount = 2

进程 A: shm_unlink("/hft_data")  → 名字删除，但内存还在
进程 B: 仍可正常访问 → refcount = 2

进程 A: munmap() → refcount = 1
进程 B: munmap() + close() → refcount = 0 → 内存释放

清理策略

策略 说明 适用
创建者 unlink 创建后立即 unlink 临时共享（无需其他进程通过名字访问）
最后使用者 unlink 使用完成后 unlink 长期共享
启动时清理 程序启动时 unlink 残留 防崩溃残留
系统重启 tmpfs 重启清空 终极清理

常见问题

问题 原因 解决
shm_open EEXIST 对象已存在且用了 O_EXCL 先 unlink 或不用 O_EXCL
泄漏 创建但未 unlink 定期清理 /dev/shm
权限错误 权限不匹配 fchmod 修改
大小为 0 忘记 ftruncate 创建后立即 ftruncate

命令行管理

# 查看
ls -l /dev/shm/
df -h /dev/shm

# 删除
rm /dev/shm/hft_data    # 等价 shm_unlink
# 或
python3 -c "import posix_ipc; posix_ipc.unlink_shared_memory('hft_data')"

# 批量清理
rm /dev/shm/hft_*       # 谨慎！

HFT / 嵌入式关联

  • 启动清理：HFT 服务启动时先 shm_unlink 可能残留的旧对象
  • 立即 unlink：创建后立即 shm_unlink（fd 已持有），防止崩溃残留
  • 大小监控：df -h /dev/shm 监控共享内存总量
  • 嵌入式：/dev/shm 空间有限，需严格管理

代码示例

#include <sys/mman.h>
#include <fcntl.h>
#include <stdio.h>

/* 安全创建模式：创建后立即 unlink */
void *create_temp_shm(const char *name, size_t size) {
    int fd = shm_open(name, O_CREAT | O_RDWR | O_EXCL, 0600);
    if (fd == -1) {
        /* 可能上次崩溃残留，清理后重试 */
        shm_unlink(name);
        fd = shm_open(name, O_CREAT | O_RDWR, 0600);
        if (fd == -1) return NULL;
    }
    ftruncate(fd, size);
    void *p = mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);
    shm_unlink(name);   /* 立即 unlink，fd 仍有效 */
    return p;
}

/* 正常创建模式：使用者负责清理 */
void *create_persistent_shm(const char *name, size_t size) {
    int fd = shm_open(name, O_CREAT | O_RDWR, 0644);
    if (fd == -1) return NULL;
    ftruncate(fd, size);
    void *p = mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);
    /* 不 unlink，其他进程可通过名字访问 */
    return p;
}

代码自测

1. shm_unlink 后正在使用的进程会怎样？

不受影响。shm_unlink 只删除名字，不影响已打开的 fd 和已映射的内存。正在使用的进程可以继续读写。只有当所有 fd 关闭且所有 mmap 解除后，内存才被真正回收。这是"延迟删除"设计。

2. 什么时候应该立即 unlink？

如果共享内存只用于已知的固定进程组（不通过名字被新进程发现），可以在所有使用者打开 fd 后立即 shm_unlink。好处：(1) 程序崩溃不留残留；(2) 不占用 /dev/shm 名字空间。fd 仍然有效直到 close + munmap。

#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    /* 1. 创建 + 设置大小 */
    int fd = shm_open("/life_shm", O_CREAT | O_RDWR, 0644);
    ftruncate(fd, 128);

    /* 2. 映射 */
    char *p = mmap(NULL, 128, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);

    /* 3. 使用 */
    strcpy(p, "lifecycle demo");

    /* 4. 解除映射（数据仍在 tmpfs 中） */
    munmap(p, 128);
    printf("munmap: mapping removed, object still exists\n");

    /* 5. 关闭 fd（对象仍存在） */
    close(fd);
    printf("close: fd closed, object still exists\n");

    /* 6. 重新打开验证数据持久 */
    fd = shm_open("/life_shm", O_RDWR, 0);
    p = mmap(NULL, 128, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    printf("reopen: data = '%s'\n", p);
    munmap(p, 128);
    close(fd);

    /* 7. unlink 删除（所有引用关闭后释放） */
    shm_unlink("/life_shm");
    printf("shm_unlink: object name removed\n");
    return 0;
}

/* POSIX 共享内存 vs SysV 共享内存对比 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/mman.h>
#include <sys/shm.h>
#include <unistd.h>

int main(void) {
    /* --- POSIX 方式 --- */
    int pfd = shm_open("/vs_shm", O_CREAT | O_RDWR, 0644);
    ftruncate(pfd, 256);
    char *posix_shm = mmap(NULL, 256, PROT_READ | PROT_WRITE, MAP_SHARED, pfd, 0);
    close(pfd);
    strcpy(posix_shm, "POSIX shm");
    printf("POSIX: name='/vs_shm', mmap, data='%s'\n", posix_shm);
    munmap(posix_shm, 256);
    shm_unlink("/vs_shm");

    /* --- SysV 方式（对比） --- */
    key_t key = IPC_PRIVATE;  /* 简化：用私有 key */
    int shmid = shmget(key, 256, IPC_CREAT | 0644);
    if (shmid >= 0) {
        char *sysv_shm = shmat(shmid, NULL, 0);
        if ((long)sysv_shm != -1) {
            strcpy(sysv_shm, "SysV shm");
            printf("SysV:  key=IPC_PRIVATE, shmget+shmat, data='%s'\n", sysv_shm);
            shmdt(sysv_shm);
        }
        shmctl(shmid, IPC_RMID, NULL);
    }

    printf("\n对比:\n");
    printf("  POSIX: name-based, shm_open+ftruncate+mmap\n");
    printf("  SysV:  key-based,  shmget+shmat+shmdt+shmctl\n");
    return 0;
}

#include <sys/mman.h>
#include <fcntl.h>
#include <semaphore.h>
#include <stdio.h>
#include <unistd.h>

typedef struct {
    int counter;
    sem_t sem;
} shared_data_t;

int main(void) {
    /* 创建共享内存 */
    int fd = shm_open("/hft_counter", O_CREAT|O_RDWR, 0644);
    ftruncate(fd, sizeof(shared_data_t));
    shared_data_t *sd = mmap(NULL, sizeof(*sd),
                             PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);

    sem_init(&sd->sem, 1, 1);  /* pshared=1, 初始=1 (mutex) */
    sd->counter = 0;

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程：递增 */
        for (int i = 0; i < 100000; i++) {
            sem_wait(&sd->sem);
            sd->counter++;
            sem_post(&sd->sem);
        }
        _exit(0);
    }
    /* 父进程：递减 */
    for (int i = 0; i < 100000; i++) {
        sem_wait(&sd->sem);
        sd->counter--;
        sem_post(&sd->sem);
    }
    wait(NULL);

    printf("final counter: %d (expected 0)\n", sd->counter);
    sem_destroy(&sd->sem);
    munmap(sd, sizeof(*sd));
    shm_unlink("/hft_counter");
    return 0;
}

        md->volume = 1000 + i;

}

练习 3 参考代码

void *safe_shm_create(const char *name, size_t size) {
    /* 尝试创建，O_EXCL 确保新建 */
    int fd = shm_open(name, O_CREAT|O_EXCL|O_RDWR, 0600);
    if (fd == -1 && errno == EEXIST) {
        /* 残留对象，清理后重建 */
        shm_unlink(name);
        fd = shm_open(name, O_CREAT|O_EXCL|O_RDWR, 0600);
    }
    if (fd == -1) return NULL;

    ftruncate(fd, size);
    void *p = mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);
    if (p == MAP_FAILED) { shm_unlink(name); return NULL; }

    /* 立即 unlink 防止再次残留 */
    shm_unlink(name);
    return p;
}
