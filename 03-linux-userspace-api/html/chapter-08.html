章节目标

理解 Linux 用户/组；掌握 passwd/group/shadow、查询 API、crypt 身份校验；为 Ch9 进程凭证铺垫。

Ch8 vs Ch9 速查

Ch8 Users and Groups Ch9 Process Credentials
焦点 账户数据库（文件 + 查询） 进程运行时 UID/GID 集合
内容 passwd/group/shadow、crypt RUID/EUID/SUID、setuid、切换
不讲 setuid 程序、权限切换 账户文件字段细节

易错清单

  1. passwd 可读；shadow 严格限制。
  2. getpwnam/getpwuid/crypt 静态缓冲，连续调用覆盖。
  3. pw_passwd 仅占位；真密文在 shadow。
  4. 本章 无 RUID/EUID/saved UID / setuid 程序。
  5. getpwnam 失败时先清 errno 再区分「不存在」vs「出错」。

章节链路

Ch6  进程是谁在跑
  → Ch8  系统里有哪些用户/组（数据库）
  → Ch9  这个进程此刻用哪套 UID/GID（凭证）
  → Ch15 文件 rwx 如何用 UID/GID 判定

双线提示

路线
嵌入式 设备节点/配置文件属主；勿把 shadow 逻辑塞进普通服务
HFT 少直接碰账户库；跑单用户/专用账号即可；权限切换见 Ch9

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>
#include <grp.h>
#include <pwd.h>

/* Ch8 用户与组 — getuid/getgid/getgroups + getpwuid/getgrgid。
 * 演示获取当前进程的用户/组信息。
 * 编译: gcc -o ch8_demo ch8_demo.c */

int main(void) {
    uid_t uid = getuid();
    gid_t gid = getgid();
    printf("uid=%u, gid=%u\n", uid, gid);

    /* 获取用户名 */
    struct passwd *pw = getpwuid(uid);
    if (pw)
        printf("user: %s (%s)\n", pw->pw_name, pw->pw_gecos);

    /* 获取组名 */
    struct group *gr = getgrgid(gid);
    if (gr)
        printf("group: %s\n", gr->gr_name);

    /* 获取所有附属组 */
    int ngroups = getgroups(0, NULL);
    if (ngroups > 0) {
        gid_t groups[32];
        getgroups(ngroups, groups);
        printf("supplementary groups (%d):", ngroups);
        for (int i = 0; i < ngroups; i++) {
            struct group *g = getgrgid(groups[i]);
            printf(" %s", g ? g->gr_name : "?");
        }
        printf("\n");
    }
    return 0;
}

`/etc/passwd`（全局可读）

本章：TLPI 第 08 章 — Users and Groups - 上一节：8.1 UID & GID - 下一节：8.3 /etc/group（全局可读）

本节讲什么

格式（7 字段，冒号分隔）： name:x:uid:gid:comment:home:shell #include

要点

格式（7 字段，冒号分隔）：

name:x:uid:gid:comment:home:shell
字段 含义
name 登录名
x 密码占位；真密文在 /etc/shadow
uid / gid 用户 ID / 主组 ID
comment GECOS / finger
home 家目录
shell 登录 shell；/sbin/nologin 等禁止登录 
#include <pwd.h>
struct passwd {
    char *pw_name;
    char *pw_passwd;   /* 占位，非真密文 */
    uid_t pw_uid;
    gid_t pw_gid;
    char *pw_gecos;
    char *pw_dir;
    char *pw_shell;
};

查询 API

struct passwd *getpwnam(const char *name);
struct passwd *getpwuid(uid_t uid);

void setpwent(void);
struct passwd *getpwent(void);
void endpwent(void);

找不到用户：返回 NULL，可能不改 errno。范式：

errno = 0;
pwd = getpwnam("nobody");
if (pwd == NULL) {
    if (errno == 0) { /* 不存在 */ }
    else { /* 系统错误 */ }
}

返回值指向静态缓冲；连续调用会覆盖；多线程不安全。

代码示例

#include <pwd.h>
#include <stdio.h>
#include <errno.h>

int main(void) {
    /* 按 UID 查询用户信息 */
    errno = 0;
    struct passwd *pw = getpwuid(getuid());
    if (pw == NULL) {
        if (errno == 0) printf("用户不存在\n");
        else perror("getpwuid");
        return 1;
    }

    printf("用户名: %s\n", pw->pw_name);
    printf("UID: %d\n", pw->pw_uid);
    printf("GID: %d\n", pw->pw_gid);
    printf("家目录: %s\n", pw->pw_dir);
    printf("Shell: %s\n", pw->pw_shell);

    /* 按名字查询 */
    errno = 0;
    struct passwd *pw2 = getpwnam("root");
    if (pw2 != NULL) {
        printf("\nroot 的 UID = %d\n", pw2->pw_uid);
    }

    /* 注意: 返回值指向静态缓冲, 连续调用会覆盖 */
    return 0;
}

代码示例

#include <stdio.h>
#include <shadow.h>
#include <pwd.h>
#include <unistd.h>
/* /etc/shadow: 存储密码哈希, 需要 root 权限读 */
int main(void) {
    /* getspnam: 读取 shadow 条目 (需要 root) */
    struct spwd *sp = getspnam(getpwuid(getuid())->pw_name);
    if (sp)
        printf("shadow entry found, last change: %ld\n", (long)sp->sp_lstchg);
    else
        printf("Cannot read shadow (need root)\n");
    return 0;
}

#include <grp.h>

int main(void) {
    /* 按 GID 查询组信息 */
    struct group *gr = getgrgid(getgid());

    return 0;

本节讲什么

进程可有一组附属 GID，访问多组资源时一并参与校验（与主组一起）。 #include int getgroups(int size, gid_t list[]);

要点

进程可有一组附属 GID，访问多组资源时一并参与校验（与主组一起）。

#include <grp.h>
int getgroups(int size, gid_t list[]);
int setgroups(size_t size, const gid_t *list);  /* 特权 */

附属组 ≠ 主组；文件权限看有效 GID + 全部附属组。细节与「当前进程凭证」仍属 Ch9。

代码示例

#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>
#include <grp.h>
/* 附属组: getgroups/setgroups */
int main(void) {
    int n = getgroups(0, NULL);
    printf("supplementary groups count: %d\n", n);

    if (n > 0) {
        gid_t groups[32];
        getgroups(n, groups);
        for (int i = 0; i < n; i++) {
            struct group *g = getgrgid(groups[i]);
            printf("  [%d] gid=%u name=%s\n",
                   i, (unsigned)groups[i], g ? g->gr_name : "?");
        }
    }
    return 0;
}

`crypt()` — 密码加密

本节讲什么

define _XOPEN_SOURCE #include char crypt(const char key, const char *salt);

要点

#define _XOPEN_SOURCE
#include <unistd.h>
char *crypt(const char *key, const char *salt);
key 明文密码
salt 取自 shadow 密文头部（算法标识 + 盐）
返回 静态缓冲；勿 free；非线程安全

认证：getspnam → 取 sp_pwdp 作 salt → crypt → 与密文比字符串。
验证后立刻覆盖清空明文缓冲。

Demo：code/check_password.c（需能读 shadow，通常 root）

代码示例

#include <stdio.h>
#include <crypt.h>
#include <unistd.h>
/* crypt(): 密码哈希 (SHA-512) */
int main(void) {
    /* $6$ = SHA-512, salt = "abcdefgh" */
    const char *salt = "$6$abcdefgh";
    const char *hash = crypt("password123", salt);
    if (hash)
        printf("hash = %s\n", hash);
    else
        perror("crypt");
    return 0;
}

代码自测

1. 为什么引入 /etc/shadow？
  1. 安全性：/etc/passwd 必须全局可读（很多程序需要查 UID→用户名），如果把密码哈希放在里面，任何用户都能获取哈希进行离线破解
  2. 密码策略：shadow 文件支持过期、警告、锁定等策略字段，passwd 无法表达
  3. 引入时机：早期 Unix 密码直接在 passwd 第二字段；随着 CPU 增强，离线破解变得可行，引入 shadow 将哈希隔离到仅 root 可读
2. getpwuid 返回的指针需要 free 吗？

不需要。getpwuid 返回的是静态分配的 struct passwd 指针，下次调用同一函数会被覆盖。因此不是线程安全的——多线程同时调用会导致竞争。线程安全版本是 getpwuid_r，需调用者提供缓冲区。

3. 附属组（supplementary groups）的作用？

一个用户可以属于多个组。主组在 /etc/passwd 的 GID 字段，附属组在 /etc/group 的成员列表中。进程启动时加载所有组到 supplementary groups 列表，用于权限检查。这样用户无需切换组即可访问多个组的资源。

练习 1 参考代码

#include "tlpi_hdr.h"
#include <pwd.h>
#include <grp.h>

int main(int argc, char *argv[]) {
    if (argc != 2)
        usageErr("usage: %s <uid>\n", argv[0]);

    uid_t uid = atoi(argv[1]);

    /* 查用户信息 */
    struct passwd *pw = getpwuid(uid);
    if (pw == NULL) {
        fprintf(stderr, "UID %d not found\n", uid);
        exit(EXIT_FAILURE);
    }

    /* 查主组 */
    struct group *gr = getgrgid(pw->pw_gid);
    printf("user=%s  uid=%d  primary_group=%s(%d)\n",
           pw->pw_name, pw->pw_uid,
           gr ? gr->gr_name : "???", pw->pw_gid);

    /* 查附属组 */
    int n = getgroups(0, NULL);
    if (n > 0) {
        gid_t *groups = malloc(n * sizeof(gid_t));
        getgroups(n, groups);

        printf("supplementary groups:");
        for (int i = 0; i < n; i++) {
            struct group *g = getgrgid(groups[i]);
            printf(" %s(%d)", g ? g->gr_name : "???", groups[i]);
        }
        printf("\n");
        free(groups);
    }
    return 0;
}

练习 2 答案

没有 shadow 的问题： 1. /etc/passwd 必须全局可读——ls -l 需要查 UID→用户名，sshd 需要验证用户存在 2. 密码哈希在 passwd 中 → 任何用户可获取哈希 → 离线字典攻击/彩虹表 3. 无密码过期策略，无法强制定期修改

shadow 的优势： 1. 哈希仅 root 可读（权限 600 或 640 + shadow 组） 2. 支持过期、锁定、宽限等策略 3. 即使非 root 用户获取 passwd 也无法看到哈希

练习 3 参考代码

#include <pwd.h>
#include <stdio.h>

int main(void) {
    struct passwd *pw;
    setpwent();                 /* 回到文件开头 */
    while ((pw = getpwent()) != NULL) {
        printf("%5d  %s\n", pw->pw_uid, pw->pw_name);
    }
    endpwent();                 /* 关闭 */
    return 0;
}
