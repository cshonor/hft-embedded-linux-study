## 代码示例

```c
#include <sys/capability.h>
#include <unistd.h>
#include <stdio.h>
/* Linux capabilities: 把 root 权限细分为多项
 * CAP_NET_BIND_SERVICE: 绑定 <1024 端口
 * CAP_KILL: 发信号
 * CAP_SYS_ADMIN: 系统管理
 * 需要 -lcap
 */
int main(void) {
    printf("UID=%d EUID=%d\n", getuid(), geteuid());
    /* 查看 capability */
    cap_t caps = cap_get_proc();
    if (caps) {
        char *text = cap_to_text(caps, NULL);
        if (text) { printf("capabilities: %s\n", text); cap_free(text); }
        cap_free(caps);
    }
    return 0;
}
```
