## 要点

`/dev/fd/n` 访问约等于操作 fd `n`；shell 重定向常用。

## 代码示例

```c
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    /* /dev/fd/N 打开约等于操作 fd N */
    /* 常用于 shell 重定向，C 中也可用 */

    /* 打开 /dev/fd/0 相当于复制 stdin */
    int fd = open("/dev/fd/0", O_RDONLY);
    if (fd < 0) { perror("open /dev/fd/0"); return 1; }
    printf("open(\"/dev/fd/0\") = fd %d\n", fd);

    /* 验证：和 stdin 指向同一个打开文件描述 */
    int fd2 = open("/dev/fd/1", O_WRONLY);
    if (fd2 < 0) { perror("open /dev/fd/1"); return 1; }
    write(fd2, "via /dev/fd/1\n", 13);

    close(fd);
    close(fd2);
    return 0;
}
```
