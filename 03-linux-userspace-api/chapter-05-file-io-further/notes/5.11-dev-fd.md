## 本节讲什么

`/dev/fd` 让你**用路径名的形式**去引用一个已经打开的 fd。本节回答四个问题：

1. `/dev/fd/N` 在 Linux 上到底是什么（信号链怎么走）？
2. `open("/dev/fd/N", ...)` 与 `dup(N)` **等价吗**？（本节有实测，结论和流传的说法不一样）
3. 已经 `unlink` 掉的文件，还能通过 `/proc/self/fd/N` 打开吗？
4. 哪些程序靠这个机制吃饭（以及它为什么是「管道过滤器」的关键）？

---

## 1. `/dev/fd` 与 `/proc/self/fd`

官方 `proc_pid_fd(5)` 的原话（man-pages 6.19）：

> `/proc/pid/fd/` — This is a subdirectory containing one entry for each file which the process has open, named by its file descriptor, and **which is a symbolic link to the actual file**.

> `/proc/self/fd/N` is approximately the same as `/dev/fd/N` in some UNIX and UNIX-like systems. **Most Linux MAKEDEV scripts symbolically link `/dev/fd` to `/proc/self/fd`**, in fact.

所以链子是：

```
/dev/fd  ──(符号链接)──▶  /proc/self/fd  ──(目录)──▶  N  ──(magic symlink)──▶  真正的文件
```

**在本仓库的 CE 沙箱里 `/dev/fd` 不存在**（实测 `stat` 返回 `ENOENT`）——它只是个符号链接，缺了就缺了。同级能力可以直接走 `/proc/self/fd`。

## 2. `proc_fd_link()`：内核只做了一件事

`/proc/self/fd/N` 这个条目的 inode 是**符号链接**，注册在 `fs/proc/fd.c:211-214`：

```c
inode->i_op = &proc_pid_link_inode_operations;
inode->i_size = 64;

ei->op.proc_get_link = proc_fd_link;
```

它指向的 `proc_fd_link()` 全文（`fs/proc/fd.c:169-190`）：

```c
static int proc_fd_link(struct dentry *dentry, struct path *path)
{
        struct task_struct *task;
        int ret = -ENOENT;

        task = get_proc_task(d_inode(dentry));
        if (task) {
                unsigned int fd = proc_fd(d_inode(dentry));
                struct file *fd_file;

                fd_file = fget_task(task, fd);
                if (fd_file) {
                        *path = fd_file->f_path;
                        path_get(&fd_file->f_path);
                        ret = 0;
                        fput(fd_file);
                }
                put_task_struct(task);
        }

        return ret;
}
```

**关键在 `:181-182` 那两行**：它把目标 fd 的 `f_path`（`dentry` + `mnt`）**复制到 `path` 输出参数**，然后 `fput(fd_file)` 把临时引用放掉。**它没有复制 `struct file`，也没有 `get_file()` 留下的长寿引用。**

于是后续 `open` 走的是「在 `f_path` 上打开文件」的常规路径 → **创建一个全新的 `struct file`（新的打开文件表项，新的 `f_pos`）**。

## 3. 实测：和 `dup` 一样吗？

完整源码 [`code/c5_11_dev_fd.c`](../code/c5_11_dev_fd.c)：

```c
/* c5_11_dev_fd.c — §5.11 /dev/fd 目录：用「fd 的名字」当路径名来用
 *
 * 在正常 Linux 上 /dev/fd 是指向 /proc/self/fd 的符号链接，于是有三件事能做：
 *   1) open("/dev/fd/N", ...) —— 相当于把 N 号 fd 再打开一份
 *   2) cat /dev/fd/N         —— 把某个 fd 当文件读
 *   3) 程序可以用「文件名参数」收到自己的 stdin/stdout（例如 sort /dev/fd/0）
 *
 * 本 demo 演示五件事：
 *   1) CE 沙箱里没有 /dev/fd（ENOENT）—— 它只是符号链接，缺了就缺了；
 *      同级能力改走 /proc/self/fd
 *   2) readlink("/proc/self/fd/N") 能看到这个 fd 指向哪个文件
 *   3) open("/proc/self/fd/N") 之后再读，看它和 N 号是不是共享偏移（实测说话）
 *   4) unlink 掉文件名之后，还能不能通过 /proc/self/fd/N 打开同一个文件
 *   5) 打开目录 /proc/self/fd 列一遍 fd
 *
 * 编译: gcc -O0 -Wall -Wextra -o c5_11_dev_fd c5_11_dev_fd.c
 */
#define _GNU_SOURCE
#include <dirent.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/stat.h>
#include <unistd.h>
#include <errno.h>

static void list_fds(void)
{
    DIR *d = opendir("/proc/self/fd");
    struct dirent *e;

    if (d == NULL) {
        printf("    opendir 失败: %s\n", strerror(errno));
        return;
    }
    printf("    /proc/self/fd = ");
    while ((e = readdir(d)) != NULL) {
        if (e->d_name[0] != '.')
            printf("%s ", e->d_name);
    }
    printf("\n");
    closedir(d);
}

int main(void)
{
    const char *p = "/tmp/c5_devfd.txt";
    char link[256];
    int fd, dup_fd;
    ssize_t n;

    printf("== 1. /dev/fd 存在吗？==\n");
    {
        struct stat s;

        errno = 0;
        if (stat("/dev/fd", &s) == -1)
            printf("    stat(\"/dev/fd\") -> -1 errno=%d (%s)：本沙箱没有这个符号链接\n",
                   errno, strerror(errno));
        else
            printf("    stat(\"/dev/fd\") 成功，模式 = 0%o\n", s.st_mode);
        errno = 0;
        if (stat("/proc/self/fd", &s) == -1)
            printf("    stat(\"/proc/self/fd\") -> -1 errno=%d (%s)\n", errno, strerror(errno));
        else
            printf("    stat(\"/proc/self/fd\") 成功 —— 真正干活的是它\n");
    }

    fd = open(p, O_RDWR | O_CREAT | O_TRUNC, 0644);
    if (fd < 0) {
        perror("open");
        return 1;
    }
    write(fd, "0123456789", 10);

    printf("\n== 2. readlink(\"/proc/self/fd/N\") ==\n");
    {
        char path[64];
        ssize_t r;

        snprintf(path, sizeof path, "/proc/self/fd/%d", fd);
        r = readlink(path, link, sizeof link - 1);
        if (r >= 0)
            link[r] = '\0';
        printf("    fd %d 指向 \"%s\"\n", fd, r >= 0 ? link : "?");
    }

    printf("\n== 3. 列出当前所有 fd ==\n");
    list_fds();

    printf("\n== 4. open(\"/proc/self/fd/N\") 之后，偏移共享吗？（实测）==\n");
    {
        char path[64];

        snprintf(path, sizeof path, "/proc/self/fd/%d", fd);
        dup_fd = open(path, O_RDWR);
        printf("    原 fd = %d，经由 %s 打开的新 fd = %d\n", fd, path, dup_fd);
        if (dup_fd >= 0) {
            printf("    参考：dup(%d) 得到的最小可用号 = %d\n", fd, dup(fd));
            lseek(fd, 0, SEEK_SET);
            {
                char b[8];
                ssize_t got = read(dup_fd, b, 4);     /* 通过新 fd 读 4 字节 */

                printf("    read(新 fd, 4) -> %ld 字节\n", (long) got);
                printf("    原 fd 的 f_pos 现在 = %ld\n", (long) lseek(fd, 0, SEEK_CUR));
                printf("    → 若为 4 则两者共享打开文件表项（等价 dup）；若为 0 则是各自独立的表项\n");
            }
        }
    }

    printf("\n== 5. unlink 掉文件名之后，还能通过 /proc/self/fd/N 打开吗？==\n");
    if (unlink(p) == 0) {
        char path[64];
        ssize_t r;

        snprintf(path, sizeof path, "/proc/self/fd/%d", fd);
        r = readlink(path, link, sizeof link - 1);
        if (r >= 0)
            link[r] = '\0';
        printf("    unlink 后 readlink -> \"%s\"\n", r >= 0 ? link : "?");
        errno = 0;
        {
            int re = open(path, O_RDWR);

            printf("    open(\"%s\") -> %d errno=%d (%s)\n", path, re, errno, strerror(errno));
            if (re >= 0) {
                char b[8];

                lseek(re, 0, SEEK_SET);
                n = read(re, b, 4);
                b[n > 0 ? n : 0] = '\0';
                printf("    从它读回来 -> \"%s\"（文件已经没有名字了，但内容还在）\n", b);
                close(re);
            }
        }
    }

    if (dup_fd >= 0)
        close(dup_fd);
    close(fd);
    return 0;
}
```

编译运行：

```bash
gcc -O0 -Wall -Wextra -o c5_11_dev_fd c5_11_dev_fd.c && ./c5_11_dev_fd
```

CE（gcc 13.3.0 / x86-64 / Ubuntu 24.04）实测：

```text
== 1. /dev/fd 存在吗？==
    stat("/dev/fd") -> -1 errno=2 (No such file or directory)：本沙箱没有这个符号链接
    stat("/proc/self/fd") 成功 —— 真正干活的是它

== 2. readlink("/proc/self/fd/N") ==
    fd 3 指向 "/tmp/c5_devfd.txt"

== 3. 列出当前所有 fd ==
    /proc/self/fd = 0 1 2 3 4 

== 4. open("/proc/self/fd/N") 之后，偏移共享吗？（实测）==
    原 fd = 3，经由 /proc/self/fd/3 打开的新 fd = 4
    参考：dup(3) 得到的最小可用号 = 5
    read(新 fd, 4) -> 4 字节
    原 fd 的 f_pos 现在 = 0
    → 若为 4 则两者共享打开文件表项（等价 dup）；若为 0 则是各自独立的表项

== 5. unlink 掉文件名之后，还能通过 /proc/self/fd/N 打开吗？==
    unlink 后 readlink -> "/tmp/c5_devfd.txt (deleted)"
    open("/proc/self/fd/3") -> 6 errno=0 (Success)
    从它读回来 -> "0123"（文件已经没有名字了，但内容还在）
```

逐条看：

| 观察 | 说明 |
|------|------|
| `stat("/dev/fd")` → `ENOENT`；`/proc/self/fd` 成功 | 沙箱没有那个 MAKEDEV 符号链接（[沙箱环境表](5.13-summary.md)）。**直接走 `/proc/self/fd` 才是可靠做法** |
| `readlink("/proc/self/fd/3")` → `/tmp/c5_devfd.txt` | 这就是「符号链接到实际文件」的字面体现。`readlink` 受 `PTRACE_MODE_READ_FSCREDS` 检查约束（官方 man page 原文） |
| `/proc/self/fd = 0 1 2 3 4` | 目录里一项一个 fd。（内核对目录 `stat` 还会把 `st_size` 填成**打开的 fd 个数**：`proc_fd_getattr` → `proc_readfd_count`，`fs/proc/fd.c:348`/`:284`） |
| 经由 `/proc/self/fd/3` 打开得到 fd **4**，而从它读 4 字节后，**原 fd 的 `f_pos` 仍是 0** | **核心结论**：`open("/proc/self/fd/N")` **不与原 fd 共享打开文件表项**——偏移各自独立。对照 `dup(3)` 会拿到 5 号，且偏移立刻变成 4 |
| `unlink` 后 `readlink` 显示 `"/tmp/c5_devfd.txt (deleted)"` | 名字没了、inode 还活着（还有 fd 引用着它，`f_count`/`i_count` > 0）。` (deleted)` 后缀是内核 `d_path()` 加的 |
| `open("/proc/self/fd/3")` → **6**，`errno=0`；读回 `"0123"` | **取消链接之后依然能重新打开**。这是 `/proc/self/fd/N` 最有价值的性质之一：它引用的是 **inode**，不是名字 |

> ### ⚠️ 与流传说法的分歧
>
> 「`open("/dev/fd/N")` 等价于 `dup(N)`」是相当常见的表述。**本仓库的实测与源码都不支持这个说法**（在 Linux 上）：
>
> | | `dup(N)` | `open("/proc/self/fd/N")` |
> |---|---|---|
> | 新 fd 指向 | **同一个** `struct file` | **新的** `struct file` |
> | 偏移（`f_pos`） | **共享** | **独立**（实测：原 fd 仍是 0） |
> | 状态标志（`F_GETFL`） | 共享 | 独立（按 `open` 的入参设定） |
> | 内核依据 | `fs/file.c:1286` `do_dup2` / `get_file()` | `fs/proc/fd.c:181-182` 只复制 `f_path` |
>
> 官方 `proc_pid_fd(5)` 的用词也支持这一点：它说这些条目是「**a symbolic link to the actual file**」，并在结尾把这件事描述为「**创建新的文件描述符**去读那个 inode」（"cat does not have permission to create a new file descriptor to read from that inode"）。**没有一处说它等于 `dup`。**
>
> 「等价于 dup」的表述可能来自其他 UNIX 系统的 `/dev/fd` 实现（那里 `/dev/fd/N` 由伪设备驱动提供、行为是 dup）——这是推测，**本仓库不作为事实陈述**。在 Linux 上，请以「打开那个文件」来理解它。

## 4. 访问模式的约束从哪来

原 fd 的 `f_mode` 会影响 `/proc/self/fd/N` 这个条目的权限位——`tid_fd_update_inode()`（`fs/proc/fd.c:123-136`）：

```c
if (S_ISLNK(inode->i_mode)) {
        unsigned i_mode = S_IFLNK;
        if (f_mode & FMODE_READ)
                i_mode |= S_IRUSR | S_IXUSR;
        if (f_mode & FMODE_WRITE)
                i_mode |= S_IWUSR | S_IXUSR;
        inode->i_mode = i_mode;
}
```

这就是「原 fd 只读时，用 `/proc/self/fd/N` 也拿不到写权限」的机制——**符号链接的权限位由原 fd 的 `f_mode` 推导出来**。另外每个条目在每次查找时都会 `tid_fd_revalidate`（`:139`）：原 fd 已经关闭的话，条目就失效（返回 0，导致 `-ENOENT`/`-ESTALE`）。

## 5. `/proc/self/fdinfo/N`：把打开文件表项打印出来

同一个目录下还有 `fdinfo`，内容由 `seq_show()`（`fs/proc/fd.c:23-75`）生成，格式字符串就在 `:59-62`：

```c
seq_printf(m, "pos:\t%lli\nflags:\t0%o\nmnt_id:\t%i\nino:\t%lu\n",
           (long long)file->f_pos, f_flags,
           real_mount(file->f_path.mnt)->mnt_id,
           file_inode(file)->i_ino);
```

其中 `flags:` 的一个细节非常值得注意（`:45-46`）：

```c
f_flags = file->f_flags;
if (close_on_exec(fd, fdt))
        f_flags |= O_CLOEXEC;
```

**内核在打印时把 `FD_CLOEXEC`「合成」进 `flags` 里** ——但 `FD_CLOEXEC` 其实不在 `f_flags` 里，它在 **fd 表项**上（[5.3](5.3-open-file-status-flags.md) 实测过：`F_GETFL` 里看不到它）。所以：

- `fcntl(fd, F_GETFL)` **看不到** `O_CLOEXEC`；
- `cat /proc/self/fdinfo/N` **能看到**（内核拼给你看）；
- 两者来自**不同的层**，但被摆在了同一行里。

`pos:` 字段则是观察「偏移共享」的另一个窗口——`dup` 出来的两个 fd 应该有同一个 `pos:`。

## 6. 常见坑

| 坑 | 说明 |
|------|------|
| 以为 `open("/dev/fd/N")` 等于 `dup(N)` | Linux 上**不等价**：新 fd 有独立的偏移与标志（实测 + `fs/proc/fd.c:181-182`） |
| 假设 `/dev/fd` 一定存在 | 它只是 `MAKEDEV` 建的符号链接，精简系统/容器里常缺（CE 沙箱就缺）。用 `/proc/self/fd` 更可靠 |
| 把 `/dev/stdin` 当成 stdin 的 dup | 它是 `/proc/self/fd/0` 的符号链接，同样是「打开那个文件」的语义 |
| 在 `unlink` 后以为文件没救了 | 只要还有 fd 打开着，`/proc/self/fd/N` 仍能重新打开，内容是完整的（实测读回 `"0123"`） |
| 用 `/proc/PID/fd/N` 去动别的进程 | 受 `PTRACE_MODE_READ_FSCREDS` 检查（`fs/proc/fd.c:85`、`:329`）；`/proc/self/fd` 有专门的特例放行（`:339-343` 的 `same_thread_group`） |
| 以为 `/proc/self/fd` 里的条目总是有效 | 原 fd 关闭后 `tid_fd_revalidate`（`:139`）让它失效 → `ENOENT`/`ESTALE` |
| 用它去读**因 inode 权限不足**而无法重新打开的对象 | 管道/socket 的 inode 属主可能与本进程不同 → `EACCES`（官方 man page 给了 `sudo -u nobody cat /proc/self/fd/0` 的例子） |

## 7. HFT / 嵌入式关联

| 场景 | 本节机制的兑现 |
|------|----------------|
| 管道过滤器 | `foobar -i /proc/self/fd/0 -o /proc/self/fd/1` 让「只认文件名的程序」也能吃 stdin/stdout（官方 man page 的原文例子）。**这是把流式工具塞进管道链的关键技巧** |
| 恢复已删文件 | 日志轮转/临时文件被误删时，只要还有进程握着 fd，就能从 `/proc/self/fd/N` 把内容捞出来（实测可行） |
| 排查「谁占着这个文件」 | `ls -l /proc/PID/fd` + `readlink` 是排查 fd 泄漏与文件占用的第一手段；`fdinfo` 的 `pos:`/`flags:` 能看出共享关系与被谁改了标志 |
| 容器/精简系统 | 没有 `/dev/fd` 时不要慌，`/proc/self/fd` 才是本体；写脚本时优先用 `/proc/self/fd` |
| 嵌入式调试 | 没有 `strace` 的环境里，`/proc/PID/fd` 是判断「这个进程握着哪些设备节点」的最廉价办法 |
| 安全边界 | 跨进程用 `/proc/PID/fd/N` 需要 ptrace 权限；嵌入式上 `dumpable=0` 或 setuid 程序会让该目录变得不可访问 |

<summary>代码自测（Q&A）</summary>

**Q1：`/dev/fd/N` 和 `/proc/self/fd/N` 是什么关系？**
A：在 Linux 上 `/dev/fd` 一般就是指向 `/proc/self/fd` 的**符号链接**（官方 `proc_pid_fd(5)` 原文：「Most Linux MAKEDEV scripts symbolically link `/dev/fd` to `/proc/self/fd`」）。所以两者等价，但 `/dev/fd` **不一定存在**——CE 沙箱里 `stat("/dev/fd")` 就返回 `ENOENT`。

**Q2：`open("/dev/fd/N")` 等价于 `dup(N)` 吗？**
A：**在 Linux 上不等价。** 实测：经由 `/proc/self/fd/3` 打开并读 4 字节后，**原 fd 的 `f_pos` 仍是 0**（若是 dup 应变成 4）。源码依据：`proc_fd_link()` 只把目标 fd 的 `f_path` 复制出来（`fs/proc/fd.c:181-182`），**没有复制 `struct file`**，所以后续 `open` 会创建一个全新的打开文件表项。官方 `proc_pid_fd(5)` 也只说它是「a symbolic link to the actual file」。

**Q3：为什么 `unlink` 之后还能通过 `/proc/self/fd/N` 打开？**
A：因为它引用的是 **inode**（`f_path` 里的 dentry 指向那个 inode），不是文件名。只要还有 fd 持有着它，inode 就不会被回收；实测 `unlink` 后 `open` 成功（返回 6）并读回 `"0123"`，`readlink` 显示路径带 ` (deleted)` 后缀。

**Q4：怎么观察两个 fd 是否共享打开文件表项？**
A：三种办法：① 读/写之后看另一个 fd 的 `f_pos`（本节实测用法）；② 看 `/proc/self/fdinfo/N` 里的 `pos:` 字段（`fs/proc/fd.c:59-60`）；③ `fcntl(F_GETFL)` 对比状态标志。**`dup`/`fork` 共享，各自 `open`（以及 `open("/proc/self/fd/N")`）不共享。**

**Q5：`/proc/self/fdinfo/N` 里为什么会有 `O_CLOEXEC`？它不该在 `F_GETFL` 里吗？**
A：内核在打印时**合成**的：`f_flags = file->f_flags; if (close_on_exec(fd, fdt)) f_flags |= O_CLOEXEC;`（`fs/proc/fd.c:45-46`）。`FD_CLOEXEC` 真正的家在 **fd 表项**（第 ① 层），所以 `fcntl(fd, F_GETFL)` 看不到它，但 `fdinfo` 把它拼进 `flags` 行展示。**两者来自不同的层。**

**Q6：为什么打开 `/proc/PID/fd/N` 可能需要特殊权限？**
A：因为读取/解引用这些符号链接受 `PTRACE_MODE_READ_FSCREDS` 检查（官方 man page 原文；内核侧在 `fs/proc/fd.c:85` 的 `proc_fdinfo_access_allowed` 与 `:329` 的 `proc_fd_permission`）。`/proc/self/fd` 有特例：同线程组进程直接放行（`:339-343`）。

**Q7：`readlink("/proc/self/fd/3")` 对管道/socket 显示什么？**
A：不是路径，而是 `type:[inode]` 形式（官方 man page 原文给出 `socket:[2248868]` 的例子）。对没有对应 inode 的匿名对象（`epoll`、`eventfd`、`timerfd` 等）显示 `anon_inode:file-type`。**所以脚本里不能假设 `readlink` 的结果一定是个路径。**
