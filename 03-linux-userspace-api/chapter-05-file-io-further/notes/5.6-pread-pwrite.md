# 5.6 `pread` / `pwrite`（指定偏移读写）

> 本章：[TLPI 第 05 章 — File I/O: Further Details](../README.md)

## 本节讲什么

`pread` / `pwrite` 与 `read` / `write` 的**唯一区别**是：**不碰 `f_pos`**。这一个区别带来两件事——**原子性**（定位与读写合成一步）和**多线程友好**（无共享状态）。本节回答四个问题：

1. `pread` 为什么「不碰 `f_pos`」？内核里是怎么做到的（不是「约定」，是**传了一个局部变量**）？
2. 它比 `lseek` + `read` 好在哪？给出一个具体的竞态场景。
3. 偏移超过文件末尾时，`pread` 和 `pwrite` 各自怎么表现？
4. 为什么管道上 `pread` 会 `ESPIPE`——源码里在哪里判断的？

---

## 1. 原型

```c
#include <unistd.h>
ssize_t pread (int fd, void *buf, size_t count, off_t offset);
ssize_t pwrite(int fd, const void *buf, size_t count, off_t offset);
/* 成功返回实际读/写的字节数；失败返回 -1 并设 errno */
```

`offset` 是**第四个参数**——它不是「先 seek 到这里」，而是「在这次操作里就用这个位置」。调用返回后 `fd` 的 `f_pos` **原封不动**。

## 2. 内核：为什么「不碰 `f_pos`」是结构性的

对比两条路径，一眼看出区别：

```c
/* —— 普通 read：位置来自 struct file->f_pos —— */
ssize_t ksys_read(unsigned int fd, char __user *buf, size_t count)
{
        struct fd f = fdget_pos(fd);              /* fs/read_write.c:604 */
        ssize_t ret = -EBADF;

        if (f.file) {
                loff_t pos, *ppos = file_ppos(f.file);
                if (ppos) {
                        pos = *ppos;              /* 读出 f_pos            */
                        ppos = &pos;
                }
                ret = vfs_read(f.file, buf, count, ppos);
                if (ret >= 0 && ppos)
                        f.file->f_pos = pos;      /* 回写 f_pos  fs/read_write.c:615 */
                fdput_pos(f);
        }
        return ret;
}
```

```c
/* —— pread：位置来自参数，是一个栈上的局部变量 —— */
ssize_t ksys_pread64(unsigned int fd, char __user *buf, size_t count,
                     loff_t pos)
{
        struct fd f;
        ssize_t ret = -EBADF;

        if (pos < 0)
                return -EINVAL;                   /* fs/read_write.c:658-659 */

        f = fdget(fd);                            /* 注意：fdget，不是 fdget_pos */
        if (f.file) {
                ret = -ESPIPE;
                if (f.file->f_mode & FMODE_PREAD) /* fs/read_write.c:664 */
                        ret = vfs_read(f.file, buf, count, &pos);  /* 传的是 &pos */
                fdput(f);
        }
        return ret;
}
```

三处差异，每处都有后果：

| 差异 | 普通 `read` | `pread` | 后果 |
|------|------------|---------|------|
| 取 fd 的函数 | `fdget_pos`（`fs/read_write.c:604`） | `fdget`（`:661`） | `fdget_pos` 会加 `f_pos_lock`，把同一文件描述上的读**串行化**；`pread` 不加锁，可并发 |
| 传给 `vfs_read` 的位置 | `&pos`（`f_pos` 的**副本**） | `&pos`（**参数**） | 普通读完成后**回写** `f_pos`（`:615`）；`pread` 没有回写这一步 |
| 能力检查 | 无 | `FMODE_PREAD`（`:664`） | 不支持随机读的对象（管道）→ `ESPIPE` |

> **「`pread` 不碰 `f_pos`」不是约定俗成的礼貌，而是「它压根没有回写那一行」。** 把 `ksys_pread64` 从头读到尾，找不到任何写 `f.file->f_pos` 的语句。

## 3. 代码：五段实测

完整源码 [`code/c5_6_pread_pwrite.c`](../code/c5_6_pread_pwrite.c)：

```c
/* c5_6_pread_pwrite.c — §5.6 在指定偏移读写：pread() / pwrite()
 *
 * 和 read()/write() 的唯一区别：**不动 f_pos**。于是「定位 + 读写」合成一个
 * 原子动作。多线程读同一个 fd 时，用 lseek()+read() 会因为偏移被别人改而错位；
 * pread() 没有这个窗口——它把偏移当参数传，压根不碰共享状态。
 *
 * 演示五件事：
 *   1) pread() 读完 f_pos 不变；lseek()+read() 读完 f_pos 变了
 *   2) pwrite() 写完 f_pos 也不变
 *   3) 偏移超过文件末尾 → pread 返回 0（EOF），pwrite 则把文件撑长、中间是洞
 *   4) 管道上 pread → ESPIPE
 *   5) 用 pread 读「同一个 fd」的多个区间，互不干扰
 *
 * 编译: gcc -O0 -Wall -Wextra -o c5_6_pread_pwrite c5_6_pread_pwrite.c
 */
#define _GNU_SOURCE
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/stat.h>
#include <unistd.h>
#include <errno.h>

static long pos(int fd)
{
    return (long) lseek(fd, 0, SEEK_CUR);
}

static void show_file(const char *tag)
{
    int r = open("/tmp/c5_pread.txt", O_RDONLY);
    char b[64];
    ssize_t n = r < 0 ? -1 : read(r, b, sizeof b - 1);

    if (n < 0)
        n = 0;
    b[n] = '\0';
    if (r >= 0)
        close(r);
    printf("    [%s] 文件 = \"%s\"（%ld 字节）\n", tag, b, (long) n);
}

int main(void)
{
    const char *p = "/tmp/c5_pread.txt";
    int fd = open(p, O_RDWR | O_CREAT | O_TRUNC, 0644);
    char buf[16];
    ssize_t n;

    if (fd < 0) {
        perror("open");
        return 1;
    }
    write(fd, "ABCDEFGHIJ", 10);
    printf("== 0. 准备：写入 \"ABCDEFGHIJ\"，f_pos=%ld ==\n", pos(fd));

    printf("\n== 1. pread() 不碰 f_pos，lseek()+read() 会碰 ==\n");
    memset(buf, 0, sizeof buf);
    n = pread(fd, buf, 4, 0);
    printf("    pread(fd, 4, 0)        -> %ld 字节 \"%.*s\"，随后 f_pos=%ld（没动）\n",
           (long) n, (int) n, buf, pos(fd));

    lseek(fd, 0, SEEK_SET);
    memset(buf, 0, sizeof buf);
    n = read(fd, buf, 4);
    printf("    lseek(0)+read(4)       -> %ld 字节 \"%.*s\"，随后 f_pos=%ld（动了）\n",
           (long) n, (int) n, buf, pos(fd));

    printf("\n== 2. pwrite() 也不碰 f_pos ==\n");
    lseek(fd, 10, SEEK_SET);                 /* 先把游标挪到末尾 */
    printf("    写前 f_pos=%ld\n", pos(fd));
    n = pwrite(fd, "xyz", 3, 5);             /* 写到偏移 5，游标不该变 */
    printf("    pwrite(fd, \"xyz\", 3, 5) -> %ld，写后 f_pos=%ld（还是 10）\n",
           (long) n, pos(fd));
    show_file("pwrite 后");

    printf("\n== 3. 偏移超过文件末尾 ==\n");
    memset(buf, 0, sizeof buf);
    n = pread(fd, buf, 4, 1000);
    printf("    pread(offset=1000)（文件才 10 字节）-> %ld（0 = EOF）\n", (long) n);
    n = pwrite(fd, "Z", 1, 1000);            /* 撑出一个洞 */
    printf("    pwrite(offset=1000) -> %ld\n", (long) n);
    {
        struct stat st;

        fstat(fd, &st);
        printf("    文件现在 st_size=%lld，物理占用 st_blocks*512=%lld\n",
               (long long) st.st_size, (long long) st.st_blocks * 512);
        printf("    ↑ 逻辑长了，但 1000 这个偏移仍落在同一个 4096 字节块内，所以物理没省；\n");
        printf("      偏移跨出块边界时才真正产生「洞」（见 5.8 的 ftruncate 对照）\n");
    }

    printf("\n== 4. 管道上 pread ==\n");
    {
        int pp[2];

        if (pipe(pp) == 0) {
            errno = 0;
            n = pread(pp[0], buf, 4, 0);
            printf("    pread(pipe) -> %ld errno=%d (%s)  ← 管道没有「偏移」这个概念\n",
                   (long) n, errno, strerror(errno));
            close(pp[0]);
            close(pp[1]);
        }
    }

    printf("\n== 5. 同一个 fd 上并发读多个区间，各自不干扰 ==\n");
    {
        char a[4], b[4], c[4];

        pread(fd, a, 3, 0);
        pread(fd, b, 3, 5);
        pread(fd, c, 3, 10);
        a[3] = b[3] = c[3] = '\0';
        printf("    区间 [0,3)=\"%s\"  [5,8)=\"%s\"  [10,13)=\"%s\"   f_pos 始终 = %ld\n",
               a, b, c, pos(fd));
    }

    close(fd);
    return 0;
}
```

编译运行：

```bash
gcc -O0 -Wall -Wextra -o c5_6_pread_pwrite c5_6_pread_pwrite.c && ./c5_6_pread_pwrite
```

CE（gcc 13.3.0 / x86-64 / Ubuntu 24.04）实测：

```text
== 0. 准备：写入 "ABCDEFGHIJ"，f_pos=10 ==

== 1. pread() 不碰 f_pos，lseek()+read() 会碰 ==
    pread(fd, 4, 0)        -> 4 字节 "ABCD"，随后 f_pos=10（没动）
    lseek(0)+read(4)       -> 4 字节 "ABCD"，随后 f_pos=4（动了）

== 2. pwrite() 也不碰 f_pos ==
    写前 f_pos=10
    pwrite(fd, "xyz", 3, 5) -> 3，写后 f_pos=10（还是 10）
    [pwrite 后] 文件 = "ABCDExyzIJ"（10 字节）

== 3. 偏移超过文件末尾 ==
    pread(offset=1000)（文件才 10 字节）-> 0（0 = EOF）
    pwrite(offset=1000) -> 1
    文件现在 st_size=1001，物理占用 st_blocks*512=4096
    ↑ 逻辑长了，但 1000 这个偏移仍落在同一个 4096 字节块内，所以物理没省；
      偏移跨出块边界时才真正产生「洞」（见 5.8 的 ftruncate 对照）

== 4. 管道上 pread ==
    pread(pipe) -> -1 errno=29 (Illegal seek)  ← 管道没有「偏移」这个概念

== 5. 同一个 fd 上并发读多个区间，各自不干扰 ==
    区间 [0,3)="ABC"  [5,8)="xyz"  [10,13)=""   f_pos 始终 = 10
```

逐条看：

| 观察 | 说明 |
|------|------|
| `pread(fd,4,0)` 读到 `ABCD`，`f_pos` 仍是 **10** | 起始 `f_pos` 是 10（刚写了 10 字节），`pread` 在偏移 0 读 4 字节，读完游标没被碰 |
| `lseek(0)+read(4)` 读到 `ABCD`，`f_pos` 变 **4** | 对照组：同样读偏移 0 起的 4 字节，但游标被推到 4 |
| `pwrite(fd,"xyz",3,5)` → 3，`f_pos` 仍是 **10** | 在偏移 5 写 3 字节，文件变 `ABCDExyzIJ`（原来的 `FGH` 被覆盖），游标不动 |
| `pread(offset=1000)` → **0** | 偏移越过 EOF 时 `pread` 返回 0——**不是错误**，就是「读到文件末尾了」 |
| `pwrite(offset=1000)` → **1**，`st_size` 变 **1001** | 越尾写会把文件**撑长**，中间没写过的字节读回来是 0 |
| `st_blocks*512 = 4096`（物理没超过 4 KiB） | **诚实说明**：1000 这个偏移仍落在同一个 4096 字节块内，所以是「逻辑长了、物理没省」。「洞」要等偏移跨出块边界才真的出现——那一段放在 [5.8](5.8-truncate-ftruncate.md) 用 `ftruncate` 做干净对照（8192 逻辑 / 4096 物理） |
| `pread(pipe)` → `-1 errno=29 (ESPIPE)` | 管道不支持 `FMODE_PREAD`（`fs/read_write.c:664`）→ 走 `ret = -ESPIPE;` 那条路 |
| 三次 `pread` 读 `[0,3)` / `[5,8)` / `[10,13)`，`f_pos` 始终 **10** | 这就是「同一 fd 上并发读多区间」的安全基础——**没有共享状态可被踩** |

## 4. 它解决的竞态到底是什么

用 `lseek + read` 读同一 fd 的多个区间，两个线程会互相踩：

```
线程 A: lseek(fd, 100, SEEK_SET)      → f_pos = 100
线程 B: lseek(fd, 500, SEEK_SET)      → f_pos = 500      ← 把 A 的位置改掉了
线程 A: read(fd, buf, 10)             → 实际从 500 读！错误
```

`pread(fd, buf, 10, 100)` 里位置是**参数**，不经过 `f_pos`，所以两个线程各读各的、互不干扰。**这就是「原子性」在这里的具体含义**：不是「更快」，而是「定位 + 读取」之间没有可被插入的共享状态。

| 需求 | 该用哪个 |
|------|---------|
| 顺序读取（流式解析、`copy`） | `read`（`f_pos` 帮你推进，少一个参数） |
| 随机访问 / 多线程读同一 fd / 需要幂等重读 | `pread` |
| 追加写日志 | `write` + `O_APPEND`（[5.1](5.1-atomicity-race-conditions.md)） |
| 固定位置改字段 / 多线程写不同区间 | `pwrite` |

## 5. 常见坑

| 坑 | 说明 |
|----|------|
| 在管道 / FIFO / socket 上用 `pread` | 一律 `ESPIPE`（`errno=29`）。这些对象没有「偏移」概念 |
| 以为 `pread` 会改变 `f_pos`，特意 `lseek` 去「复位」 | `pread` **从不**改 `f_pos`，多余的 `lseek` 反而引入竞态 |
| 越尾 `pread` 当错误处理 | 返回 `0` 是**正常结果**（EOF），和 `read` 一样要循环 |
| 以为越尾 `pwrite` 一定省磁盘 | 只撑长逻辑长度；是否省物理取决于偏移有没有跨出块边界（实测 1000 没跨，`st_blocks*512` 仍是 4096） |
| 混用 `read` 与 `pread` | 可以混（都作用于同一个打开文件表项，只是 `pread` 不动游标），但推理时要想清「游标由谁推进」 |
| 32 位平台上的 `_FILE_OFFSET_BITS` | `pread`/`pwrite` 的 `offset` 是 `off_t`；32 位要开 `_FILE_OFFSET_BITS=64` 才能处理大偏移（[5.10](5.10-large-files.md)） |

## 6. HFT / 嵌入式关联

| 场景 | 本节机制的兑现 |
|------|----------------|
| 多线程回测读历史数据 | 多个回测线程共享一个 `open` 的 fd，各用 `pread` 读自己那段区间——省 fd、无锁、无竞态，这是回测引擎的标准做法 |
| 定长记录文件 | 第 N 条记录的偏移是 `N * REC_SIZE`，`pread` 一次取一条，无需 seek（也不污染别的读取者） |
| 内存映射前的过渡方案 | `pread` 能做「随机访问」而不用 `mmap`，嵌入式上省地址空间（[Ch49](../../chapter-49-memory-mappings/README.md)） |
| 行情快照 / 订单簿还原 | `pwrite` 把某个校验和字段就地改写，游标不受影响——顺序写日志与就地修补可以共存 |
| 串口 / 管道上误用 | 嵌入式串口（`/dev/ttyS*`）在某些驱动下不支持 `pread` → 报 `ESPIPE`，得老实 `read` |
| 零拷贝 vs 一次拷贝 | `pread` 仍是「内核 → 用户缓冲」的一次拷贝；极致低延迟场景会考虑 `mmap` 或 `io_uring`（[Ch63](../../chapter-63-alternative-i-o-models/notes/63.4-the-epoll-api.md)） |

<summary>代码自测（Q&A）</summary>

**Q1：`pread` 为什么不改变文件偏移？**
A：因为它的位置是**参数**，内核把它放在栈上的局部变量 `pos` 里，`vfs_read(f.file, buf, count, &pos)` 用的是这个局部变量的地址（`fs/read_write.c:665`），**没有回写 `f_pos` 的语句**。对比 `ksys_read`（`:602`）在 `:615` 有 `f.file->f_pos = pos;`。

**Q2：`pread` 比 `lseek` + `read` 好在哪？**
A：两点。① **原子性**：定位与读取之间没有共享状态可被插入；② **多线程友好**：`pread` 用 `fdget` 而不是 `fdget_pos`（`fs/read_write.c:661` vs `:604`），不加 `f_pos_lock`，多个线程可以并发读同一 fd 的不同区间。实测三次 `pread` 后 `f_pos` 始终是 10。

**Q3：`pread` 在管道上返回什么？为什么？**
A：`-1 errno=29 (ESPIPE)`。`ksys_pread64` 里 `if (f.file->f_mode & FMODE_PREAD)`（`fs/read_write.c:664`）不成立，于是保留 `ret = -ESPIPE;`（`:663`）。管道没有偏移概念，不支持随机读。

**Q4：偏移超过文件末尾时，`pread` 和 `pwrite` 分别怎样？**
A：`pread` 返回 **0**（EOF，不是错误）；`pwrite` **把文件撑长**到 `offset + count`，中间未写的字节读出来是 0。实测 `pwrite(offset=1000)` 后 `st_size` 从 10 变 1001。

**Q5：第 3 段实测里 `st_blocks*512` 只有 4096，说明「有洞」了吗？**
A：**没有真正省磁盘**。文件逻辑 1001 字节、物理 4096（一个块），因为偏移 1000 落在第一个块内。真正的「逻辑大、物理小」要看 [5.8](5.8-truncate-ftruncate.md) 的 `ftruncate(fd, 8192)`：逻辑 8192 / 物理仍 4096。**判断洞要用 `st_size` 与 `st_blocks*512` 对比，不能只看 `st_size`。**

**Q6：`pwrite` 能不能用来做「就地改一个字段」？**
A：能，而且很合适：`pwrite(fd, &newval, sizeof newval, offset_of_field)` 不打扰任何人的游标，多线程写不同字段互不干扰。注意 fd 仍必须是**可写**的（`FMODE_PWRITE`，`fs/read_write.c:698`）。

**Q7：`read` 和 `pread` 混用会出问题吗？**
A：不会出错，但要想清「游标由谁推进」。`pread` 不影响 `f_pos`，所以混用时后续 `read` 会从**上一次 `read` 留下的位置**继续——这既可以是特性（顺序流 + 随机抽查），也可以是 bug 源（以为 `pread` 把位置挪走了）。
