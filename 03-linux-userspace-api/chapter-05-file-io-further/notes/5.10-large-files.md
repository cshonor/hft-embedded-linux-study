## 要点

历史：32 位 `off_t` 约 2GB 上限。

| 做法 | |
|------|--|
| 推荐 | 编译 `-D_FILE_OFFSET_BITS=64`，`off_t` 变 64 位 |
| 备选 | `off64_t` / `open64` / `lseek64` |

## 代码示例

```c
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

/* 编译: gcc -D_FILE_OFFSET_BITS=64 lfs_demo.c */
/* 加了 _FILE_OFFSET_BITS=64 后 off_t 自动变 64 位 */
int main(void) {
    printf("sizeof(off_t) = %zu\n", sizeof(off_t));

    /* 创建一个大文件（用 lseek 跳到 5GB 处写 1 字节） */
    int fd = open("/tmp/bigfile.dat", O_WRONLY | O_CREAT, 0644);
    if (fd < 0) { perror("open"); return 1; }

    off_t target = (off_t)5 * 1024 * 1024 * 1024;  /* 5 GB */
    if (lseek(fd, target, SEEK_SET) < 0) {
        perror("lseek");
        close(fd);
        return 1;
    }
    write(fd, "x", 1);  /* 空洞文件：实际只占 1 个磁盘块 */

    /* 验证文件大小 */
    off_t size = lseek(fd, 0, SEEK_END);
    printf("文件大小: %lld bytes (%.1f GB)\n",
           (long long)size, (double)size / (1024.0 * 1024 * 1024));

    close(fd);
    return 0;
}
```
