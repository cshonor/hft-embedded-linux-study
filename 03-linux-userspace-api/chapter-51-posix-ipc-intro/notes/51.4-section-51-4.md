## 代码示例

```c

/* POSIX IPC 权限模型：mode 参数 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>
#include <unistd.h>

int main(void) {
    /* mode 0644 = owner rw, group r, other r */
    mode_t old_umask = umask(0);  /* 清除 umask 以确保 mode 生效 */

    mqd_t mq = mq_open("/perm_demo", O_CREAT | O_RDWR, 0640, NULL);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }
    printf("MQ created with mode 0640 (owner rw, group r)\n");

    /* 查看实际权限 */
    struct stat sb;
    if (stat("/dev/mqueue/perm_demo", &sb) == 0) {
        printf("actual permissions: %04o\n", sb.st_mode & 0777);
    }

    umask(old_umask);  /* 恢复 umask */
    mq_close(mq);
    mq_unlink("/perm_demo");
    return 0;
}

```
