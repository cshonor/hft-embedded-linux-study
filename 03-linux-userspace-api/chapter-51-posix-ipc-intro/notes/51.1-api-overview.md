## 代码示例

```c

/* POSIX IPC vs SysV IPC 对比：创建与删除 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>

int main(void) {
    /* POSIX IPC：名字以 / 开头，用 mq_open / sem_open / shm_open */
    struct mq_attr attr = { .mq_maxmsg = 10, .mq_msgsize = 256 };
    mqd_t mq = mq_open("/posix_demo", O_CREAT | O_RDWR, 0644, &attr);
    if (mq == (mqd_t)-1) { perror("mq_open"); exit(1); }
    printf("POSIX MQ created: /posix_demo (name-based, no key needed)\n");

    /* 查看属性 */
    struct mq_attr cur;
    mq_getattr(mq, &cur);
    printf("  maxmsg=%ld, msgsize=%ld, curmsgs=%ld\n",
           cur.mq_maxmsg, cur.mq_msgsize, cur.mq_curmsgs);

    /* 对比：SysV 需要 ftok() 生成 key，再用 msgget(key, ...) */
    /* SysV: key_t key = ftok("/tmp/file", 'A'); int id = msgget(key, IPC_CREAT); */
    printf("SysV IPC needs ftok()+msgget(), POSIX uses name string directly\n");

    mq_close(mq);
    mq_unlink("/posix_demo");
    return 0;
}

```
