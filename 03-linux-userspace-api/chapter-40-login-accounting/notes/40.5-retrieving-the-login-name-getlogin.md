## 代码示例

```c
#include <utmpx.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>
int main(void) {
    /* 写登录记录 */
    struct utmpx ut = {0};
    ut.ut_type = USER_PROCESS;
    ut.ut_pid = getpid();
    strcpy(ut.ut_user, "myapp");
    strcpy(ut.ut_line, "pts/0");

    setutxent();
    /* pututxline(&ut); */  /* 实际写入 */
    endutxent();
    printf("utmp 更新 API: setutxent→pututxline→endutxent\n");
    return 0;
}
```
