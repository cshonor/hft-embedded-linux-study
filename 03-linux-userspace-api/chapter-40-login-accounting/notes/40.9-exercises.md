### 练习 1：列出当前登录用户

**题目**：使用 `utmpx` API 实现简化版 `who` 命令。

<details>
<summary>参考答案</summary>

```c
#include <utmpx.h>
#include <stdio.h>
#include <time.h>

int main() {
    setutxent();
    struct utmpx *ut;
    while ((ut = getutxent()) != NULL) {
        if (ut->ut_type != USER_PROCESS) continue;
        printf("%-10s %-8s %-16s %s",
               ut->ut_user, ut->ut_line,
               ut->ut_host[0] ? ut->ut_host : "-",
               ctime((time_t*)&ut->ut_tv.tv_sec));
    }
    endutxent();
    return 0;
}
```

</details>

### 练习 2：模拟登录记录

**题目**：编写程序向 wtmp 追加一条登录记录和一条注销记录。

<details>
<summary>参考答案</summary>

```c
#include <utmpx.h>
#include <string.h>
#include <time.h>
#include <unistd.h>

void write_wtmp(short type, const char *user, const char *line) {
    struct utmpx ut = {0};
    ut.ut_type = type;
    ut.ut_pid = getpid();
    strncpy(ut.ut_user, user, sizeof(ut.ut_user));
    strncpy(ut.ut_line, line, sizeof(ut.ut_line));
    ut.ut_tv.tv_sec = time(NULL);

    utmpxname("/var/log/wtmp");
    setutxent();
    pututxline(&ut);
    endutxent();
}

/* 写入登录 + 注销 */
write_wtmp(USER_PROCESS, "hft", "pts/0");
/* ... 一段时间后 ... */
write_wtmp(DEAD_PROCESS, "", "pts/0");
```

</details>
