## 本节讲什么

两端宜各自 bind。包大小受缓冲限制。 ---

---

## 要点

| | 网络 UDP | UNIX DGRAM |
|--|----------|------------|
| 可靠/有序 | 否 | **是**（内核转发） |
| 边界 | 有 | 有 |
| 满队列 | 可丢 | 发送方可**阻塞** |

两端宜各自 bind。包大小受缓冲限制。

## 代码示例

```c

/* AF_UNIX SOCK_DGRAM 数据报通信 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <sys/wait.h>

int main(void) {
    const char *srv_path = "/tmp/ch57_dgram_srv.sock";
    const char *cli_path = "/tmp/ch57_dgram_cli.sock";
    unlink(srv_path); unlink(cli_path);

    int sfd = socket(AF_UNIX, SOCK_DGRAM, 0);
    struct sockaddr_un saddr = { .sun_family = AF_UNIX };
    strncpy(saddr.sun_path, srv_path, sizeof(saddr.sun_path) - 1);
    bind(sfd, (struct sockaddr *)&saddr, sizeof(saddr));

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_UNIX, SOCK_DGRAM, 0);
        struct sockaddr_un caddr = { .sun_family = AF_UNIX };
        strncpy(caddr.sun_path, cli_path, sizeof(caddr.sun_path) - 1);
        bind(cfd, (struct sockaddr *)&caddr, sizeof(caddr));

        struct sockaddr_un dest = { .sun_family = AF_UNIX };
        strncpy(dest.sun_path, srv_path, sizeof(dest.sun_path) - 1);

        /* sendto 指定目标地址 */
        sendto(cfd, "dgram msg", 10, 0,
               (struct sockaddr *)&dest, sizeof(dest));
        printf("client: sent dgram\n");

        char buf[64];
        ssize_t n = recvfrom(cfd, buf, sizeof(buf) - 1, 0, NULL, NULL);
        if (n > 0) { buf[n] = '\0'; printf("client: reply='%s'\n", buf); }
        close(cfd);
        _exit(0);
    }

    /* recvfrom 获取发送方地址 */
    char buf[64];
    struct sockaddr_un src;
    socklen_t srclen = sizeof(src);
    ssize_t n = recvfrom(sfd, buf, sizeof(buf) - 1, 0,
                         (struct sockaddr *)&src, &srclen);
    if (n > 0) { buf[n] = '\0'; printf("server: received='%s' from %s\n", buf, src.sun_path); }

    /* 回复 */
    sendto(sfd, "dgram reply", 12, 0,
           (struct sockaddr *)&src, srclen);

    wait(NULL);
    close(sfd);
    unlink(srv_path); unlink(cli_path);
    return 0;
}

```
