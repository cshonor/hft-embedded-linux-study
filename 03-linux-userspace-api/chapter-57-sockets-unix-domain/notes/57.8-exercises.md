## 练习题

### 练习 1：Unix 域 echo 服务器

实现 Unix 域套接字 echo 服务器，客户端发送什么服务器回显什么。

### 练习 2：fd 传递

实现两个进程通过 Unix 域套接字传递文件描述符。

### 练习 3：抽象地址绑定

使用 Linux 抽象地址命名空间创建 Unix 域套接字服务器。

## 参考答案

### 练习 1 参考代码

```c
#include "tlpi_hdr.h"
#include <sys/un.h>
#include <sys/socket.h>

#define SOCK_PATH "/tmp/echo.sock"

int main(void) {
    int sfd = socket(AF_UNIX, SOCK_STREAM, 0);
    struct sockaddr_un addr;
    memset(&addr, 0, sizeof(addr));
    addr.sun_family = AF_UNIX;
    strncpy(addr.sun_path, SOCK_PATH, sizeof(addr.sun_path)-1);

    unlink(SOCK_PATH);  /* 清理旧 */
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("listening on %s\n", SOCK_PATH);
    int cfd = accept(sfd, NULL, NULL);

    char buf[4096];
    ssize_t n;
    while ((n = recv(cfd, buf, sizeof(buf), 0)) > 0)
        send(cfd, buf, n, 0);  /* echo back */

    close(cfd);
    close(sfd);
    unlink(SOCK_PATH);
    return 0;
}
```

### 练习 2 参考代码

```c
/* 发送 fd */
int send_fd(int sock, int fd) {
    struct msghdr msg = {0};
    char buf[CMSG_SPACE(sizeof(int))];
    struct iovec iov;
    char dummy = '\0';

    iov.iov_base = &dummy;
    iov.iov_len = 1;
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = buf;
    msg.msg_controllen = sizeof(buf);

    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    cmsg->cmsg_level = SOL_SOCKET;
    cmsg->cmsg_type = SCM_RIGHTS;
    cmsg->cmsg_len = CMSG_LEN(sizeof(int));
    memcpy(CMSG_DATA(cmsg), &fd, sizeof(int));

    return sendmsg(sock, &msg, 0);
}

/* 接收 fd */
int recv_fd(int sock) {
    struct msghdr msg = {0};
    char buf[CMSG_SPACE(sizeof(int))];
    char dummy;
    struct iovec iov = { &dummy, 1 };

    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = buf;
    msg.msg_controllen = sizeof(buf);

    recvmsg(sock, &msg, 0);
    struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
    int fd;
    memcpy(&fd, CMSG_DATA(cmsg), sizeof(int));
    return fd;
}
```

### 练习 3 参考代码

```c
#include <sys/un.h>
#include <sys/socket.h>
#include <string.h>

int main(void) {
    int sfd = socket(AF_UNIX, SOCK_STREAM, 0);

    struct sockaddr_un addr;
    memset(&addr, 0, sizeof(addr));
    addr.sun_family = AF_UNIX;
    /* 抽象地址：第一个字节为 \0 */
    addr.sun_path[0] = '\0';
    strncpy(addr.sun_path + 1, "hft_abstract", sizeof(addr.sun_path) - 2);

    bind(sfd, (struct sockaddr *)&addr,
         offsetof(struct sockaddr_un, sun_path) + 1 + strlen("hft_abstract"));
    listen(sfd, 5);

    printf("listening on abstract namespace\n");
    /* 不需要 unlink（无文件） */
    /* close(sfd) 后自动消失 */
    return 0;
}
```
