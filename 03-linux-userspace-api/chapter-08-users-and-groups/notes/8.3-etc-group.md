## 要点

```
groupname:x:gid:member1,member2
```

```c
#include <grp.h>
struct group {
    char *gr_name;
    char *gr_passwd;
    gid_t gr_gid;
    char **gr_mem;   /* 附属组成员名列表 */
};
```

```c
struct group *getgrnam(const char *name);
struct group *getgrgid(gid_t gid);
void setgrent(void);
struct group *getgrent(void);
void endgrent(void);
```
