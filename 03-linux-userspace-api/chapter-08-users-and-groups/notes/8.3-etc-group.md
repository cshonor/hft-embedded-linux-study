## 要点

```
groupname:x:gid:member1,member2
```

```c
#include <grp.h>
struct group {
    char *gr_name;
    char *gr_passwd;
    gid_t gr_gid;
    char **gr_mem;   /* 附属组成员名列表 */
};
```

```c
struct group *getgrnam(const char *name);
struct group *getgrgid(gid_t gid);
void setgrent(void);
struct group *getgrent(void);
void endgrent(void);
```

Demo：[`code/users_groups.c`](./code/users_groups.c)

## 代码示例

```c
#include <grp.h>
#include <stdio.h>

int main(void) {
    /* 按 GID 查询组信息 */
    struct group *gr = getgrgid(getgid());
    if (gr == NULL) { perror("getgrgid"); return 1; }

    printf("组名: %s\n", gr->gr_name);
    printf("GID: %d\n", gr->gr_gid);

    /* 打印组成员 */
    if (gr->gr_mem != NULL) {
        printf("成员: ");
        for (char **m = gr->gr_mem; *m != NULL; m++) {
            printf("%s ", *m);
        }
        printf("\n");
    }

    /* 按名字查询 */
    struct group *gr2 = getgrnam("root");
    if (gr2 != NULL) {
        printf("root 组 GID = %d\n", gr2->gr_gid);
    }

    return 0;
}
```
