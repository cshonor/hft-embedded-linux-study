## 要点

```c
getaddrinfo(host, service, &hints, &res);  /* 链表；用完 freeaddrinfo */
getnameinfo(addr, len, host, ..., serv, ..., flags);
```

| hints | |
|-------|--|
| `AI_PASSIVE` | 服务端；host=NULL → ANY |
| `AI_NUMERICHOST` / `NUMERICSERV` | 禁 DNS / 禁服务名查表 |
| `AF_UNSPEC` | 双栈友好 |

`ai_addr` 可直接 bind/connect。

---

## 代码示例

```c

/* getaddrinfo / getnameinfo：现代地址解析 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <netdb.h>
#include <arpa/inet.h>
#include <sys/socket.h>

int main(void) {
    /* getaddrinfo: 主机名 + 服务名 → sockaddr */
    struct addrinfo hints;
    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;    /* IPv4 或 IPv6 */
    hints.ai_socktype = SOCK_STREAM;

    struct addrinfo *res, *rp;
    int err = getaddrinfo("localhost", "8080", &hints, &res);
    if (err != 0) {
        fprintf(stderr, "getaddrinfo: %s\n", gai_strerror(err));
        exit(1);
    }

    printf("getaddrinfo('localhost', '8080'):\n");
    for (rp = res; rp != NULL; rp = rp->ai_next) {
        char host[NI_MAXHOST], serv[NI_MAXSERV];
        getnameinfo(rp->ai_addr, rp->ai_addrlen,
                    host, sizeof(host), serv, sizeof(serv),
                    NI_NUMERICHOST | NI_NUMERICSERV);
        int family = rp->ai_family;
        printf("  %s:%s (family=%s, socktype=%d)\n",
               host, serv,
               family == AF_INET ? "IPv4" : "IPv6",
               rp->ai_socktype);
    }

    freeaddrinfo(res);

    /* getnameinfo: sockaddr → 主机名+服务名 */
    struct sockaddr_in sa;
    memset(&sa, 0, sizeof(sa));
    sa.sin_family = AF_INET;
    sa.sin_port = htons(443);
    inet_pton(AF_INET, "93.184.216.34", &sa.sin_addr);

    char host[NI_MAXHOST], serv[NI_MAXSERV];
    getnameinfo((struct sockaddr *)&sa, sizeof(sa),
                host, sizeof(host), serv, sizeof(serv), 0);
    printf("\ngetnameinfo(93.184.216.34:443): host=%s, serv=%s\n", host, serv);
    return 0;
}

```
