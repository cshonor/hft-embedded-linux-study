## 代码示例

```c

/* 主机名与地址转换：inet_pton + inet_ntop 实战 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>

int main(void) {
    /* IPv4 地址转换循环 */
    const char *ips[] = { "127.0.0.1", "10.0.0.1", "192.168.1.100", "::1" };

    for (int i = 0; i < 4; i++) {
        const char *ip = ips[i];
        /* 尝试 IPv4 */
        struct in_addr a4;
        if (inet_pton(AF_INET, ip, &a4) == 1) {
            char buf[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &a4, buf, sizeof(buf));
            printf("IPv4: %s -> 0x%08x -> %s\n", ip, ntohl(a4.s_addr), buf);
            continue;
        }
        /* 尝试 IPv6 */
        struct in6_addr a6;
        if (inet_pton(AF_INET6, ip, &a6) == 1) {
            char buf[INET6_ADDRSTRLEN];
            inet_ntop(AF_INET6, &a6, buf, sizeof(buf));
            printf("IPv6: %s -> %s\n", ip, buf);
            continue;
        }
        printf("invalid: %s\n", ip);
    }
    return 0;
}

```
