## 本节讲什么

UDP：`sendto`/`recvfrom`；可选 `connect` 固定对端后用 send/recv。 TCP 迭代服务：`socket→bind→listen→循环 accept→处理→close(conn)`；listener 常留。 DNS / `/etc/services`：由 getaddrinfo 经 nsswitch 使用。

## 代码示例

```c

/* UDP 服务器 + 客户端示例 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/wait.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_DGRAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9997) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    printf("UDP server: listening on 0.0.0.0:9997\n");

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_INET, SOCK_DGRAM, 0);
        struct sockaddr_in sa = { .sin_family = AF_INET, .sin_port = htons(9997) };
        inet_pton(AF_INET, "127.0.0.1", &sa.sin_addr);
        /* UDP 不需要 connect，直接 sendto */
        sendto(cfd, "udp hello", 10, 0, (struct sockaddr *)&sa, sizeof(sa));
        printf("UDP client: sent 'udp hello'\n");

        char buf[64];
        struct sockaddr_in from;
        socklen_t fromlen = sizeof(from);
        ssize_t n = recvfrom(cfd, buf, sizeof(buf)-1, 0,
                             (struct sockaddr *)&from, &fromlen);
        if (n > 0) { buf[n]='\0'; printf("UDP client: reply='%s'\n", buf); }
        close(cfd);
        _exit(0);
    }

    char buf[64];
    struct sockaddr_in cli;
    socklen_t clilen = sizeof(cli);
    ssize_t n = recvfrom(sfd, buf, sizeof(buf)-1, 0,
                         (struct sockaddr *)&cli, &clilen);
    if (n > 0) { buf[n]='\0'; printf("UDP server: received='%s'\n", buf); }
    sendto(sfd, "udp reply", 10, 0, (struct sockaddr *)&cli, clilen);

    wait(NULL);
    close(sfd);
    return 0;
}

```
