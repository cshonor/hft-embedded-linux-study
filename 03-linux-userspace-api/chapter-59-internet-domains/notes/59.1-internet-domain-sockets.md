## 代码示例

```c

/* 网络数据序列化：发送结构体要注意字节序和对齐 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <arpa/inet.h>
#include <stdint.h>

/* 网络传输用 packed 结构或手动序列化 */
#pragma pack(push, 1)
struct net_msg {
    uint32_t magic;    /* 网络字节序 */
    uint16_t version;  /* 网络字节序 */
    uint16_t length;   /* 网络字节序 */
    char data[16];
};
#pragma pack(pop)

int main(void) {
    printf("sizeof(net_msg) = %zu (packed, no padding)\n", sizeof(struct net_msg));

    /* 序列化：主机序 → 网络序 */
    struct net_msg msg;
    msg.magic = htonl(0xDEADBEEF);
    msg.version = htons(1);
    msg.length = htons(sizeof("payload") - 1);
    strncpy(msg.data, "payload", sizeof(msg.data) - 1);

    printf("serialized: magic=0x%08x, ver=%u, len=%u, data='%s'\n",
           ntohl(msg.magic), ntohs(msg.version), ntohs(msg.length), msg.data);

    /* 反序列化：网络序 → 主机序 */
    uint32_t rcv_magic = ntohl(msg.magic);
    uint16_t rcv_ver = ntohs(msg.version);
    printf("deserialized: magic=0x%08x, ver=%u\n", rcv_magic, rcv_ver);

    printf("\n要点:\n");
    printf("  1. 所有多字节整数用 htons/htonl 转换\n");
    printf("  2. 结构体用 #pragma pack(1) 消除填充\n");
    printf("  3. 不同平台对齐可能不同\n");
    return 0;
}

```
