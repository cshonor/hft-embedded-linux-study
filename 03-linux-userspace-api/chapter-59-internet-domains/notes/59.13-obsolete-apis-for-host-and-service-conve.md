## 要点

## 代码示例

```c

/* SIGPIPE 处理：对端关闭后写 socket 会触发 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <signal.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <sys/wait.h>

int main(void) {
    /* 方法 1: 忽略 SIGPIPE（简单但不精确） */
    signal(SIGPIPE, SIG_IGN);
    printf("SIGPIPE ignored (global)\n");

    /* 方法 2: send 时用 MSG_NOSIGNAL（更推荐） */
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9998) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1;
    setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 1);

    pid_t pid = fork();
    if (pid == 0) {
        int cfd = socket(AF_INET, SOCK_STREAM, 0);
        struct sockaddr_in sa = { .sin_family = AF_INET, .sin_port = htons(9998) };
        inet_pton(AF_INET, "127.0.0.1", &sa.sin_addr);
        connect(cfd, (struct sockaddr *)&sa, sizeof(sa));
        close(cfd);  /* 立即关闭连接 */
        _exit(0);
    }

    int afd = accept(sfd, NULL, NULL);
    wait(NULL);

    /* 对端已关闭，此时 send 会触发 SIGPIPE */
    /* 用 MSG_NOSIGNAL 避免 */
    ssize_t ret = send(afd, "data", 5, MSG_NOSIGNAL);
    if (ret < 0) {
        perror("send after peer closed (MSG_NOSIGNAL)");
    }

    close(afd);
    close(sfd);
    return 0;
}

```
