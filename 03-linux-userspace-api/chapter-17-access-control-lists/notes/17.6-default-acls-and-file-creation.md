## 要点

```bash
getfacl file
getfacl -d dir

setfacl -m u:alice:rw file
setfacl -m g:dev:r-x dir
setfacl -d -m u:bob:rwx /project   # default ACL

setfacl -x u:alice file
setfacl -b file                    # 清 access ACL
setfacl -k dir                     # 仅清 default ACL
```

## 代码示例

```c
#include <sys/acl.h>
#include <stdio.h>

/* ACL 速查:
 * API: acl_get_file / acl_set_file / acl_get_entry / acl_get_tag_type
 * 类型: ACL_TYPE_ACCESS (访问), ACL_TYPE_DEFAULT (默认, 目录专用)
 * mask: 限制 ACL 条目的最大有效权限
 * gcc -lacl
 */
int main(void) {
    acl_t acl = acl_get_file("/tmp", ACL_TYPE_ACCESS);
    if (acl) {
        printf("ACL 获取成功 (%d 条目)\n",
               (int)acl_to_text(acl, NULL) ? 1 : 0);
        acl_free(acl);
    }
    printf("ACL: 比 rwx 更细粒度的权限控制\n");
    return 0;
}
```
