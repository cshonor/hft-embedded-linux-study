## 代码示例

```c

/* termios 标志位详解与修改 */
#include <stdio.h>
#include <stdlib.h>
#include <termios.h>
#include <unistd.h>

int main(void) {
    struct termios tio;
    tcgetattr(STDIN_FILENO, &tio);

    /* c_lflag 标志位 */
    printf("Local flags (c_lflag):\n");
    printf("  ICANON: %d (line buffering, signal editing)\n", !!(tio.c_lflag & ICANON));
    printf("  ECHO:   %d (echo input characters)\n", !!(tio.c_lflag & ECHO));
    printf("  ECHOE:  %d (echo erase as BS-SP-BS)\n", !!(tio.c_lflag & ECHOE));
    printf("  ISIG:   %d (generate signals: INTR, QUIT, SUSP)\n", !!(tio.c_lflag & ISIG));

    /* c_iflag 标志位 */
    printf("\nInput flags (c_iflag):\n");
    printf("  ICRNL:  %d (translate CR to NL)\n", !!(tio.c_iflag & ICRNL));
    printf("  IXON:   %d (XON/XOFF flow control)\n", !!(tio.c_iflag & IXON));
    printf("  IGNBRK: %d (ignore break condition)\n", !!(tio.c_iflag & IGNBRK));

    /* 修改：关闭 ICRNL（CR 不转 NL，二进制模式常用） */
    tio.c_iflag &= ~ICRNL;
    tcsetattr(STDIN_FILENO, TCSANOW, &tio);
    printf("\nICRNL cleared (CR no longer translated to NL)\n");

    /* tcflush: 刷新缓冲区 */
    tcflush(STDIN_FILENO, TCIFLUSH);
    printf("tcflush: input buffer flushed\n");

    return 0;
}

```
