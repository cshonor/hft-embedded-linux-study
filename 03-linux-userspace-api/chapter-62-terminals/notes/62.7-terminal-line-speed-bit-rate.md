## 代码示例

```c

/* 终端大小检测：ioctl TIOCGWINSZ */
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>

int main(void) {
    struct winsize ws;

    /* TIOCGWINSZ: 获取终端窗口大小 */
    if (ioctl(STDOUT_FILENO, TIOCGWINSZ, &ws) == 0) {
        printf("Terminal size:\n");
        printf("  rows:    %d\n", ws.ws_row);
        printf("  columns: %d\n", ws.ws_col);
        printf("  xpixel:  %d\n", ws.ws_xpixel);
        printf("  ypixel:  %d\n", ws.ws_ypixel);
    } else {
        perror("ioctl TIOCGWINSZ");
        printf("(not a terminal, or ioctl not supported)\n");
    }

    /* 也可以通过环境变量获取 */
    char *lines = getenv("LINES");
    char *cols = getenv("COLUMNS");
    printf("\nEnvironment: LINES=%s, COLUMNS=%s\n",
           lines ? lines : "(null)", cols ? cols : "(null)");

    return 0;
}

```
