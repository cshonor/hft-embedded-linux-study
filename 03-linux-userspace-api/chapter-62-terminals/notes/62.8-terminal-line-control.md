## 本节讲什么

会话最多一控制终端；**前台进程组**收 Ctrl+C/Z。 后台读 → `SIGTTIN`；后台写 → `SIGTTOU`（作业控制基础）。 SIGINT / SIGQUIT / SIGTSTP / SIGTTIN / SIGTTOU。

## 代码示例

```c

/* 终端产生的信号：SIGINT / SIGQUIT / SIGTSTP */
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <termios.h>
#include <unistd.h>

static volatile sig_atomic_t got_sigint = 0;
static volatile sig_atomic_t got_sigquit = 0;
static volatile sig_atomic_t got_sigtstp = 0;

static void handler(int sig) {
    switch (sig) {
        case SIGINT:  got_sigint = 1; break;
        case SIGQUIT: got_sigquit = 1; break;
        case SIGTSTP: got_sigtstp = 1; break;
    }
}

int main(void) {
    /* 终端按键产生的信号 */
    signal(SIGINT, handler);   /* Ctrl+C */
    signal(SIGQUIT, handler);  /* Ctrl+\ */
    signal(SIGTSTP, handler);  /* Ctrl+Z */

    printf("Terminal signals demo:\n");
    printf("  Ctrl+C  -> SIGINT  (interrupt)\n");
    printf("  Ctrl+\\  -> SIGQUIT (quit + core dump)\n");
    printf("  Ctrl+Z  -> SIGTSTP (terminal stop)\n");
    printf("  Ctrl+D  -> EOF (not a signal, closes stdin)\n\n");
    printf("Waiting... (run in a terminal to test)\n");

    /* 等待信号 */
    for (int i = 0; i < 30; i++) {
        if (got_sigint) { printf("Got SIGINT\n"); break; }
        if (got_sigquit) { printf("Got SIGQUIT\n"); break; }
        if (got_sigtstp) { printf("Got SIGTSTP\n"); break; }
        sleep(1);
    }

    return 0;
}

```
