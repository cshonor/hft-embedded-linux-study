## 代码示例

```c
#include <stdio.h>
#include <unistd.h>
/* 管道 + stdio: fdopen 将 fd 包装为 FILE* */
int main(void) {
    int fd[2];
    pipe(fd);

    if (fork() == 0) {
        close(fd[0]);
        /* fdopen: 将 fd 包装为 FILE*, 可以用 fprintf */
        FILE *fp = fdopen(fd[1], "w");
        fprintf(fp, "line 1 via fprintf\n");
        fprintf(fp, "line 2 via fprintf\n");
        fflush(fp);  /* 必须 flush, 否则数据在 stdio buffer 中 */
        fclose(fp);
        _exit(0);
    }

    close(fd[1]);
    FILE *fp = fdopen(fd[0], "r");
    char line[128];
    while (fgets(line, sizeof(line), fp))
        printf("read: %s", line);
    fclose(fp);
    wait(NULL);
    return 0;
}
```
