# 34.9 Exercises

> 本章：[TLPI 第 34 章 — Process Groups and Sessions](../README.md)

## 练习题

### 练习 1：创建进程组

编写程序，`fork()` 后子进程创建新进程组，打印 PID、PGID 和 SID。

### 练习 2：daemon 化

实现 `become_daemon()` 函数，完成标准 daemon 化流程。

### 练习 3：信号投递到进程组

编写程序，创建多个子进程放入同一进程组，父进程用 `killpg` 向整组发送信号。

## 参考答案

### 练习 1 参考代码

```c
#include "tlpi_hdr.h"

int main(void) {
    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程创建新进程组 */
        if (setsid() == -1) errExit("setsid");
        printf("child:  pid=%d pgid=%d sid=%d\n",
               (int)getpid(), (int)getpgrp(), (int)getsid(0));
        sleep(3);
        _exit(0);
    }
    printf("parent: pid=%d pgid=%d sid=%d\n",
           (int)getpid(), (int)getpgrp(), (int)getsid(0));
    wait(NULL);
    return 0;
}
```

### 练习 2 参考代码

```c
#include "tlpi_hdr.h"
#include <sys/stat.h>

int become_daemon(void) {
    /* 1. fork，父退出 */
    pid_t pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 2. setsid（成为会话首） */
    if (setsid() == -1) return -1;

    /* 3. 第二次 fork，不再是会话首 */
    pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 4. chdir("/") */
    if (chdir("/") == -1) return -1;

    /* 5. umask(0) */
    umask(0);

    /* 6. 关闭所有 fd，重定向 stdin/stdout/stderr */
    int maxfd = sysconf(_SC_OPEN_MAX);
    if (maxfd == -1) maxfd = 1024;
    for (int fd = 0; fd < maxfd; fd++) close(fd);

    /* 重定向到 /dev/null */
    int devnull = open("/dev/null", O_RDWR);
    if (devnull != 0) return -1;
    if (dup2(0, 1) == -1) return -1;
    if (dup2(0, 2) == -1) return -1;

    return 0;
}
```

### 练习 3 参考代码

```c
#include "tlpi_hdr.h"
#include <signal.h>

void handler(int sig) {
    printf("pid=%d received signal %d\n", (int)getpid(), sig);
    _exit(0);
}

int main(void) {
    /* 创建 3 个子进程放入同一进程组 */
    pid_t group_leader = -1;
    for (int i = 0; i < 3; i++) {
        pid_t pid = fork();
        if (pid == 0) {
            signal(SIGTERM, handler);
            setpgid(0, 0);  /* 第一个子进程创建新组 */
            printf("child %d: pid=%d pgid=%d\n", i, (int)getpid(), (int)getpgrp());
            for (;;) pause();
        }
        if (i == 0) group_leader = pid;
        else setpgid(pid, group_leader);
    }
    /* 父进程也设置子进程组 */
    setpgid(group_leader, group_leader);

    sleep(1);
    printf("parent: sending SIGTERM to group %d\n", group_leader);
    killpg(group_leader, SIGTERM);

    for (int i = 0; i < 3; i++) wait(NULL);
    return 0;
}
```
