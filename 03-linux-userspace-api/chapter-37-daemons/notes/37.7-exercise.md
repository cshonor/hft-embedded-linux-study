## 参考答案

### 练习 1 + 3 参考代码

```c
#include "tlpi_hdr.h"
#include <sys/stat.h>
#include <syslog.h>

#define PID_FILE "/var/run/mydaemon.pid"

int write_pid_file(void) {
    int fd = open(PID_FILE, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) return -1;
    char buf[16];
    snprintf(buf, sizeof(buf), "%d\n", (int)getpid());
    write(fd, buf, strlen(buf));
    close(fd);
    return 0;
}

int become_daemon(void) {
    /* 1. fork + 父 exit */
    pid_t pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 2. setsid */
    if (setsid() == -1) return -1;

    /* 3. 第二次 fork */
    pid = fork();
    if (pid == -1) return -1;
    if (pid != 0) _exit(0);

    /* 4. chdir + umask */
    chdir("/");
    umask(0);

    /* 5. 关闭所有 fd */
    int maxfd = sysconf(_SC_OPEN_MAX);
    if (maxfd == -1) maxfd = 1024;
    for (int fd = 0; fd < maxfd; fd++) close(fd);

    /* 重定向到 /dev/null */
    open("/dev/null", O_RDWR);  /* stdin = fd 0 */
    dup2(0, 1);                 /* stdout */
    dup2(0, 2);                 /* stderr */

    /* 6. syslog */
    openlog("mydaemon", LOG_PID, LOG_DAEMON);
    return 0;
}

void cleanup(void) {
    unlink(PID_FILE);
    closelog();
}

int main(void) {
    if (become_daemon() != 0) {
        fprintf(stderr, "become_daemon failed\n");
        exit(1);
    }
    atexit(cleanup);
    write_pid_file();

    syslog(LOG_INFO, "daemon started, pid=%d", (int)getpid());

    /* 主循环 */
    for (int i = 0; i < 60; i++) {
        syslog(LOG_INFO, "tick %d", i);
        sleep(1);
    }

    syslog(LOG_INFO, "daemon exiting");
    return 0;
}
```

### 练习 2 SIGHUP 处理

```c
#include <signal.h>
#include <stdatomic.h>

static atomic_int reload_flag = 0;

void sighup_handler(int sig) {
    (void)sig;
    reload_flag = 1;
}

/* 在 main 中 */
struct sigaction sa;
sa.sa_handler = sighup_handler;
sigemptyset(&sa.sa_mask);
sa.sa_flags = SA_RESTART;
sigaction(SIGHUP, &sa, NULL);

/* 主循环 */
while (running) {
    if (reload_flag) {
        reload_flag = 0;
        syslog(LOG_INFO, "reloading config...");
        /* reload_config(); */
        syslog(LOG_INFO, "config reloaded");
    }
    sleep(1);
}
```

## HFT / 嵌入式关联

- **HFT daemon**：行情/策略/风控进程都需 daemon 化 + PID 文件
- **systemd 服务**：现代 Linux 用 systemd 管理，`Type=simple` 不需程序自 daemon 化
- **嵌入式**：start-stop-daemon 或 init 脚本管理生命周期
- **日志**：HFT 高性能日志不用 syslog（延迟不可控），用自建 mmap 日志
