## 代码示例

```c
#include <unistd.h>
#include <stdio.h>
/* 守护进程特征:
 * - 无控制终端 (PPID=1)
 * - 在后台运行
 * - 通常 PPID 是 init/systemd
 */
int main(void) {
    printf("PID=%d PPID=%d PGID=%d SID=%d\n",
           getpid(), getppid(), getpgrp(), getsid(0));
    if (getppid() == 1) printf("→ 守护进程 (被 init 收养)\n");
    else printf("→ 普通进程\n");
    return 0;
}
```
