## 代码示例

```c
#include <unistd.h>
#include <fcntl.h>
#include <stdio.h>

/* TOCTOU: Time-of-Check to Time-of-Use 竞态
 * access() 检查和 open() 使用之间文件可能被替换
 */
int main(void) {
    const char *path = "/tmp/toctou_demo.txt";

    /* 不安全: access + open 之间有间隙 */
    if (access(path, W_OK) == 0) {
        /* ← 攻击者可在此刻替换文件为符号链接指向 /etc/passwd */
        int fd = open(path, O_WRONLY);
        if (fd >= 0) {
            write(fd, "oops", 4);  /* 可能写到了错误文件! */
            close(fd);
        }
    }

    /* 安全: 直接 open, 检查错误 */
    int fd = open(path, O_WRONLY | O_NOFOLLOW);  /* O_NOFOLLOW 防符号链接 */
    if (fd < 0) {
        printf("安全方式: 直接 open 并检查错误 (无 TOCTOU)\n");
    } else {
        write(fd, "safe", 4);
        close(fd);
    }

    printf("TOCTOU: 检查和使用之间有间隙 → 用原子操作避免\n");
    return 0;
}
```
