## 要点

```c
int sigaltstack(const stack_t *ss, stack_t *old_ss);
/* sa_flags |= SA_ONSTACK */
```

默认栈耗尽时仍要跑 `SIGSEGV` handler → 预分配独立栈（大小常用 `SIGSTKSZ`）。

## 代码示例

```c
#include <signal.h>
#include <stdio.h>
void handler(int sig) {
    char buf[64];
    int n = snprintf(buf, sizeof(buf), "alt stack handler sig=%d\n", sig);
    write(STDOUT_FILENO, buf, n);
}
int main(void) {
    /* 分配备用信号栈 (栈溢出时仍能执行 handler) */
    stack_t ss;
    ss.ss_sp = malloc(SIGSTKSZ);
    ss.ss_size = SIGSTKSZ;
    ss.ss_flags = 0;
    sigaltstack(&ss, NULL);

    struct sigaction sa = {0};
    sa.sa_handler = handler;
    sa.sa_flags = SA_ONSTACK;  /* 在备用栈上执行 */
    sigaction(SIGSEGV, &sa, NULL);

    printf("备用信号栈已设置 (SA_ONSTACK)\n");
    free(ss.ss_sp);
    return 0;
}
```
