## 代码示例

```c

/* 咨询锁 vs 强制锁概念演示 */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>

int main(void) {
    int fd = open("/tmp/ch55_advisory.txt", O_RDWR | O_CREAT, 0644);
    write(fd, "advisory lock demo", 18);

    /* flock 是咨询锁：只有协作进程才检查 */
    if (flock(fd, LOCK_EX | LOCK_NB) < 0) {
        perror("flock");
    } else {
        printf("Advisory lock acquired (flock LOCK_EX)\n");
        printf("Another process CAN still read/write via different fd\n");
        printf("(unless it also calls flock and respects the lock)\n");
    }

    /* 咨询锁不阻止不检查锁的进程 */
    int fd2 = open("/tmp/ch55_advisory.txt", O_RDONLY);
    char buf[32];
    int n = read(fd2, buf, sizeof(buf));
    buf[n] = '\0';
    printf("Uncooperative read succeeded: '%s'\n", buf);

    flock(fd, LOCK_UN);
    close(fd);
    close(fd2);
    return 0;
}

```
