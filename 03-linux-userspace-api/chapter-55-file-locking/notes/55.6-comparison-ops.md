## 代码示例

```c

/* flock vs fcntl 锁比较与死锁避免 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/file.h>

int main(void) {
    printf("=== flock vs fcntl comparison ===\n\n");

    printf("flock:\n");
    printf("  - whole-file lock (no byte ranges)\n");
    printf("  - advisory only (Linux)\n");
    printf("  - same fd: lock upgraded/downgraded\n");
    printf("  - fork: child can unlock parent's lock\n\n");

    printf("fcntl:\n");
    printf("  - byte-range record lock\n");
    printf("  - POSIX standard\n");
    printf("  - fork: child does NOT inherit locks\n");
    printf("  - close: releases all locks on that fd\n\n");

    /* 演示 fcntl 锁升级 */
    int fd = open("/tmp/ch55_cmp.txt", O_RDWR | O_CREAT, 0644);
    struct flock fl = { .l_type = F_RDLCK, .l_whence = SEEK_SET,
                        .l_start = 0, .l_len = 10 };
    fcntl(fd, F_SETLK, &fl);
    printf("fcntl: F_RDLCK (read lock) on 0-9\n");

    /* 升级为写锁 */
    fl.l_type = F_WRLCK;
    if (fcntl(fd, F_SETLK, &fl) == 0) {
        printf("fcntl: upgraded to F_WRLCK\n");
    }

    fl.l_type = F_UNLCK;
    fcntl(fd, F_SETLK, &fl);
    close(fd);
    return 0;
}

```
