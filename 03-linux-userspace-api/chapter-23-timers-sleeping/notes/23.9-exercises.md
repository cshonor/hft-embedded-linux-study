# 习题 23-1：用 `setitimer()` 实现 `alarm()`

## 看似简单，但有一个真考点

最直接的映射关系：

```c
struct itimerval newv;
newv.it_interval.tv_sec  = 0;      /* ← 必须是 0！否则就成了周期定时器 */
newv.it_interval.tv_usec = 0;
newv.it_value.tv_sec     = seconds;
newv.it_value.tv_usec    = 0;
setitimer(ITIMER_REAL, &newv, &oldv);
return oldv.it_value.tv_sec;       /* 「上一个闹钟还剩几秒」 */
```

`it_interval` 必须是 0 这一条是硬的：**`alarm()` 是「一次性」闹钟**，而 `setitimer` 只要 `it_interval` 非 0 就变成周而复始的定时器。

⭐ **但真正的考点在返回值上。** `alarm()` 的返回值是「上一个闹钟**还剩几秒**」，看似直接取 `oldv.it_value.tv_sec` 就行 —— **实测证明这两者不等价**：

| 剩余时间（真实值） | `alarm(0)` 返回 | 「裸 `setitimer` + 取 `tv_sec`」返回 |
|---|---|---|
| `4.699919` 秒 | **5** | **4** |
| ≥ 0.5 秒的任何值 | **向上取整** | **截断** |

差别的来源是内核的 `alarm_setitimer()`（见下文引文）：剩余微秒 ≥ 500000 时它会 `tv_sec++`。

```c
/* ex23_1_my_alarm.c —— TLPI 习题 23-1 的实现
 *
 * 题面（原书 §23.9，逐字）：
 *     "Although alarm() is implemented as a system call within the Linux
 *      kernel, this is redundant. Implement alarm() using setitimer()."
 *
 * 思路：alarm(seconds) 与 setitimer(ITIMER_REAL, ...) 的对应关系是
 *     it_value.tv_sec  = seconds          （单次，不倒计时重复）
 *     it_value.tv_usec = 0
 *     it_interval      = {0, 0}           ← **interval 必须是 0**，
 *                                            否则就成了周期定时器而不是 alarm
 * 而返回值（"上一个闹钟还剩几秒"）来自 setitimer() 的第三个参数 old_value。
 *
 * ⭐ 返回值是本习题的**真正考点**：alarm(2) 与 setitimer() 的返回值并不等价。
 *     实测（CE, gcc 13.3 / x86-64，冻结日志 tlpi-ch23-final*.txt）：
 *         my_alarm(4) 时上一次还剩 4.699907 秒 -> 本程序返回 4（直接取 tv_sec）
 *         alarm(5) 后过 0.3 秒，getitimer() 读到 4.699919 秒，
 *             alarm(0)                           -> glibc 返回 **5**（向上取整）
 *     差别来自内核的 alarm_setitimer()（见文件末尾引文）：剩余微秒 >= 500000
 *     时会 +1。也就是说**只照抄 setitimer(ITIMER_REAL, {it_value=tv_sec}) 并不
 *     完全等价** —— 这正是"用 setitimer 实现 alarm"这个题目值得做的原因。
 *
 * 程序做五件事：
 *   ① 空转语法：连续 my_alarm() 观察"返回值 = 上一个闹钟的剩余"
 *   ② 取消语义：my_alarm(0) 撤销闹钟并返回剩余（对应 alarm(0)）
 *   ③ 真触发：my_alarm(1) + pause()，确认 SIGALRM 真的送达
 *   ④ 与真 alarm() 对拍，把 ① 的"截断"与 glibc 的"向上取整"摆在一起
 *      —— 注意 ★ 处：真 alarm() 会**覆盖**本程序刚设的定时器，
 *         所以两者必须分开测，不能混在同一条时间线上。
 *
 * 编译：gcc -O0 -Wall -Wextra -o ex23_1_my_alarm ex23_1_my_alarm.c
 * 运行：./ex23_1_my_alarm
 *
 * 源码坐标：
 *   glibc 2.39  sysdeps/unix/sysv/linux/alarm.c  __alarm() 直接调 alarm 系统调用
 *   Linux      kernel/time/itimer.c   alarm_setitimer() —— 向上取整就在这里
 *              （v3.5 时该文件是 kernel/itimer.c；本文引用的原文取自
 *                linux-next v3.5 镜像站，**v6.6 的那一版未逐字核对**，
 *                但本机实测行为与下面这段逻辑完全吻合；逐字原文见
 *                notes/23.9-exercises.md，这里只写逻辑，免得再套一层注释）
 *
 *   内核逻辑（alarm_setitimer()，注释原文照抄）：
 *       do_setitimer(ITIMER_REAL, &it_new, &it_old);
 *
 *       .. We can't return 0 if we have an alarm pending ...  And we'd
 *          better return too much than too little anyway
 *       if ((!it_old.it_value.tv_sec && it_old.it_value.tv_usec) ||
 *           it_old.it_value.tv_usec >= 500000)
 *               it_old.it_value.tv_sec++;
 *
 *       return it_old.it_value.tv_sec;
 *
 *   即：剩余「秒为 0 但微秒非 0」或「微秒 >= 500000」时进位。
 *   所以真正的 alarm() 与"裸 setitimer + 取 tv_sec"在剩余 >= 0.5 秒时差 1。
 */
#define _POSIX_C_SOURCE 199309
#include <sys/time.h>
#include <signal.h>
#include <time.h>
#include <unistd.h>
#include <stdio.h>
#include <errno.h>

static volatile sig_atomic_t alarmFired = 0;

/* 用 nanosleep 而不是 usleep：本文件只定义 _POSIX_C_SOURCE 199309，
   glibc 在这个宏下**不声明 usleep**（它属于 _DEFAULT_SOURCE / _XOPEN_SOURCE），
   而 nanosleep 是 POSIX.1b 正牌接口。 */
static void msleep(long ms)
{
    struct timespec req;

    req.tv_sec = ms / 1000;
    req.tv_nsec = (ms % 1000) * 1000000L;
    while (nanosleep(&req, &req) == -1 && errno == EINTR)
        ;
}

static void handler(int sig)
{
    (void) sig;
    alarmFired = 1;
}

/* 习题 23-1 的实现：用 setitimer() 造一个 alarm() */
static unsigned int my_alarm(unsigned int seconds)
{
    struct itimerval newv, oldv;

    newv.it_interval.tv_sec = 0;        /* 单次 —— 这是 alarm 与周期定时器的分界 */
    newv.it_interval.tv_usec = 0;
    newv.it_value.tv_sec = (long) seconds;
    newv.it_value.tv_usec = 0;

    if (setitimer(ITIMER_REAL, &newv, &oldv) == -1)
        return 0;                       /* alarm(3) 失败也返回 0 */

    printf("    [my_alarm(%u)] 旧定时器剩余：%ld.%06ld 秒"
           " -> 返回 tv_sec = %ld\n",
           seconds, (long) oldv.it_value.tv_sec,
           (long) oldv.it_value.tv_usec, (long) oldv.it_value.tv_sec);

    return (unsigned int) oldv.it_value.tv_sec;
}

int main(void)
{
    struct sigaction sa;

    sa.sa_flags = 0;
    sa.sa_handler = handler;
    sigemptyset(&sa.sa_mask);
    if (sigaction(SIGALRM, &sa, NULL) == -1) {
        printf("sigaction 失败 errno=%d\n", errno);
        return 1;
    }

    printf("=== ① 连续 my_alarm()：返回值就是「上一个闹钟的剩余」 ===\n");
    my_alarm(5);                        /* 第一个：之前没有闹钟 => 剩余 0.000000 */
    msleep(300);                        /* 消耗 0.3 秒 */
    my_alarm(4);                        /* 应剩约 4.7 秒 */
    msleep(300);
    my_alarm(3);                        /* 应剩约 3.7 秒 */

    printf("=== ② my_alarm(0) 撤销闹钟，并返回剩余（对应 alarm(0)） ===\n");
    my_alarm(0);
    alarmFired = 0;
    sleep(1);                           /* 等 1 秒 —— 若没撤销成功，这里会收到信号 */
    printf("    撤销后等 1 秒，alarmFired = %d（期望 0，说明闹钟真的被撤销了）\n",
           (int) alarmFired);

    printf("=== ③ 真触发：my_alarm(1) + pause() ===\n");
    alarmFired = 0;
    my_alarm(1);
    while (!alarmFired)
        pause();
    printf("    收到 SIGALRM，alarmFired = %d（期望 1）\n", (int) alarmFired);

    printf("=== ④ 与真 alarm() 对拍（★ 分开测，因为两者会互相覆盖） ===\n");
    printf("  真 alarm(5)   -> 返回 %u\n", alarm(5));
    msleep(300);
    {
        struct itimerval itv;

        /* 先用 getitimer() 把「真实剩余」掏出来，再把 alarm(0) 的返回值放一起看 */
        if (getitimer(ITIMER_REAL, &itv) == 0)
            printf("  getitimer() 看到的真实剩余：%ld.%06ld 秒\n",
                   (long) itv.it_value.tv_sec, (long) itv.it_value.tv_usec);
        printf("  真 alarm(0)   -> 返回 %u   ← 与上一行比：**向上取整**，不是截断\n",
               alarm(0));
    }
    printf("  对照：① 里 my_alarm() 返回的是裸的 tv_sec（截断），"
           "两者在剩余 >= 0.5 秒时会差 1\n");

    printf("\n=== ex23_1 done ===\n");
    return 0;
}
```

## 内核侧的差别（逐字引文）

``` 内核原文（linux-next v3.5 镜像站，逐字）
	do_setitimer(ITIMER_REAL, &it_new, &it_old);

	/*
	 * We can't return 0 if we have an alarm pending ...  And we'd
	 * better return too much than too little anyway
	 */
	if ((!it_old.it_value.tv_sec && it_old.it_value.tv_usec) ||
	    it_old.it_value.tv_usec >= 500000)
		it_old.it_value.tv_sec++;

	return it_old.it_value.tv_sec;
```

两条进位规则（`||` 的两边）：

| 条件 | 含义 |
|------|------|
| `!tv_sec && tv_usec` | 剩余「**不到 1 秒但也不是 0**」（比如 0.3 秒）⇒ 返回值不能是 0，因为「还有闹钟挂着」 |
| `tv_usec >= 500000` | 剩余的小数部分 ≥ 0.5 秒 ⇒ **进位** |

内核注释那句 *we'd better return too much than too little* 就是设计意图：**宁可多报，不可少报**。

> ⚠️ **版本说明（诚实标注）**：上面这段引自 **v3.5**（当时该文件是 `kernel/itimer.c`）。v6.6 已移到 `kernel/time/itimer.c`，**本仓库没有逐字核对那一版**（在线源码站被反爬拦住）。但本页实测的 `4.699919 → 返回 5` 与这段逻辑**完全吻合**，所以行为结论是可靠的。

## 本题的结论

| 问题 | 答案 |
|------|------|
| 「用 `setitimer()` 实现 `alarm()`」能实现吗？ | 功能上能：`it_interval = {0,0}` + `it_value = {seconds, 0}` + 返回 `old_value.it_value.tv_sec` |
| 完全等价吗？ | ❌ **不**。`alarm()` 的返回值是**向上取整**的剩余秒数；照抄 `setitimer` 得到的是**截断**值。剩余 ≥ 0.5 秒时两者差 1 |
| 差这 1 秒有实际影响吗？ | 有。`alarm()` 的典型用法是「撤销时知道还剩多少」—— 依赖这个数做剩余时间估算的代码会差最多 1 秒。要**精确**剩余就调 `getitimer()`（本页 ④ 段就是这么做的） |

# 习题 23-2：`nanosleep()` 的重启漂移 vs `clock_nanosleep(TIMER_ABSTIME)`

## 题面要观察的现象

```
$ while true; do kill -INT pid; done
```

用高频 `SIGINT` 猛打一个正在 `nanosleep(60)` 的进程 —— 你会看到它**睡得比预期长得多**。

## 为什么会这样

`t_nanosleep.c` 的逻辑是「被打断就用 `remain` 续睡」：

```c
s = nanosleep(&request, &remain);
if (s == 0) break;
request = remain;          /* 用剩余时间续睡 */
```

问题出在 `remain` **几乎不可能是软件时钟粒度（jiffy）的整数倍**，而内核会把定时值**向上取整**到 jiffy 的整数倍（§23.2）。于是**每一轮重启都要多睡一截，误差逐轮累积**：

| 轮次 | 请求 | jiffy = 4 ms 时的实际睡眠 | 结果 |
|------|------|-------------------------|------|
| 1 | 1000 ms | 被打断于 1000 ms | `remain = 8.1466 ms` |
| 2 | 8.1466 ms | **向上取整到 12 ms** | 多睡 3.85 ms，且**这 3.85 ms 白扔了** |

信号来得越密，轮次越多，累积的「白扔」越多。**极端情况**（信号间隔 ≤ jiffy）下每一轮都睡不完 —— 原书原文：

> In the case of an extremely high rate of signal delivery (i.e., as or more frequent than the software clock granularity), **the process may never be able to complete its sleep**.

## 官方解答：`code/t_clock_nanosleep.c`

**官方答案文件是 `timers/t_clock_nanosleep.c`** —— 完整源码已内联在 [23.5 POSIX Clocks](23.5-posix-clocks.md)（与 Listing 23-4 的片段对照着看）。

核心改动只有两处：

```c
/* ① 不用 remain 续睡，而是每次都用**同一个绝对目标时刻** */

    clock_gettime(CLOCK_REALTIME, &request);   /* 取当前绝对时刻 */
    request.tv_sec  += 1000;  /* 加上想要偏移 —— 得到绝对目标 */  /* ← 示意，实际由 argv 给 */
    ...
}

/* ② 被打断后原样重复调用（request 不变），而不是拿 remain 当新请求 */
for (;;) {
    int s = clock_nanosleep(CLOCK_REALTIME, flags, &request, &remain);
    if (s == 0) break;
    if (s == EINTR) printf("Interrupted... ");
    ...
    printf("... Restarting\n");
}
```

## 为什么这样就对了

| 维度 | `nanosleep()` + `remain` 续睡 | `clock_nanosleep(TIMER_ABSTIME)` |
|------|------------------------------|----------------------------------|
| 请求的对象 | **相对量**（从现在起睡多久） | **绝对时刻**（睡到某个点） |
| 每轮重启 | `request = remain`（新的相对量） | `request` **不变**（同一个绝对时刻） |
| 取整误差 | **每轮累积** ⇒ 越睡越长 | **不累积** ⇒ 每轮都是「睡到那一刻」 |
| 信号极密时 | **可能永远睡不完** | 只要有一次不受干扰就立即到期 |

> ⭐ 一句话：**相对睡眠的误差会累积，绝对睡眠的误差不会。** 因为「睡到 T 时刻」这个目标**不随重启而改变**，每次重启只是重新算「还有多久到 T」，不存在把上一轮的取整余数带进下一轮的问题。

## 本仓库的 `TIMER_ABSTIME` 实测

CE 环境的执行器不向被测进程发信号，所以**无法复现题面里 `while true; do kill -INT ...; done` 那个实验**（没有手动发信号的通道）。本仓库能提供的是**同一程序在两种模式下的对照实测**：

| 模式 | 命令 | 实测输出 |
|------|------|---------|
| 相对（`nanosleep` 风格） | `t_clock_nanosleep 1 0` | `Slept: 1.000071 secs` |
| 绝对（`TIMER_ABSTIME`） | `t_clock_nanosleep 1 0 a` | `Initial CLOCK_REALTIME value: 1789619387.870112001` → `Slept: 1.000051 secs` |

完整输出见 [23.5 POSIX Clocks](23.5-posix-clocks.md)。**两次超出量同量级（+71 µs / +51 µs）** —— 这说明绝对模式解决的**不是单次精度**，而是「**多次重启时的累积漂移**」与「**信号极密时永不到期**」。要看后者必须能手动发信号（真终端 / 本地机器）。

# 习题 23-4：把 Listing 23-5 改成 `sigwaitinfo()`

## 题面要什么

> Modify the program in Listing 23-5 (`ptmr_sigev_signal.c`) to use `sigwaitinfo()` instead of a signal handler.

## 改动的本质：从「异步送达」变成「同步领取」

| 维度 | Listing 23-5（处理器） | 习题 23-4（`sigwaitinfo`） |
|------|---------------------|--------------------------|
| 接收方式 | `sigaction(TIMER_SIG, SA_SIGINFO, handler)` + `for (;;) pause();` | `sigprocmask(SIG_BLOCK, {TIMER_SIG})` + `sigwaitinfo(&set, &si)` |
| 代码跑在哪 | **异步信号上下文** | **普通同步上下文** |
| 能调什么函数 | 只能 async-signal-safe 函数（`printf` 是 **UNSAFE**） | ✅ **随便调** —— `printf`、`malloc`、加锁都行 |
| 栈 | 可能跑在信号栈上（`sigaltstack`） | 普通栈 |
| 竞态风险 | 高（信号可能在任意指令处插入） | 低（信号只在 `sigwaitinfo()` 里被取走） |

## ⭐ 为什么必须先 `sigprocmask(SIG_BLOCK)`

这是本题**唯一**容易写错的地方。`sigwaitinfo()` 只对**处于阻塞状态**的信号可靠：

| 若信号**没有**被阻塞 | 后果 |
|-------------------|------|
| 信号按**默认处置**送达 | `SIGRTMAX` 的默认动作是**终止进程** ⇒ 程序还没 `sigwaitinfo()` 就死了 |
| 即便装了处理器 | 信号可能被处理器**抢先领走** ⇒ `sigwaitinfo()` **永远等下去** |

所以正确顺序是：**先 `SIG_BLOCK`，再创建/武装定时器，最后 `sigwaitinfo()`**。

> ⚠️ 顺序反了（先武装、后阻塞）会丢通知：信号在阻塞生效之前送达会被处理器/默认处置消化掉。本仓库程序的文件头专门标注了这一点。

## 完整源码（本仓库自写）

⚠️ **原书没有这个程序的官方实现**（习题 23-4 没有解答文件）。

**实现**：`code/ex23_4_ptmr_sigwaitinfo.c`（本仓库自写，非原书内容）。

[`code/ex23_4_ptmr_sigwaitinfo.c`](../code/ex23_4_ptmr_sigwaitinfo.c)

```c
/* ex23_4_ptmr_sigwaitinfo.c —— TLPI 习题 23-4 的实现
 *
 * 题面（原书 §23.9，逐字）：
 *     "Modify the program in Listing 23-5 (ptmr_sigev_signal.c) to use
 *      sigwaitinfo() instead of a signal handler."
 *
 * 与 Listing 23-5（`code/ptmr_sigev_signal.c`）的差别只有一处，但含义很大：
 *
 *   Listing 23-5：sigaction(TIMER_SIG, SA_SIGINFO, handler) + `for (;;) pause();`
 *                 ⇒ 通知**异步**送到信号处理器里
 *   本程序     ：sigprocmask(SIG_BLOCK, {TIMER_SIG}) + `sigwaitinfo(&set, &si)`
 *                 ⇒ 通知**同步**取出来，没有信号处理器、没有异步信号上下文
 *
 * ⭐ 为什么要先 sigprocmask() 阻塞？
 *     sigwaitinfo() 只有在信号**处于阻塞状态**时才是可靠的：如果信号没被阻塞，
 *     它可能在 sigwaitinfo() 被调用**之前**就按默认处置送达（SIGRTMAX 的默认
 *     动作是终止进程）；更糟的是，即便装了处理器，通知也可能被处理器抢先领走，
 *     让 sigwaitinfo() 永远等下去。先阻塞 → 信号只会挂在 pending 集合里 →
 *     由 sigwaitinfo() 独占领取。这是 §22.10 的核心结论。
 *
 * ⭐ 由此带来的两个好处（正是 sigwaitinfo 存在的理由）：
 *     1. **没有异步信号上下文** ⇒ 可以直接调用 printf / malloc 等非异步信号
 *        安全函数，不必像 Listing 23-5 那样在注释里写 "UNSAFE"。
 *     2. **可以拿到 siginfo 的完整信息**，包括定时器专用的 `si_overrun`
 *        （Linux 扩展）——它和 timer_getoverrun() 给的是同一个数。
 *
 * ⚠️ 本程序相对题面的**唯一扩展**（明确声明，不是原书内容）：
 *     Listing 23-5 的 `for (;;) pause();` 没有退出路径；为了让它在无人值守的
 *     环境里也能跑完并留下完整输出，本程序加了一个 `-n <N>` 选项：收到 N 次
 *     通知后正常退出（`-n 0` / 不写 = 与原书一样无限等待）。
 *
 * 用法：./ex23_4_ptmr_sigwaitinfo [-n max-notify] secs[/nsecs][:int-secs[/int-nsecs]]...
 *
 * 编译：gcc -O0 -Wall -Wextra -o ex23_4_ptmr_sigwaitinfo \
 *           ex23_4_ptmr_sigwaitinfo.c curr_time.c itimerspec_from_str.c
 *
 * 源码坐标（Linux v6.6）：
 *   kernel/signal.c            sigwaitinfo() 走 do_sigtimedwait()，直接操作
 *                              pending 集合，**不经过信号投递流程**
 *   kernel/time/posix-timers.c common_timer_get() → si_overrun 的来源
 *                              （即 timer_getoverrun() 的字段级出处）
 */
#define _POSIX_C_SOURCE 199309
#include <signal.h>
#include <time.h>
#include <unistd.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "curr_time.h"                  /* Declares currTime() */
#include "itimerspec_from_str.h"        /* Declares itimerspecFromStr() */
#include "tlpi_hdr.h"

#define TIMER_SIG SIGRTMAX              /* 与 Listing 23-5 保持一致 */

int
main(int argc, char *argv[])
{
    struct itimerspec ts;
    struct sigevent   sev;
    timer_t *tidlist;
    sigset_t blockSet;
    int j, opt, maxNotify = 0, nNotify = 0;

    while ((opt = getopt(argc, argv, "n:")) != -1) {
        switch (opt) {
        case 'n':
            maxNotify = atoi(optarg);
            break;
        default:
            usageErr("%s [-n max-notify] "
                     "secs[/nsecs][:int-secs[/int-nsecs]]...\n", argv[0]);
        }
    }

    if (optind >= argc)
        usageErr("%s [-n max-notify] "
                 "secs[/nsecs][:int-secs[/int-nsecs]]...\n", argv[0]);

    tidlist = calloc(argc - optind, sizeof(timer_t));
    if (tidlist == NULL)
        errExit("calloc");

    /* ★ 关键差异 1：不做 sigaction()，改成把 TIMER_SIG 阻塞掉。
       （先阻塞、后建定时器 —— 顺序反了就会丢掉最早几次通知。） */

    sigemptyset(&blockSet);
    sigaddset(&blockSet, TIMER_SIG);
    if (sigprocmask(SIG_BLOCK, &blockSet, NULL) == -1)
        errExit("sigprocmask");

    /* 创建并启动每个命令行参数对应的一个定时器 —— 与 Listing 23-5 完全相同 */

    sev.sigev_notify = SIGEV_SIGNAL;    /* Notify via signal */
    sev.sigev_signo = TIMER_SIG;        /* Notify using this signal */

    for (j = 0; j < argc - optind; j++) {
        itimerspecFromStr(argv[j + optind], &ts);

        sev.sigev_value.sival_ptr = &tidlist[j];
                /* Allows the waiter to get ID of this timer */

        if (timer_create(CLOCK_REALTIME, &sev, &tidlist[j]) == -1)
            errExit("timer_create");
        printf("Timer ID: %ld (%s)\n", (long) tidlist[j], argv[j + optind]);

        if (timer_settime(tidlist[j], 0, &ts, NULL) == -1)
            errExit("timer_settime");
    }

    /* ★ 关键差异 2：同步等待，而不是 pause() + 处理器 */

    for (;;) {
        siginfo_t si;
        int sig;
        timer_t *tidptr;

        sig = sigwaitinfo(&blockSet, &si);
        if (sig == -1) {
            if (errno == EINTR) {
                printf("    （sigwaitinfo 被无关信号打断，重试）\n");
                continue;
            }
            errExit("sigwaitinfo");
        }

        tidptr = si.si_value.sival_ptr;

        /* 这里 printf 是**安全的** —— 同步上下文，不是异步信号处理器 */
        printf("[%s] sigwaitinfo() got signal %d\n", currTime("%T"), sig);
        printf("    *sival_ptr         = %ld\n", (long) *tidptr);
        printf("    si_overrun         = %d\n", si.si_overrun);
        printf("    timer_getoverrun() = %d\n", timer_getoverrun(*tidptr));

        nNotify++;
        if (maxNotify > 0 && nNotify >= maxNotify) {
            printf("已经收到 %d 次通知，按 -n 限定退出（注意：未阻塞前的事件数不保证）\n",
                   nNotify);
            break;
        }
    }

    for (j = 0; j < argc - optind; j++)
        timer_delete(tidlist[j]);
    free(tidlist);

    printf("\n=== ex23_4 done ===\n");
    exit(EXIT_SUCCESS);
}
```

⭐ **源码核验发现**：官方 `timers/` 目录里有三个 `.c` **在 §23.7 正文里完全没有被提到**，也没有印成任何 Listing：

4 [t=0.10 (delta: 0.05)] %CPU: 99.77
3 [t=0.15 (delta: 0.05)] %CPU: 99.82

两个多路版本的读法：

| 观察 | 结论 |
|------|------|
| pid 是 **3、4**（多进程版）或 tid 是 **3、4**（多线程版） | 都是 nsjail PID namespace 里的小号 |
| `%CPU` 几乎都是 **99.x** | 与 `cpu_burner -i 0` 不同，这两个程序**没有 idle 占比选项** ⇒ 每个子进程/线程都**贪婪烧满**它拿到的 CPU |
| 多线程版的两行**交错**出现（3、4、3、4…） | 两个线程在**同一个进程**里被调度器轮流执行，打印交错是正常的 |
| `delta: 0.05` | 与上一行报告之间的**实际间隔**；偶尔出现 `delta: 0.32` / `0.67` / `0.78` 就是**被抢占**了 |

> ⭐ 最后一行那个「偶尔冒出来的大 `delta`」是本附录最有价值的一手数据：**它把 §23.2 说的「到期 ≠ 被调度」变成了可见的数字。** 定时器说「50 ms 后叫我」，实际有时候 320 ms 才轮得到 —— 中间那 270 ms 就是调度延迟。

**Q1：习题 23-1 的「真正考点」是什么？**

A：**返回值口径不一致**。`alarm()` 返回的是「上一个闹钟**还剩几秒**」，而内核 `alarm_setitimer()` 在剩余微秒 ≥ 500000 时会 `tv_sec++`（**向上取整**）；如果你只是照抄 `setitimer()` 然后取 `oldv.it_value.tv_sec`，拿到的是**截断**值。本页实测：真实剩余 `4.699919` 秒 ⇒ `alarm(0)` 返回 **5**，而 `my_alarm()` 返回 **4**。**所以「用 `setitimer` 实现 `alarm`」在功能上可行、在返回值上不完全等价。**

**Q2：习题 23-2 里为什么「`nanosleep` + `remain` 续睡」会越来越慢？**

A：因为每一轮重启都要**向上取整到 jiffy 的整数倍**，而 `remain` 几乎不会是整数倍 ⇒ 每轮多睡一截，误差**逐轮累积**。信号越密轮次越多，累积越多；信号密到间隔 ≤ jiffy 时，**每一轮都睡不完，程序永远出不来**。`clock_nanosleep(TIMER_ABSTIME)` 请求的是**绝对时刻**，重启时 `request` 不变 ⇒ 误差不累积。

**Q3：习题 23-4 为什么必须先 `sigprocmask(SIG_BLOCK)`？**

A：`sigwaitinfo()` 只对**阻塞状态**的信号可靠。如果信号没被阻塞：① 它可能按**默认处置**送达（`SIGRTMAX` 的默认动作是**终止进程**），程序还没等到就死了；② 即便装了处理器，通知也可能被处理器**抢先领走**，`sigwaitinfo()` 永远等下去。所以顺序是：**先阻塞 → 再建/武装定时器 → 最后 `sigwaitinfo()`**。

**Q4：`sigwaitinfo()` 相对信号处理器有哪两个真实好处？**

A：① **没有异步信号上下文** —— 可以随便调 `printf`、`malloc`、加锁、`free`（Listing 23-5 的处理器里这些全是 `UNSAFE`）；② **能安全地做清理** —— 本页的自写程序在退出前能 `timer_delete()` + `free()`，而原书版本只能靠进程退出时内核回收。附带第三点：本页程序还能顺手打印 `si_overrun`（Linux 扩展字段），省掉一次 `timer_getoverrun()` 系统调用。

**Q5：`cpu_burner -i 50` 为什么实测是 46~50% 而不是精确 50%？**

A：因为 `nanosleep()` 会 **oversleep**（§23.2 实测 +51~96 µs / 次）。`cpu_burner` 的算法是「每 10 ms 里睡 5 ms」，每次睡眠都会多睡几十微秒，于是**多睡的部分就从 CPU 时间里扣掉了**。原书在源码注释里自己写明了这一点：*we may get up to around **1% less** CPU time than we were aiming for*。本页实测的 46~50 与这个描述一致。

**Q6：`cpu_multi_burner` 输出里偶尔出现的 `delta: 0.32` / `0.67` / `0.78` 说明什么？**

A：说明**这个进程被抢占了**。程序本意每 50 ms 报一次，但实际有时 320 ms、670 ms、780 ms 才轮到它 —— **多出来的那几百毫秒就是调度延迟**。这正是 §23.2 的核心结论「到期 ≠ 被调度」的直接数字证据。而这些 burner 的存在意义就是**人为制造这种竞争**，用来复现 §23.6.6 说的 overrun 场景。

**Q7：为什么本页的附录取名叫「附录 A / B」而不是给它们编节号？**

A：因为按本仓库的硬约定 —— **不伪造节号**。原书 §23.9 只有四道练习题，`cpu_*_burner` 和「`Listing 23-4` 无文件」这两件事都属于**本仓库对照官方分发素材时补出来的说明**，不是原书的节。把它们放进附录并显式标注来源，读者才不会误以为书里有这几节。
