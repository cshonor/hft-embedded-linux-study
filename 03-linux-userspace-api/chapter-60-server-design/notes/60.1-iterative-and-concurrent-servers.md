## 代码示例

```c

/* 迭代服务器：accept → 处理 → close → 循环 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

int main(void) {
    int sfd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in addr = { .sin_family = AF_INET, .sin_port = htons(9995) };
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    int opt = 1; setsockopt(sfd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    bind(sfd, (struct sockaddr *)&addr, sizeof(addr));
    listen(sfd, 5);

    printf("iterative server: port 9995\n");
    printf("accept -> handle -> close -> loop\n");
    printf("(one client at a time, others wait in backlog)\n");

    /* 实际循环（简化演示） */
    for (int i = 0; i < 1; i++) {
        int cfd = accept(sfd, NULL, NULL);
        if (cfd < 0) continue;
        char buf[64];
        ssize_t n = recv(cfd, buf, sizeof(buf)-1, 0);
        if (n > 0) { buf[n]='\0'; printf("handled: %s\n", buf); }
        send(cfd, "done", 5, 0);
        close(cfd);
    }

    close(sfd);
    return 0;
}

```
