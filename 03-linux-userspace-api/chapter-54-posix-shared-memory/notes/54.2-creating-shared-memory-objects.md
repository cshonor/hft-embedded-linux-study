## 要点

```c
int fd = shm_open("/shmbuf", O_CREAT|O_RDWR, 0600);
ftruncate(fd, len);   /* 新建为 0，必须扩展 */
void *p = mmap(NULL, len, PROT_READ|PROT_WRITE, MAP_SHARED, fd, 0);
```

- 名：`/foo`；无多层 `/`  
- 返回**普通 fd**（`fstat`/`fcntl` 可用）  
- mmap **必须 `MAP_SHARED`**（`PRIVATE` 改不可见）  
- 缩小后越过长度 → SIGBUS

## 代码示例

```c

/* shm_open / shm_unlink API 完整用法 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/mman.h>
#include <unistd.h>

int main(void) {
    /* 创建共享内存 */
    int fd = shm_open("/api_shm", O_CREAT | O_RDWR, 0644);
    if (fd < 0) { perror("shm_open"); exit(1); }
    printf("shm_open: fd=%d\n", fd);

    /* 设置大小（必须，否则 mmap 报 BUS error） */
    ftruncate(fd, 4096);

    /* 查看属性 */
    struct stat sb;
    fstat(fd, &sb);
    printf("size=%lld, mode=%04o\n", (long long)sb.st_size, sb.st_mode & 0777);

    /* 映射并使用 */
    char *p = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);  /* 映射后 fd 不再需要 */

    memset(p, 'A', 4096);
    printf("filled 4096 bytes with 'A'\n");

    munmap(p, 4096);

    /* shm_unlink 删除对象名（引用计数为 0 时释放） */
    shm_unlink("/api_shm");
    printf("shm_unlink: object marked for deletion\n");
    return 0;
}

```
