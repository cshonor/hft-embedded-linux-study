
## 代码示例

```c
#include <stdio.h>
#include <sys/sysinfo.h>
#include <unistd.h>
/* /proc 文件系统: 内核暴露的进程/系统信息 */
int main(void) {
    /* sysinfo: 系统统计 */
    struct sysinfo si;
    sysinfo(&si);
    printf("uptime=%ld, freeram=%lu MB\n",
           (long)si.uptime, (unsigned long)(si.freeram / 1024 / 1024));

    /* /proc/self/status: 进程统计 */
    printf("\n/proc/self/status (first 5 lines):\n");
    FILE *fp = fopen("/proc/self/status", "r");
    if (fp) {
        char line[256];
        for (int i = 0; i < 5 && fgets(line, sizeof(line), fp); i++)
            printf("  %s", line);
        fclose(fp);
    }
    return 0;
}
```
