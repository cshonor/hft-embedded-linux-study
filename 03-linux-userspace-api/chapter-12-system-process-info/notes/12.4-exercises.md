## 练习题

### 练习 1：读取进程内存映射

通过 `/proc/[pid]/maps` 读取指定进程的内存映射并解析打印。

**要求**：
- 接受 PID 参数
- 读取 `/proc/<pid>/maps` 文件
- 解析每行的地址范围、权限、映射对象
- 统计各段类型（text/data/heap/stack/anon/mapped）

### 练习 2：打印系统信息

使用 `uname()` 打印内核版本、架构等系统信息。

### 练习 3：进程状态监控

编写程序，读取 `/proc/self/status` 打印进程的 UID、RSS、线程数等关键信息。

## 参考答案

### 练习 1 参考代码

```c
#include "tlpi_hdr.h"
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
    unsigned long start, end;
    char perms[5];
    char path[256];
} map_entry_t;

int read_maps(pid_t pid) {
    char path[64];
    snprintf(path, sizeof(path), "/proc/%d/maps", pid);

    FILE *f = fopen(path, "r");
    if (!f) { perror("fopen"); return -1; }

    char line[512];
    int n = 0;
    while (fgets(line, sizeof(line), f)) {
        map_entry_t e;
        char offset[32], dev[32];
        long inode;

        /* 格式: start-end perms offset dev inode path */
        sscanf(line, "%lx-%lx %4s %s %s %ld %255[^\n]",
               &e.start, &e.end, e.perms, offset, dev, &inode, e.path);

        printf("%012lx-%012lx %s %s\n",
               e.start, e.end, e.perms,
               e.path[0] ? e.path : "[anon]");
        n++;
    }
    fclose(f);
    printf("total: %d mappings\n", n);
    return 0;
}

int main(int argc, char *argv[]) {
    if (argc != 2) usageErr("usage: %s <pid>\n", argv[0]);
    return read_maps(atoi(argv[1]));
}
```

### 练习 2 参考代码

```c
#include <sys/utsname.h>
#include <stdio.h>

int main(void) {
    struct utsname buf;
    if (uname(&buf) != 0) {
        perror("uname");
        return 1;
    }
    printf("OS name    : %s\n", buf.sysname);
    printf("Node name  : %s\n", buf.nodename);
    printf("Release    : %s\n", buf.release);
    printf("Version    : %s\n", buf.version);
    printf("Machine    : %s\n", buf.machine);
    printf("Domain     : %s\n", buf.domainname);
    return 0;
}
```

### 练习 3 参考代码

```c
#include <stdio.h>
#include <string.h>

int main(void) {
    FILE *f = fopen("/proc/self/status", "r");
    if (!f) { perror("fopen"); return 1; }

    char line[256];
    while (fgets(line, sizeof(line), f)) {
        if (strncmp(line, "Name:", 5) == 0 ||
            strncmp(line, "Pid:", 4) == 0 ||
            strncmp(line, "PPid:", 5) == 0 ||
            strncmp(line, "Uid:", 4) == 0 ||
            strncmp(line, "Gid:", 4) == 0 ||
            strncmp(line, "VmRSS:", 6) == 0 ||
            strncmp(line, "VmSize:", 7) == 0 ||
            strncmp(line, "Threads:", 8) == 0 ||
            strncmp(line, "State:", 6) == 0) {
            printf("%s", line);
        }
    }
    fclose(f);
    return 0;
}
```
