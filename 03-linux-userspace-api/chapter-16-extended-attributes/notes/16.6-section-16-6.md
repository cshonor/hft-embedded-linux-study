## 代码示例

```c
#include <sys/xattr.h>
#include <stdio.h>

int main(void) {
    /* xattr API 总结:
     * setxattr / getxattr / listxattr / removexattr
     * lsetxattr / lgetxattr / llistxattr / lremovexattr (不跟随符号链接)
     * fsetxattr / fgetxattr / flistxattr / fremovexattr (通过 fd)
     */

    const char *path = "/tmp/xattr_summary.txt";
    FILE *fp = fopen(path, "w"); if (fp) fclose(fp);

    setxattr(path, "user.note", "test", 4, 0);

    char buf[64];
    ssize_t n = getxattr(path, "user.note", buf, sizeof(buf));
    printf("getxattr: %.*s\n", (int)n, buf);

    removexattr(path, "user.note");
    printf("removed\n");
    return 0;
}
```
