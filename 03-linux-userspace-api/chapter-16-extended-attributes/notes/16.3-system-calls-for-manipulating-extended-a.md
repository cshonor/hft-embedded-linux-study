## 本节讲什么

六个调用（路径/fd × set/get/list）+ `removexattr` 的完整语义，四个实测钉住的点：

1. **探大小**的惯用法（buf=NULL）与三种失败（ENOATTR/ERANGE/EACCES）；
2. `listxattr` 返回的是 **NUL 拼接串**，不是指针数组；
3. 值是**二进制安全**的——double 往返无损（实测）；
4. `XATTR_CREATE` / `XATTR_REPLACE` 两个互斥语义（跨平台签名差异如何抹平）。

---

## 接口一览

```c
#include <sys/xattr.h>
/* 路径版（默认跟随链接） */              /* fd 版（无路径竞态） */
int setxattr(path, name, value, size, 0);     int fsetxattr(fd, name, value, size, 0);
ssize_t getxattr(path, name, buf, size);      ssize_t fgetxattr(fd, name, buf, size);
ssize_t listxattr(path, buf, size);           ssize_t flistxattr(fd, buf, size);
int removexattr(path, name);                  int fremovexattr(fd, name);
```

> **签名差异**：macOS 的每个函数多两个参数（`position`、`options`），且 flags 叫 `XATTR_CREATE/XATTR_REPLACE`（Linux 一致）。本目录 demo 用移植宏抹平，跨平台代码直接抄。

## 实测：探大小 + 二进制值 + list 格式

**完整源码（[`code/c16_1_xattr_basic.c`](../code/c16_1_xattr_basic.c)）：**

```c
/* c16_1_xattr_basic.c — Ch16 §16.3：setxattr/getxattr/listxattr/removexattr 四件套
 *
 * TLPI §16.3 必须实测钉住的点：
 *   ① xattr 是 name=value 对：名字带 namespace 前缀（user.），值是**二进制安全**的
 *      字节串——不解释、不补 NUL，size 自己管；
 *   ② getxattr 用 buf=NULL 探大小：返回值是"所需字节数"；缓冲小了 ERANGE，
 *      属性不存在 ENOATTR——三种返回都要分清；
 *   ③ listxattr 返回的是 **NUL 分隔的名字串拼接**（不是指针数组），
 *      返回值同样是"所需字节数"；
 *   ④ value 是二进制：存 struct、存 int 都合法（demo 里存一个 double）；
 *   ⑤ 同名重设 = 覆盖（默认语义），removexattr 之后 getxattr → ENOATTR。
 *
 * 本 demo 用移植宏抹平 Linux/macOS 的签名差异（macOS 多 position/options 参数）。
 * 编译： cc -Wall -Wextra -o c16_1_xattr_basic c16_1_xattr_basic.c
 * 取材： man-pages 6.19 xattr(7)（namespace 规则）+ getxattr(3)（ENOATTR/ERANGE）
 *       TLPI §16.1/§16.3
 */
#include <errno.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>
#include <unistd.h>

#if defined(__APPLE__)
#include <sys/xattr.h>
#define XSET(p,n,v,s)   setxattr(p,n,v,s,0,0)
#define XGET(p,n,v,s)   getxattr(p,n,v,s,0,0)
#define XLIST(p,l,s)    listxattr(p,l,s,0)
#define XRM(p,n)        removexattr(p,n,0)
#define ENOATTR_NAME    "user.comment"
#else
#include <sys/xattr.h>
#define XSET(p,n,v,s)   setxattr(p,n,v,s,0)
#define XGET(p,n,v,s)   getxattr(p,n,v,s)
#define XLIST(p,l,s)    listxattr(p,l,s)
#define XRM(p,n)        removexattr(p,n)
#define ENOATTR_NAME    ENODATA       /* Linux 上属性不存在返回 ENODATA */
#endif

int main(void)
{
    const char *path = "/tmp/tlpi_c16_1.txt";
    FILE *fp = fopen(path, "w");
    if (!fp) { perror("fopen"); return EXIT_FAILURE; }
    fputs("xattr demo file\n", fp);
    fclose(fp);

    /* ---------- ①② set + 探大小 + 取值 ---------- */
    printf("== ①② set/get：值是二进制安全的字节串 ==\n");
    const char *comment = "created by c16_1, TLPI ch16 demo";
    if (XSET(path, "user.comment", comment, strlen(comment)) == -1) {
        perror("setxattr user.comment"); return EXIT_FAILURE;
    }
    /* 存一个 double：证明 value 不解释内容 */
    double price = 12345.6789;
    if (XSET(path, "user.price", &price, sizeof price) == -1) {
        perror("setxattr user.price"); return EXIT_FAILURE;
    }

    ssize_t need = XGET(path, "user.comment", NULL, 0);   /* 探大小 */
    printf("  getxattr(buf=NULL) → %zd（属性值所需字节数）\n", need);
    char buf[128];
    ssize_t got = XGET(path, "user.comment", buf, sizeof buf);
    printf("  getxattr 实取 %zd 字节: \"%.*s\"\n", got, (int) got, buf);

    double price_back = 0;
    XGET(path, "user.price", &price_back, sizeof price_back);
    printf("  user.price 读回 double = %.4f（二进制往返无损）\n\n", price_back);

    /* ---------- 缓冲太小的 ERANGE ---------- */
    printf("== 缓冲太小：ERANGE ==\n");
    char tiny[4];
    errno = 0;
    ssize_t r = XGET(path, "user.comment", tiny, sizeof tiny);
    printf("  getxattr(4 字节缓冲) = %zd errno=%d(%s)\n\n",
           r, errno, strerror(errno));

    /* ---------- ③ listxattr：NUL 拼接串 ---------- */
    printf("== ③ listxattr：名字串 NUL 拼接，不是指针数组 ==\n");
    char list[1024];
    ssize_t lneed = XLIST(path, NULL, 0);
    ssize_t llen = XLIST(path, list, sizeof list);
    printf("  探大小=%zd，实取=%zd，逐个打印：\n", lneed, llen);
    for (ssize_t i = 0; i < llen; ) {
        const char *name = list + i;
        size_t len = strlen(name);
        if (strncmp(name, "user.", 5) == 0)
            printf("    [%s]\n", name);
        i += len + 1;
    }
    printf("  ⚠️ 名字串里 user.* 之外还可能有系统自己放的属性（如 macOS 的\n");
    printf("     com.apple.*），逐个按前缀过滤；返回值是「还需要多少字节」。\n\n");

    /* ---------- ⑤ 同名覆盖与删除 ---------- */
    printf("== ⑤ 同名重设=覆盖；removexattr 后 ENOATTR ==\n");
    const char *v2 = "shorter";
    XSET(path, "user.comment", v2, strlen(v2));
    got = XGET(path, "user.comment", buf, sizeof buf);
    printf("  重设后 user.comment = \"%.*s\"（%zd 字节，旧值被顶掉）\n",
           (int) got, buf, got);
    if (XRM(path, "user.comment") == -1) { perror("removexattr"); return EXIT_FAILURE; }
    errno = 0;
    r = XGET(path, "user.comment", buf, sizeof buf);
    printf("  删除后 getxattr = %zd errno=%d(%s)\n",
           r, errno, strerror(errno));
    printf("  ⚠️ 「属性不存在」Linux 返回 ENODATA(61)，macOS 返回 ENOATTR(93)——\n");
    printf("     名字与**数值都不同**，可移植代码必须两个宏都判（或判 errno==\n");
    printf("     ENODATA || errno==ENOATTR）。本机实测 errno=93 即 ENOATTR。\n");

    unlink(path);
    return EXIT_SUCCESS;
}
```

```c
double price = 12345.6789;
XSET(path, "user.price", &price, sizeof price);       /* 值=二进制 */
```

```text
== ①② set/get：值是二进制安全的字节串 ==
  getxattr(buf=NULL) → 32（属性值所需字节数）
  user.price 读回 double = 12345.6789（二进制往返无损）

== 缓冲太小：ERANGE ==
  getxattr(4 字节缓冲) = -1 errno=34(Result too large)

== ③ listxattr：名字串 NUL 拼接，不是指针数组 ==
  探大小=24，实取=24，逐个打印：
    [user.comment]
    [user.price]

== ⑤ 同名重设=覆盖；removexattr 后 ENOATTR ==
  重设后 user.comment = "shorter"（7 字节，旧值被顶掉）
  删除后 getxattr = -1 errno=93(Attribute not found)
  ⚠️ 「属性不存在」Linux 返回 ENODATA(61)，macOS 返回 ENOATTR(93)——
     名字与数值都不同，可移植代码必须两个宏都判。
```

**探大小惯用法**：`getxattr(path, name, NULL, 0)` 返回所需字节数 → 分配 → 再取。两步之间属性可能变大——生产代码要么给足余量再容 ERANGE 重试，要么用 fd 版在已知快照上操作。

**错误码数值实测**：macOS `ENOATTR=93`；Linux `ENODATA=61`。且 macOS 另有一个语义完全不同的 `ENODATA=96`（STREAM）——三处数值三处名，只能两宏都判。

## CREATE / REPLACE 语义

| flag | 属性已存在 | 属性不存在 |
|------|-----------|-----------|
| 无（0） | 覆盖 | 创建 |
| `XATTR_CREATE` | **EEXIST** | 创建 |
| `XATTR_REPLACE` | 覆盖 | **ENOATTR** |

用途：`CREATE` 防"意外顶掉别人的属性"（如占位键）；`REPLACE` 做"只更新已存在的"（如校验和刷新，键不存在本身就是错误）。macOS 的 options 位还提供 `XATTR_NOFOLLOW`（等价 lgetxattr 语义）。

```c
/* view_xattr.c

   Display the extended attributes of a file.

   This program is Linux (2.6 and later) specific.

   See also t_setxattr.c.
*/
#include <sys/xattr.h>
#include "tlpi_hdr.h"

#define XATTR_SIZE 10000

static void
usageError(char *progName)
{
    fprintf(stderr, "Usage: %s [-x] file...\n", progName);
    exit(EXIT_FAILURE);
}

int
main(int argc, char *argv[])
{
    char list[XATTR_SIZE], value[XATTR_SIZE];
    ssize_t listLen, valueLen;
    int ns, j, k, opt;
    Boolean hexDisplay;

    hexDisplay = 0;
    while ((opt = getopt(argc, argv, "x")) != -1) {
        switch (opt) {
        case 'x': hexDisplay = 1;       break;
        case '?': usageError(argv[0]);
        }
    }

    if (optind >= argc)
        usageError(argv[0]);

    for (j = optind; j < argc; j++) {
        listLen = listxattr(argv[j], list, XATTR_SIZE);
        if (listLen == -1)
            errExit("listxattr");

        printf("%s:\n", argv[j]);

        /* Loop through all EA names, displaying name + value */

        for (ns = 0; ns < listLen; ns += strlen(&list[ns]) + 1) {
            printf("        name=%s; ", &list[ns]);

            valueLen = getxattr(argv[j], &list[ns], value, XATTR_SIZE);
            if (valueLen == -1) {
                printf("couldn't get value");
            } else if (!hexDisplay) {
                printf("value=%.*s", (int) valueLen, value);
            } else {
                printf("value=");
                for (k = 0; k < valueLen; k++)
                    printf("%02x ", (unsigned char) value[k]);
            }

            printf("\n");
        }

        printf("\n");
    }

    exit(EXIT_SUCCESS);
}
```

（**原书 Listing 16-1**，dist 逐字镜像）遍历文件列表，`listxattr` + `getxattr` 打印所有属性（支持 `-x` 十六进制）。它依赖 `XATTR_SIZE`（自定 65536）的大缓冲一次取全——书里的简化写法；生产代码建议探大小+重试。

**macOS 注意**：`t_setxattr.c`（dist 补充程序，5 参签名）仅可在 Linux 编译；macOS 上编译请改走本目录 demo 的移植宏。镜像本身**逐字保留**（Linux 复测平台可直接编）。

## 常见坑

| 坑 | 正解 |
|----|------|
| 拿 listxattr 返回值当属性个数 | 是**字节数**；要数 NUL 自己数 |
| 值是字符串就补 NUL | xattr 不解释内容；存字符串要不要带 NUL 由你的读侧约定，get 返回的 size 才是真相 |
| ENODATA 只判 Linux 宏 | macOS 是 ENOATTR(93)，数值都不同；两宏都判 |
| listxattr 缓冲刚好等于第一次探测值 | 探与取之间属性可能变多 → ERANGE；重试循环 |
| `XATTR_CREATE` 用来做锁 | 不是原子的跨进程互斥；竞态仍在（两进程同 create 一个不存在键，仍有窗口） |

## HFT / 嵌入式关联

| 场景 | 本节机制怎么兑现 |
|------|----------------|
| 热数据标签 | `(dev,ino)` 级别的元数据（校验和、采样窗口）走 fgetxattr，inode 常驻 cache，读开销近零 |
| 写侧自检 | `fsetxattr` 写 checksum + `XATTR_CREATE` 防重放覆盖，读侧不匹配即拒收 |
| 错误处理统一 | ENOATTR/ENODATA 两宏都判（跨平台工具的必修课） |
| 嵌入式 | 属性值大小提前探测+上限断言，避免落独立块引发的额外擦写 |

<summary>代码自测（Q&A）</summary>

**Q1：getxattr 一次调用怎么同时区分"属性不存在"和"缓冲太小"？**
A：看 errno：ENOATTR/ENODATA=不存在；ERANGE=缓冲不够（此时返回值是所需字节数，可用它重试）。EACCES 则是权限问题。

**Q2：listxattr 的返回值是什么？**
A：**所需/写入的字节数**（NUL 拼接串的总长），不是属性个数。buf=NULL 时返回"需要多少"。

**Q3：值是 8 字节 double，getxattr 后可以直接转回吗？**
A：可以（实测往返无损），但生产环境建议值里带版本/长度头——xattr 是裸字节，跨机器/跨版本读侧要自校验。

**Q4：setxattr 默认语义是什么？**
A：同名覆盖（不报错）。要"必须新建"用 XATTR_CREATE（已存在则 EEXIST），要"必须已存在"用 XATTR_REPLACE（不存在则 ENOATTR）。

**Q5：为什么本目录 demo 要用宏包 xattr 调用？**
A：macOS 签名多 position/options 两个参数。宏在编译期抹平，业务代码两平台共用一份。

</details>
