## 要点

```c
#include <sys/xattr.h>

int setxattr(const char *path, const char *name,
             const void *value, size_t size, int flags);
int lsetxattr(...);   /* 不跟随链接 */
int fsetxattr(int fd, ...);

ssize_t getxattr(const char *path, const char *name, void *value, size_t size);
ssize_t lgetxattr(...);
ssize_t fgetxattr(int fd, ...);

int removexattr(const char *path, const char *name);
int lremovexattr(...);
int fremovexattr(int fd, ...);

ssize_t listxattr(const char *path, char *list, size_t size);
ssize_t llistxattr(...);
ssize_t flistxattr(int fd, ...);
```

### `setxattr` flags

| flag | 行为 |
|------|------|
| `0`（默认） | 无则建，有则换 |
| `XATTR_CREATE` | 仅新建；已存在 → `EEXIST` |
| `XATTR_REPLACE` | 仅更新；不存在 → `ENODATA`（文档/历史上也称 ENOATTR） |

## 代码示例

```c
#include <stdio.h>
#include <sys/xattr.h>
#include <string.h>
/* xattr 系统调用族: set/get/list/remove */
int main(void) {
    const char *path = "/tmp/xattr_syscall_test.txt";
    FILE *fp = fopen(path, "w"); fprintf(fp, "x"); fclose(fp);

    /* setxattr: 设置 */
    setxattr(path, "user.author", "alice", 5, 0);
    setxattr(path, "user.priority", "high", 4, 0);

    /* listxattr: 列出所有 key */
    char list[256];
    ssize_t llen = listxattr(path, list, sizeof(list));
    printf("xattr keys (%zd bytes):\n", llen);
    for (ssize_t i = 0; i < llen; ) {
        printf("  %s\n", list + i);
        i += strlen(list + i) + 1;
    }

    /* getxattr: 读取单个 */
    char val[32];
    ssize_t vlen = getxattr(path, "user.author", val, sizeof(val));
    if (vlen > 0) { val[vlen] = '\0'; printf("user.author = %s\n", val); }

    /* removexattr: 删除 */
    removexattr(path, "user.priority");
    printf("After removexattr(user.priority):\n");
    llen = listxattr(path, list, sizeof(list));
    printf("  remaining keys: %zd bytes\n", llen);

    remove(path);
    return 0;
}
```
