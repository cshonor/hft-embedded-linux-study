## 要点

初值 1；P=`-1`；V=`+1`（常加 `SEM_UNDO`）。  
⚠️ **无所有权**：任意进程可 V「别人」持有的锁（异于 `pthread_mutex`）。

---

## 代码示例

```c
#include <stdio.h>
#include <sys/sem.h>
#include <unistd.h>
#include <sys/wait.h>
/* 用信号量实现互斥锁 (mutex) */
union semun { int val; };

int main(void) {
    int semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666);
    /* 初始化为 1 (未锁定) */
    union semun arg = {.val = 1};
    semctl(semid, 0, SETVAL, arg.val);

    /* 两个子进程争抢锁 */
    for (int i = 0; i < 2; i++) {
        if (fork() == 0) {
            struct sembuf lock = {0, -1, SEM_UNDO};   /* P = lock */
            struct sembuf unlock = {0, 1, SEM_UNDO};   /* V = unlock */

            semop(semid, &lock, 1);    /* 获取锁 */
            printf("Process %d: acquired lock\n", i);
            sleep(1);                   /* 临界区 */
            printf("Process %d: releasing lock\n", i);
            semop(semid, &unlock, 1);  /* 释放锁 */
            _exit(0);
        }
    }

    for (int i = 0; i < 2; i++) wait(NULL);
    printf("All done, final value=%d\n", semctl(semid, 0, GETVAL));
    semctl(semid, 0, IPC_RMID);
    return 0;
}
```
