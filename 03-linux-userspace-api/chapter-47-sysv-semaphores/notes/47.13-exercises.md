### 练习 1：二元互斥锁

**题目**：使用 SysV 信号量实现二元互斥锁（初始值为1），包含 `SEM_UNDO` 保护。

<details>
<summary>参考答案</summary>

```c
#include <sys/sem.h>

union semun { int val; struct semid_ds *buf; unsigned short *array; };

typedef struct {
    int semid;
    int semnum;
} sem_mutex_t;

int sem_mutex_init(sem_mutex_t *m, key_t key) {
    m->semid = semget(key, 1, IPC_CREAT | 0666);
    if (m->semid == -1) return -1;
    m->semnum = 0;

    /* 初始化为 1（解锁状态） */
    union semun arg;
    arg.val = 1;
    return semctl(m->semid, 0, SETVAL, arg);
}

int sem_mutex_lock(sem_mutex_t *m) {
    struct sembuf op = {
        .sem_num = m->semnum,
        .sem_op = -1,        /* P: 值减1 */
        .sem_flg = SEM_UNDO, /* 退出时自动恢复 */
    };
    int ret;
    do {
        ret = semop(m->semid, &op, 1);
    } while (ret == -1 && errno == EINTR);
    return ret;
}

int sem_mutex_unlock(sem_mutex_t *m) {
    struct sembuf op = {
        .sem_num = m->semnum,
        .sem_op = +1,        /* V: 值加1 */
        .sem_flg = SEM_UNDO,
    };
    return semop(m->semid, &op, 1);
}

void sem_mutex_destroy(sem_mutex_t *m) {
    semctl(m->semid, 0, IPC_RMID);
}
```

</details>

### 练习 2：生产者-消费者

**题目**：用 SysV 信号量集（3个信号量）实现生产者-消费者同步。

<details>
<summary>参考答案</summary>

```c
/* 信号量集：
 * sem[0] = mutex (1, 互斥访问缓冲区)
 * sem[1] = empty (N, 空位数量)
 * sem[2] = full  (0, 满位数量)
 */
#define SEM_MUTEX 0
#define SEM_EMPTY 1
#define SEM_FULL  2
#define BUF_SIZE 10

void producer(int semid) {
    struct sembuf ops[2];
    /* P(empty), P(mutex) */
    ops[0] = (struct sembuf){SEM_EMPTY, -1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_MUTEX, -1, SEM_UNDO};
    semop(semid, ops, 2);

    /* 写入数据到缓冲区 */
    /* ... */

    /* V(mutex), V(full) */
    ops[0] = (struct sembuf){SEM_MUTEX, +1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_FULL,  +1, SEM_UNDO};
    semop(semid, ops, 2);
}

void consumer(int semid) {
    struct sembuf ops[2];
    /* P(full), P(mutex) */
    ops[0] = (struct sembuf){SEM_FULL,  -1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_MUTEX, -1, SEM_UNDO};
    semop(semid, ops, 2);

    /* 从缓冲区读取数据 */
    /* ... */

    /* V(mutex), V(empty) */
    ops[0] = (struct sembuf){SEM_MUTEX, +1, SEM_UNDO};
    ops[1] = (struct sembuf){SEM_EMPTY, +1, SEM_UNDO};
    semop(semid, ops, 2);
}
```

### 练习 3：初始化竞争

**题目**：以下代码有什么竞争？如何修复？

```c
int semid = semget(key, 1, IPC_CREAT | 0666);
semctl(semid, 0, SETVAL, 1);  /* 初始化 */
```

<details>
<summary>参考答案</summary>

**竞争**：两个进程同时执行此代码，`semget` 返回同一个 semid（已存在），但两者都调用 `SETVAL` 设置为 1。如果在这期间有第三个进程执行 `semop(P)`，可能看到值 0（未初始化）。

**修复**：使用 `IPC_EXCL` 确保只有创建者初始化：

```c
int semid = semget(key, 1, IPC_CREAT | IPC_EXCL | 0666);
if (semid != -1) {
    /* 我们是创建者，负责初始化 */
    union semun arg = {.val = 1};
    semctl(semid, 0, SETVAL, arg);
} else if (errno == EEXIST) {
    /* 已存在，打开即可 */
    semid = semget(key, 1, 0666);
    /* 等待创建者初始化完成 */
    struct semid_ds ds;
    union semun arg = {.buf = &ds};
    while (ds.sem_otime == 0) {
        semctl(semid, 0, IPC_STAT, arg);
        usleep(1000);
    }
}
```

</details>
