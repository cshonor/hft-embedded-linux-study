## 要点

```c
if (fork() == 0) {
    /* redirect / close fds */
    execvp(...);
    _exit(127);   /* 禁止 exit() */
}
```

## 代码示例

```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <fcntl.h>
#include <string.h>
/* fork+exec 范式: 父进程通过管道与 exec 后的子进程通信 */
int main(void) {
    int pipefd[2];
    pipe(pipefd);

    pid_t pid = fork();
    if (pid == 0) {
        /* 子进程: 重定向 stdout 到管道写端, 然后 exec */
        close(pipefd[0]);
        dup2(pipefd[1], STDOUT_FILENO);
        close(pipefd[1]);

        /* exec 后 fd 继承: pipefd[1] 现在是 stdout */
        execlp("ls", "ls", "-la", "/tmp", NULL);
        perror("exec");
        _exit(127);
    }

    /* 父进程: 从管道读端读取子进程输出 */
    close(pipefd[1]);
    char buf[4096];
    ssize_t n;
    while ((n = read(pipefd[0], buf, sizeof(buf))) > 0)
        write(STDOUT_FILENO, buf, n);
    close(pipefd[0]);

    int status;
    waitpid(pid, &status, 0);
    printf("\n[child exit status: %d]\n", WEXITSTATUS(status));
    return 0;
}
```
