## 代码示例

```c
#include <unistd.h>
#include <stdio.h>
/* shebang 内核处理:
 * exec 读取文件前 2 字节 → "#!" → 提取解释器路径
 * 用解释器 + 脚本路径重新 exec
 * 限制: 路径长度 ≤128 字符, SUID 无效
 */
int main(void) {
    pid_t pid = fork();
    if (pid == 0) {
        char *args[] = {"/bin/sh", "-c", "echo $0", "myscript", NULL};
        execv("/bin/sh", args);
        _exit(1);
    }
    wait(NULL);
    return 0;
}
```
