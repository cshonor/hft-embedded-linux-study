| 路线 | 要点 |

## 代码示例

```c
#include <stdio.h>
#include <sys/shm.h>
#include <unistd.h>
/* 共享内存配额与操作限制 */
int main(void) {
    /* 系统限制: 查看配置 */
    FILE *fp = fopen("/proc/sys/kernel/shmmax", "r");
    if (fp) {
        char buf[32];
        fgets(buf, sizeof(buf), fp);
        printf("SHMMAX (max segment size): %s", buf);
        fclose(fp);
    }

    fp = fopen("/proc/sys/kernel/shmall", "r");
    if (fp) {
        char buf[32];
        fgets(buf, sizeof(buf), fp);
        printf("SHMALL (total pages): %s", buf);
        fclose(fp);
    }

    /* 尝试创建超过 SHMMAX 的段 */
    int shmid = shmget(IPC_PRIVATE, 1UL << 32, IPC_CREAT | 0666);
    if (shmid < 0)
        printf("\nshmget(4GB) failed (exceeds SHMMAX)\n");
    else
        shmctl(shmid, IPC_RMID, NULL);

    /* 查看当前 IPC 对象 */
    printf("\nUse 'ipcs -m' to list shared memory segments\n");
    printf("Use 'ipcrm -m <id>' to remove\n");
    return 0;
}
```
