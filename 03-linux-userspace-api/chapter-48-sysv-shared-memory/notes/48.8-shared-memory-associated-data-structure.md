# 48.8 Shared Memory Associated Data Structure

> 本章：[TLPI 第 48 章](./README.md)

## 本节讲什么

共享内存段关联的 `shmid_ds` 结构，包含权限、时间戳和统计信息。通过 `shmctl(IPC_STAT)` 获取，用于监控和调试。

## 要点

### struct shmid_ds

```c
struct shmid_ds {
    struct ipc_perm shm_perm;   /* 权限和属主 */
    size_t          shm_segsz;  /* 段大小（字节） */
    time_t          shm_atime;   /* 最后 shmat 时间 */
    time_t          shm_dtime;   /* 最后 shmdt 时间 */
    time_t          shm_ctime;   /* 最后 shmctl 修改时间 */
    pid_t           shm_cpid;   /* 创建者 PID */
    pid_t           shm_lpid;   /* 最后操作 PID */
    shmatt_t        shm_nattch;  /* 当前映射进程数 */
};
```

### 字段用途

| 字段 | 监控用途 |
|------|---------|
| `shm_segsz` | 段大小，验证分配 |
| `shm_nattch` | 映射数，检测泄漏（应=预期进程数） |
| `shm_atime` | 最近 attach，活跃度 |
| `shm_dtime` | 最近 detach，活跃度 |
| `shm_ctime` | 最近权限变更 |
| `shm_cpid` | 创建者，审计 |
| `shm_lpid` | 最近操作者，审计 |

### struct ipc_perm

```c
struct ipc_perm {
    key_t  __key;   /* IPC key */
    uid_t  uid;     /* 所有者 UID */
    gid_t  gid;     /* 所有者 GID */
    uid_t  cuid;    /* 创建者 UID */
    gid_t  cgid;    /* 创建者 GID */
    unsigned short mode;   /* 权限位 */
    unsigned short __seq;  /* 序列号 */
};
```

### ipcs -m 输出对应

```
$ ipcs -m
------ Shared Memory Segments --------
key        shmid owner  perms  bytes   nattch  status
0xabcd1234 0     hft    660    4096    2
```

| 列 | 对应字段 |
|----|---------|
| key | `shm_perm.__key` |
| shmid | shmget 返回值 |
| owner | `shm_perm.uid` |
| perms | `shm_perm.mode` |
| bytes | `shm_segsz` |
| nattch | `shm_nattch` |
| status | dest=已标记删除 |

### 系统级信息 (SHM_INFO)

```c
struct shminfo info;
shmctl(0, SHM_INFO, (struct shmid_ds *)&info);
/* info.shmmni = 系统最大段数 */
/* info.shmmax = 最大段大小 */
```

### 系统限制

| 参数 | 文件 | 典型值 |
|------|------|--------|
| `SHMMNI` | `/proc/sys/kernel/shmmni` | 4096 |
| `SHMMAX` | `/proc/sys/kernel/shmmax` | 18EB |
| `SHMALL` | `/proc/sys/kernel/shmall` | 页数 |

## HFT / 嵌入式关联

- HFT 监控共享内存映射数：nattch 应等于预期进程数
- 嵌入式系统 `ipcs -m` 定期检查残留段
- 权限审计：确保只有授权进程可映射

## 代码示例

```c
#include <sys/shm.h>
#include <stdio.h>
#include <time.h>

void dump_shm_info(int shmid) {
    struct shmid_ds ds;
    if (shmctl(shmid, IPC_STAT, &ds) == -1) {
        perror("shmctl");
        return;
    }

    time_t now = time(NULL);
    printf("=== Shared Memory %d ===\n", shmid);
    printf("Key:           0x%x\n", ds.shm_perm.__key);
    printf("Size:          %zu bytes\n", ds.shm_segsz);
    printf("Attaches:      %lu\n", ds.shm_nattch);
    printf("Creator PID:   %d\n", ds.shm_cpid);
    printf("Last op PID:   %d\n", ds.shm_lpid);
    printf("Owner:         %d:%d\n", ds.shm_perm.uid, ds.shm_perm.gid);
    printf("Creator:       %d:%d\n", ds.shm_perm.cuid, ds.shm_perm.cgid);
    printf("Permissions:   %04o\n", ds.shm_perm.mode);

    if (ds.shm_atime)
        printf("Last attach:   %ld sec ago\n", now - ds.shm_atime);
    else
        printf("Last attach:   never\n");

    if (ds.shm_dtime)
        printf("Last detach:   %ld sec ago\n", now - ds.shm_dtime);
    else
        printf("Last detach:   never\n");

    printf("Last change:   %ld sec ago\n", now - ds.shm_ctime);
}

/* 列出系统所有共享内存段 */
void list_all_shm(void) {
    struct shmid_ds ds;
    int maxid;
    /* SHM_INFO 获取系统级信息 */
    maxid = shmctl(0, SHM_INFO, &ds);
    printf("Max shmid index: %d\n", maxid);

    /* 遍历可能的 shmid */
    for (int i = 0; i <= maxid; i++) {
        if (shmctl(i, SHM_STAT, &ds) != -1) {
            printf("shmid=%d  key=0x%x  size=%zu  nattch=%lu\n",
                   i, ds.shm_perm.__key,
                   ds.shm_segsz, ds.shm_nattch);
        }
    }
}
```

## 代码自测

<details>
<summary>shm_nattch 为 0 但段仍然存在，这是什么情况？</summary>

`shm_nattch` 为 0 表示没有进程映射该段，但段尚未被 `IPC_RMID` 删除。SysV 共享内存是内核持久的——即使没有使用者，段也保留在内核中直到显式删除。这是 SysV IPC 的设计缺陷：进程崩溃后不会自动清理，导致资源泄漏。需要用 `ipcrm -m <shmid>` 手动清理。

</details>
