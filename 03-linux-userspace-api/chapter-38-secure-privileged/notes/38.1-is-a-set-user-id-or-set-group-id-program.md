## 代码示例

```c
#include <unistd.h>
#include <stdio.h>
/* 编写 setuid 程序的安全原则:
 * 1. 最小权限: 用完特权立即丢弃
 * 2. 净化环境: 清除 LD_PRELOAD 等危险变量
 * 3. 检查所有返回值
 * 4. 不用 system(), 用 execve
 */
int main(void) {
    printf("EUID=%d\n", geteuid());
    if (geteuid() == 0) {
        printf("特权程序! 需格外小心\n");
        /* 永久丢弃特权 */
        setuid(getuid());
        printf("特权已丢弃: EUID=%d\n", geteuid());
    }
    return 0;
}
```
