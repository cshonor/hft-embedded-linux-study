## 代码示例

```c
#include <unistd.h>
#include <signal.h>
#include <stdio.h>
int main(void) {
    pid_t pgid = getpgrp();
    printf("进程组 ID = %d\n", pgid);
    printf("进程 PID = %d\n", getpid());
    printf("会话 ID = %d\n", getsid(0));

    /* setsid: 创建新会话+进程组 */
    pid_t pid = fork();
    if (pid == 0) {
        setsid();
        printf("子进程: 新 PGID=%d, SID=%d\n", getpgrp(), getsid(0));
        _exit(0);
    }
    wait(NULL);
    return 0;
}
```
